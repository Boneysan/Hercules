// ── Shared Arc 16 helpers ──────────────────────────────────────────────────

function	script	DM_Arc16AddClue	{
	.@bit = getarg(0);
	if (!(dm_arc16_evidence_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc16_evidence_mask", dm_arc16_evidence_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc16_evidence_mask);
	callfunc("DM_DMJStoryObjective", 20201, 2, "Explore", .@n, 3, (.@n >= 3), getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc16_clue_" + .@bit))
		callfunc("DM_PartyExp", 200000, 100000);
	if (.@n >= 2 && questprogress(20204) == 0)
		callfunc("DM_InstanceQuestStart", 20204);
	if (.@n >= 3)
		callfunc("DM_PartyMes", "^3355FF[All three clues. The route into the vault is known, and one more witness is out of Heine's reach of the cult.]^000000");
	else if (.@n == 2)
		callfunc("DM_PartyMes", "^3355FF[Two clues. The infiltration route is clear. Matron Rina can be answered.]^000000");
	else
		callfunc("DM_PartyMes", "^3355FF[One clue recorded.]^000000");
	return .@n;
}

// mode: 1 expose, 2 monitored dismantling, 3 defection with protection terms. Final.
function	script	DM_Arc16Commit	{
	.@mode = getarg(0);
	if (dm_arc16_rina_resolution)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc16_rina_resolution", .@mode);
	callfunc("DM_InstanceClearFlag", "dm_arc16_rina_exposed");
	callfunc("DM_InstanceClearFlag", "dm_arc16_rina_rolled");
	callfunc("DM_InstanceClearFlag", "dm_arc16_rina_defected");
	if (.@mode == 1) {
		callfunc("DM_InstanceSetFlag", "dm_arc16_rina_exposed", 1);
	} else if (.@mode == 2) {
		callfunc("DM_InstanceSetFlag", "dm_arc16_rina_rolled", 1);
	} else {
		callfunc("DM_InstanceSetFlag", "dm_arc16_rina_defected", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc16_release_known", 1);
	}
	callfunc("DM_InstanceQuestComplete", 20204);
	callfunc("DM_DMJStoryObjective", 20204, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc16_resolution"))
		callfunc("DM_PartyExp", 400000, 200000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1605);
	return 1;
}

// The arc closes one of two ways, never both. freed=1 only after Maret's Cell
// was opened; otherwise Bijou was killed.
function	script	DM_Arc16Finish	{
	.@freed = getarg(0);
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc16_bijou", .@party))
		return 0;
	callfunc("DM_InstanceQuestComplete", 20201);
	if (.@freed) {
		callfunc("DM_InstanceClearFlag", "dm_arc16_bijou_killed");
		callfunc("DM_InstanceSetFlag", "dm_arc16_maret_freed", 1);
	} else {
		callfunc("DM_InstanceClearFlag", "dm_arc16_maret_freed");
		callfunc("DM_InstanceSetFlag", "dm_arc16_bijou_killed", 1);
	}
	callfunc("DM_PartyExp", 2000000, 1000000, .@party);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1699);
	if (.@freed)
		instance_announce(instance_id(), "^3355FF[DM] Maret is out of the cell. The cascade signal in the prison breaks. She surfaces for the first time in three years and does not know what month it is.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] Bijou falls silent. The cascade loses its court-facing relay. The royal army is secured, what remains of it, for the final war.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The court is secured. Varmundt's research is the last piece before Morroc's ruins.^000000", bc_all);
	return 1;
}
