// ── Shared Arc 17 helpers ──────────────────────────────────────────────────

function	script	DM_Arc17Step	{
	.@what$ = getarg(0);
	.@n = (dm_arc17_design_recovered ? 1 : 0) + (dm_arc17_chamber_isolated ? 1 : 0) + callfunc("DM_CountBits", dm_arc17_fault_mask);
	callfunc("DM_DMJStoryObjective", 20211, 2, "Explore", .@n, 5, (.@n >= 5), getcharid(CHAR_ID_PARTY));
	callfunc("DM_PartyMes", "^3355FF[" + .@what$ + " Test progress: " + .@n + " of 5.]^000000");
	return .@n;
}

function	script	DM_Arc17FaultFixed	{
	.@bit = getarg(0);
	if (!(dm_arc17_fault_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc17_fault_mask", dm_arc17_fault_mask | .@bit);
	if (callfunc("DM_ClaimGrant", "arc17_fault_" + .@bit))
		callfunc("DM_PartyExp", 80000, 40000);
	callfunc("DM_Arc17Step", "Fault cleared.");
	if (dm_arc17_fault_mask == 7 && !dm_arc17_prototype_verified) {
		callfunc("DM_InstanceSetFlag", "dm_arc17_prototype_verified", 1);
		if (callfunc("DM_ClaimGrant", "arc17_prototype")) {
			callfunc("DM_PartyExp", 160000, 85000);
			callfunc("DM_InstanceQuestStart", 20214);
		}
		callfunc("DM_PartyMes", "^3355FF[Prototype cycle complete. The counter-frequency holds against all three faults. The feasibility record is written.]^000000");
	}
	return 1;
}

// mode: 1 export then shut down, 2 retain under limits, 3 broader partnership. Final.
function	script	DM_Arc17Commit	{
	.@mode = getarg(0);
	if (dm_arc17_admin_resolution)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc17_admin_resolution", .@mode);
	callfunc("DM_InstanceClearFlag", "dm_arc17_admin_purged");
	callfunc("DM_InstanceClearFlag", "dm_arc17_admin_negotiated");
	callfunc("DM_InstanceClearFlag", "dm_arc17_admin_running");
	if (.@mode == 1)
		callfunc("DM_InstanceSetFlag", "dm_arc17_admin_purged", 1);
	else if (.@mode == 2)
		callfunc("DM_InstanceSetFlag", "dm_arc17_admin_negotiated", 1);
	else
		callfunc("DM_InstanceSetFlag", "dm_arc17_admin_running", 1);
	callfunc("DM_InstanceQuestComplete", 20214);
	callfunc("DM_DMJStoryObjective", 20214, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc17_resolution"))
		callfunc("DM_PartyExp", 440000, 215000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1705);
	return 1;
}
