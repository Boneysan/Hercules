// Final commitment. Re-checks eligibility and claims the transition once per
// party, so two conversations cannot commit two endings. Returns 1, or a
// negative reason: -1 not ready, -2 volunteer absent, -3 already committed.
function	script	DM_Arc19Commit	{
	.@e = getarg(0);
	if (callfunc("DM_Arc19Ready", .@e) != 2)
		return -1;
	if (.@e == 4 && dm_arc19_volunteer_kind == 1 && !callfunc("DM_Arc19VolunteerPresent"))
		return -2;
	if (dm_arc19_final_choice)
		return -3;
	if (!callfunc("DM_ClaimGrant", "arc19_choice"))
		return -3;
	callfunc("DM_InstanceSetFlag", "dm_arc19_final_choice", .@e);
	callfunc("DM_InstanceClearFlag", "dm_finale_shared_seal");
	callfunc("DM_InstanceClearFlag", "dm_finale_reforged_seal");
	callfunc("DM_InstanceClearFlag", "dm_finale_queens_bargain");
	callfunc("DM_InstanceClearFlag", "dm_finale_thanatos_road");
	callfunc("DM_InstanceClearFlag", "dm_finale_ragnarok_unbound");
	switch (.@e) {
	case 1: callfunc("DM_InstanceSetFlag", "dm_finale_shared_seal", 1); break;
	case 2: callfunc("DM_InstanceSetFlag", "dm_finale_reforged_seal", 1); break;
	case 3: callfunc("DM_InstanceSetFlag", "dm_finale_queens_bargain", 1); break;
	case 4: callfunc("DM_InstanceSetFlag", "dm_finale_thanatos_road", 1); break;
	case 5: callfunc("DM_InstanceSetFlag", "dm_finale_ragnarok_unbound", 1); break;
	}
	return 1;
}

		break;
	}
	next();
	return;

// Three outcome cards from actual state: a place, a named person, a new obligation.
S_Cards:
	.@e = getarg(0);
	switch (.@e) {
	case 1:
		.@place$ = (dm_arc12_survey_agreement == 1) ? "The New World camp" : ((dm_arc07_labor_resolution == 1 || dm_arc07_labor_resolution == 2) ? "The south foundry at Einbroch" : "The ward stones above Hugel");

		.@person$ = (dm_arc17_maintenance_secured == 1) ? "Greta Holm" : "Doctor Mira Tressa";
		.@duty$ = "The machine is kept running by the team that agreed. If they stop, it stops, and everyone should know that.";
		break;
	case 3:
		.@place$ = "The northern gate of Niflheim";
		.@person$ = "Himmelmez";
		.@duty$ = "Her pact: the boundary, her hours, and the living remembering her. It is a pact, not a guarantee.";

		.@place$ = "The Tower at Aldebaran";
		.@person$ = getd("$dm_arc19_vol_name_" + getcharid(CHAR_ID_PARTY) + "$");

// ── Epilogue: what the earlier choices left standing ──────────────────────
S_Epilogue:
	.@ending = getarg(0, 0);
	.@volunteer$ = getarg(1, "");
	.@said = 0;
	callfunc("DM_PartyMes", "[Loki The Voice]");
	if (dm_arc15_pratt_exposed) {
		callfunc("DM_PartyMes", "Pratt's list has names on it now, and each of them was asked. Thanatos never had that list.");
	} else if (dm_arc15_pratt_delayed) {
		callfunc("DM_PartyMes", "Pratt's council review is still open. The method has not been used on anyone yet. That is as much as anyone can honestly say.");
	}
	switch (.@ending) {
	case 1:
		if (dm_arc19_ally_turnout == 0)
			callfunc("DM_PartyMes", "Nobody came when the seal needed carriers. It holds on the people who were standing there, and they will have to find more.");
		else if (dm_arc19_ally_turnout < 3)
			callfunc("DM_PartyMes", "A few groups carry it, and they know each other's names.");
		else
			callfunc("DM_PartyMes", "Enough groups carry it that no one of them is the lock.");
		if (dm_arc16_rina_defected) {
			callfunc("DM_PartyMes", "Rina is on the roster. She walked free once. She will be asked again.");
		}
		if (dm_arc12_vance_exposed) {
			callfunc("DM_PartyMes", "Vance pulled his team back and filed an honest report. His superiors have not answered it.");
		} else if (dm_arc12_vance_helped) {
			callfunc("DM_PartyMes", "Vance's crews are on stabilization work instead of resource survey. They are the only outside hands the seal has.");
		}
		break;
	case 2:
		if (dm_arc17_admin_purged)
			callfunc("DM_PartyMes", "The Administrator is dark. The transfer was clean, and nothing is left that can answer when the counter-frequency drifts.");
		else if (dm_arc17_admin_negotiated)
			callfunc("DM_PartyMes", "The Administrator still answers queries. Whoever keeps the machine running can ask it how.");
		else if (dm_arc17_admin_running)
			callfunc("DM_PartyMes", "The Biosphere Core is still running, a defence nobody can switch off, in the same place as the machine that holds the seal.");
		else
			callfunc("DM_PartyMes", "Someone will have to keep it running. Nobody has been asked yet.");
		if (dm_arc07_strike_supported) {
			callfunc("DM_PartyMes", "Einbroch's machinists keep it tuned. They remember who stood with them in the strike.");
		} else if (dm_arc07_strike_broken) {
			callfunc("DM_PartyMes", "Einbroch's machinists keep it running too. They remember who broke the strike, and nothing obliges them to be quick about repairs.");
		} else if (dm_arc07_strike_negotiated) {
			callfunc("DM_PartyMes", "Einbroch's machinists keep it running under a monitored arrangement both sides signed. Neither trusts the other. It holds.");
		}
		break;
	case 3:
		if (dm_arc18_himmelmez_killed)
			callfunc("DM_PartyMes", "Nobody can tell you whether this was a bargain at all. The only party to it was not there to agree.");
		else if (dm_arc18_himmelmez_bargained)
			callfunc("DM_PartyMes", "Himmelmez chose this, aloud, in front of you. That is more than most of the dead were given.");
		else
			callfunc("DM_PartyMes", "Himmelmez was never asked in the end. The pact is held on the strength of an old promise.");
		break;
	case 4:
		if (.@volunteer$ != "")
			callfunc("DM_PartyMes", .@volunteer$ + " goes into the lock knowing what it costs. They said so out loud, and the party read the cost before they did. That is the only thing about this seal that Thanatos's did not have.");
		else
			callfunc("DM_PartyMes", "Whoever goes into the lock goes knowing what it costs. That is the only thing about this seal that Thanatos's did not have.");
		break;
	case 5:
		if (dm_arc19_ally_turnout > 0)
			callfunc("DM_PartyMes", "The groups who owe you passage are standing at the edge of the rift. They did not agree to this either.");
		else
			callfunc("DM_PartyMes", "Nobody else is there. Whatever rises will be met by the people who chose to let it.");
		break;
	}
	next();
	callfunc("DM_PartyMes", "[Loki]");
	callfunc("DM_PartyMes", "That is what you leave behind. Not what you meant. What you did.");
	return;
}
