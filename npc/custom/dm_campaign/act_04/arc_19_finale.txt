	callfunc("DM_PartyMes", "There are five ways to close the cascade. You have done the work to make all five possible. Only you have done enough of that work to know which one is right.");
	next();
	callfunc("DM_PartyMes", "[Loki]");
	callfunc("DM_PartyMes", "Face Surt. He holds the focal point open. When the focal point closes, the choice will be available. The Ash Vacuum Rift at moc_fild22.");
	next();
	callfunc("DM_PartyMes", "^3355FF[The Choice He Never Had - go to moc_fild22 and face Surt. The Central Choice follows.]^000000");

moc_fild22,170,140,0	script	The Ash Vacuum Rift#dm	HIDDEN_NPC,{
	end;

OnRiftStart:
	stopnpctimer;
	'rift_map$ = strnpcinfo(NPC_MAP);
	'rift_party_id = $@dm_arc19_rift_party_id;
	'rift_damage = $@dm_arc19_rift_damage;
	'rift_status = $@dm_arc19_rift_status;
	'rift_status_ms = $@dm_arc19_rift_status_ms;
	if ('rift_party_id > 0) {
		mapannounce 'rift_map$, "[Ash Vacuum Rift] The focal point is open. The seal is screaming.", bc_map, 0xFF6600;
		initnpctimer;
	}
	end;

OnRiftStop:
	stopnpctimer;
	end;

OnTimer8000:
	mapannounce 'rift_map$, "[Ash Vacuum Rift] The cascade hammers against the old seal.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'rift_map$, 170, 140, 10, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	end;

OnTimer16000:
	mapannounce 'rift_map$, "[Ash Vacuum Rift] A pressure wave rolls out from Surt's focal point.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'rift_map$, 170, 140, 10, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	end;

OnTimer24000:
	mapannounce 'rift_map$, "[Ash Vacuum Rift] The old lock flexes and nearly gives way.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'rift_map$, 170, 140, 10, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	end;

OnTimer32000:
	mapannounce 'rift_map$, "[Ash Vacuum Rift] The rift spends its last violent surge.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'rift_map$, 170, 140, 10, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	stopnpctimer;
	end;

OnSurtDead:
	stopnpctimer;
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc19_surt", .@party))
		end;
	callfunc("DM_InstanceQuestComplete", 20231);
	callfunc("DM_InstanceSetFlag", "dm_arc19_surt_defeated", 1);
	callfunc("DM_PartyExp", 5000000, 3000000, .@party);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1902);
	if (dm_arc18_himmelmez_bargained)
		instance_announce(instance_id(), "^3355FF[DM] Surt disperses. The focal point closes. From Niflheim, Himmelmez says nothing — but the dead stop walking north for the first time in three years.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] Surt disperses. The focal point closes. The cascade's resonance drops to silence for the first time in three hundred years. The silence is louder than you expected.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The seal point is open. The Central Choice is ahead. This is what everything was for.^000000", bc_all);
	end;
}

	switch (select("The Shared Seal — many carry what one could not hold alone.", "The Reforged Seal — Varmundt's counter-frequency rebuilds the lock.", "The Queen's Bargain — Himmelmez holds the seal from the other side.", "Thanatos's Road — one person chooses the seal, knowing the cost.", "Ragnarok Unbound — the seal is not rebuilt. Let what is beneath rise and be faced.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_finale_shared_seal", 1);
		callfunc("DM_PartyMes", "^3355FF[The Shared Seal]^000000");
		callfunc("DM_PartyMes", "The party distributes the resonance across a network of willing carriers. No single person becomes the lock. The seal holds through collective intention. It requires maintenance. It requires trust. It will last exactly as long as the trust does.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_finale_reforged_seal", 1);
		callfunc("DM_PartyMes", "^3355FF[The Reforged Seal]^000000");
		callfunc("DM_PartyMes", "Varmundt's inverse-frequency counter-broadcasts at the cascade's exact harmonic inverse. The seal closes mechanically. No living lock. No collective. A machine that cancels itself. It requires power and maintenance. It will last as long as someone keeps it running.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_finale_queens_bargain", 1);
		callfunc("DM_PartyMes", "^3355FF[The Queen's Bargain]^000000");
		callfunc("DM_PartyMes", "Himmelmez holds the seal from Niflheim's side. The dead become the lock. This is what the cascade feared — that the dead and the living would cooperate on the seal rather than divide it between them. The seal is permanent while Himmelmez holds it. The cost is hers. She already knew that when she made the pact.");
		break;
	case 4:
		callfunc("DM_InstanceSetFlag", "dm_finale_thanatos_road", 1);
		callfunc("DM_PartyMes", "^3355FF[Thanatos's Road]^000000");
		callfunc("DM_PartyMes", "One person chooses the seal, knowing the full cost. Not ignorantly, the way Thanatos could not have known. With full knowledge of what three centuries in the lock means. The seal is perfect. The person is permanent. Scholar Pratt's volunteers are gone. The choice is for whoever is willing to make it knowing.");
		break;
	case 5:
		callfunc("DM_InstanceSetFlag", "dm_finale_ragnarok_unbound", 1);
		callfunc("DM_PartyMes", "^3355FF[Ragnarok Unbound]^000000");
		callfunc("DM_PartyMes", "The seal is not rebuilt. What Thanatos sealed rises. It has been in there for three hundred years. The cascade's amplification has been feeding it. What rises is not what went in. It may be reasoned with. It may not. Midgard faces what it has been avoiding facing and finds out what that is.");
		break;
	}
	next();
	callfunc("DM_PartyMes", "^FF6600[DM] The Seal Cascade ends. The party has shaped the fate of Midgard.^000000");
	callfunc("DM_InstanceQuestComplete", 20233);
	callfunc("DM_InstanceSetFlag", "dm_campaign_complete", 1);
	if (callfunc("DM_ClaimGrant", "arc19_finale"))
		callfunc("DM_PartyExp", 1000000, 500000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1999);
	announce("[DM] The Seal Cascade ends. The choice has been made. The shape of the next age is yours.", bc_all, 0xFF6600);
	close();

	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20233, 1;
	end;
