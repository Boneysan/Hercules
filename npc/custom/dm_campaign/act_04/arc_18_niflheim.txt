// ── Shared Arc 18 helpers ──────────────────────────────────────────────────

function	script	DM_Arc18AddAudience	{
	.@bit = getarg(0);
	if (!(dm_arc18_audience_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc18_audience_mask", dm_arc18_audience_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc18_audience_mask);
	callfunc("DM_DMJStoryObjective", 20221, 2, "Talk", .@n, 2, (.@n >= 2), getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc18_audience_" + .@bit))
		callfunc("DM_PartyExp", 450000, 220000);
	if (.@n >= 2)
		callfunc("DM_PartyMes", "^3355FF[Both audiences heard. Himmelmez will receive you.]^000000");
	else
		callfunc("DM_PartyMes", "^3355FF[One audience heard.]^000000");
	return .@n;
}

// The one place the arc closes and pays, once, whichever way it went.
function	script	DM_Arc18Close	{
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc18_himmelmez", .@party))
		return 0;
	callfunc("DM_InstanceQuestComplete", 20221);
	callfunc("DM_PartyExp", 2000000, 1200000, .@party);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1899);
	return 1;
}
