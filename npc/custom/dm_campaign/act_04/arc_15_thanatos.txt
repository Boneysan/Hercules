// ── Shared Arc 15 helpers ──────────────────────────────────────────────────

function	script	DM_Arc15AddMemory	{
	.@bit = getarg(0);
	if (!(dm_arc15_memory_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc15_memory_mask", dm_arc15_memory_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc15_memory_mask);
	callfunc("DM_DMJStoryObjective", 20191, 2, "Explore", .@n, 3, (.@n >= 3), getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc15_memory_" + .@bit))
		callfunc("DM_PartyExp", 200000, 100000);
	if (.@n >= 2 && questprogress(20194) == 0)
		callfunc("DM_InstanceQuestStart", 20194);
	if (.@n >= 3) {
		callfunc("DM_InstanceSetFlag", "dm_arc15_cost_disclosed", 1);
		callfunc("DM_PartyMes", "^3355FF[All three memories seen. What the seal costs is on the record: a life, held for as long as it holds, with no relief and no end.]^000000");
	} else if (.@n == 2) {
		callfunc("DM_PartyMes", "^3355FF[Two memories seen. Scholar Pratt can be answered now. The third completes the record.]^000000");
	} else {
		callfunc("DM_PartyMes", "^3355FF[One memory seen.]^000000");
	}
	return .@n;
}

// mode: 1 stop and name, 2 delay, 3 full record and consent procedure. Final.
function	script	DM_Arc15Commit	{
	.@mode = getarg(0);
	if (dm_arc15_pratt_resolution)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc15_pratt_resolution", .@mode);
	callfunc("DM_InstanceClearFlag", "dm_arc15_pratt_exposed");
	callfunc("DM_InstanceClearFlag", "dm_arc15_pratt_delayed");
	callfunc("DM_InstanceClearFlag", "dm_arc15_pratt_challenged");
	if (.@mode == 1)
		callfunc("DM_InstanceSetFlag", "dm_arc15_pratt_exposed", 1);
	else if (.@mode == 2)
		callfunc("DM_InstanceSetFlag", "dm_arc15_pratt_delayed", 1);
	else
		callfunc("DM_InstanceSetFlag", "dm_arc15_pratt_challenged", 1);
	callfunc("DM_InstanceQuestComplete", 20194);
	callfunc("DM_DMJStoryObjective", 20194, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc15_resolution"))
		callfunc("DM_PartyExp", 400000, 200000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1505);
	return 1;
}

	'resonance_party_id = $@dm_arc15_resonance_party_id;
	'resonance_damage = $@dm_arc15_resonance_damage;
	'resonance_status = $@dm_arc15_resonance_status;
	'resonance_status_ms = $@dm_arc15_resonance_status_ms;
	if ('resonance_party_id > 0) {
		mapannounce 'resonance_map$, "[Thanatos Memory] The tower begins replaying the cost of the seal.", bc_map, 0xCC99FF;
		initnpctimer;
	}
	end;

OnResonanceStop:
	stopnpctimer;
	end;

OnTimer6000:
	mapannounce 'resonance_map$, "[Thanatos Memory] A remembered oath crushes the chamber.", bc_map, 0xCC99FF;
	callfunc("DM_HazardArea", 'resonance_map$, 150, 150, 9, 'resonance_damage, 'resonance_status, 'resonance_status_ms, 'resonance_party_id);
	end;

OnTimer12000:
	mapannounce 'resonance_map$, "[Thanatos Memory] Four absent volunteers answer in echo.", bc_map, 0xCC99FF;
	callfunc("DM_HazardArea", 'resonance_map$, 150, 150, 9, 'resonance_damage, 'resonance_status, 'resonance_status_ms, 'resonance_party_id);
	end;

OnTimer18000:
	mapannounce 'resonance_map$, "[Thanatos Memory] The seal's pressure finds every living heartbeat.", bc_map, 0xCC99FF;
	callfunc("DM_HazardArea", 'resonance_map$, 150, 150, 9, 'resonance_damage, 'resonance_status, 'resonance_status_ms, 'resonance_party_id);
	end;

OnTimer24000:
	mapannounce 'resonance_map$, "[Thanatos Memory] The tower resonance fractures and fades.", bc_map, 0xCC99FF;
	callfunc("DM_HazardArea", 'resonance_map$, 150, 150, 9, 'resonance_damage, 'resonance_status, 'resonance_status_ms, 'resonance_party_id);
	stopnpctimer;
	end;

OnThanatosDead:
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc15_thanatos", .@party))
		end;
	stopnpctimer;
	killmonster strnpcinfo(NPC_MAP), "Memory of Thanatos#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20191);
	callfunc("DM_InstanceSetFlag", "dm_arc15_thanatos_killed", 1);
	callfunc("DM_PartyExp", 2000000, 1000000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1599);
	if (dm_arc15_pratt_challenged)
		instance_announce(instance_id(), "^3355FF[DM] The Memory stills. Thanatos looks at you from inside the seal. He knows you know what he is still doing. That is the most alone thing in the world.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] The Memory fades. Thanatos did not slay Morroc. He became the seal. He is still in there. He has been in there for three hundred years.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The truth is yours. The next seal must be different. Prontera's court is the next thread.^000000", bc_all);
	end;

OnAddDead:
	end;
