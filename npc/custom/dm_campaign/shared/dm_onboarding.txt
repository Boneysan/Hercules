iz_int,23,28,5	script	Campaign Guide#dm_start	4_F_01,{
	mes("[Campaign Guide]");
	mes("If you are here for Seal Cascade, you do not need to clear the training route first.");
	next();
	mes("[Campaign Guide]");
	mes("The campaign gathers at the Session Board near the Prontera fountain.");
	next();
	switch (select("Take me to the Session Board.", "Show me where it is.", "I'll stay here.")) {
	case 1:
		close2();
		warp("prontera", 155, 185);
		end;
	case 2:
		mes("[Campaign Guide]");
		mesf("Look for %s.", mesnavigation("[Session Board]", "prontera", 155, 185));
		close();
	case 3:
		mes("[Campaign Guide]");
		mes("No trouble. The regular training route is still open.");
		close();
	}
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_JOB, Job_Novice;
	end;
}
