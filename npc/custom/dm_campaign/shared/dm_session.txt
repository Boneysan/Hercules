// Seal Cascade session tools: cleanup, scene claim, encounter records,
// and Act I "this arc is actually finished" helpers.
//
// Scene claim stops two party members from committing the same menu.
// Encounter records stop instance copies and @dm cleanup from awarding
// a fight that never belonged to this party.

function	script	DM_CleanupMap	{
	.@map$ = getarg(0, callfunc("DM_CurrentMap"));

	killmonster(.@map$, "DM_Console::OnDMKilled", 1);
	killmonster(.@map$, "DM_Console::OnDMBossKilled", 1);
	killmonster(.@map$, "DM_Console::OnDMCleanup", 1);
	@dm_hazard_ticks = 0;
	callfunc("DM_EncSweepMap", .@map$);
	mapannounce(.@map$, "[DM] Encounter cleanup complete.", bc_map, 0xAAAAAA);
	return 1;
}

// ---------------------------------------------------------------------------
// Scene claim
//
// Store party, owner char, and unix tick under $dm_sc_*. A claim older than
// 60 seconds is abandoned so a disconnected speaker cannot brick the scene.
// ---------------------------------------------------------------------------

function	script	DM_SceneKeyOk	{
	.@s$ = getarg(0);
	.@n = getstrlen(.@s$);
	if (.@n < 3)
		return 0;
	for (.@i = 0; .@i < .@n; ++.@i) {
		if (!callfunc("DM_IsIdentChar", substr(.@s$, .@i, .@i)))
			return 0;
	}
	return 1;
}

function	script	DM_SceneRelease	{
	.@s$ = getarg(0);
	if (!callfunc("DM_SceneKeyOk", .@s$))
		return 0;
	setd("$dm_sc_p_" + .@s$, 0);
	setd("$dm_sc_o_" + .@s$, 0);
	setd("$dm_sc_t_" + .@s$, 0);
	return 1;
}

function	script	DM_SceneBusy	{
	.@s$ = getarg(0);
	if (!callfunc("DM_SceneKeyOk", .@s$))
		return 0;
	.@p = getd("$dm_sc_p_" + .@s$);
	if (.@p <= 0)
		return 0;
	.@tick = getd("$dm_sc_t_" + .@s$);
	// 180s: a table can talk through a long scene. Abandoned means the
	// owner has been gone, not that a pause to discuss should brick it.
	if (gettimetick(2) - .@tick > 180) {
		callfunc("DM_SceneRelease", .@s$);
		return 0;
	}
	return 1;
}

function	script	DM_SceneTouch	{
	.@s$ = getarg(0);
	if (!callfunc("DM_SceneBusy", .@s$))
		return 0;
	if (getd("$dm_sc_o_" + .@s$) != getcharid(CHAR_ID_CHAR))
		return 0;
	setd("$dm_sc_t_" + .@s$, gettimetick(2));
	return 1;
}

function	script	DM_SceneOwnerParty	{
	.@s$ = getarg(0);
	if (!callfunc("DM_SceneBusy", .@s$))
		return 0;
	return getd("$dm_sc_p_" + .@s$);
}

function	script	DM_SceneClaim	{
	.@s$ = getarg(0);
	.@party = getarg(1, getcharid(CHAR_ID_PARTY));
	.@cid = getcharid(CHAR_ID_CHAR);
	if (!callfunc("DM_SceneKeyOk", .@s$))
		return 0;
	if (.@party <= 0)
		.@party = .@cid;
	if (callfunc("DM_SceneBusy", .@s$)) {
		if (getd("$dm_sc_p_" + .@s$) == .@party && getd("$dm_sc_o_" + .@s$) == .@cid) {
			setd("$dm_sc_t_" + .@s$, gettimetick(2));
			return 1;
		}
		return 0;
	}
	setd("$dm_sc_p_" + .@s$, .@party);
	setd("$dm_sc_o_" + .@s$, .@cid);
	setd("$dm_sc_t_" + .@s$, gettimetick(2));
	return 1;
}

function	script	DM_SceneJoin	{
	.@s$ = getarg(0);
	.@party = getarg(1, getcharid(CHAR_ID_PARTY));
	if (!callfunc("DM_SceneBusy", .@s$))
		return 0;
	if (.@party <= 0)
		.@party = getcharid(CHAR_ID_CHAR);
	if (getd("$dm_sc_p_" + .@s$) != .@party)
		return 0;
	setd("$dm_sc_t_" + .@s$, gettimetick(2));
	return 1;
}

function	script	DM_SceneFail	{
	.@s$ = getarg(0);
	callfunc("DM_PartyMes", "^777777Someone in the party is already speaking this scene. Wait, or listen.^000000");
	return 0;
}

// ---------------------------------------------------------------------------
// Encounter record
// ---------------------------------------------------------------------------

function	script	DM_EncStart	{
	.@s$ = getarg(0);
	.@party = getarg(1, getcharid(CHAR_ID_PARTY));
	.@map$ = getarg(2, strnpcinfo(NPC_MAP));
	.@boss$ = getarg(3, "");
	.@add$ = getarg(4, "");
	.@npc$ = getarg(5, strnpcinfo(NPC_NAME));
	if (!callfunc("DM_SceneKeyOk", .@s$))
		return 0;
	.@gen = getd("$dm_enc_g_" + .@s$) + 1;
	setd("$dm_enc_g_" + .@s$, .@gen);
	setd("$dm_enc_p_" + .@s$, .@party);
	setd("$dm_enc_m_" + .@s$ + "$", .@map$);
	setd("$dm_enc_b_" + .@s$ + "$", .@boss$);
	setd("$dm_enc_a_" + .@s$ + "$", .@add$);
	setd("$dm_enc_n_" + .@s$ + "$", .@npc$);
	return .@gen;
}

function	script	DM_EncOnDeath	{
	.@s$ = getarg(0);
	.@party = getarg(1, getcharid(CHAR_ID_PARTY));
	if (!callfunc("DM_SceneKeyOk", .@s$))
		return 0;
	if (getd("$dm_enc_g_" + .@s$) <= 0)
		return 0;
	if (getd("$dm_enc_p_" + .@s$) != .@party)
		return 0;
	return 1;
}

function	script	DM_EncCleanup	{
	.@s$ = getarg(0);
	if (!callfunc("DM_SceneKeyOk", .@s$))
		return 0;
	.@map$ = getd("$dm_enc_m_" + .@s$ + "$");
	.@boss$ = getd("$dm_enc_b_" + .@s$ + "$");
	.@add$ = getd("$dm_enc_a_" + .@s$ + "$");
	.@npc$ = getd("$dm_enc_n_" + .@s$ + "$");
	if (.@map$ != "") {
		if (.@boss$ != "")
			killmonster(.@map$, .@boss$, 1);
		if (.@add$ != "")
			killmonster(.@map$, .@add$, 1);
	}
	if (.@npc$ != "")
		set getvariableofnpc('boss_up, .@npc$), 0;
	setd("$dm_enc_p_" + .@s$, 0);
	setd("$dm_enc_m_" + .@s$ + "$", "");
	setd("$dm_enc_b_" + .@s$ + "$", "");
	setd("$dm_enc_a_" + .@s$ + "$", "");
	setd("$dm_enc_n_" + .@s$ + "$", "");
	callfunc("DM_SceneRelease", .@s$);
	return 1;
}

function	script	DM_EncSweepMap	{
	.@map$ = getarg(0);
	setarray .@scenes$[0],
		"arc01_holt", "arc02_voss", "arc02_spirit",
		"arc03_osiris", "arc03_amon", "arc03_ckpt_a", "arc03_ckpt_b",
		"arc04_baphomet", "arc05_hold", "arc05_tao";
	.@n = getarraysize(.@scenes$);
	for (.@i = 0; .@i < .@n; ++.@i) {
		if (getd("$dm_enc_m_" + .@scenes$[.@i] + "$") == .@map$)
			callfunc("DM_EncCleanup", .@scenes$[.@i]);
	}
	return 1;
}

function	script	DM_Arc01Clues	{
	.@m = getarg(0, dm_arc01_clue_mask);
	.@n = 0;
	if (.@m & 1) ++.@n;
	if (.@m & 2) ++.@n;
	if (.@m & 4) ++.@n;
	return .@n;
}

function	script	DM_Arc01ChamberReady	{
	if (callfunc("DM_Arc01Clues") >= 2)
		return 1;
	if ((dm_arc01_clue_mask & 3) && dm_arc01_tibbets_befriended)
		return 1;
	return 0;
}

function	script	DM_CountBits	{
	.@m = getarg(0);
	.@n = 0;
	.@b = 1;
	while (.@b > 0 && .@b <= .@m) {
		if (.@m & .@b)
			++.@n;
		.@b *= 2;
	}
	return .@n;
}

// OnInit walk audit. Unwalkable campaign NPCs print [Error] DM walk
// so tools/check-campaign.sh fails instead of shipping a click-through wall.
function	script	DM_AssertWalk	{
	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	if (.@m$ == "")
		return 1;
	if (checkcell(.@m$, .@x, .@y, cell_chkpass))
		return 1;
	consolemes(CONSOLEMES_ERROR, "DM walk: %s %s,%d,%d", strnpcinfo(NPC_NAME), .@m$, .@x, .@y);
	for (.@r = 1; .@r <= 4; ++.@r) {
		for (.@dx = -.@r; .@dx <= .@r; ++.@dx) {
			for (.@dy = -.@r; .@dy <= .@r; ++.@dy) {
				if (.@dx == 0 && .@dy == 0)
					continue;
				if (checkcell(.@m$, .@x + .@dx, .@y + .@dy, cell_chkpass)) {
					consolemes(CONSOLEMES_ERROR, "DM walk hint: %s try %s,%d,%d", strnpcinfo(NPC_NAME), .@m$, .@x + .@dx, .@y + .@dy);
					return 0;
				}
			}
		}
	}
	return 0;
}
