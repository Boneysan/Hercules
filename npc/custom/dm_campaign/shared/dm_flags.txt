// Seal Cascade story flag helpers.

function	script	DM_SetFlag	{
	.@flag$ = getarg(0);
	.@value = getarg(1, 1);

	if (.@flag$ == "")
		return 0;

	setd(.@flag$, .@value);
	return .@value;
}

function	script	DM_GetFlag	{
	.@flag$ = getarg(0);

	if (.@flag$ == "")
		return 0;

	return getd(.@flag$);
}

function	script	DM_ClearFlag	{
	.@flag$ = getarg(0);

	if (.@flag$ == "")
		return 0;

	setd(.@flag$, 0);
	return 1;
}

function	script	DM_ArcFlagCount	{
	.@arc = getarg(0, 0);

	if (.@arc < 1)
		return 0;

	return getelementofarray(getvariableofnpc(.arc_count, "DM_FlagRegistry"), .@arc);
}

function	script	DM_ArcFlagName	{
	.@arc = getarg(0, 0);
	.@i = getarg(1, 0);
	.@count = callfunc("DM_ArcFlagCount", .@arc);

	if (.@i < 0 || .@i >= .@count)
		return "";

	.@start = getelementofarray(getvariableofnpc(.arc_start, "DM_FlagRegistry"), .@arc);
	return getelementofarray(getvariableofnpc(.flag_names$, "DM_FlagRegistry"), .@start + .@i);
}

function	script	DM_ArcFlagMax	{
	return getvariableofnpc(.arc_max, "DM_FlagRegistry");
}

function	script	DM_ClearArcFlags	{
	.@arc = getarg(0, 0);
	.@count = callfunc("DM_ArcFlagCount", .@arc);

	for (.@i = 0; .@i < .@count; ++.@i) {
		.@flag$ = callfunc("DM_ArcFlagName", .@arc, .@i);
		if (.@flag$ != "")
			.@done += callfunc("DM_ClearFlag", .@flag$);
	}
	return .@done;
}

function	script	DM_PrintArcFlags	{
	.@arc = getarg(0, 0);
	.@count = callfunc("DM_ArcFlagCount", .@arc);
	.@arc$ = (.@arc < 10 ? "0" : "") + .@arc;

	if (.@count < 1) {
		dispbottom("[DM] Arc " + .@arc$ + " has no registered flags.");
		return 0;
	}

	dispbottom("[DM] Arc " + .@arc$ + " flags:");
	.@line$ = "  ";
	for (.@i = 0; .@i < .@count; ++.@i) {
		.@flag$ = callfunc("DM_ArcFlagName", .@arc, .@i);
		if (.@flag$ == "")
			continue;
		.@entry$ = .@flag$ + "=" + callfunc("DM_GetFlag", .@flag$);
		if (.@line$ == "  ")
			.@line$ = "  " + .@entry$;
		else
			.@line$ = .@line$ + " " + .@entry$;
		if ((.@i % 4) == 3) {
			dispbottom(.@line$);
			.@line$ = "  ";
		}
	}
	if (.@line$ != "  ")
		dispbottom(.@line$);
	return .@count;
}

function	script	DM_ClearAllArcFlags	{
	.@max = callfunc("DM_ArcFlagMax");

	for (.@arc = 1; .@arc <= .@max; ++.@arc)
		.@done += callfunc("DM_ClearArcFlags", .@arc);
	return .@done;
}

function	script	DM_PartyClearArcFlags	{
	.@arc = getarg(0, 0);
	.@party_id = getarg(1, getcharid(CHAR_ID_PARTY));
	.@count = callfunc("DM_ArcFlagCount", .@arc);

	for (.@i = 0; .@i < .@count; ++.@i) {
		.@flag$ = callfunc("DM_ArcFlagName", .@arc, .@i);
		if (.@flag$ != "")
			.@done += callfunc("DM_PartyClearFlag", .@flag$, .@party_id);
	}
	return .@done;
}

function	script	DM_FlagSyncFrom	{
	.@source$ = getarg(0, "");
	.@party_id = getarg(1, getcharid(CHAR_ID_PARTY));
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@source$ == "") {
		dispbottom("[DM] Usage: @dm flag sync <online_party_member>");
		return 0;
	}
	if (.@party_id <= 0) {
		dispbottom("[DM] Join or create a party first.");
		return 0;
	}

	.@source_aid = getcharid(CHAR_ID_ACCOUNT, .@source$);
	if (.@source_aid <= 0) {
		dispbottom("[DM] Player '" + .@source$ + "' is not online.");
		return 0;
	}
	if (!attachrid(.@source_aid)) {
		if (.@origin_aid)
			attachrid(.@origin_aid);
		dispbottom("[DM] Could not attach to '" + .@source$ + "'.");
		return 0;
	}

	.@source_cid = getcharid(CHAR_ID_CHAR);
	.@source_name$ = strcharinfo(PC_NAME);
	if (getcharid(CHAR_ID_PARTY) != .@party_id) {
		detachrid();
		if (.@origin_aid)
			attachrid(.@origin_aid);
		dispbottom("[DM] " + .@source_name$ + " is not in your current party.");
		return 0;
	}

	.@n = 0;
	.@max = callfunc("DM_ArcFlagMax");
	for (.@arc = 1; .@arc <= .@max; ++.@arc) {
		.@count = callfunc("DM_ArcFlagCount", .@arc);
		for (.@i = 0; .@i < .@count; ++.@i) {
			.@flag$ = callfunc("DM_ArcFlagName", .@arc, .@i);
			if (.@flag$ == "")
				continue;
			.@flags$[.@n] = .@flag$;
			.@values[.@n] = getd(.@flag$);
			++.@n;
		}
	}
	detachrid();

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (.@charid[.@i] == .@source_cid)
			continue;
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		if (!attachrid(.@accid[.@i]))
			continue;
		for (.@f = 0; .@f < .@n; ++.@f) {
			setd(.@flags$[.@f], .@values[.@f]);
			++.@writes;
		}
		detachrid();
		++.@members;
	}

	if (.@origin_aid)
		attachrid(.@origin_aid);

	dispbottom("[DM] Synced " + .@n + " flags from " + .@source_name$ + " to " + .@members + " member(s) (" + .@writes + " flag writes).");
	return .@members;
}

S_RegisterArc:
	.@arc = getarg(0);
	.@count = getarraysize(getarg(1));
	.arc_start[.@arc] = getarraysize(.flag_names$);
	.arc_count[.@arc] = .@count;

	for (.@i = 0; .@i < .@count; ++.@i)
		.flag_names$[.arc_start[.@arc] + .@i] = getelementofarray(getarg(1), .@i);
	return;
}
