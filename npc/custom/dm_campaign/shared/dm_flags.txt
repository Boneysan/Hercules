// Seal Cascade story flag helpers.

function	script	DM_SetFlag	{
	.@flag$ = getarg(0);
	.@value = getarg(1, 1);

	if (.@flag$ == "")
		return 0;

	setd(.@flag$, .@value);
	return .@value;
}

function	script	DM_GetFlag	{
	.@flag$ = getarg(0);

	if (.@flag$ == "")
		return 0;

	return getd(.@flag$);
}

function	script	DM_ClearFlag	{
	.@flag$ = getarg(0);

	if (.@flag$ == "")
		return 0;

	setd(.@flag$, 0);
	return 1;
}

function	script	DM_ClearArc01Flags	{
	callfunc("DM_ClearFlag", "dm_arc01_started");
	callfunc("DM_ClearFlag", "dm_arc01_tibbets_befriended");
