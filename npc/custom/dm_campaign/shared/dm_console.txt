// Seal Cascade in-game DM command console.
//
// NOTE: bindatcmd fills .@atcmd_parameters$ / .@atcmd_numparameters in the event
// scope. callsub()/callfunc() start a *new* .@ scope, so those vars are empty
// inside S_* labels. We bridge them into temporary character vars (@dm_atcmd_*)
// before every callsub that needs parameters.

-	script	DM_Console	FAKE_NPC,{
	end;

OnInit:
	bindatcmd("dm", strnpcinfo(NPC_NAME_UNIQUE)+"::OnDM", 1, 99, 1);
	bindatcmd("dmreward", strnpcinfo(NPC_NAME_UNIQUE)+"::OnReward", 1, 99, 1);
	bindatcmd("dmflag", strnpcinfo(NPC_NAME_UNIQUE)+"::OnFlag", 1, 99, 1);
	bindatcmd("dmquest", strnpcinfo(NPC_NAME_UNIQUE)+"::OnQuest", 1, 99, 1);
	bindatcmd("dmbeat", strnpcinfo(NPC_NAME_UNIQUE)+"::OnBeat", 1, 99, 1);
	bindatcmd("dmguide", strnpcinfo(NPC_NAME_UNIQUE)+"::OnGuide", 1, 99, 1);
	bindatcmd("dmstory", strnpcinfo(NPC_NAME_UNIQUE)+"::OnStory", 1, 99, 1);
	bindatcmd("dmcleanup", strnpcinfo(NPC_NAME_UNIQUE)+"::OnCleanup", 1, 99, 1);
	bindatcmd("dmwarp", strnpcinfo(NPC_NAME_UNIQUE)+"::OnWarp", 1, 99, 1);
	bindatcmd("dmrecall", strnpcinfo(NPC_NAME_UNIQUE)+"::OnRecall", 1, 99, 1);
	bindatcmd("dminstance", strnpcinfo(NPC_NAME_UNIQUE)+"::OnInstance", 1, 99, 1);
	bindatcmd("dmmode", strnpcinfo(NPC_NAME_UNIQUE)+"::OnMode", 1, 99, 1);
	bindatcmd("dmhazard", strnpcinfo(NPC_NAME_UNIQUE)+"::OnHazard", 1, 99, 1);
	bindatcmd("dmexp", strnpcinfo(NPC_NAME_UNIQUE)+"::OnExp", 1, 99, 1);
	bindatcmd("dmreset", strnpcinfo(NPC_NAME_UNIQUE)+"::OnReset", 1, 99, 1);
	bindatcmd("dmstatus", strnpcinfo(NPC_NAME_UNIQUE)+"::OnStatus", 1, 99, 1);
	bindatcmd("roll", strnpcinfo(NPC_NAME_UNIQUE)+"::OnRoll", 0, 99, 1);
	// Skill checks are player-facing: anyone may roll their own or help.
	bindatcmd("check", strnpcinfo(NPC_NAME_UNIQUE)+"::OnCheck", 0, 99, 1);
	bindatcmd("assist", strnpcinfo(NPC_NAME_UNIQUE)+"::OnAssist", 0, 99, 1);
	bindatcmd("dmcheck", strnpcinfo(NPC_NAME_UNIQUE)+"::OnDMCheck", 1, 99, 1);
	bindatcmd("dmstakes", strnpcinfo(NPC_NAME_UNIQUE)+"::OnStakes", 1, 99, 1);
	bindatcmd("dmenc", strnpcinfo(NPC_NAME_UNIQUE)+"::OnEnc", 1, 99, 1);
	end;

OnDM:
	if (!callfunc("DM_RequireDM"))
		end;

	// Bridge into @ temps before any callsub (callsub opens a new .@ scope).
	deletearray @dm_atcmd_p$[0], 128;
	copyarray @dm_atcmd_p$[0], .@atcmd_parameters$[0], .@atcmd_numparameters;
	@dm_atcmd_n = .@atcmd_numparameters;

	if (.@atcmd_numparameters < 1) {
		callsub(S_Help);
		end;
	}

	.@sub$ = .@atcmd_parameters$[0];

	if (.@sub$ == "help") {
		callsub(S_Help);
		end;
	}

	if (.@sub$ == "reward") {
		callsub(S_Reward, 1);
		end;
	}

	if (.@sub$ == "flag") {
		callsub(S_Flag, 1);
		end;
	}

	if (.@sub$ == "quest") {
		callsub(S_Quest, 1);
		end;
	}

	if (.@sub$ == "beat") {
		callfunc("DM_BeatMenu");
		end;
	}
	if (.@sub$ == "guide" || .@sub$ == "handbook") {
		callfunc("DM_HandbookMenu");
		end;
	}

	if (.@sub$ == "story") {
		.@text$ = "";
		for (.@i = 1; .@i < .@atcmd_numparameters; ++.@i)
			.@text$ += (.@i > 1 ? " " : "") + .@atcmd_parameters$[.@i];
		callfunc("DM_MapStory", .@text$);
		end;
	}

	if (.@sub$ == "globalstory") {
		.@text$ = "";
		for (.@i = 1; .@i < .@atcmd_numparameters; ++.@i)
			.@text$ += (.@i > 1 ? " " : "") + .@atcmd_parameters$[.@i];
		callfunc("DM_GlobalStory", .@text$);
		end;
	}

	if (.@sub$ == "spawn") {
		callsub(S_Spawn, 1);
		end;
	}

	if (.@sub$ == "cleanup") {
		callfunc("DM_CleanupMap");
		end;
	}

	if (.@sub$ == "warp") {
		callsub(S_Warp, 1);
		end;
	}

	if (.@sub$ == "recall") {
		callsub(S_Recall);
		end;
	}

	if (.@sub$ == "instance") {
		callsub(S_Instance, 1);
		end;
	}

	if (.@sub$ == "mode") {
		callsub(S_Mode, 1);
		end;
	}

	if (.@sub$ == "hazard") {
		callsub(S_Hazard, 1);
		end;
	}

	if (.@sub$ == "exp") {
		callsub(S_Exp, 1);
		end;
	}

	if (.@sub$ == "reset") {
		callsub(S_Reset, 1);
		end;
	}

	if (.@sub$ == "status") {
		callsub(S_Status);
		end;
	}

	if (.@sub$ == "check") {
		callsub(S_Check, 1);
		end;
	}

	if (.@sub$ == "stakes") {
		callsub(S_Stakes, 1);
		end;
	}

	if (.@sub$ == "enc" || .@sub$ == "encounter") {
		callsub(S_Enc, 1);
		end;
	}

	if (.@sub$ == "stealth") {
		.@dc = atoi(@dm_atcmd_p$[1]);
		if (.@dc <= 0)
			.@dc = 15;
		callfunc("DM_StealthScene", .@dc);
		end;
	}

	if (.@sub$ == "wake") {
		callfunc("DM_StealthWake", 0);
		dispbottom("[Stealth] Everything is awake again.");
		end;
	}

	dispbottom("[DM] Unknown subcommand: " + .@sub$);
	callsub(S_Help);
	end;

OnBeat:
	if (!callfunc("DM_RequireDM"))
		end;

	// callsub), so no @dm_atcmd bridge is needed. No/invalid arg = full menu.
	.@arc = atoi(.@atcmd_parameters$[0]);
	if (.@arc >= 1 && .@arc <= 19) {
		callfunc("DM_BeatArc" + (.@arc < 10 ? "0" : "") + .@arc);
		end;
	}
	callfunc("DM_BeatMenu");
	end;

	end;

OnRoll:
	.@input$ = "";
	.@hidden = 0;
	.@start_idx = 0;

	if (.@atcmd_parameters$[0] == "fudge" || .@atcmd_parameters$[0] == "override") {
		if (getgmlevel() < 1) {
			dispbottom("[DM] Roll overrides are for DMs only.");
			end;
		}
		.@total = atoi(.@atcmd_parameters$[1]);
		if (.@atcmd_parameters$[1] == "") {
			dispbottom("[DM] Usage: @roll fudge <total> [note]");
			end;
		}
		.@note$ = "";
		for (.@i = 2; .@i < .@atcmd_numparameters; ++.@i)
			.@note$ += (.@i > 2 ? " " : "") + .@atcmd_parameters$[.@i];
		.@msg$ = "[Roll] " + strcharinfo(PC_NAME) + " set a DM override result: " + .@total;
		if (.@note$ != "")
			.@msg$ += " (" + .@note$ + ")";
		mapannounce strcharinfo(PC_MAP), .@msg$, bc_map, 0xFFAA33;
		end;
	}

	if (.@atcmd_parameters$[0] == "hidden") {
		if (getgmlevel() >= 1) {
			.@hidden = 1;
			.@start_idx = 1;
		} else {
			dispbottom("[DM] Hidden rolls are for DMs only.");
			end;
		}
	}

	for (.@i = .@start_idx; .@i < .@atcmd_numparameters; ++.@i)
		.@input$ += .@atcmd_parameters$[.@i];

	if (.@input$ == "") {
		dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
		end;
	}

	.@num = 1;
	.@sides = 20;
	.@mod = 0;

	// Pick the format by the modifier sign actually present: sscanf on a
	// modifier-less input against "%dd%d+%d" leaks a spurious modifier
	// instead of failing (found by the headless dm-roll-bounds scenario).
	if (compare(.@input$, "+")) {
		if (sscanf(.@input$, "%dd%d+%d", .@num, .@sides, .@mod) < 3) {
			dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
			end;
		}
	} else if (compare(.@input$, "-")) {
		if (sscanf(.@input$, "%dd%d-%d", .@num, .@sides, .@mod) < 3) {
			dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
			end;
		}
		.@mod = -.@mod;
	} else if (sscanf(.@input$, "%dd%d", .@num, .@sides) < 2) {
		dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
		end;
	}

	if (.@num < 1) .@num = 1;
	if (.@num > 100) .@num = 100;
	if (.@sides < 2) .@sides = 2;
	if (.@sides > 1000) .@sides = 1000;

	.@sum = 0;
	.@detail$ = "";
	for (.@i = 0; .@i < .@num; ++.@i) {
		.@roll = rand(1, .@sides);
		.@sum += .@roll;
		if (.@num <= 20)
			.@detail$ += (.@i > 0 ? ", " : "") + .@roll;
	}

	.@total = .@sum + .@mod;

	.@msg$ = "[Roll] " + strcharinfo(PC_NAME) + " rolled " + .@input$ + ": " + .@total;
	if (.@detail$ != "")
		.@msg$ += " [" + .@detail$ + "]";
	if (.@mod != 0 || .@num > 20)
		.@msg$ += " (Base: " + .@sum + ( .@mod != 0 ? " Mod: " + (.@mod > 0 ? "+" : "") + .@mod : "" ) + ")";

	if (.@hidden)
		dispbottom("[Hidden] " + .@msg$);
	else
		mapannounce strcharinfo(PC_MAP), .@msg$, bc_map, 0xFFAA33;
	end;

S_Enc:
	// @dmenc list | info <name> | spawn <name> [formation]
	.@offset = getarg(0, 0);
	.@action$ = strtolower(@dm_atcmd_p$[.@offset]);

	if (.@action$ == "" || .@action$ == "list") {
		callfunc("DM_EncList");
		return;
	}
	if (.@action$ == "info") {
		if (@dm_atcmd_p$[.@offset + 1] == "") {
			dispbottom("[Enc] Usage: @dmenc info <name>");
			return;
		}
		callfunc("DM_EncInfo", @dm_atcmd_p$[.@offset + 1]);
		return;
	}
	if (.@action$ == "spawn") {
		if (@dm_atcmd_p$[.@offset + 1] == "") {
			dispbottom("[Enc] Usage: @dmenc spawn <name> [ambush|line|scatter]");
			return;
		}
		callfunc("DM_EncSpawn", @dm_atcmd_p$[.@offset + 1], @dm_atcmd_p$[.@offset + 2]);
		return;
	}

	dispbottom("[Enc] Unknown action: " + .@action$);
	callfunc("DM_EncList");
	return;

S_Stakes:
	// @dm stakes <success text> | <failure text>
	// Announced BEFORE the check. A roll whose price nobody knows is just a
	// number; naming both outcomes first is the cheapest way to make a check
	// matter at the table.
	.@offset = getarg(0, 0);
	.@text$ = "";
	for (.@i = .@offset; .@i < @dm_atcmd_n; ++.@i)
		.@text$ += (.@i > .@offset ? " " : "") + @dm_atcmd_p$[.@i];

	if (.@text$ == "") {
		dispbottom("[DM] Usage: @dm stakes <what success means> | <what failure means>");
		dispbottom("[DM] Announced to the map before you call the check.");
		return;
	}

	.@bar = 0;
	.@len = getstrlen(.@text$);
	for (.@i = 0; .@i < .@len; ++.@i) {
		if (substr(.@text$, .@i, .@i) == "|") {
			.@bar = .@i;
			break;
		}
	}

	if (.@bar > 0) {
		// substr() is inclusive at both ends, so bar-1 / bar+1 split cleanly.
		.@win$ = substr(.@text$, 0, .@bar - 1);
		.@lose$ = substr(.@text$, .@bar + 1, .@len - 1);
		mapannounce strcharinfo(PC_MAP), "[Stakes] On success: " + .@win$, bc_map, 0x88FF88;
		mapannounce strcharinfo(PC_MAP), "[Stakes] On failure: " + .@lose$, bc_map, 0xFF8844;
	} else {
		mapannounce strcharinfo(PC_MAP), "[Stakes] " + .@text$, bc_map, 0xFFCC33;
	}
	return;

S_Check:
	// @dm check <who> <tag|stat> <dc> [flag]
	// who = a character name, "me", or "party".
	.@offset = getarg(0, 0);
	.@who$  = @dm_atcmd_p$[.@offset];
	.@tag$  = @dm_atcmd_p$[.@offset + 1];
	.@dc    = atoi(@dm_atcmd_p$[.@offset + 2]);
	.@flag$ = @dm_atcmd_p$[.@offset + 3];

	if (.@who$ == "" || .@tag$ == "" || .@dc <= 0) {
		dispbottom("[DM] Usage: @dm check <player|me|party> <tag|stat> <dc> [dm_flag]");
		dispbottom("[DM] The flag, if given, is set party-wide on any success.");
		callfunc("DM_CheckDCHelp");
		return;
	}
	if (.@flag$ != "" && !callfunc("DM_ValidFlagName", .@flag$)) {
		dispbottom("[DM] Not a valid flag name: " + .@flag$ + " (must start dm_)");
		return;
	}

	.@origin = getcharid(CHAR_ID_ACCOUNT);
	.@ok = 0;

	if (strtolower(.@who$) == "party") {
		.@ok = callfunc("DM_CheckParty", .@tag$, .@dc);
	} else if (strtolower(.@who$) == "me") {

			return;
		.@ok = (.@band >= 1);
	} else {
		.@aid = getcharid(CHAR_ID_ACCOUNT, .@who$);
		if (.@aid <= 0) {
			dispbottom("[DM] No online character named '" + .@who$ + "'.");
			return;
		}
		if (!attachrid(.@aid)) {
			dispbottom("[DM] Could not attach to '" + .@who$ + "'.");
			return;
		}
		.@band = callfunc("DM_CheckRoll", .@tag$, .@dc);
		detachrid();
		attachrid(.@origin);
		if (.@band == -2)
			return;
		.@ok = (.@band >= 1);
	}

	if (.@flag$ == "")
		return;

	if (.@ok) {
		callfunc("DM_PartySetFlag", .@flag$, 1);
		dispbottom("[DM] " + .@flag$ + " set party-wide on the successful check.");
	} else {
		dispbottom("[DM] Check failed; " + .@flag$ + " left unset.");
	}
	return;

S_Reset:
	.@offset = getarg(0, 0);
	.@confirm$ = @dm_atcmd_p$[.@offset];
	if (.@confirm$ != "confirm") {
		dispbottom("[DM] @dm reset confirm — wipes all campaign flags and quests for a fresh playthrough.");
		dispbottom("[DM] Warning: clears all 19 arcs for every online party member. Cannot be undone.");
		return;
	}
	// Hercules kills a script that spends more than script_config.check_gotocount
	// (2048) gotos in one run, and every callfunc AND its return spends one. A
	// full reset is far past that: per member, 19 arc-clears plus 136
	// DM_ClearFlag calls, each of which now also calls DM_ValidFlagName -- and
	// then 90 quest ids, three functions deep, once per online member.
	//
	// It crossed the line quietly. The reset was fine when the campaign was
	// smaller, and the validation the security pass added to DM_ClearFlag is
	// what tipped it: the map-server logs "run_script: infinity loop !" and
	// abandons the script, so the flags are half cleared and the DM sees no
	// message at all -- the command simply does nothing and says nothing.
	//
	// freeloop is the sanctioned escape hatch for a script that legitimately
	// does this much work. Safe here because the work is BOUNDED by
	// construction: 19 arcs, a fixed list of 90 quest ids, and however many
	// members are online. There is no input that makes this loop forever.
	// Turned back off before returning so the guard covers everything after.
	freeloop(true);

	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	if (.@party_id > 0) {
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i]))
				continue;
			if (!attachrid(.@accid[.@i]))
				continue;
			callfunc("DM_ClearArc01Flags"); callfunc("DM_ClearArc02Flags"); callfunc("DM_ClearArc03Flags");
			callfunc("DM_ClearArc04Flags"); callfunc("DM_ClearArc05Flags"); callfunc("DM_ClearArc06Flags");
			callfunc("DM_ClearArc07Flags"); callfunc("DM_ClearArc08Flags"); callfunc("DM_ClearArc09Flags");
			callfunc("DM_ClearArc10Flags"); callfunc("DM_ClearArc11Flags"); callfunc("DM_ClearArc12Flags");
			callfunc("DM_ClearArc13Flags"); callfunc("DM_ClearArc14Flags"); callfunc("DM_ClearArc15Flags");
			callfunc("DM_ClearArc16Flags"); callfunc("DM_ClearArc17Flags"); callfunc("DM_ClearArc18Flags");
			callfunc("DM_ClearArc19Flags");
			detachrid();
			++.@cleared;
		}
		if (.@origin_aid)
			attachrid(.@origin_aid);
	} else {
		callfunc("DM_ClearArc01Flags"); callfunc("DM_ClearArc02Flags"); callfunc("DM_ClearArc03Flags");
		callfunc("DM_ClearArc04Flags"); callfunc("DM_ClearArc05Flags"); callfunc("DM_ClearArc06Flags");
		callfunc("DM_ClearArc07Flags"); callfunc("DM_ClearArc08Flags"); callfunc("DM_ClearArc09Flags");
		callfunc("DM_ClearArc10Flags"); callfunc("DM_ClearArc11Flags"); callfunc("DM_ClearArc12Flags");
		callfunc("DM_ClearArc13Flags"); callfunc("DM_ClearArc14Flags"); callfunc("DM_ClearArc15Flags");
		callfunc("DM_ClearArc16Flags"); callfunc("DM_ClearArc17Flags"); callfunc("DM_ClearArc18Flags");
		callfunc("DM_ClearArc19Flags");
		.@cleared = 1;
	}
	for (.@q = 20001; .@q <= 20030; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20101; .@q <= 20105; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20111; .@q <= 20115; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20121; .@q <= 20124; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20131; .@q <= 20134; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20141; .@q <= 20144; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20151; .@q <= 20155; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20161; .@q <= 20165; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20171; .@q <= 20175; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20181; .@q <= 20185; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20191; .@q <= 20194; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20201; .@q <= 20204; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20211; .@q <= 20214; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20221; .@q <= 20223; ++.@q) callfunc("DM_InstanceQuestErase", .@q);

	return;

	.@action$ = @dm_atcmd_p$[.@offset];
	.@quest_id = atoi(@dm_atcmd_p$[.@offset + 1]);

	if (.@quest_id <= 0) {
		dispbottom("[DM] Usage: @dmquest <start|complete|erase> <quest_id>");
		return;
	}
	if (!callfunc("DM_IsCampaignQuest", .@quest_id)) {
		dispbottom("[DM] Quest " + .@quest_id + " is outside the campaign range (20001-20233).");
		return;
	}

	if (.@action$ == "start" || .@action$ == "set") {
		.@done = callfunc("DM_InstanceQuestStart", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " started for " + .@done + " party member(s).");
		return;
	}

	if (.@action$ == "complete") {
		.@done = callfunc("DM_InstanceQuestComplete", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " completed for " + .@done + " party member(s).");
		return;
	}

	if (.@action$ == "erase" || .@action$ == "clear") {
		.@done = callfunc("DM_InstanceQuestErase", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " erased for " + .@done + " party member(s).");
		return;
	}

	dispbottom("[DM] Unknown quest action: " + .@action$);
	return;

S_Spawn:
	.@offset = getarg(0, 0);
	.@mob_id = atoi(@dm_atcmd_p$[.@offset]);
	.@count = atoi(@dm_atcmd_p$[.@offset + 1]);
	.@name$ = @dm_atcmd_p$[.@offset + 2];

	if (.@mob_id <= 0) {
		dispbottom("[DM] Usage: @dm spawn <mob_id> [count] [name]");
		return;
	}

	if (.@count <= 0)
		.@count = 1;
	if (.@count > 20)
		.@count = 20;
	if (.@name$ == "")
		.@name$ = "--ja--";

	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	areamonster(.@map$, .@x - 3, .@y - 3, .@x + 3, .@y + 3, .@name$, .@mob_id, .@count, "DM_Console::OnDMKilled");
	dispbottom("[DM] Spawned " + .@count + " mob(s) near you on " + .@map$ + ".");
	return;

}
