// Seal Cascade in-game DM command console.

-	script	DM_Console	FAKE_NPC,{
	end;

OnInit:
	bindatcmd("dm", strnpcinfo(NPC_NAME_UNIQUE)+"::OnDM", 60, 99, 1);
	bindatcmd("dmreward", strnpcinfo(NPC_NAME_UNIQUE)+"::OnReward", 60, 99, 1);
	bindatcmd("dmflag", strnpcinfo(NPC_NAME_UNIQUE)+"::OnFlag", 60, 99, 1);
	bindatcmd("dmquest", strnpcinfo(NPC_NAME_UNIQUE)+"::OnQuest", 60, 99, 1);
	bindatcmd("dmbeat", strnpcinfo(NPC_NAME_UNIQUE)+"::OnBeat", 60, 99, 1);
	bindatcmd("dmstory", strnpcinfo(NPC_NAME_UNIQUE)+"::OnStory", 60, 99, 1);
	bindatcmd("dmcleanup", strnpcinfo(NPC_NAME_UNIQUE)+"::OnCleanup", 60, 99, 1);
	bindatcmd("dmwarp", strnpcinfo(NPC_NAME_UNIQUE)+"::OnWarp", 60, 99, 1);
	bindatcmd("dmrecall", strnpcinfo(NPC_NAME_UNIQUE)+"::OnRecall", 60, 99, 1);
	bindatcmd("dminstance", strnpcinfo(NPC_NAME_UNIQUE)+"::OnInstance", 60, 99, 1);
	end;

OnDM:
	if (!callfunc("DM_RequireDM"))
		end;

	if (.@atcmd_numparameters < 1) {
		callsub(S_Help);
		end;
	}

	.@sub$ = .@atcmd_parameters$[0];

	if (.@sub$ == "help") {
		callsub(S_Help);
		end;
	}

	if (.@sub$ == "reward") {
		callsub(S_Reward, 1);
		end;
	}

	if (.@sub$ == "flag") {
		callsub(S_Flag, 1);
		end;
	}

	if (.@sub$ == "quest") {
		callsub(S_Quest, 1);
		end;
	}

	if (.@sub$ == "beat") {
		callsub(S_Beat);
		end;
	}

	if (.@sub$ == "story") {
		.@text$ = "";
		for (.@i = 1; .@i < .@atcmd_numparameters; ++.@i)
			.@text$ += (.@i > 1 ? " " : "") + .@atcmd_parameters$[.@i];
		callfunc("DM_MapStory", .@text$);
		end;
	}

	if (.@sub$ == "globalstory") {
		.@text$ = "";
		for (.@i = 1; .@i < .@atcmd_numparameters; ++.@i)
			.@text$ += (.@i > 1 ? " " : "") + .@atcmd_parameters$[.@i];
		callfunc("DM_GlobalStory", .@text$);
		end;
	}

	if (.@sub$ == "spawn") {
		callsub(S_Spawn, 1);
		end;
	}

	if (.@sub$ == "cleanup") {
		callfunc("DM_CleanupMap");
		end;
	}

	if (.@sub$ == "warp") {
		callsub(S_Warp, 1);
		end;
	}

	if (.@sub$ == "recall") {
		callsub(S_Recall);
		end;
	}

	if (.@sub$ == "instance") {
		callsub(S_Instance, 1);
		end;
	}

	dispbottom("[DM] Unknown subcommand: " + .@sub$);
	callsub(S_Help);
	end;

OnReward:
	if (!callfunc("DM_RequireDM"))
		end;
	callsub(S_Reward, 0);
	end;

OnFlag:
	if (!callfunc("DM_RequireDM"))
		end;
	callsub(S_Flag, 0);
	end;

OnQuest:
	if (!callfunc("DM_RequireDM"))
		end;
	callsub(S_Quest, 0);
	end;

OnBeat:
	if (!callfunc("DM_RequireDM"))
		end;
	callsub(S_Beat);
	end;

OnStory:
	if (!callfunc("DM_RequireDM"))
		end;
	.@text$ = "";
	for (.@i = 0; .@i < .@atcmd_numparameters; ++.@i)
		.@text$ += (.@i > 0 ? " " : "") + .@atcmd_parameters$[.@i];
	callfunc("DM_MapStory", .@text$);
	end;

OnCleanup:
	if (!callfunc("DM_RequireDM"))
		end;
	callfunc("DM_CleanupMap");
	end;

OnWarp:
	if (!callfunc("DM_RequireDM"))
		end;
	callsub(S_Warp, 0);
	end;

OnRecall:
	if (!callfunc("DM_RequireDM"))
		end;
	callsub(S_Recall);
	end;

OnInstance:
	if (!callfunc("DM_RequireDM"))
		end;
	callsub(S_Instance, 0);
	end;

OnDMKilled:
	end;

OnDMBossKilled:
	mapannounce(strcharinfo(PC_MAP), "[DM] The set-piece foe falls.", bc_map, 0xFFAA33);
	end;

OnDMCleanup:
	end;

S_Help:
	dispbottom("[DM] @dm reward [arc] [tier] [dryrun]");
	dispbottom("[DM] @dm flag <get|set|clear|arc01|cleararc01|arc02|cleararc02|arc03|cleararc03|arc04|cleararc04|arc05|cleararc05> [name] [value]");
	dispbottom("[DM] @dm quest <start|complete|erase> <quest_id>   (current party)");
	dispbottom("[DM] @dm beat                 (menu: location/NPC/story tracking)");
	dispbottom("[DM] @dm story <message> | @dm globalstory <message>");
	dispbottom("[DM] @dm spawn <mob_id> [count] [name]");
	dispbottom("[DM] @dm warp <map> [x] [y]   (warps your whole party)");
	dispbottom("[DM] @dm recall              (pulls party to you)");
	dispbottom("[DM] @dm instance <start <map> [x] [y] [label] | end>");
	dispbottom("[DM] Shortcuts: @dmreward, @dmflag, @dmquest, @dmbeat, @dmstory, @dmcleanup, @dmwarp, @dmrecall, @dminstance");
	return;

S_Warp:
	.@offset = getarg(0, 0);
	.@map$ = .@atcmd_parameters$[.@offset];
	.@x = atoi(.@atcmd_parameters$[.@offset + 1]);
	.@y = atoi(.@atcmd_parameters$[.@offset + 2]);

	if (.@map$ == "") {
		dispbottom("[DM] Usage: @dm warp <map> [x] [y]");
		return;
	}

	.@moved = callfunc("DM_WarpParty", .@map$, .@x, .@y, getcharid(CHAR_ID_PARTY));
	dispbottom("[DM] Warped " + .@moved + " party member(s) to " + .@map$ + ".");
	return;

S_Recall:
	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	.@moved = callfunc("DM_WarpParty", .@map$, .@x, .@y, getcharid(CHAR_ID_PARTY));
	dispbottom("[DM] Recalled " + .@moved + " party member(s) to your position on " + .@map$ + ".");
	return;

S_Instance:
	.@offset = getarg(0, 0);
	.@action$ = .@atcmd_parameters$[.@offset];

	if (.@action$ == "end") {
		callfunc("DM_InstanceEnd", getcharid(CHAR_ID_PARTY));
		return;
	}

	if (.@action$ == "start") {
		.@src$ = .@atcmd_parameters$[.@offset + 1];
		if (.@src$ == "") {
			dispbottom("[DM] Usage: @dm instance start <source_map> [x] [y] [label]");
			return;
		}
		.@x = atoi(.@atcmd_parameters$[.@offset + 2]);
		.@y = atoi(.@atcmd_parameters$[.@offset + 3]);

		.@label$ = "";
		for (.@i = .@offset + 4; .@i < .@atcmd_numparameters; ++.@i)
			.@label$ += (.@i > .@offset + 4 ? " " : "") + .@atcmd_parameters$[.@i];
		if (.@label$ == "")
			.@label$ = "Seal Cascade - " + .@src$;

		callfunc("DM_InstanceStart", .@label$, .@src$, .@x, .@y, getcharid(CHAR_ID_PARTY));
		return;
	}

	dispbottom("[DM] Usage: @dm instance <start <map> [x] [y] [label] | end>");
	return;

S_Reward:
	.@offset = getarg(0, 0);
	.@arc = atoi(.@atcmd_parameters$[.@offset]);
	if (.@arc <= 0)
		.@arc = 1;

	.@tier = callfunc("DM_ParseTier", .@atcmd_parameters$[.@offset + 1]);
	.@dryrun = atoi(.@atcmd_parameters$[.@offset + 2]);
	.@party_id = getcharid(CHAR_ID_PARTY);

	callfunc("DM_GenerateLoot", .@arc, BaseLevel, .@tier, .@party_id, .@dryrun);
	return;

S_Flag:
	.@offset = getarg(0, 0);
	.@action$ = .@atcmd_parameters$[.@offset];

	if (.@action$ == "arc01") {
		callfunc("DM_PrintArc01Flags");
		return;
	}

	if (.@action$ == "cleararc01") {
		callfunc("DM_ClearArc01Flags");
		dispbottom("[DM] Arc 01 flags cleared for this character.");
		return;
	}

	if (.@action$ == "arc02") {
		callfunc("DM_PrintArc02Flags");
		return;
	}

	if (.@action$ == "cleararc02") {
		callfunc("DM_ClearArc02Flags");
		dispbottom("[DM] Arc 02 flags cleared for this character.");
		return;
	}

	if (.@action$ == "arc03") {
		callfunc("DM_PrintArc03Flags");
		return;
	}

	if (.@action$ == "cleararc03") {
		callfunc("DM_ClearArc03Flags");
		dispbottom("[DM] Arc 03 flags cleared for this character.");
		return;
	}

	if (.@action$ == "arc04") {
		callfunc("DM_PrintArc04Flags");
		return;
	}

	if (.@action$ == "cleararc04") {
		callfunc("DM_ClearArc04Flags");
		dispbottom("[DM] Arc 04 flags cleared for this character.");
		return;
	}

	if (.@action$ == "arc05") {
		callfunc("DM_PrintArc05Flags");
		return;
	}

	if (.@action$ == "cleararc05") {
		callfunc("DM_ClearArc05Flags");
		dispbottom("[DM] Arc 05 flags cleared for this character.");
		return;
	}

	.@flag$ = .@atcmd_parameters$[.@offset + 1];
	if (.@flag$ == "") {
		dispbottom("[DM] Usage: @dmflag <get|set|clear|arc01|cleararc01|arc02|cleararc02|arc03|cleararc03|arc04|cleararc04|arc05|cleararc05> [name] [value]");
		return;
	}

	if (.@action$ == "get") {
		dispbottom("[DM] " + .@flag$ + " = " + callfunc("DM_GetFlag", .@flag$));
		return;
	}

	if (.@action$ == "set") {
		.@value = atoi(.@atcmd_parameters$[.@offset + 2]);
		.@done = callfunc("DM_InstanceSetFlag", .@flag$, .@value);
		dispbottom("[DM] " + .@flag$ + " set to " + .@value + " for " + .@done + " party member(s).");
		return;
	}

	if (.@action$ == "clear") {
		.@done = callfunc("DM_InstanceClearFlag", .@flag$);
		dispbottom("[DM] " + .@flag$ + " cleared for " + .@done + " party member(s).");
		return;
	}

	dispbottom("[DM] Unknown flag action: " + .@action$);
	return;

S_Quest:
	.@offset = getarg(0, 0);
	.@action$ = .@atcmd_parameters$[.@offset];
	.@quest_id = atoi(.@atcmd_parameters$[.@offset + 1]);

	if (.@quest_id <= 0) {
		dispbottom("[DM] Usage: @dmquest <start|complete|erase> <quest_id>");
		return;
	}

	if (.@action$ == "start" || .@action$ == "set") {
		.@done = callfunc("DM_InstanceQuestStart", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " started for " + .@done + " party member(s).");
		return;
	}

	if (.@action$ == "complete") {
		.@done = callfunc("DM_InstanceQuestComplete", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " completed for " + .@done + " party member(s).");
		return;
	}

	if (.@action$ == "erase" || .@action$ == "clear") {
		.@done = callfunc("DM_InstanceQuestErase", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " erased for " + .@done + " party member(s).");
		return;
	}

	dispbottom("[DM] Unknown quest action: " + .@action$);
	return;

S_Beat:
	mes("[DM Beat Director]");
	mes("Choose a campaign beat. These update the current party's quest log and story flags.");
	next();
	switch (select("Act I / Arc 1 - Prontera", "Act I / Arc 2 - Payon", "Act I / Arc 3 - Morroc", "Act I / Arc 4 - Geffen", "Act I / Arc 5 - Alberta", "Cancel")) {
	case 1:
		callsub(S_BeatArc01);
		close();
	case 2:
		callsub(S_BeatArc02);
		close();
	case 3:
		callsub(S_BeatArc03);
		close();
	case 4:
		callsub(S_BeatArc04);
		close();
	case 5:
		callsub(S_BeatArc05);
		close();
	case 6:
		close();
	}
	return;

S_Spawn:
	.@offset = getarg(0, 0);
	.@mob_id = atoi(.@atcmd_parameters$[.@offset]);
	.@count = atoi(.@atcmd_parameters$[.@offset + 1]);
	.@name$ = .@atcmd_parameters$[.@offset + 2];

	if (.@mob_id <= 0) {
		dispbottom("[DM] Usage: @dm spawn <mob_id> [count] [name]");
		return;
	}

	if (.@count <= 0)
		.@count = 1;
	if (.@name$ == "")
		.@name$ = "--ja--";

	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	areamonster(.@map$, .@x - 3, .@y - 3, .@x + 3, .@y + 3, .@name$, .@mob_id, .@count, "DM_Console::OnDMKilled");
	dispbottom("[DM] Spawned " + .@count + " mob(s) near you on " + .@map$ + ".");
	return;
