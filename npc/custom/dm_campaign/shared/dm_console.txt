OnDM:
	if (!callfunc("DM_RequireDM"))
		end;

	if (.@atcmd_numparameters < 1) {
		callsub(S_Help);
		end;
	}

	.@sub$ = .@atcmd_parameters$[0];

	if (.@sub$ == "help") {
		callsub(S_Help);
		end;
	}

	if (.@sub$ == "reward") {
		callsub(S_Reward, 1);
		end;
	}

	if (.@sub$ == "flag") {
		callsub(S_Flag, 1);
		end;
	}

	if (.@sub$ == "quest") {
		callsub(S_Quest, 1);
		end;
	}

	if (.@sub$ == "beat") {
		callfunc("DM_BeatMenu");
		end;
	}

	if (.@sub$ == "story") {
		.@text$ = "";
		for (.@i = 1; .@i < .@atcmd_numparameters; ++.@i)
			.@text$ += (.@i > 1 ? " " : "") + .@atcmd_parameters$[.@i];
		callfunc("DM_MapStory", .@text$);
		end;
	}

	if (.@sub$ == "globalstory") {
		.@text$ = "";
		for (.@i = 1; .@i < .@atcmd_numparameters; ++.@i)
			.@text$ += (.@i > 1 ? " " : "") + .@atcmd_parameters$[.@i];
		callfunc("DM_GlobalStory", .@text$);
		end;
	}

	if (.@sub$ == "spawn") {
		callsub(S_Spawn, 1);
		end;
	}

	if (.@sub$ == "cleanup") {
		callfunc("DM_CleanupMap");
		end;
	}

	if (.@sub$ == "warp") {
		callsub(S_Warp, 1);
		end;

	if (.@sub$ == "recall") {
		callsub(S_Recall);
		end;
	}

		end;

		callsub(S_Mode, 1);
		end;

		callsub(S_Hazard, 1);
		end;

		callsub(S_Exp, 1);
		end;

		callsub(S_ExpRate, 1);
		end;

		end;

		callsub(S_Reset, 1);
		end;

		callsub(S_Status);
		end;
	}

	end;

OnRoll:
	.@input$ = "";
	.@hidden = 0;
	.@start_idx = 0;

	if (.@atcmd_parameters$[0] == "fudge" || .@atcmd_parameters$[0] == "override") {
		if (getgmlevel() < 1) {
			dispbottom("[DM] Roll overrides are for DMs only.");
			end;
		}
		.@total = atoi(.@atcmd_parameters$[1]);
		if (.@atcmd_parameters$[1] == "") {
			dispbottom("[DM] Usage: @roll fudge <total> [note]");
			end;
		}
		.@note$ = "";
		for (.@i = 2; .@i < .@atcmd_numparameters; ++.@i)
			.@note$ += (.@i > 2 ? " " : "") + .@atcmd_parameters$[.@i];
		.@msg$ = "[Roll] " + strcharinfo(PC_NAME) + " set a DM override result: " + .@total;
		if (.@note$ != "")
			.@msg$ += " (" + .@note$ + ")";
		mapannounce strcharinfo(PC_MAP), .@msg$, bc_map, 0xFFAA33;
		end;
	}

	if (.@atcmd_parameters$[0] == "hidden") {
		if (getgmlevel() >= 1) {
			.@hidden = 1;
			.@start_idx = 1;
		} else {
			dispbottom("[DM] Hidden rolls are for DMs only.");
			end;
		}
	}

	for (.@i = .@start_idx; .@i < .@atcmd_numparameters; ++.@i)
		.@input$ += .@atcmd_parameters$[.@i];

	if (.@input$ == "") {
		dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
		end;
	}

	.@num = 1;
	.@sides = 20;
	.@mod = 0;

	if (sscanf(.@input$, "%dd%d+%d", .@num, .@sides, .@mod) >= 3) {
	} else if (sscanf(.@input$, "%dd%d-%d", .@num, .@sides, .@mod) >= 3) {
		.@mod = -.@mod;
	} else if (sscanf(.@input$, "%dd%d", .@num, .@sides) >= 2) {
		.@mod = 0;
	} else {
		dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
		end;
	}

	if (.@num < 1) .@num = 1;
	if (.@num > 100) .@num = 100;
	if (.@sides < 2) .@sides = 2;
	if (.@sides > 1000) .@sides = 1000;

	.@sum = 0;
	.@detail$ = "";
	for (.@i = 0; .@i < .@num; ++.@i) {
		.@roll = rand(1, .@sides);
		.@sum += .@roll;
		if (.@num <= 20)
			.@detail$ += (.@i > 0 ? ", " : "") + .@roll;
	}

	.@total = .@sum + .@mod;

	.@msg$ = "[Roll] " + strcharinfo(PC_NAME) + " rolled " + .@input$ + ": " + .@total;
	if (.@detail$ != "")
		.@msg$ += " [" + .@detail$ + "]";
	if (.@mod != 0 || .@num > 20)
		.@msg$ += " (Base: " + .@sum + ( .@mod != 0 ? " Mod: " + (.@mod > 0 ? "+" : "") + .@mod : "" ) + ")";

	if (.@hidden)
		dispbottom("[Hidden] " + .@msg$);
	else
		mapannounce strcharinfo(PC_MAP), .@msg$, bc_map, 0xFFAA33;
	end;

OnHazardTick:
	if (@dm_hazard_ticks <= 0)
		end;

	callfunc("DM_HazardArea", @dm_hazard_map$, @dm_hazard_x, @dm_hazard_y, @dm_hazard_range, @dm_hazard_damage, @dm_hazard_status, @dm_hazard_status_ms, @dm_hazard_party_id);

		addtimer(@dm_hazard_interval, strnpcinfo(NPC_NAME_UNIQUE)+"::OnHazardTick");
	else
		dispbottom("[DM] Hazard expired.");
	end;

S_Reset:
	.@offset = getarg(0, 0);
	.@confirm$ = .@atcmd_parameters$[.@offset];
	if (.@confirm$ != "confirm") {
		dispbottom("[DM] @dm reset confirm — wipes all campaign flags and quests for a fresh playthrough.");
		dispbottom("[DM] Warning: clears all 19 arcs for every online party member. Cannot be undone.");
		return;
	}
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	if (.@party_id > 0) {
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i]))
				continue;
			if (!attachrid(.@accid[.@i]))
				continue;
			callfunc("DM_ClearArc01Flags"); callfunc("DM_ClearArc02Flags"); callfunc("DM_ClearArc03Flags");
			callfunc("DM_ClearArc04Flags"); callfunc("DM_ClearArc05Flags"); callfunc("DM_ClearArc06Flags");
			callfunc("DM_ClearArc07Flags"); callfunc("DM_ClearArc08Flags"); callfunc("DM_ClearArc09Flags");
			callfunc("DM_ClearArc10Flags"); callfunc("DM_ClearArc11Flags"); callfunc("DM_ClearArc12Flags");
			callfunc("DM_ClearArc13Flags"); callfunc("DM_ClearArc14Flags"); callfunc("DM_ClearArc15Flags");
			callfunc("DM_ClearArc16Flags"); callfunc("DM_ClearArc17Flags"); callfunc("DM_ClearArc18Flags");
			callfunc("DM_ClearArc19Flags");
			detachrid();
			++.@cleared;
		}
		if (.@origin_aid)
			attachrid(.@origin_aid);
	} else {
		callfunc("DM_ClearArc01Flags"); callfunc("DM_ClearArc02Flags"); callfunc("DM_ClearArc03Flags");
		callfunc("DM_ClearArc04Flags"); callfunc("DM_ClearArc05Flags"); callfunc("DM_ClearArc06Flags");
		callfunc("DM_ClearArc07Flags"); callfunc("DM_ClearArc08Flags"); callfunc("DM_ClearArc09Flags");
		callfunc("DM_ClearArc10Flags"); callfunc("DM_ClearArc11Flags"); callfunc("DM_ClearArc12Flags");
		callfunc("DM_ClearArc13Flags"); callfunc("DM_ClearArc14Flags"); callfunc("DM_ClearArc15Flags");
		callfunc("DM_ClearArc16Flags"); callfunc("DM_ClearArc17Flags"); callfunc("DM_ClearArc18Flags");
		callfunc("DM_ClearArc19Flags");
		.@cleared = 1;
	}
	for (.@q = 20000; .@q <= 20030; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20101; .@q <= 20105; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20111; .@q <= 20115; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20121; .@q <= 20124; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20131; .@q <= 20134; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20141; .@q <= 20144; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20151; .@q <= 20155; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20161; .@q <= 20165; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20171; .@q <= 20175; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20181; .@q <= 20185; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20191; .@q <= 20194; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20201; .@q <= 20204; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20211; .@q <= 20214; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20221; .@q <= 20223; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	for (.@q = 20231; .@q <= 20233; ++.@q) callfunc("DM_InstanceQuestErase", .@q);
	if (.@party_id > 0)
		setd("$dm_inst_" + .@party_id, 0);
	$dm_mode = 0;
	$dm_active_party = 0;
	$@dm_exp_party = 0;
	$@dm_exp_rate = 100;
	dispbottom("[DM] Campaign reset complete for " + .@cleared + " member(s). All arcs cleared, mode off.");
	return;

S_Exp:
	.@offset = getarg(0, 0);
	.@base = atoi(.@atcmd_parameters$[.@offset]);
	.@job  = atoi(.@atcmd_parameters$[.@offset + 1]);

	if (.@base <= 0 && .@job <= 0) {
		dispbottom("[DM] Usage: @dm exp <base> [job]");
		dispbottom("[DM] Example: @dm exp 50000 20000");
		return;
	}

	.@given = callfunc("DM_PartyExp", .@base, .@job);
	dispbottom("[DM] Granted " + .@base + " base / " + .@job + " job EXP to " + .@given + " party member(s).");
	return;

S_ExpRate:
	.@offset = getarg(0, 0);
	.@action$ = strtolower(.@atcmd_parameters$[.@offset]);
	.@party_id = getcharid(CHAR_ID_PARTY);

	if (.@action$ == "" || .@action$ == "status") {
		if ($@dm_exp_party > 0 && $@dm_exp_rate > 0)
			dispbottom("[DM] Monster EXP rate is " + $@dm_exp_rate + "% for party " + $@dm_exp_party + ".");
		else
			dispbottom("[DM] Monster EXP rate is normal.");
		dispbottom("[DM] Usage: @dm exprate <percent|off|status>  Example: @dm exprate 200");
		return;
	}

	if (.@action$ == "off" || .@action$ == "clear" || .@action$ == "normal") {
		$@dm_exp_party = 0;
		$@dm_exp_rate = 100;
		dispbottom("[DM] Monster EXP rate reset to normal.");
		return;
	}

	.@rate = atoi(.@action$);
	if (.@rate < 1 || .@rate > 1000) {
		dispbottom("[DM] EXP rate must be 1-1000 percent, or off/status. 200 = 2x.");
		return;
	}

	if (.@party_id <= 0) {
		dispbottom("[DM] Join or create a party before setting a party EXP rate.");
		return;
	}

	$@dm_exp_party = .@party_id;
	$@dm_exp_rate = .@rate;
	dispbottom("[DM] Monster EXP rate set to " + .@rate + "% for party " + .@party_id + ".");
	dispbottom("[DM] This affects monster/MVP EXP only. Use @dm exprate off to reset.");
	return;

S_Levels:

	dispbottom("[DM] A11 88  A12 90  A13 92  A14 94  A15 95");
	dispbottom("[DM] A16 96  A17 97  A18 98  A19 99");
	dispbottom("[DM] These are not start requirements; they drive @dmreward scaling.");
	return;

S_Mode:
	.@offset = getarg(0, 0);
	.@action$ = .@atcmd_parameters$[.@offset];
	if (.@action$ == "on") {
		if ($dm_active_party > 0 && $dm_active_party != getcharid(CHAR_ID_PARTY))
			callfunc("DM_InstanceQuestErase", 20000, $dm_active_party);
		$dm_mode = 1;
		$dm_active_party = getcharid(CHAR_ID_PARTY);
		callfunc("DM_InstanceQuestStart", 20000, $dm_active_party);
		dispbottom("[DM] DnD mode enabled. MVP spawns will be suppressed.");
		dispbottom("[DM] Campaign NPCs are active for party " + $dm_active_party + ".");
		mapannounce "prontera", "A campaign session has begun. Check the Session Board near the fountain.", bc_map, 0xFFAA33;
		return;
	}
	if (.@action$ == "off") {
		callfunc("DM_InstanceQuestErase", 20000, $dm_active_party);
		$dm_mode = 0;
		$dm_active_party = 0;
		$@dm_exp_party = 0;
		$@dm_exp_rate = 100;
		dispbottom("[DM] DnD mode disabled. MVP spawns will resume naturally.");
		return;
	}
	dispbottom("[DM] Mode is currently " + ($dm_mode ? "ON" : "OFF") + ($dm_mode ? " (party " + $dm_active_party + ")" : ""));
	dispbottom("[DM] Usage: @dm mode <on|off>");
	return;

S_HazardStatus:
	.@status$ = strtolower(getarg(0, ""));
	if (.@status$ == "" || .@status$ == "0" || .@status$ == "none")
		return 0;
	if (.@status$ == "poison" || .@status$ == "sc_poison")
		return SC_POISON;
	if (.@status$ == "freeze" || .@status$ == "frozen" || .@status$ == "sc_freeze")
		return SC_FREEZE;
	if (.@status$ == "stun" || .@status$ == "stunned" || .@status$ == "sc_stun")
		return SC_STUN;
	if (.@status$ == "sleep" || .@status$ == "sc_sleep")
		return SC_SLEEP;
	if (.@status$ == "curse" || .@status$ == "cursed" || .@status$ == "sc_curse")
		return SC_CURSE;
	if (.@status$ == "confusion" || .@status$ == "confused" || .@status$ == "sc_confusion")
		return SC_CONFUSION;
	if (.@status$ == "blind" || .@status$ == "sc_blind")
		return SC_BLIND;
	.@status = atoi(.@status$);
	if (.@status > 0)
		return .@status;
	return -1;

S_Hazard:
	.@offset = getarg(0, 0);
	.@action$ = .@atcmd_parameters$[.@offset];

	if (.@action$ == "clear" || .@action$ == "stop") {
		deltimer(strnpcinfo(NPC_NAME_UNIQUE)+"::OnHazardTick");
		@dm_hazard_ticks = 0;
		dispbottom("[DM] Hazard cleared.");
		return;
	}

	.@range = atoi(.@atcmd_parameters$[.@offset]);
	.@damage = atoi(.@atcmd_parameters$[.@offset + 1]);
	.@ticks = atoi(.@atcmd_parameters$[.@offset + 2]);
	.@interval = atoi(.@atcmd_parameters$[.@offset + 3]);
	.@status = callsub(S_HazardStatus, .@atcmd_parameters$[.@offset + 4]);
	.@status_ms = atoi(.@atcmd_parameters$[.@offset + 5]);

	if (.@status < 0) {
		dispbottom("[DM] Unknown hazard status: " + .@atcmd_parameters$[.@offset + 4]);
		dispbottom("[DM] Use a numeric SC id, or: poison, freeze, stun, sleep, curse, confusion, blind, none.");
		return;
	}

	if (.@range <= 0)
		.@range = 3;
	if (.@range > 20)
		.@range = 20;
	if (.@damage < 0)
		.@damage = 0;
	if (.@damage > 100)
		.@damage = 100;
	if (.@ticks <= 0)
		.@ticks = 3;
	if (.@ticks > 60)
		.@ticks = 60;
	if (.@interval <= 0)
		.@interval = 3000;
	if (.@interval < 1000)
		.@interval = 1000;
	if (.@status_ms <= 0)
		.@status_ms = 10000;

	deltimer(strnpcinfo(NPC_NAME_UNIQUE)+"::OnHazardTick");
	getmapxy(@dm_hazard_map$, @dm_hazard_x, @dm_hazard_y, UNITTYPE_PC);
	@dm_hazard_range = .@range;
	@dm_hazard_damage = .@damage;
	@dm_hazard_ticks = .@ticks;
	@dm_hazard_interval = .@interval;
	@dm_hazard_status = .@status;
	@dm_hazard_status_ms = .@status_ms;
	@dm_hazard_party_id = getcharid(CHAR_ID_PARTY);

	dispbottom("[DM] Hazard placed at " + @dm_hazard_map$ + " " + @dm_hazard_x + "," + @dm_hazard_y + " for " + .@ticks + " tick(s).");
	callfunc("DM_HazardArea", @dm_hazard_map$, @dm_hazard_x, @dm_hazard_y, @dm_hazard_range, @dm_hazard_damage, @dm_hazard_status, @dm_hazard_status_ms, @dm_hazard_party_id);
	--@dm_hazard_ticks;

	if (@dm_hazard_ticks > 0)
		addtimer(@dm_hazard_interval, strnpcinfo(NPC_NAME_UNIQUE)+"::OnHazardTick");
	return;

S_Warp:
	.@offset = getarg(0, 0);
	.@map$ = .@atcmd_parameters$[.@offset];
	.@x = atoi(.@atcmd_parameters$[.@offset + 1]);
	.@y = atoi(.@atcmd_parameters$[.@offset + 2]);

	if (.@map$ == "") {
		dispbottom("[DM] Usage: @dm warp <map> [x] [y]");
		return;
	}

	.@moved = callfunc("DM_WarpParty", .@map$, .@x, .@y, getcharid(CHAR_ID_PARTY));
	dispbottom("[DM] Warped " + .@moved + " party member(s) to " + .@map$ + ".");
	return;

S_Recall:
	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	.@moved = callfunc("DM_WarpParty", .@map$, .@x, .@y, getcharid(CHAR_ID_PARTY));
	dispbottom("[DM] Recalled " + .@moved + " party member(s) to your position on " + .@map$ + ".");
	return;

S_Instance:
	.@offset = getarg(0, 0);
	.@action$ = .@atcmd_parameters$[.@offset];

	if (.@action$ == "end") {
		callfunc("DM_InstanceEnd", getcharid(CHAR_ID_PARTY));
		return;
	}

	if (.@action$ == "start") {
		.@src$ = .@atcmd_parameters$[.@offset + 1];
		if (.@src$ == "") {
			dispbottom("[DM] Usage: @dm instance start <source_map> [x] [y] [label]");
			return;
		}
		.@x = atoi(.@atcmd_parameters$[.@offset + 2]);
		.@y = atoi(.@atcmd_parameters$[.@offset + 3]);

		.@label$ = "";
		for (.@i = .@offset + 4; .@i < .@atcmd_numparameters; ++.@i)
			.@label$ += (.@i > .@offset + 4 ? " " : "") + .@atcmd_parameters$[.@i];
		if (.@label$ == "")
			.@label$ = "Seal Cascade - " + .@src$;

		callfunc("DM_InstanceStart", .@label$, .@src$, .@x, .@y, getcharid(CHAR_ID_PARTY));
		return;
	}

	dispbottom("[DM] Usage: @dm instance <start <map> [x] [y] [label] | end>");
	return;

S_Reward:
	.@offset = getarg(0, 0);
	.@arc = atoi(.@atcmd_parameters$[.@offset]);
	if (.@arc <= 0)
		.@arc = 1;

	.@tier = callfunc("DM_ParseTier", .@atcmd_parameters$[.@offset + 1]);
	.@mode$ = strtolower(.@atcmd_parameters$[.@offset + 2]);
	.@dryrun = (.@mode$ == "preview" || .@mode$ == "dryrun" || .@mode$ == "roll" || atoi(.@mode$) > 0);
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@reward_level = callfunc("DM_RewardArcLevel", .@arc);

	callfunc("DM_GenerateLoot", .@arc, .@reward_level, .@tier, .@party_id, .@dryrun);
	return;

		return;

		.@done = callfunc("DM_InstanceQuestStart", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " started for " + .@done + " party member(s).");
		return;
	}

		.@done = callfunc("DM_InstanceQuestComplete", .@quest_id);

		.@done = callfunc("DM_InstanceQuestErase", .@quest_id);

S_Spawn:
	.@offset = getarg(0, 0);
	.@mob_id = atoi(.@atcmd_parameters$[.@offset]);
	.@count = atoi(.@atcmd_parameters$[.@offset + 1]);
	.@name$ = .@atcmd_parameters$[.@offset + 2];

	if (.@mob_id <= 0) {
		dispbottom("[DM] Usage: @dm spawn <mob_id> [count] [name]");
		return;
	}

	if (.@count <= 0)
		.@count = 1;
	if (.@name$ == "")
		.@name$ = "--ja--";

	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	areamonster(.@map$, .@x - 3, .@y - 3, .@x + 3, .@y + 3, .@name$, .@mob_id, .@count, "DM_Console::OnDMKilled");
	dispbottom("[DM] Spawned " + .@count + " mob(s) near you on " + .@map$ + ".");
	return;
