OnRoll:
	.@input$ = "";
	.@hidden = 0;
	.@start_idx = 0;

	if (.@atcmd_parameters$[0] == "fudge" || .@atcmd_parameters$[0] == "override") {
		if (getgmlevel() < 1) {
			dispbottom("[DM] Roll overrides are for DMs only.");
			end;
		}
		.@total = atoi(.@atcmd_parameters$[1]);
		if (.@atcmd_parameters$[1] == "") {
			dispbottom("[DM] Usage: @roll fudge <total> [note]");
			end;
		}
		.@note$ = "";
		for (.@i = 2; .@i < .@atcmd_numparameters; ++.@i)
			.@note$ += (.@i > 2 ? " " : "") + .@atcmd_parameters$[.@i];
		.@msg$ = "[Roll] " + strcharinfo(PC_NAME) + " set a DM override result: " + .@total;
		if (.@note$ != "")
			.@msg$ += " (" + .@note$ + ")";
		mapannounce strcharinfo(PC_MAP), .@msg$, bc_map, 0xFFAA33;
		end;
	}

	if (.@atcmd_parameters$[0] == "hidden") {
		if (getgmlevel() >= 1) {
			.@hidden = 1;
			.@start_idx = 1;
		} else {
			dispbottom("[DM] Hidden rolls are for DMs only.");
			end;
		}
	}

	for (.@i = .@start_idx; .@i < .@atcmd_numparameters; ++.@i)
		.@input$ += .@atcmd_parameters$[.@i];

	if (.@input$ == "") {
		dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
		end;
	}

	.@num = 1;
	.@sides = 20;
	.@mod = 0;

	if (sscanf(.@input$, "%dd%d+%d", .@num, .@sides, .@mod) >= 3) {
	} else if (sscanf(.@input$, "%dd%d-%d", .@num, .@sides, .@mod) >= 3) {
		.@mod = -.@mod;
	} else if (sscanf(.@input$, "%dd%d", .@num, .@sides) >= 2) {
		.@mod = 0;
	} else {
		dispbottom("[DM] Usage: @roll [hidden] <NdX[+/-mod]> (e.g. 1d20+3)");
		end;
	}

	if (.@num < 1) .@num = 1;
	if (.@num > 100) .@num = 100;
	if (.@sides < 2) .@sides = 2;
	if (.@sides > 1000) .@sides = 1000;

	.@sum = 0;
	.@detail$ = "";
	for (.@i = 0; .@i < .@num; ++.@i) {
		.@roll = rand(1, .@sides);
		.@sum += .@roll;
		if (.@num <= 20)
			.@detail$ += (.@i > 0 ? ", " : "") + .@roll;
	}

	.@total = .@sum + .@mod;

	.@msg$ = "[Roll] " + strcharinfo(PC_NAME) + " rolled " + .@input$ + ": " + .@total;
	if (.@detail$ != "")
		.@msg$ += " [" + .@detail$ + "]";
	if (.@mod != 0 || .@num > 20)
		.@msg$ += " (Base: " + .@sum + ( .@mod != 0 ? " Mod: " + (.@mod > 0 ? "+" : "") + .@mod : "" ) + ")";

	if (.@hidden)
		dispbottom("[Hidden] " + .@msg$);
	else
		mapannounce strcharinfo(PC_MAP), .@msg$, bc_map, 0xFFAA33;
	end;

S_Reset:
	.@offset = getarg(0, 0);
	.@confirm$ = .dm_params$[.@offset];
	if (.@confirm$ != "confirm") {
		dispbottom("[DM] @dm reset confirm — wipes all campaign flags and quests for a fresh playthrough.");
		dispbottom("[DM] Warning: clears all 19 arcs for every online party member. Cannot be undone.");
		return;
	}
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	if (.@party_id > 0) {
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i]))
				continue;
			if (!attachrid(.@accid[.@i]))
				continue;
			callfunc("DM_ClearAllArcFlags");
			detachrid();
			++.@cleared;
		}
		if (.@origin_aid)
			attachrid(.@origin_aid);
	} else {
		callfunc("DM_ClearAllArcFlags");
		.@cleared = 1;
	}
	if (.@party_id > 0)
		setd("$dm_inst_" + .@party_id, 0);
	callfunc("DM_CutsceneParty", 0, "", 0, .@party_id);
	callfunc("DM_EraseAllCampaignQuests");
	$dm_mode = 0;
	$dm_active_party = 0;
	$@dm_exp_party = 0;
	$@dm_exp_char = 0;
	$@dm_exp_rate = 100;
	callfunc("DM_EncounterReset");
	dispbottom("[DM] Campaign reset complete for " + .@cleared + " member(s). All arcs cleared, mode off.");
	return;

	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	.@enc_last_gid = @dm_enc_last_gid;

	.@inst_party = (.@party_id > 0 ? .@party_id : $dm_active_party);
	.@inst_id = (.@inst_party > 0 ? getd("$dm_inst_" + .@inst_party) : 0);
	.@exp_rate = ($@dm_exp_rate > 0 ? $@dm_exp_rate : 100);
	.@exp_scope$ = "normal";
	if ($@dm_exp_party > 0 && $@dm_exp_rate > 0)
		.@exp_scope$ = "party " + $@dm_exp_party;
	else if ($@dm_exp_char > 0 && $@dm_exp_rate > 0)
		.@exp_scope$ = "char " + $@dm_exp_char;
	else if (.@exp_rate != 100)
		.@exp_scope$ = "unscoped";

		return;

	if (.@action$ == "erase" || .@action$ == "clear") {
		.@done = callfunc("DM_InstanceQuestErase", .@quest_id);
		dispbottom("[DM] Quest " + .@quest_id + " erased for " + .@done + " party member(s).");
		return;
	}

	dispbottom("[DM] Unknown quest action: " + .@action$);
	dispbottom("[DM] Usage: @dmquest <start|complete|erase> <quest_id>");
	dispbottom("[DM]        @dmquest sync <forward|back>");
	dispbottom("[DM]        @dmquest refresh");
	return;

S_QuestRefresh:
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@party_id <= 0) {
		dispbottom("[DM] Join or create a party first.");
		return;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	.@nm = 0;
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		.@maccid[.@nm] = .@accid[.@i];
		++.@nm;
	}

	// Build campaign quest ID list.
	.@n = 0;
	for (.@q = 20000; .@q <= 20030; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20101; .@q <= 20105; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20111; .@q <= 20115; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20121; .@q <= 20124; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20131; .@q <= 20134; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20141; .@q <= 20144; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20151; ++.@n;
	for (.@q = 20153; .@q <= 20155; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20161; ++.@n;
	for (.@q = 20163; .@q <= 20165; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20171; ++.@n;
	for (.@q = 20173; .@q <= 20175; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20181; ++.@n;
	for (.@q = 20183; .@q <= 20185; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20191; .@q <= 20194; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20201; .@q <= 20204; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20211; .@q <= 20214; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20221; .@q <= 20223; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20231; .@q <= 20233; ++.@q) { .@qids[.@n] = .@q; ++.@n; }

	.@refreshed = 0;
	for (.@i = 0; .@i < .@nm; ++.@i) {
		if (!attachrid(.@maccid[.@i])) continue;
		for (.@qi = 0; .@qi < .@n; ++.@qi) {
			if (questprogress(.@qids[.@qi]) == 1) {
				erasequest(.@qids[.@qi]);
				setquest(.@qids[.@qi]);
				++.@refreshed;
			}
		}
		detachrid();
	}

	if (.@origin_aid) attachrid(.@origin_aid);
	dispbottom("[DM] Refreshed " + .@refreshed + " in-progress quest(s) across " + .@nm + " member(s).");
	return;

S_QuestSync:
	.@dir$ = getarg(0, "forward");
	.@forward = (.@dir$ != "back");
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@party_id <= 0) {
		dispbottom("[DM] Join or create a party first.");
		return;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	.@nm = 0;
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		.@maccid[.@nm] = .@accid[.@i];
		++.@nm;
	}

	if (.@nm < 2) {
		dispbottom("[DM] Need at least 2 online party members to sync.");
		if (.@origin_aid) attachrid(.@origin_aid);
		return;
	}

	// Build list of all campaign quest IDs.
	.@n = 0;
	for (.@q = 20000; .@q <= 20030; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20101; .@q <= 20105; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20111; .@q <= 20115; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20121; .@q <= 20124; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20131; .@q <= 20134; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20141; .@q <= 20144; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20151; ++.@n;
	for (.@q = 20153; .@q <= 20155; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20161; ++.@n;
	for (.@q = 20163; .@q <= 20165; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20171; ++.@n;
	for (.@q = 20173; .@q <= 20175; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	.@qids[.@n] = 20181; ++.@n;
	for (.@q = 20183; .@q <= 20185; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20191; .@q <= 20194; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20201; .@q <= 20204; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20211; .@q <= 20214; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20221; .@q <= 20223; ++.@q) { .@qids[.@n] = .@q; ++.@n; }
	for (.@q = 20231; .@q <= 20233; ++.@q) { .@qids[.@n] = .@q; ++.@n; }

	.@changed = 0;
	for (.@qi = 0; .@qi < .@n; ++.@qi) {
		.@qid = .@qids[.@qi];
		.@best = (.@forward ? 0 : 2);

		// Pass 1: find furthest (forward) or least (back) progress.
		for (.@i = 0; .@i < .@nm; ++.@i) {
			if (!attachrid(.@maccid[.@i])) continue;
			.@prog = questprogress(.@qid);
			detachrid();
			if (.@forward && .@prog > .@best) .@best = .@prog;
			else if (!.@forward && .@prog < .@best) .@best = .@prog;
		}

		// Pass 2: apply target progress to every member.
		for (.@i = 0; .@i < .@nm; ++.@i) {
			if (!attachrid(.@maccid[.@i])) continue;
			.@cur = questprogress(.@qid);
			if (.@cur != .@best) {
				if (.@cur > 0) erasequest(.@qid);
				if (.@best >= 1) setquest(.@qid);
				if (.@best == 2) completequest(.@qid);
				++.@changed;
			}
			detachrid();
		}
	}

	if (.@origin_aid) attachrid(.@origin_aid);
	dispbottom("[DM] Quest sync (" + .@dir$ + ") done — " + .@changed + " update(s) across " + .@nm + " member(s).");
	return;

S_ResetStat:
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@done = 0;
	if (.@party_id > 0) {
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i])) continue;
			if (!attachrid(.@accid[.@i])) continue;
			resetstatus();
			detachrid();
			++.@done;
		}
		if (.@origin_aid) attachrid(.@origin_aid);
	} else {
		resetstatus();
		.@done = 1;
	}
	dispbottom("[DM] Status points reset for " + .@done + " member(s).");
	return;

S_ResetSkill:

S_SavePoint:
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	.@party_id = getcharid(CHAR_ID_PARTY);
	getmapxy(.@save_map$, .@save_x, .@save_y, UNITTYPE_PC);
	.@done = 0;

	if (.@party_id > 0) {
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i]))
				continue;
			if (!attachrid(.@accid[.@i]))
				continue;
			savepoint(.@save_map$, .@save_x, .@save_y);
			detachrid();
			++.@done;
		}
		if (.@origin_aid)
			attachrid(.@origin_aid);
	} else {
		savepoint(.@save_map$, .@save_x, .@save_y);
		.@done = 1;
	}

	dispbottom("[DM] Save point set to " + .@save_map$ + " " + .@save_x + "," + .@save_y + " for " + .@done + " member(s).");
	return;

S_Novice:
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@granted = 0;
	// Tutorial quest IDs from the old novice training grounds (new_1-1 through new_1-3).
	// Erasing them removes any gating in case a player landed there before skipping.
	setarray .@tut_quests[0], 7117, 7118, 7119, 7120, 7121, 7122, 7123, 7124, 7126, 7127;
	if (.@party_id > 0) {
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i]))
				continue;
			if (!attachrid(.@accid[.@i]))
				continue;
			skill "NV_FIRSTAID", 1, 0;
			skill "NV_TRICKDEAD", 1, 0;
			for (.@qi = 0; .@qi < getarraysize(.@tut_quests); ++.@qi) {
				if (questprogress(.@tut_quests[.@qi]))
					erasequest(.@tut_quests[.@qi]);
			}
			detachrid();
			++.@granted;
		}
		if (.@origin_aid)
			attachrid(.@origin_aid);
	} else {
		skill "NV_FIRSTAID", 1, 0;
		skill "NV_TRICKDEAD", 1, 0;
		for (.@qi = 0; .@qi < getarraysize(.@tut_quests); ++.@qi) {
			if (questprogress(.@tut_quests[.@qi]))
				erasequest(.@tut_quests[.@qi]);
		}
		.@granted = 1;
	}
	dispbottom("[DM] Granted First Aid + Play Dead and cleared tutorial flags for " + .@granted + " member(s).");
	return;

S_Spawn:
	.@offset = getarg(0, 0);
	.@mob_id = atoi(.dm_params$[.@offset]);
	.@count = atoi(.dm_params$[.@offset + 1]);
	.@name$ = .dm_params$[.@offset + 2];

	if (.@mob_id <= 0) {
		dispbottom("[DM] Usage: @dm spawn <mob_id> [count] [name]");
		return;
	}

	if (.@count <= 0)
		.@count = 1;
	if (.@name$ == "")
		.@name$ = "--ja--";

	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	for (.@i = 0; .@i < .@count; ++.@i) {
		.@sx = .@x + rand(7) - 3;
		.@sy = .@y + rand(7) - 3;
		if (.@sx < 1)
			.@sx = .@x;
		if (.@sy < 1)
			.@sy = .@y;

		.@gid = monster(.@map$, .@sx, .@sy, .@name$, .@mob_id, 1, "DM_Console::OnDMKilled");
		if (.@gid <= 0)
			continue;
		callfunc("DM_EncounterTrack", .@gid, .@name$, .@mob_id, 0);
		++.@spawned;
	}
	dispbottom("[DM] Spawned " + .@spawned + " mob(s) near you on " + .@map$ + ". Use @dm encounter status for GIDs.");
	return;

S_HoldSpawn:
	.@offset = getarg(0, 0);
	.@mob_id = atoi(.dm_params$[.@offset]);
	.@count = atoi(.dm_params$[.@offset + 1]);
	.@name$ = .dm_params$[.@offset + 2];

	if (.@mob_id <= 0) {
		dispbottom("[DM] Usage: @dm holdspawn <mob_id> [count] [name]");
		return;
	}

	if (.@count <= 0)
		.@count = 1;
	if (.@name$ == "")
		.@name$ = "--ja--";

	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	for (.@i = 0; .@i < .@count; ++.@i) {
		.@sx = .@x + rand(7) - 3;
		.@sy = .@y + rand(7) - 3;
		if (.@sx < 1)
			.@sx = .@x;
		if (.@sy < 1)
			.@sy = .@y;

		.@gid = monster(.@map$, .@sx, .@sy, .@name$, .@mob_id, 1, "DM_Console::OnDMKilled");
		if (.@gid <= 0)
			continue;

		.@mode = getunitdata(.@gid, UDT_MODE);
		@dm_hold_gid[@dm_hold_count] = .@gid;
		@dm_hold_mode[@dm_hold_count] = .@mode;
		++@dm_hold_count;
		++.@held;

		callfunc("DM_EncounterTrack", .@gid, .@name$, .@mob_id, 1);
		setunitdata(.@gid, UDT_MODE, .@mode & ~(MD_AGGRESSIVE|MD_ANGRY|MD_CANATTACK));
	}

	dispbottom("[DM] Held " + .@held + " staged mob(s). Use @dm release when combat begins, or @dm encounter status for GIDs.");
	return;

S_Release:
	.@offset = getarg(0, 0);
	.@which$ = .dm_params$[.@offset];
	if (.@which$ == "")
		.@which$ = "all";

	if (@dm_hold_count <= 0) {
		dispbottom("[DM] No held mobs tracked for you.");
		return;
	}

	if (.@which$ == "last") {
		.@start = @dm_hold_count - 1;
		.@end = @dm_hold_count;
	} else {
		.@start = 0;
		.@end = @dm_hold_count;
	}

	for (.@i = .@start; .@i < .@end; ++.@i) {
		if (@dm_hold_gid[.@i] <= 0)
			continue;
		setunitdata(@dm_hold_gid[.@i], UDT_MODE, @dm_hold_mode[.@i] | MD_AGGRESSIVE | MD_CANATTACK | MD_CANMOVE);
		callfunc("DM_EncounterMarkHeld", @dm_hold_gid[.@i], 0);
		++.@released;
	}

	if (.@which$ == "last") {
		--@dm_hold_count;
	} else {
		deletearray @dm_hold_gid[0], getarraysize(@dm_hold_gid);
		deletearray @dm_hold_mode[0], getarraysize(@dm_hold_mode);
		@dm_hold_count = 0;
	}

	dispbottom("[DM] Released " + .@released + " held mob(s).");
	return;

S_HoldClear:
	if (@dm_hold_count <= 0)
		return;

	for (.@i = 0; .@i < @dm_hold_count; ++.@i) {
		if (@dm_hold_gid[.@i] > 0) {
			unitkill(@dm_hold_gid[.@i]);
			callfunc("DM_EncounterForgetGid", @dm_hold_gid[.@i]);
		}
	}
	deletearray @dm_hold_gid[0], getarraysize(@dm_hold_gid);
	deletearray @dm_hold_mode[0], getarraysize(@dm_hold_mode);
	@dm_hold_count = 0;
	dispbottom("[DM] Cleared held staged mobs.");
	return;
