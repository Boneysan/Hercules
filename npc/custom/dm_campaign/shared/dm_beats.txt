//===== Seal Cascade Campaign ================================
//= DM Beat Director
//===== Description: =========================================
//= The @dmbeat menu tree, extracted from dm_console.txt. One
//= global function per menu level: DM_BeatMenu -> DM_BeatAct1..4
//= -> DM_BeatArc01..19. Each arc beat warps the party and/or
//= applies story flags/quests via the shared DM_ helpers.
//= Invoked from dm_console.txt OnBeat via callfunc("DM_BeatMenu").
//===========================================================

function	script	DM_BeatMenu	{
	mes("[DM Beat Director]");
	mes("Choose an act.");
	next();
	switch (select("Act I (Arcs 1-5)", "Act II (Arcs 6-10)", "Act III (Arcs 11-14)", "Act IV (Arcs 15-19)", "Cancel")) {
	case 1: callfunc("DM_BeatAct1"); close();
	case 2: callfunc("DM_BeatAct2"); close();
	case 3: callfunc("DM_BeatAct3"); close();
	case 4: callfunc("DM_BeatAct4"); close();
	case 5: close();
	}
	return;

}
