// Seal Cascade shared DM helpers.

function	script	DM_IsDM	{
	return getgmlevel() >= getarg(0, 1);
}

function	script	DM_RequireDM	{
	if (callfunc("DM_IsDM", getarg(0, 1)))
		return 1;

	dispbottom("[DM] You do not have permission to use this tool.");
	return 0;
}

// Session is active only when a real party owns it. Party 0 used to match
// every unpartied player (0 == 0) and unlocked the campaign for the whole map.
function	script	DM_SessionAllows	{
	return $dm_mode && $dm_active_party > 0 && getcharid(CHAR_ID_PARTY) == $dm_active_party;
}

function	script	DM_SessionParty	{
	if ($dm_mode && $dm_active_party > 0)
		return $dm_active_party;
	return getcharid(CHAR_ID_PARTY);
}

// Hercules treats $ inside a quoted string as a variable, so a regex $
// anchor cannot be written. Walk the name instead.
function	script	DM_IsIdentChar	{
	.@c$ = getarg(0);
	if (.@c$ == "_")
		return 1;
	if (.@c$ >= "0" && .@c$ <= "9")
		return 1;
	if (.@c$ >= "A" && .@c$ <= "Z")
		return 1;
	if (.@c$ >= "a" && .@c$ <= "z")
		return 1;
	return 0;
}

function	script	DM_ValidFlagName	{
	.@flag$ = getarg(0);
	.@len = getstrlen(.@flag$);
	if (.@len < 4)
		return 0;
	if (substr(.@flag$, 0, 2) != "dm_")
		return 0;
	for (.@i = 3; .@i < .@len; ++.@i) {
		if (!callfunc("DM_IsIdentChar", substr(.@flag$, .@i, .@i)))
			return 0;
	}
	return 1;
}

function	script	DM_ValidGrantKey	{
	.@key$ = getarg(0);
	.@len = getstrlen(.@key$);
	if (.@len < 1)
		return 0;
	for (.@i = 0; .@i < .@len; ++.@i) {
		if (!callfunc("DM_IsIdentChar", substr(.@key$, .@i, .@i)))
			return 0;
	}
	return 1;
}

// Party-scoped "already paid" latch. Set before the grant. Returns 1 the first
// time this party claims the key, 0 afterwards. Cleared when @dmmode turns off.
function	script	DM_ClaimGrant	{
	.@key$ = getarg(0);
	.@party_id = getarg(1, callfunc("DM_SessionParty"));

	if (!callfunc("DM_ValidGrantKey", .@key$))
		return 0;
	if (.@party_id <= 0)
		return 0;

	.@var$ = "$dm_paid_" + .@party_id + "_" + .@key$;
	if (getd(.@var$))
		return 0;
	setd(.@var$, 1);
	return 1;
}

function	script	DM_ClearPartyGrants	{
	.@party_id = getarg(0, $dm_active_party);
	if (.@party_id <= 0)
		return 0;
	// query_sql is unused in campaign scripts; drop matching $dm_paid_* vars

		"sigil_ring", "arc01_deviruchi",

		"arc06_krenn_zeny", "arc06_mistress",
		"arc07_rsx",
		"arc08_darklord",
		"arc09_gloom",
		"arc10_kiel",
		"arc11_randgris",
		"arc12_naght",
		"arc13_beelzebub",
		"arc14_ifrit",
		"arc15_thanatos",
		"arc16_bijou",
		"arc17_beta",
		"arc19_surt", "arc19_finale";
	.@n = getarraysize(.@keys$);
	for (.@i = 0; .@i < .@n; ++.@i)
		setd("$dm_paid_" + .@party_id + "_" + .@keys$[.@i], 0);
	return 1;
}

function	script	DM_IsCampaignQuest	{
	.@quest_id = getarg(0);
	return .@quest_id >= 20001 && .@quest_id <= 20233;
}

		"einbroch", "einbech", "ein_dun02",
		"glast_01", "gl_cas02", "gl_chyard",
		"rachel", "ra_san05", "ra_temple",
		"lighthalzen", "lhz_dun03", "lhz_in01",
		"hugel", "odin_tem03",
		"splendide", "mid_camp", "man_fild01",

function	script	DM_CurrentParty	{
	return getcharid(CHAR_ID_PARTY);
}

function	script	DM_PartyOnlineCount	{
	.@party_id = getarg(0, getcharid(CHAR_ID_PARTY));
	if (.@party_id <= 0)
		return 1;

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	for (.@i = 0; .@i < .@count; ++.@i) {
		if (isloggedin(.@accid[.@i], .@charid[.@i]))
			++.@online;
	}

	return .@online;
}

function	script	DM_CurrentMap	{
	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	return .@map$;
}

function	script	DM_WarpParty	{
	.@map$ = getarg(0);
	.@x = getarg(1, 0);
	.@y = getarg(2, 0);
	.@party_id = getarg(3, getcharid(CHAR_ID_PARTY));

	if (.@map$ == "")
		return 0;

	// Solo / no party: warp the attached character only.
	if (.@party_id <= 0) {
		warp(.@map$, .@x, .@y);
		return 1;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	for (.@i = 0; .@i < .@count; ++.@i) {
		if (isloggedin(.@accid[.@i], .@charid[.@i])) {
			warpchar(.@map$, .@x, .@y, .@charid[.@i]);
			++.@moved;
		}
	}

	return .@moved;
}

function	script	DM_JoinArgs	{
	.@start = getarg(0, 0);
	.@end = getarg(1, .@atcmd_numparameters);
	.@text$ = "";

	for (.@i = .@start; .@i < .@end; ++.@i)
		.@text$ += (.@i > .@start ? " " : "") + .@atcmd_parameters$[.@i];

	return .@text$;
}

function	script	DM_ParseTier	{
	.@tier$ = getarg(0, "common");

	if (.@tier$ == "common")
		return 0;
	if (.@tier$ == "uncommon")
		return 1;
	if (.@tier$ == "rare")
		return 2;
	if (.@tier$ == "boss")
		return 3;

	return atoi(.@tier$);
}

function	script	DM_TierName	{
	switch (getarg(0, 0)) {
		case 1: return "uncommon";
		case 2: return "rare";
		case 3: return "boss";
	}

	return "common";
}

function	script	DM_Color	{
	switch (getarg(0, 0)) {
		case 1: return 0x88CCFF;
		case 2: return 0xD0A0FF;
		case 3: return 0xFFAA33;
	}

	return 0xAAFFAA;
}

// Fire an event on a hazard/encounter NPC by name, resolving to the
// instance-specific clone when the caller's party has a live instance.
function	script	DM_TriggerEvent	{
	.@npc$   = getarg(0);
	.@label$ = getarg(1);

	.@party_id = getcharid(CHAR_ID_PARTY);
	if (.@party_id > 0) {
		.@stored = getd("$dm_inst_" + .@party_id);
		if (.@stored > 0) {
			// Storage is iid + 1 so instance 0 is trackable. instance_npcname
			// wants the real id; passing the stored value fired the next
			// party's clone (or a missing dup_1_* on the first instance).
			.@inst_npc$ = instance_npcname(.@npc$, .@stored - 1);
			if (.@inst_npc$ != "")
				.@npc$ = .@inst_npc$;
		}
	}

	donpcevent(.@npc$ + "::" + .@label$);
	return;
}
