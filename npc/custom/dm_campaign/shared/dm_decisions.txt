function	script	DM_DecisionFind	{
	callfunc("DM_DecisionInit");
	.@key$ = strtolower(getarg(0, ""));
	.@outcome$ = strtolower(getarg(1, ""));

	for (.@i = 0; .@i < getarraysize($@dm_dec_key$); ++.@i) {
		if ($@dm_dec_key$[.@i] == .@key$ && $@dm_dec_outcome$[.@i] == .@outcome$)
			return .@i;
	}
	return -1;
}

function	script	DM_DecisionParseArc	{
	.@s$ = strtolower(getarg(0, ""));
	if (.@s$ == "")
		return 0;
	if (getstrlen(.@s$) >= 4 && substr(.@s$, 0, 2) == "arc")
		return atoi(substr(.@s$, 3, getstrlen(.@s$) - 1));
	return atoi(.@s$);
}

		return .@arc;
	}

	for (.@i = 1; .@i <= 19; ++.@i) {
		.@p = questprogress(callfunc("DM_ArcTrackerId", .@i));
		if (.@p == 1)
			return .@i;
		if (.@p == 2)
			.@last = .@i;
	}
	return .@last;
}

function	script	DM_PartyAddFlag	{
	.@flag$ = getarg(0);
	.@delta = getarg(1, 1);
	.@party_id = getarg(2, getcharid(CHAR_ID_PARTY));
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@flag$ == "")
		return 0;

	if (.@party_id <= 0) {
		setd(.@flag$, getd(.@flag$) + .@delta);
		return 1;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		if (!attachrid(.@accid[.@i]))
			continue;
		setd(.@flag$, getd(.@flag$) + .@delta);
		detachrid();
		++.@done;
	}

	if (.@origin_aid)
		attachrid(.@origin_aid);

	return .@done;
}

function	script	DM_DecisionFlagName	{
	.@entry$ = getarg(0, "");
	if (.@entry$ == "")
		return "";

	deletearray .@parts$[0], getarraysize(.@parts$);
	explode(.@parts$, .@entry$, "+=");
	if (getarraysize(.@parts$) > 1)
		return "";

	deletearray .@parts$[0], getarraysize(.@parts$);
	explode(.@parts$, .@entry$, "=");
	if (getarraysize(.@parts$) > 1)
		return .@parts$[0];

	return .@entry$;
}

function	script	DM_DecisionApplySetList	{
	.@list$ = getarg(0, "");
	.@party_id = getarg(1, getcharid(CHAR_ID_PARTY));
	if (.@list$ == "")
		return 0;

	deletearray .@entries$[0], getarraysize(.@entries$);
	explode(.@entries$, .@list$, ",");
	for (.@i = 0; .@i < getarraysize(.@entries$); ++.@i) {
		.@entry$ = .@entries$[.@i];
		if (.@entry$ == "")
			continue;

		deletearray .@parts$[0], getarraysize(.@parts$);
		explode(.@parts$, .@entry$, "+=");
		if (getarraysize(.@parts$) > 1) {
			.@done += callfunc("DM_PartyAddFlag", .@parts$[0], atoi(.@parts$[1]), .@party_id);
			continue;
		}

		deletearray .@parts$[0], getarraysize(.@parts$);
		explode(.@parts$, .@entry$, "=");
		if (getarraysize(.@parts$) > 1)
			.@done += callfunc("DM_PartySetFlag", .@parts$[0], atoi(.@parts$[1]), .@party_id);
		else
			.@done += callfunc("DM_PartySetFlag", .@entry$, 1, .@party_id);
	}
	return .@done;
}

function	script	DM_DecisionApplyClearList	{
	.@list$ = getarg(0, "");
	.@party_id = getarg(1, getcharid(CHAR_ID_PARTY));
	if (.@list$ == "")
		return 0;

	deletearray .@entries$[0], getarraysize(.@entries$);
	explode(.@entries$, .@list$, ",");
	for (.@i = 0; .@i < getarraysize(.@entries$); ++.@i) {
		.@flag$ = callfunc("DM_DecisionFlagName", .@entries$[.@i]);
		if (.@flag$ != "")
			.@done += callfunc("DM_PartyClearFlag", .@flag$, .@party_id);
	}
	return .@done;
}

function	script	DM_DecisionApplyQuestList	{
	.@action$ = getarg(0, "");
	.@list$ = getarg(1, "");
	.@party_id = getarg(2, getcharid(CHAR_ID_PARTY));
	if (.@action$ == "" || .@list$ == "")
		return 0;

	deletearray .@quests$[0], getarraysize(.@quests$);
	explode(.@quests$, .@list$, ",");
	for (.@i = 0; .@i < getarraysize(.@quests$); ++.@i) {
		.@quest_id = atoi(.@quests$[.@i]);
		if (.@quest_id <= 0)
			continue;
		if (.@action$ == "start" || .@action$ == "set")
			.@done += callfunc("DM_PartyQuestSet", .@quest_id, .@party_id);
		else if (.@action$ == "complete")
			.@done += callfunc("DM_PartyQuestComplete", .@quest_id, .@party_id);
		else if (.@action$ == "erase" || .@action$ == "clear")
			.@done += callfunc("DM_PartyQuestErase", .@quest_id, .@party_id);
	}
	return .@done;
}

function	script	DM_DecisionSetListMatches	{
	.@list$ = getarg(0, "");
	if (.@list$ == "")
		return 0;

	deletearray .@entries$[0], getarraysize(.@entries$);
	explode(.@entries$, .@list$, ",");
	for (.@i = 0; .@i < getarraysize(.@entries$); ++.@i) {
		.@entry$ = .@entries$[.@i];
		if (.@entry$ == "")
			continue;

		deletearray .@parts$[0], getarraysize(.@parts$);
		explode(.@parts$, .@entry$, "+=");
		if (getarraysize(.@parts$) > 1)
			continue;

		deletearray .@parts$[0], getarraysize(.@parts$);
		explode(.@parts$, .@entry$, "=");
		if (getarraysize(.@parts$) > 1) {
			if (callfunc("DM_GetFlag", .@parts$[0]) != atoi(.@parts$[1]))
				return 0;
		} else if (!callfunc("DM_GetFlag", .@entry$)) {
			return 0;
		}
	}
	return 1;
}

function	script	DM_DecisionChosenOutcome	{
	callfunc("DM_DecisionInit");
	.@key$ = strtolower(getarg(0, ""));

	for (.@i = 0; .@i < getarraysize($@dm_dec_key$); ++.@i) {
		if ($@dm_dec_key$[.@i] == .@key$ && callfunc("DM_DecisionSetListMatches", $@dm_dec_set$[.@i]))
			return $@dm_dec_outcome$[.@i];
	}
	return "";
}

function	script	DM_Decide	{
	callfunc("DM_DecisionInit");
	.@key$ = strtolower(getarg(0, ""));
	.@outcome$ = strtolower(getarg(1, ""));
	.@idx = callfunc("DM_DecisionFind", .@key$, .@outcome$);
	.@party_id = getcharid(CHAR_ID_PARTY);

	if (.@idx < 0) {
		dispbottom("[DM] Unknown decision/outcome: " + .@key$ + " " + .@outcome$);
		callfunc("DM_DecideStatus", callfunc("DM_DecisionArcForKey", .@key$));
		return 0;
	}

	callfunc("DM_DecisionUndo", .@key$, 0);
	callfunc("DM_DecisionApplyClearList", $@dm_dec_clear$[.@idx], .@party_id);
	.@flag_writes = callfunc("DM_DecisionApplySetList", $@dm_dec_set$[.@idx], .@party_id);
	.@quest_writes = callfunc("DM_DecisionApplyQuestList", $@dm_dec_quest_action$[.@idx], $@dm_dec_quests$[.@idx], .@party_id);

	if ($@dm_dec_base_exp[.@idx] > 0 || $@dm_dec_job_exp[.@idx] > 0)
		callfunc("DM_PartyExp", $@dm_dec_base_exp[.@idx], $@dm_dec_job_exp[.@idx]);

	if ($@dm_dec_beat[.@idx] > 0)
		callfunc("DM_PartyApplyFlag", "dm_story_beat", $@dm_dec_beat[.@idx], .@party_id);

	if ($@dm_dec_story$[.@idx] != "")
		callfunc("DM_MapStory", $@dm_dec_story$[.@idx]);

	dispbottom("[DM] Decision " + .@key$ + " -> " + .@outcome$ + " applied. Flags=" + .@flag_writes + " quest_ops=" + .@quest_writes + ".");
	if ($@dm_dec_confirm$[.@idx] != "")
		dispbottom("[DM] " + $@dm_dec_confirm$[.@idx]);
	return 1;
}

function	script	DM_DecisionArcForKey	{
	callfunc("DM_DecisionInit");
	.@key$ = strtolower(getarg(0, ""));
	for (.@i = 0; .@i < getarraysize($@dm_dec_key$); ++.@i) {
		if ($@dm_dec_key$[.@i] == .@key$)
			return $@dm_dec_arc[.@i];
	}
	return callfunc("DM_CurrentArc");
}

function	script	DM_DecisionUndo	{
	callfunc("DM_DecisionInit");
	.@key$ = strtolower(getarg(0, ""));
	.@announce = getarg(1, 1);
	.@party_id = getcharid(CHAR_ID_PARTY);

	if (.@key$ == "")
		return 0;

	for (.@i = 0; .@i < getarraysize($@dm_dec_key$); ++.@i) {
		if ($@dm_dec_key$[.@i] != .@key$)
			continue;
		++.@matched;
		.@done += callfunc("DM_DecisionApplyClearList", $@dm_dec_set$[.@i], .@party_id);
		.@done += callfunc("DM_DecisionApplyClearList", $@dm_dec_clear$[.@i], .@party_id);
	}

	if (.@announce) {
		if (.@matched)
			dispbottom("[DM] Decision " + .@key$ + " reset to PENDING. Flag writes=" + .@done + ".");
		else
			dispbottom("[DM] Unknown decision key: " + .@key$);
	}
	return .@matched;
}

function	script	DM_DecideStatus	{
	callfunc("DM_DecisionInit");
	.@arc = getarg(0, callfunc("DM_CurrentArc"));
	if (.@arc <= 0)
		.@arc = callfunc("DM_CurrentArc");

	dispbottom("[DM] Decisions for Arc " + .@arc + ":");
	for (.@i = 0; .@i < getarraysize($@dm_dec_key$); ++.@i) {
		if ($@dm_dec_arc[.@i] != .@arc)
			continue;

		.@seen = 0;
		for (.@j = 0; .@j < .@i; ++.@j) {
			if ($@dm_dec_key$[.@j] == $@dm_dec_key$[.@i]) {
				.@seen = 1;
				break;
			}
		}
		if (.@seen)
			continue;

		.@chosen$ = callfunc("DM_DecisionChosenOutcome", $@dm_dec_key$[.@i]);
		dispbottom("  " + $@dm_dec_key$[.@i] + " (" + $@dm_dec_prompt$[.@i] + ") -> " + (.@chosen$ == "" ? "PENDING" : .@chosen$));
		++.@count;
	}

	if (!.@count)
		dispbottom("  No registered decisions for this arc.");
	return .@count;
}

function	script	DM_DecideMenu	{
	callfunc("DM_DecisionInit");
	.@arc = getarg(0, callfunc("DM_CurrentArc"));
	if (.@arc <= 0)
		.@arc = callfunc("DM_CurrentArc");

	.@count = 0;
	for (.@i = 0; .@i < getarraysize($@dm_dec_key$); ++.@i) {
		if ($@dm_dec_arc[.@i] != .@arc)
			continue;

		.@seen = 0;
		for (.@j = 0; .@j < .@count; ++.@j) {
			if (.@keys$[.@j] == $@dm_dec_key$[.@i]) {
				.@seen = 1;
				break;
			}
		}
		if (.@seen)
			continue;

		.@keys$[.@count] = $@dm_dec_key$[.@i];
		.@chosen$ = callfunc("DM_DecisionChosenOutcome", $@dm_dec_key$[.@i]);
		.@menu$[.@count] = $@dm_dec_prompt$[.@i] + " [" + (.@chosen$ == "" ? "PENDING" : .@chosen$) + "]";
		++.@count;
	}

	if (!.@count) {
		mes("[DM Decisions]");
		mes("No registered decisions for Arc " + .@arc + ".");
		close;
	}

	mes("[DM Decisions]");
	mes("Arc " + .@arc + " decisions.");
	next();
	.@pick = select(implode(.@menu$, ":") + ":Status:Cancel");
	if (.@pick == .@count + 1) {
		callfunc("DM_DecideStatus", .@arc);
		close;
	}
	if (.@pick == .@count + 2)
		close;

	callfunc("DM_DecideOutcomeMenu", .@keys$[.@pick - 1]);
	close;
}

function	script	DM_DecideOutcomeMenu	{
	callfunc("DM_DecisionInit");
	.@key$ = strtolower(getarg(0, ""));
	.@count = 0;

	for (.@i = 0; .@i < getarraysize($@dm_dec_key$); ++.@i) {
		if ($@dm_dec_key$[.@i] != .@key$)
			continue;
		if (.@prompt$ == "")
			.@prompt$ = $@dm_dec_prompt$[.@i];
		.@outcomes$[.@count] = $@dm_dec_outcome$[.@i];
		.@menu$[.@count] = $@dm_dec_label$[.@i] + " (" + $@dm_dec_outcome$[.@i] + ")";
		++.@count;
	}

	if (!.@count) {
		mes("[DM Decisions]");
		mes("Unknown decision: " + .@key$);
		close;
	}

	mes("[DM Decisions]");
	mes(.@prompt$);
	mes("Current: " + (callfunc("DM_DecisionChosenOutcome", .@key$) == "" ? "PENDING" : callfunc("DM_DecisionChosenOutcome", .@key$)));
	next();
	.@pick = select(implode(.@menu$, ":") + ":Undo / set PENDING:Cancel");
	if (.@pick <= .@count) {
		callfunc("DM_Decide", .@key$, .@outcomes$[.@pick - 1]);
		return 1;
	}
	if (.@pick == .@count + 1)
		callfunc("DM_DecisionUndo", .@key$);
	return 0;
}

function	script	DM_DecideCommand	{
	.@num = getarg(0, 0);
	.@a$ = strtolower(getarg(1, ""));
	.@b$ = strtolower(getarg(2, ""));
	.@c$ = strtolower(getarg(3, ""));

	if (.@num <= 0 || .@a$ == "") {
		callfunc("DM_DecideMenu", callfunc("DM_CurrentArc"));
		return 1;
	}

	if (.@a$ == "status") {
		.@arc = callfunc("DM_DecisionParseArc", .@b$);
		if (.@arc <= 0)
			.@arc = callfunc("DM_CurrentArc");
		callfunc("DM_DecideStatus", .@arc);
		return 1;
	}

	if (.@a$ == "undo") {
		if (.@b$ == "") {
			dispbottom("[DM] Usage: @dm decide undo <key>");
			return 0;
		}
		callfunc("DM_DecisionUndo", .@b$);
		return 1;
	}

	.@arc = callfunc("DM_DecisionParseArc", .@a$);
	if (.@arc > 0 && .@b$ == "") {
		callfunc("DM_DecideMenu", .@arc);
		return 1;
	}

	if (.@b$ == "") {
		callfunc("DM_DecideOutcomeMenu", .@a$);
		return 1;
	}

	return callfunc("DM_Decide", .@a$, .@b$);
}
