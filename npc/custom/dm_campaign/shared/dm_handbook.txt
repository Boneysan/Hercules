// Private in-game GM Handbook. Browsing never mutates campaign state.

function	script	DM_HandbookMenu	{
	mes("[GM Handbook]");
	mes("Private cues. SEND is the only player-visible action. Browsing never mutates.");
	next();
	switch (select("Arc 1 - Prontera", "Arc 2 - Payon", "Arc 3 - Morroc", "Arc 4 - Geffen", "Arc 5 - Alberta", "Open Beat Director", "Close")) {
	case 1: callfunc("DM_HandbookArc01"); break;
	case 2: callfunc("DM_HandbookArc02"); break;
	case 3: callfunc("DM_HandbookArc03"); break;
	case 4: callfunc("DM_HandbookArc04"); break;
	case 5: callfunc("DM_HandbookArc05"); break;
	case 6: callfunc("DM_BeatMenu"); break;
	case 7: close();
	}
	return;
}

function	script	DM_HandbookArc	{
	.@arc = getarg(0, 1);
	if (.@arc == 1) callfunc("DM_HandbookArc01");
	else if (.@arc == 2) callfunc("DM_HandbookArc02");
	else if (.@arc == 3) callfunc("DM_HandbookArc03");
	else if (.@arc == 4) callfunc("DM_HandbookArc04");
	else if (.@arc == 5) callfunc("DM_HandbookArc05");
	else {
		mes("[GM Handbook - Arc " + .@arc + "]");
		mes("Use the quest-giver and CAMPAIGN.md. Act I is the reference slice.");
		next();
		callfunc("DM_BeatMenu");
	}
	return;
}

function	script	DM_HandbookArc01	{
	mes("[Arc 1 - Omens at the Fountain]");
	mes("Promise: the small people we help change what happens under the city.");
	mes("Main path: clues (paint, Mira, Tibbets) -> unofficial job -> optional drain/binding -> Holt.");
	next();
	switch (select("Wynne", "Mother / Mira / soup", "Tibbets / wheel / Hlin", "Listening Chamber", "Resolution", "Party State", "Open Arc 1 Beats", "Back")) {
	case 1: callfunc("DM_HandbookArc01Scene", 1); break;
	case 2: callfunc("DM_HandbookArc01Scene", 2); break;
	case 3: callfunc("DM_HandbookArc01Scene", 3); break;
	case 4: callfunc("DM_HandbookArc01Scene", 4); break;
	case 5: callfunc("DM_HandbookArc01Scene", 5); break;
	case 6:
		mes("[Arc 1 Party State]");
		mes("clues=" + dm_arc01_clue_mask + " child=" + dm_arc01_child_found + " route=" + dm_arc01_rescue_route);
		mes("key=" + dm_arc01_key_access + " drained=" + dm_arc01_chamber_drained + " binding=" + dm_arc01_binding_applied);
		mes("approach=" + dm_arc01_holt_approach + " fate=" + dm_arc01_holt_fate + " ring=" + dm_arc01_sigil_ring_obtained);
		next(); callfunc("DM_HandbookArc01"); break;
	case 7: callfunc("DM_BeatArc01"); break;
	case 8: callfunc("DM_HandbookMenu"); break;
	}
	return;
}

function	script	DM_HandbookArc01Scene	{
	.@scene = getarg(0);
	if (.@scene == 1) {
		mes("[Wynne]");
		mes("Starts 20001 only. Optional hunts from a menu. 20005 after two clues or one clue + Tibbets.");
		.@read$ = "Prontera's fountain throws clean water into a city pretending the southern road is not full of refugees. Wynne opens an ink-stained ledger.";
	} else if (.@scene == 2) {
		mes("[Mother / Mira]");
		mes("Mother cannot finish 20004 until Mira is found at the soup line. Quiet walk / report / confront. Later is safe.");
		.@read$ = "A gray pot, a gray man, a girl with a crayon goat. The mother will not pretend the child is home until you have seen her.";
	} else if (.@scene == 3) {
		mes("[Tibbets / wheel / Hlin]");
		mes("Friendship gives the key, not the drain. Force the catch or ask him. Failed drain pulses and retries. Binding is optional starve.");
		.@read$ = "The Culvert hums through the sluice like struck glass. Tibbets will not go down with you.";
	} else if (.@scene == 4) {
		mes("[Listening Chamber]");
		mes("Fate commits after the fight from holt_approach: spare / custody / lethal. Adds: quiet or confront -2, report +1, drain -1, binding -1, floor 1.");
		.@read$ = "A Deviruchi is chained to a cracked goat-head slab. Cassell speaks like water and leaves through the grate.";
	} else {
		mes("[Resolution]");
		mes("20001 and 20005 close together. Point to Sun-Hwa, payon 181,104.");
		.@read$ = "The chamber falls silent. The Sigil Ring grows colder. Far below, something vast shifts once in its sleep.";
	}
	next();
	switch (select("SEND read-aloud to current map", "Open Arc 1 Beats", "Back", "Close")) {
	case 1: callfunc("DM_MapStory", .@read$); dispbottom("[GM Handbook] Narration sent; no story state changed."); close();
	case 2: callfunc("DM_BeatArc01"); break;
	case 3: callfunc("DM_HandbookArc01"); break;
	case 4: close();
	}
	return;
}
