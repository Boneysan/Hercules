// Seal Cascade Trap and Hazard Helpers
//
// These functions provide standard puzzle resets, area hazards,
// and trigger tile abstractions for the DM campaign.

// DM_HazardArea("<map>", x, y, range, hp_percent_loss, status_type,
//               status_duration{, party_id{, "<save_tag>", save_dc}})
// Inflicts damage and/or status effects to the caller's party in the area.
//
// With a save tag and DC, each member in the area rolls that check first:
// passing halves the damage and avoids the status entirely, failing takes it
// whole. This is what turns a hazard from a tax into a decision — the party
// can position, brace, or send the VIT character in first. Both arguments are
// optional and every pre-existing caller (Arcs 4, 7, 10, 12, 14, 15, 19) keeps
// its original unconditional behaviour.
function	script	DM_HazardArea	{
	.@map$ = getarg(0);
	.@x = getarg(1);
	.@y = getarg(2);
	.@range = getarg(3);
	.@percent_damage = getarg(4, 0);
	.@status_type = getarg(5, 0);
	.@status_dur = getarg(6, 10000);
	.@party_id = getarg(7, getcharid(CHAR_ID_PARTY));
	.@save_tag$ = getarg(8, "");
	.@save_dc = getarg(9, 0);

	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@party_id <= 0) {
		getmapxy(.@pc_map$, .@pc_x, .@pc_y, UNITTYPE_PC);
		.@dx = .@pc_x - .@x;
		.@dy = .@pc_y - .@y;
		if (.@dx < 0)
			.@dx = -.@dx;
		if (.@dy < 0)
			.@dy = -.@dy;
		if (.@pc_map$ == .@map$ && .@dx <= .@range && .@dy <= .@range) {
			.@dmg = .@percent_damage;
			.@sc = .@status_type;
			if (.@save_tag$ != "" && .@save_dc > 0
				&& callfunc("DM_CheckRoll", .@save_tag$, .@save_dc) >= 1) {
				.@dmg = .@dmg / 2;
				.@sc = 0;
			}
			if (.@dmg > 0)
				percentheal -(.@dmg), 0;
			if (.@sc > 0)
				sc_start .@sc, .@status_dur, 1;
			return 1;
		}
		return 0;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@party_count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	for (.@i = 0; .@i < .@party_count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		if (!attachrid(.@accid[.@i]))
			continue;

		getmapxy(.@pc_map$, .@pc_x, .@pc_y, UNITTYPE_PC);
		.@dx = .@pc_x - .@x;
		.@dy = .@pc_y - .@y;
		if (.@dx < 0)
			.@dx = -.@dx;
		if (.@dy < 0)
			.@dy = -.@dy;
		if (.@pc_map$ != .@map$ || .@dx > .@range || .@dy > .@range) {
			detachrid();
			continue;
		}

		.@dmg = .@percent_damage;
		.@sc = .@status_type;
		if (.@save_tag$ != "" && .@save_dc > 0
			&& callfunc("DM_CheckRoll", .@save_tag$, .@save_dc) >= 1) {
			.@dmg = .@dmg / 2;
			.@sc = 0;
		}
		if (.@dmg > 0)
			percentheal -(.@dmg), 0;
		if (.@sc > 0)
			sc_start .@sc, .@status_dur, 1;
		++.@hit;
		detachrid();
	}

	if (.@origin_aid > 0)
		attachrid(.@origin_aid);
	else
		detachrid();

	return .@hit;
}

// One signaled pulse. Wrong glyph configs and failed drain repairs use this
// instead of a looping hazard. Recoverable; does not start a timer.
function	script	DM_PulseWarn	{
	.@map$ = getarg(0, strnpcinfo(NPC_MAP));
	.@x = getarg(1, 0);
	.@y = getarg(2, 0);
	.@pct = getarg(3, 3);
	.@line$ = getarg(4, "Pressure vents. The attempt resets. Try a different order.");
	mapannounce .@map$, "[Seal] " + .@line$, bc_map, 0xFFAA33;
	callfunc("DM_HazardArea", .@map$, .@x, .@y, 8, .@pct, SC_CURSE, 3000, getcharid(CHAR_ID_PARTY));
	return 1;
}

// DM_ResetPuzzleFlag("<flag_prefix>", count)
// Sets a sequence of indexed flags back to 0. Useful for resettable puzzles.
function	script	DM_ResetPuzzleFlag	{
	.@prefix$ = getarg(0);
	.@count = getarg(1);
	for (.@i = 1; .@i <= .@count; ++.@i) {
		callfunc("DM_InstanceClearFlag", .@prefix$ + .@i);
	}
	return 1;
}

// DM_CleanupEncounter("<label>", "<map>")
// A reusable path to kill monsters spawned by an encounter label.
function	script	DM_CleanupEncounter	{
	.@label$ = getarg(0);
	.@map$ = getarg(1);
	if (.@map$ == "")
		.@map$ = strnpcinfo(NPC_MAP);
	killmonster .@map$, .@label$;
	return 1;
}
