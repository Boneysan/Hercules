// These functions provide standard resettable puzzle step logic and
// riddle checking for campaign set-pieces.

// DM_PuzzleStep("<flag_prefix>", step, total, "<wrong_event$>")
// Use for sequences of levers/buttons that must be hit in order.
// Sets prefix + step if all prior steps are set.
// Otherwise resets all prefix flags (1..total) and optionally triggers the wrong_event label.
// Returns 1 on success, 0 on failure.
function	script	DM_PuzzleStep	{
	.@prefix$ = getarg(0);
	.@step = getarg(1);
	.@total = getarg(2);
	.@wrong_event$ = getarg(3, "");

	.@ok = 1;
	for (.@i = 1; .@i < .@step; ++.@i) {
		if (callfunc("DM_GetFlag", .@prefix$ + .@i) == 0) {
			.@ok = 0;
			break;
		}
	}

	if (.@ok) {
		callfunc("DM_InstanceSetFlag", .@prefix$ + .@step, 1);
		return 1;
	} else {
		callfunc("DM_ResetPuzzleFlag", .@prefix$, .@total);
		if (.@wrong_event$ != "") {
			donpcevent .@wrong_event$;
		}
		return 0;
	}
}

// DM_PuzzleRiddle("<solved_flag>", "<attempts_flag>", max_attempts, "<fail_event$>", is_correct)
// Tracks riddle attempts. If is_correct is 1, sets solved_flag and returns 1.
// If 0, increments attempts_flag. If attempts >= max, fires fail_event$ and returns -1.
// Otherwise returns 0.
function	script	DM_PuzzleRiddle	{
	.@flag$ = getarg(0);
	.@attempts_flag$ = getarg(1);
	.@max_attempts = getarg(2);
	.@fail_event$ = getarg(3, "");
	.@is_correct = getarg(4, 0);

	if (callfunc("DM_GetFlag", .@flag$)) {
		return 1;
	}

	if (.@is_correct) {
		callfunc("DM_InstanceSetFlag", .@flag$, 1);
		return 1;
	}

	.@attempts = callfunc("DM_GetFlag", .@attempts_flag$) + 1;
	callfunc("DM_InstanceSetFlag", .@attempts_flag$, .@attempts);

	if (.@attempts >= .@max_attempts) {
		if (.@fail_event$ != "") donpcevent .@fail_event$;
		return -1;
	}

	return 0;
}
