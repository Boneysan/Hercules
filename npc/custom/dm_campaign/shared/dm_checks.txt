// Seal Cascade DM skill checks / saving throws.
//
// RO-native math: roll d20 + (stat / 10) versus a DC. A stat of 30 = +3,
// 99 = +9, so higher-level heroes with more stat points are naturally better.
//
//   @dm check <player|party> <stat> <DC> [adv|dis]
//     stat: str agi vit int dex luk
//     adv  = roll twice, keep high;  dis = roll twice, keep low
//   @dm inspire <player> [+n|spend|clear|set n]
//   @dm inspire party
//
// Results are announced to everyone on the DM's map so the whole table sees the
// drama. Nat-20 always succeeds, nat-1 always fails. If a character has
// dm_inspiration and the check is rolled with adv, one token is consumed.

function	script	DM_StatConst	{
	.@s$ = strtolower(getarg(0, ""));
	if (.@s$ == "str") return bStr;
	if (.@s$ == "agi") return bAgi;
	if (.@s$ == "vit") return bVit;
	if (.@s$ == "int") return bInt;
	if (.@s$ == "dex") return bDex;
	if (.@s$ == "luk") return bLuk;
	return -1;
}

// DM_RollCheck(<stat_id>, <dc>, <mode$>)
// Pure roll: d20 (+adv/dis) + (readparam(stat)/10) vs dc.
// Respects/ consumes inspiration on adv. Forces nat20=pass, nat1=fail.
// Returns 1=pass, 0=fail.
// Also sets @dm_check_* temps on the attached RID for callers that need
// the exact numbers (e.g. traps, initiative) to build custom announces.
// Caller must be attached to the character whose stats are read.
function	script	DM_RollCheck	{
	.@stat  = getarg(0);
	.@dc    = getarg(1, 10);
	.@mode$ = strtolower(getarg(2, ""));

	.@d1 = rand(1, 20);
	.@note$ = "";
	if (.@mode$ == "adv") {
		if (dm_inspiration > 0) {
			--dm_inspiration;
			.@note$ = " Inspiration";
		}
		.@d2 = rand(1, 20);
		.@roll = (.@d1 > .@d2 ? .@d1 : .@d2);
		.@note$ += " adv[" + .@d1 + "/" + .@d2 + "]";
	} else if (.@mode$ == "dis") {
		.@d2 = rand(1, 20);
		.@roll = (.@d1 < .@d2 ? .@d1 : .@d2);
		.@note$ = " dis[" + .@d1 + "/" + .@d2 + "]";
	} else {
		.@roll = .@d1;
	}

	.@mod = readparam(.@stat) / 10;
	.@total = .@roll + .@mod;
	.@pass = (.@total >= .@dc);

	if (.@roll == 20) {
		.@pass = 1;
	} else if (.@roll == 1) {
		.@pass = 0;
	}

	@dm_check_roll  = .@roll;
	@dm_check_mod   = .@mod;
	@dm_check_total = .@total;
	@dm_check_dc    = .@dc;
	@dm_check_note$ = .@note$;
	@dm_check_pass  = .@pass;  // redundant with return but handy
	return .@pass;
}

function	script	DM_CheckList	{
	dispbottom("Str = bStr, Agi = bAgi, Vit = bVit, Int = bInt, Dex = bDex, Luk = bLuk");
	return;
}

function	script	DM_Initiative	{
	.@party_id = getcharid(CHAR_ID_PARTY);
	if (.@party_id <= 0) {
		dispbottom("[DM] You must be in a party for initiative.");
		return;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	.@n = 0;
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		if (!attachrid(.@accid[.@i]))
			continue;

		callfunc("DM_RollCheck", bAgi, 0, "");
		.@rolls[.@n] = @dm_check_total;
		.@names$[.@n] = strcharinfo(PC_NAME);
		++.@n;
		detachrid();
	}

	// Sort by roll descending. Bubble sort.
	for (.@i = 0; .@i < .@n - 1; ++.@i) {
		for (.@j = 0; .@j < .@n - .@i - 1; ++.@j) {
			if (.@rolls[.@j] < .@rolls[.@j + 1]) {
				// Swap
				.@t = .@rolls[.@j];
				.@rolls[.@j] = .@rolls[.@j + 1];
				.@rolls[.@j + 1] = .@t;

				.@t$ = .@names$[.@j];
				.@names$[.@j] = .@names$[.@j + 1];
				.@names$[.@j + 1] = .@t$;
			}
		}
	}

	.@map$ = callfunc("DM_CurrentMap");
	mapannounce .@map$, "[Initiative Order]", bc_map, 0x00FFFF;
	for (.@i = 0; .@i < .@n; ++.@i) {
		mapannounce .@map$, " " + (.@i + 1) + ". " + .@names$[.@i] + " (" + .@rolls[.@i] + ")", bc_map, 0xFFFFFF;
	}
	
	return;
}

function	script	DM_Check	{
	.@target$ = getarg(0, "");
	.@stat$   = getarg(1, "");
	.@dc      = getarg(2, 10);
	.@mode$   = strtolower(getarg(3, ""));
	.@announce_map$ = strcharinfo(PC_MAP);
	.@origin = getcharid(CHAR_ID_ACCOUNT);

	.@stat = callfunc("DM_StatConst", .@stat$);
	if (.@stat < 0) {
		dispbottom("[DM] Usage: @dm check <player|party> <str|agi|vit|int|dex|luk> <DC> [adv|dis]");
		return 0;
	}
	if (.@dc <= 0)
		.@dc = 10;

	.@label$ = strtoupper(strtolower(.@stat$));

	if (.@target$ == "party" || .@target$ == "all") {
		.@party_id = getcharid(CHAR_ID_PARTY);
		if (.@party_id <= 0) {
			callsub(S_Roll, strcharinfo(PC_NAME), .@stat, .@label$, .@dc, .@mode$, .@announce_map$);
			return 1;
		}

		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);

		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i]))
				continue;
			if (!attachrid(.@accid[.@i]))
				continue;
			callsub(S_Roll, strcharinfo(PC_NAME), .@stat, .@label$, .@dc, .@mode$, .@announce_map$);
			detachrid();
			++.@done;
		}

		if (.@origin)
			attachrid(.@origin);
		if (!.@done)
			dispbottom("[DM] No online party members to check.");
		return .@done;
	}

	// Single named player (must be online for a stat read).
	.@accid = getcharid(CHAR_ID_ACCOUNT, .@target$);
	if (.@accid <= 0) {
		dispbottom("[DM] Player '" + .@target$ + "' is not online.");
		return 0;
	}
	if (!attachrid(.@accid)) {
		dispbottom("[DM] Could not attach to '" + .@target$ + "'.");
		return 0;
	}
	callsub(S_Roll, strcharinfo(PC_NAME), .@stat, .@label$, .@dc, .@mode$, .@announce_map$);
	detachrid();
	if (.@origin)
		attachrid(.@origin);
	return 1;

S_Roll:
	.@name$  = getarg(0);
	.@st     = getarg(1);
	.@lbl$   = getarg(2);
	.@dc2    = getarg(3);
	.@md$    = getarg(4);
	.@map$   = getarg(5);

	.@pass = callfunc("DM_RollCheck", .@st, .@dc2, .@md$);
	.@roll  = @dm_check_roll;
	.@mod   = @dm_check_mod;
	.@total = @dm_check_total;
	.@note$ = @dm_check_note$;

	.@res$ = (.@pass ? "SUCCESS" : "FAILURE");
	if (.@roll == 20) {
		.@res$ = "CRIT SUCCESS";
	} else if (.@roll == 1) {
		.@res$ = "CRIT FAIL";
	}

	.@modstr$ = (.@mod >= 0 ? "+" : "") + .@mod;
	.@msg$ = "[Check] " + .@name$ + "  " + .@lbl$ + " vs DC" + .@dc2 + ":  d20=" + .@roll + .@note$ + " " + .@modstr$ + " = " + .@total + "  -> " + .@res$;

	mapannounce .@map$, .@msg$, bc_map, (.@pass ? 0x66FF66 : 0xFF6666);
	return;
}

function	script	DM_Inspire	{
	.@target$ = getarg(0, "");
	.@action$ = strtolower(getarg(1, ""));
	.@value$ = getarg(2, "");
	.@origin = getcharid(CHAR_ID_ACCOUNT);
	.@announce_map$ = strcharinfo(PC_MAP);

	if (.@target$ == "" || .@target$ == "help") {
		dispbottom("[DM] @dm inspire <player> [+n|spend|clear|set n]");
		dispbottom("[DM] @dm inspire party   (list current party tokens)");
		return 0;
	}

	if (.@target$ == "party" || .@target$ == "all") {
		.@party_id = getcharid(CHAR_ID_PARTY);
		if (.@party_id <= 0) {
			dispbottom("[DM] You are not in a party. Your Inspiration: " + dm_inspiration);
			return 1;
		}

		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@line$ = "[DM] Inspiration:";
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (!isloggedin(.@accid[.@i], .@charid[.@i]))
				continue;
			if (!attachrid(.@accid[.@i]))
				continue;
			.@line$ += " " + strcharinfo(PC_NAME) + "=" + dm_inspiration;
			detachrid();
			++.@done;
		}
		if (.@origin)
			attachrid(.@origin);
		if (!.@done)
			dispbottom("[DM] No online party members to list.");
		else
			dispbottom(.@line$);
		return .@done;
	}

	.@accid = getcharid(CHAR_ID_ACCOUNT, .@target$);
	if (.@accid <= 0) {
		dispbottom("[DM] Player '" + .@target$ + "' is not online.");
		return 0;
	}

	if (!attachrid(.@accid)) {
		dispbottom("[DM] Could not attach to '" + .@target$ + "'.");
		return 0;
	}

	.@name$ = strcharinfo(PC_NAME);
	if (.@action$ == "" || .@action$ == "grant") {
		.@delta = 1;
	} else if (.@action$ == "spend" || .@action$ == "use") {
		if (dm_inspiration > 0) {
			--dm_inspiration;
			.@spent = 1;
		} else {
			.@empty = 1;
		}
	} else if (.@action$ == "clear" || .@action$ == "reset") {
		dm_inspiration = 0;
		.@cleared = 1;
	} else if (.@action$ == "set") {
		dm_inspiration = atoi(.@value$);
		.@set = 1;
	} else {
		.@delta = atoi(.@action$);
		if (.@delta == 0 && getstrlen(.@action$) > 1 && charat(.@action$, 0) == "+")
			.@delta = atoi(substr(.@action$, 1, getstrlen(.@action$) - 1));
	}

	if (.@delta != 0)
		dm_inspiration += .@delta;
	if (dm_inspiration < 0)
		dm_inspiration = 0;
	if (dm_inspiration > 9)
		dm_inspiration = 9;

	.@tokens = dm_inspiration;
	detachrid();
	if (.@origin)
		attachrid(.@origin);

	if (.@spent) {
		mapannounce .@announce_map$, "[DM] " + .@name$ + " spends Inspiration. (" + .@tokens + " left)", bc_map, 0xFFAA33;
		return 1;
	}
	if (.@empty) {
		dispbottom("[DM] " + .@name$ + " has no Inspiration to spend.");
		return 0;
	}
	if (.@cleared) {
		dispbottom("[DM] " + .@name$ + "'s Inspiration cleared.");
		return 1;
	}
	if (.@set) {
		dispbottom("[DM] " + .@name$ + " Inspiration set to " + .@tokens + ".");
		return 1;
	}
	if (.@delta != 0) {
		mapannounce .@announce_map$, "[DM] " + .@name$ + " gains Inspiration. (" + .@tokens + " token" + (.@tokens == 1 ? "" : "s") + ")", bc_map, 0xFFAA33;
		return 1;
	}

	dispbottom("[DM] Usage: @dm inspire <player> [+n|spend|clear|set n]");
	return 0;
}

-	script	DM_CheckConsole	FAKE_NPC,{
	end;

OnInit:
	bindatcmd("dmcheck", strnpcinfo(NPC_NAME_UNIQUE)+"::OnCheck", 1, 99, 1);
	bindatcmd("dminspire", strnpcinfo(NPC_NAME_UNIQUE)+"::OnInspire", 1, 99, 1);
	bindatcmd("dminitiative", strnpcinfo(NPC_NAME_UNIQUE)+"::OnInitiative", 1, 99, 1);
	end;

OnCheck:
	if (!callfunc("DM_RequireDM"))
		end;

	callfunc("DM_Check", .@atcmd_parameters$[0], .@atcmd_parameters$[1], atoi(.@atcmd_parameters$[2]), .@atcmd_parameters$[3]);
	end;

OnInspire:
	if (!callfunc("DM_RequireDM"))
		end;

	callfunc("DM_Inspire", .@atcmd_parameters$[0], .@atcmd_parameters$[1], .@atcmd_parameters$[2]);
	end;

OnInitiative:
	if (!callfunc("DM_RequireDM"))
		end;

	callfunc("DM_Initiative");
	end;
}
