// Seal Cascade arc symptom director.
//
// Converts the updated Obsidian per-arc "MVP Encounter & Symptom Mechanics"
// notes into live DM tools:
//
//   @dm symptom <arc> [pulse|setup|read|clear]
//   @dmsymptom <arc> [pulse|setup|read|clear]
//
// `pulse` announces the arc symptom and applies an RO-native proxy near the DM.
// `setup` only announces the table rule/concept.
// `read` prints concise boss read-aloud lines for the updated arc climax.
// `clear` clears map weather created by symptoms such as fog/ash.

function	script	DM_SymptomTitle	{
	switch (getarg(0, 0)) {
		case 1: return "The Tremors";
		case 2: return "Restless Spirits";
		case 3: return "Unnatural Heat";
		case 4: return "Wild Magic Surges";
		case 5: return "The Churning Tides";
		case 6: return "Failing Gravity";
		case 7: return "Toxic Smog";
		case 8: return "Aura of Decay";
		case 9: return "Oppressive Holiness";
		case 10: return "Chemical Paranoia";
		case 11: return "Dragon's Wrath";
		case 12: return "Dimensional Sickness";
		case 13: return "The Stifling Fog";
		case 14: return "Volcanic Ash";
		case 15: return "Time Anomalies";
		case 16: return "Martial Law";
		case 17: return "Rogue Leylines";
		case 18: return "The Grasp of the Dead";
		case 19: return "The Apocalypse";
	}
	return "";
}

function	script	DM_Symptom	{
	.@arc = atoi(getarg(0, ""));
	.@mode$ = strtolower(getarg(1, "pulse"));

	if (.@arc < 1 || .@arc > 19 || .@mode$ == "help") {
		dispbottom("[DM] @dm symptom <arc 1-19> [pulse|setup|read|clear]");
		dispbottom("[DM] pulse applies an RO proxy near you; setup announces the rule; read prints boss read-aloud.");
		return 0;
	}

	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	if (.@map$ == "") {
		dispbottom("[DM] Could not resolve your current map.");
		return 0;
	}

	if (.@mode$ == "read" || .@mode$ == "boss") {
		return callfunc("DM_SymptomBossRead", .@arc);
	}

	if (.@mode$ == "clear" || .@mode$ == "off") {
		callfunc("DM_SceneWeatherClear", .@map$);
		dispbottom("[DM] Symptom weather cleared on " + .@map$ + ".");
		return 1;
	}

	callfunc("DM_SymptomDescribe", .@arc, .@map$);

	if (.@mode$ == "setup" || .@mode$ == "info")
		return 1;
	if (.@mode$ != "pulse") {
		dispbottom("[DM] Unknown symptom mode '" + .@mode$ + "'. Use pulse, setup, read, or clear.");
		return 0;
	}

	.@party_id = getcharid(CHAR_ID_PARTY);

	switch (.@arc) {
		case 1:
			callfunc("DM_Check", "party", "dex", 12, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 14, 3, SC_STUN, 1500, .@party_id);
			break;
		case 2:
			callfunc("DM_Check", "party", "luk", 13, "");
			areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Restless Spirit", 1028, rand(2, 4);
			areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Restless Familiar", 1005, rand(1, 3);
			break;
		case 3:
			callfunc("DM_Check", "party", "vit", 14, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 16, 4, SC_BURNING, 4000, .@party_id);
			break;
		case 4:
			callfunc("DM_Check", "party", "int", 12, "");
			.@status = rand(1, 4);
			if (.@status == 1) .@sc = SC_CONFUSION;
			else if (.@status == 2) .@sc = SC_STUN;
			else if (.@status == 3) .@sc = SC_BLIND;
			else .@sc = SC_CURSE;
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 12, 0, .@sc, 5000, .@party_id);
			break;
		case 5:
			callfunc("DM_Check", "party", "str", 13, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 14, 4, SC_STUN, 1500, .@party_id);
			break;
		case 6:
			callfunc("DM_Check", "party", "dex", 12, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 14, 0, SC_SLOWDOWN, 5000, .@party_id);
			break;
		case 7:
			callfunc("DM_Check", "party", "vit", 13, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 16, 2, SC_POISON, 10000, .@party_id);
			break;
		case 8:
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 16, 3, SC_CURSE, 8000, .@party_id);
			break;
		case 9:
			callfunc("DM_Check", "party", "int", 13, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 14, 0, SC_FEAR, 4000, .@party_id);
			break;
		case 10:
			callfunc("DM_Check", "party", "int", 12, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 14, 0, SC_CONFUSION, 8000, .@party_id);
			break;
		case 11:
			callfunc("DM_Check", "party", "dex", 14, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 10, 5, SC_BURNING, 5000, .@party_id);
			break;
		case 12:
			callfunc("DM_Check", "party", "luk", 12, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 16, 0, SC_CONFUSION, 6000, .@party_id);
			break;
		case 13:
			callfunc("DM_SceneWeather", .@map$, "fog");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 18, 0, SC_BLIND, 10000, .@party_id);
			break;
		case 14:
			callfunc("DM_SceneWeather", .@map$, "clouds");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 16, 3, SC_BLIND, 8000, .@party_id);
			break;
		case 15:
			callfunc("DM_Check", "party", "luk", 15, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 14, 0, SC_SLOWDOWN, 6000, .@party_id);
			break;
		case 16:
			.@count = rand(2, 5);
			areamonster .@map$, .@x - 6, .@y - 6, .@x + 6, .@y + 6, "Royal Patrol", 1163, .@count;
			mapannounce .@map$, "[Martial Law] A patrol answers the disturbance.", bc_map, 0xFFAA33;
			break;
		case 17:
			callfunc("DM_Check", "party", "dex", 14, "");
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 14, 0, SC_ANKLESNARE, 5000, .@party_id);
			break;
		case 18:
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 16, 3, SC_SLOWDOWN, 8000, .@party_id);
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 16, 0, SC_CURSE, 8000, .@party_id);
			break;
		case 19:
			callfunc("DM_HazardArea", .@map$, .@x, .@y, 24, 6, 0, 0, .@party_id);
			break;
	}

	dispbottom("[DM] Symptom pulse fired for Arc " + .@arc + " on " + .@map$ + ".");
	return 1;
}
