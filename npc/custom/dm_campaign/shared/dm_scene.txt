// Seal Cascade DM ambience director.
//
// Sets mood on the DM's current map in one command: weather (map-wide mapflags),
// background music (custom tracks in the client BGM/ folder), and an optional
// illustration/portrait shown to the whole party.
//
//   @dm scene <preset> [portrait]
//     presets: clear dread boss calm holy ruin snow fest
//   @dm scene weather <fog|snow|sakura|leaves|clouds|clouds2|fireworks|none> [portrait]
//
// BGM note: this client (pre-20220504) cannot be told to STOP a BGM from script,
// so `@dm scene clear` only clears weather + portrait; play a calm scene to
// override the music. Preset BGM filenames (dm_dread, dm_boss, ...) are custom
// mp3s the DM distributes to players -- see planning/dm-live-table.md.

function	script	DM_SceneWeatherClear	{
	.@map$ = getarg(0);
	removemapflag(.@map$, MF_FOG);
	removemapflag(.@map$, MF_SNOW);
	removemapflag(.@map$, MF_SAKURA);
	removemapflag(.@map$, MF_LEAVES);
	removemapflag(.@map$, MF_CLOUDS);
	removemapflag(.@map$, MF_CLOUDS2);
	removemapflag(.@map$, MF_FIREWORKS);
	return;
}

function	script	DM_SceneWeather	{
	.@map$ = getarg(0);
	.@w$ = strtolower(getarg(1, ""));

	callfunc("DM_SceneWeatherClear", .@map$);

	if (.@w$ == "" || .@w$ == "none" || .@w$ == "clear")
		return 1;
	if (.@w$ == "fog")       { setmapflag(.@map$, MF_FOG);       return 1; }
	if (.@w$ == "snow")      { setmapflag(.@map$, MF_SNOW);      return 1; }
	if (.@w$ == "sakura")    { setmapflag(.@map$, MF_SAKURA);    return 1; }
	if (.@w$ == "leaves")    { setmapflag(.@map$, MF_LEAVES);    return 1; }
	if (.@w$ == "clouds")    { setmapflag(.@map$, MF_CLOUDS);    return 1; }
	if (.@w$ == "clouds2")   { setmapflag(.@map$, MF_CLOUDS2);   return 1; }
	if (.@w$ == "fireworks") { setmapflag(.@map$, MF_FIREWORKS); return 1; }

	dispbottom("[DM] Unknown weather '" + .@w$ + "'. Use: fog snow sakura leaves clouds clouds2 fireworks none.");
	return 0;
}

// Show (or clear, with empty filename) a cutin portrait for every online party
// member -- cutin is per-client, so it must be looped over the party.
function	script	DM_ScenePortrait	{
	.@img$ = getarg(0, "");
	.@pos = (.@img$ == "" ? 255 : 2);

	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin = getcharid(CHAR_ID_ACCOUNT);

	if (.@party_id <= 0) {
		cutin(.@img$, .@pos);
		return;
	}

	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);

	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		if (!attachrid(.@accid[.@i]))
			continue;
		cutin(.@img$, .@pos);
		detachrid();
	}

	if (.@origin)
		attachrid(.@origin);
	return;
}

function	script	DM_Scene	{
	.@sub$ = strtolower(getarg(0, ""));
	.@a$ = getarg(1, "");
	.@b$ = getarg(2, "");

	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	if (.@map$ == "") {
		dispbottom("[DM] Could not resolve your current map.");
		return 0;
	}

	if (.@sub$ == "" || .@sub$ == "help") {
		dispbottom("[DM] @dm scene <clear|dread|boss|calm|holy|ruin|snow|fest> [portrait]");
		dispbottom("[DM] @dm scene weather <fog|snow|sakura|leaves|clouds|clouds2|fireworks|none> [portrait]");
		return 0;
	}

	if (.@sub$ == "clear" || .@sub$ == "off") {
		callfunc("DM_SceneWeatherClear", .@map$);
		callfunc("DM_ScenePortrait", "");
		dispbottom("[DM] Scene cleared on " + .@map$ + ". (BGM cannot be stopped on this client; use a calm scene to override.)");
		return 1;
	}

	if (.@sub$ == "weather") {
		if (!callfunc("DM_SceneWeather", .@map$, .@a$))
			return 0;
		if (.@b$ != "")
			callfunc("DM_ScenePortrait", .@b$);
		dispbottom("[DM] Weather '" + .@a$ + "' set on " + .@map$ + ".");
		return 1;
	}

	// Named presets: weather + custom BGM (+ optional portrait in slot .@a$).
	.@weather$ = "";
	.@bgm$ = "";
	if (.@sub$ == "dread")      { .@weather$ = "fog";       .@bgm$ = "dm_dread"; }
	else if (.@sub$ == "boss")  { .@weather$ = "clouds2";   .@bgm$ = "dm_boss";  }
	else if (.@sub$ == "calm")  { .@weather$ = "none";      .@bgm$ = "dm_calm";  }
	else if (.@sub$ == "holy")  { .@weather$ = "sakura";    .@bgm$ = "dm_holy";  }
	else if (.@sub$ == "ruin")  { .@weather$ = "fog";       .@bgm$ = "dm_ruin";  }
	else if (.@sub$ == "snow")  { .@weather$ = "snow";      .@bgm$ = "dm_snow";  }
	else if (.@sub$ == "fest")  { .@weather$ = "fireworks"; .@bgm$ = "dm_fest";  }
	else {
		dispbottom("[DM] Unknown scene '" + .@sub$ + "'. Try: clear dread boss calm holy ruin snow fest, or 'weather <type>'.");
		return 0;
	}

	callfunc("DM_SceneWeather", .@map$, .@weather$);
	if (.@bgm$ != "")
		playbgmall(.@bgm$, .@map$);
	if (.@a$ != "")
		callfunc("DM_ScenePortrait", .@a$);

	dispbottom("[DM] Scene '" + .@sub$ + "' on " + .@map$ + "  (weather=" + .@weather$ + ", bgm=" + .@bgm$ + (.@a$ != "" ? ", portrait=" + .@a$ : "") + ").");
	return 1;
}

-	script	DM_SceneConsole	FAKE_NPC,{
	end;

OnInit:
	bindatcmd("dmscene", strnpcinfo(NPC_NAME_UNIQUE)+"::OnScene", 1, 99, 1);
	end;

OnScene:
	if (!callfunc("DM_RequireDM"))
		end;

	callfunc("DM_Scene", .@atcmd_parameters$[0], .@atcmd_parameters$[1], .@atcmd_parameters$[2]);
	end;
}
