// Seal Cascade party-wide quest helpers.
//
// setquest/completequest/erasequest only affect the attached character, so
// these loop the online party (attaching each account RID in turn) and then
// restore the original attachment. Use them from boss-death labels and DM
// tools so quest credit lands on the whole party, not just the killer.

function	script	DM_PartyApplyQuest	{
	.@action$   = getarg(0);                                   // "set" | "complete" | "erase"
	.@quest_id  = getarg(1);
	.@party_id  = getarg(2, getcharid(CHAR_ID_PARTY));
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@quest_id <= 0)
		return 0;

	// Solo / no party: apply to the attached character only.
	if (.@party_id <= 0) {
		callsub(S_Apply, .@action$, .@quest_id);
		return 1;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		if (!attachrid(.@accid[.@i]))
			continue;
		callsub(S_Apply, .@action$, .@quest_id);
		detachrid();
		++.@done;
	}

	// Re-attach the original caller so the calling script keeps its context.
	if (.@origin_aid)
		attachrid(.@origin_aid);

	return .@done;

S_Apply:
	.@a$ = getarg(0);
	.@q = getarg(1);

	if (.@a$ == "set") {
		if (questprogress(.@q) == 0)
			setquest(.@q);
		return;
	}
	if (.@a$ == "complete") {
		if (questprogress(.@q) == 0)
			setquest(.@q);
		if (questprogress(.@q) == 1)
			completequest(.@q);
		return;
	}
	if (.@a$ == "erase") {
		if (questprogress(.@q) > 0)
			erasequest(.@q);
		return;
	}
	return;
}

function	script	DM_PartyQuestComplete	{
	return callfunc("DM_PartyApplyQuest", "complete", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_PartyQuestSet	{
	return callfunc("DM_PartyApplyQuest", "set", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_PartyQuestErase	{
	return callfunc("DM_PartyApplyQuest", "erase", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

// Instance quest aliases. These intentionally use the caller's party; the
// instance itself is party-owned, and the same helpers safely fall back to solo.
function	script	DM_InstanceQuestStart	{
	return callfunc("DM_PartyQuestSet", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_InstanceQuestComplete	{
	return callfunc("DM_PartyQuestComplete", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_InstanceQuestErase	{
	return callfunc("DM_PartyQuestErase", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

// Story flags are character vars too. Use these for party/instance choices so
// later NPC gates see the same branch for every online member in the group.
function	script	DM_PartyApplyFlag	{
	.@flag$ = getarg(0);
	.@value = getarg(1, 1);
	.@party_id = getarg(2, getcharid(CHAR_ID_PARTY));
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@flag$ == "")
		return 0;

	if (.@party_id <= 0) {
		setd(.@flag$, .@value);
		return 1;
	}

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@accid[.@i], .@charid[.@i]))
			continue;
		if (!attachrid(.@accid[.@i]))
			continue;
		setd(.@flag$, .@value);
		detachrid();
		++.@done;
	}

	if (.@origin_aid)
		attachrid(.@origin_aid);

	return .@done;
}

function	script	DM_PartySetFlag	{
	return callfunc("DM_PartyApplyFlag", getarg(0), getarg(1, 1), getarg(2, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_PartyClearFlag	{
	return callfunc("DM_PartyApplyFlag", getarg(0), 0, getarg(1, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_InstanceSetFlag	{
	return callfunc("DM_PartySetFlag", getarg(0), getarg(1, 1), getarg(2, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_InstanceClearFlag	{
	return callfunc("DM_PartyClearFlag", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

// Erases every campaign quest (20000-20233 range blocks) for the caller's party.
// Used by @dm reset so the list of IDs lives in one place.
function	script	DM_EraseAllCampaignQuests	{
	for (.@arc = 1; .@arc <= 19; ++.@arc) {
		.@count = callfunc("DM_ArcQuestCount", .@arc);
		for (.@i = 0; .@i < .@count; ++.@i) {
			callfunc("DM_InstanceQuestErase", callfunc("DM_ArcQuestId", .@arc, .@i));
		}
	}
	return;
}

function	script	DM_ArcQuestCount	{
	.@arc = getarg(0);
	.@var$ = sprintf(".arc%02d_id", .@arc);
	return getarraysize(getvariableofnpc(getd(.@var$), "DM_QuestRegistry"));
}

function	script	DM_ArcQuestId	{
	.@arc = getarg(0);
	.@i = getarg(1);
	.@var$ = sprintf(".arc%02d_id", .@arc);
	return getelementofarray(getvariableofnpc(getd(.@var$), "DM_QuestRegistry"), .@i);
}

function	script	DM_ArcQuestName	{
	.@arc = getarg(0);
	.@i = getarg(1);
	.@var$ = sprintf(".arc%02d_nm$", .@arc);
	return getelementofarray(getvariableofnpc(getd(.@var$), "DM_QuestRegistry"), .@i);
}

function	script	DM_QuestNameById	{
	.@id = getarg(0);
	for (.@arc = 1; .@arc <= 19; ++.@arc) {
		.@count = callfunc("DM_ArcQuestCount", .@arc);
		for (.@i = 0; .@i < .@count; ++.@i) {
			if (callfunc("DM_ArcQuestId", .@arc, .@i) == .@id) {
				return callfunc("DM_ArcQuestName", .@arc, .@i);
			}
		}
	}
	return "Unknown Quest";
}

function	script	DM_ArcTrackerId	{
	return callfunc("DM_ArcQuestId", getarg(0), 0);
}
