// Seal Cascade party-wide quest helpers.
//
// setquest/completequest/erasequest only affect the attached character, so
// these loop the online party (attaching each account RID in turn) and then
// restore the original attachment. Use them from boss-death labels and DM
// tools so quest credit lands on the whole party, not just the killer.

// Remember one party quest or story flag. `name` is a quest id or a dm_* flag.
function	script	DM_StatePut	{
	.@party_id = getarg(0);
	.@kind$ = getarg(1);
	.@name$ = getarg(2);
	.@value = getarg(3);
	if (.@party_id <= 0 || .@kind$ == "" || .@name$ == "")
		return 0;
	.@campaign$ = callfunc("DM_CheckpointCampaign");
	// A later "start" must not replace a stored "complete". Erase passes 0 and
	// does replace. Flags always replace, because a choice can change.
	// A positive quest value never replaces a higher one. Zero and negative
	// values replace, so an erase (-1) cannot be undone by an old copy.
	if (.@value > 0 && (.@kind$ == "quest" || .@name$ == "dm_story_beat" || .@name$ == "dm_catechism_tally"))
		query_sql("INSERT INTO `dm_campaign_party_state` (`campaign_id`,`party_id`,`kind`,`name`,`value`) VALUES ('" + .@campaign$ + "'," + .@party_id + ",'quest','" + .@name$ + "'," + .@value + ") ON DUPLICATE KEY UPDATE `value` = GREATEST(`value`, " + .@value + ")");
	else
		query_sql("INSERT INTO `dm_campaign_party_state` (`campaign_id`,`party_id`,`kind`,`name`,`value`) VALUES ('" + .@campaign$ + "'," + .@party_id + ",'" + .@kind$ + "','" + .@name$ + "'," + .@value + ") ON DUPLICATE KEY UPDATE `value` = " + .@value);
	return 1;
}
// Write every online member's quests and flags into the party store, then
// copy that store onto anyone who is still missing it.
function	script	DM_PublishOnlineParty	{
	.@party_id = $dm_active_party;
	if (.@party_id <= 0)
		.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	if (.@party_id <= 0 || .@origin_aid <= 0)
		return 0;

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
	.@done = 0;
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (isloggedin(.@accid[.@i], .@charid[.@i]))
			++.@done;
	}
	freeloop(true);
	callfunc("DM_CatchUpFromParty", 1);
	campaigncatchup;
	freeloop(false);
	return .@done;
}

function	script	DM_StateGet	{
	.@party_id = getarg(0);
	.@kind$ = getarg(1);
	.@name$ = getarg(2);
	if (.@party_id <= 0)
		return 0;
	.@campaign$ = callfunc("DM_CheckpointCampaign");
	.@rows = query_sql("SELECT `value` FROM `dm_campaign_party_state` WHERE `campaign_id` = '" + .@campaign$ + "' AND `party_id` = " + .@party_id + " AND `kind` = '" + .@kind$ + "' AND `name` = '" + .@name$ + "'", .@value);
	if (.@rows != 1)
		return 0;
	return .@value;
}

function	script	DM_PartyApplyQuest	{
	.@action$   = getarg(0);                                   // "set" | "complete" | "erase"
	.@quest_id  = getarg(1);
	.@party_id  = getarg(2, getcharid(CHAR_ID_PARTY));
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@quest_id <= 0)
		return 0;
	if (callfunc("DM_SessionAllows") && .@party_id == $dm_active_party) {
		.@event_kind$ = "quest_" + .@action$;
		if (callfunc("DM_CampaignEventAppend", .@party_id, .@event_kind$, .@quest_id + "", .@quest_id) <= 0) {
			dispbottom("[DM] Campaign event could not be saved; quest state was not changed.");
			return 0;
		}
	}

	// Solo / no party: apply to the attached character only.
	if (.@party_id <= 0) {
		callsub(S_Apply, .@action$, .@quest_id, 0);
		return 1;
	}

	// The talker is updated here. Everyone else gets a queued catch-up so an
	// open conversation is not replaced by this script.
	callsub(S_Apply, .@action$, .@quest_id, .@party_id);
	// The actor has already applied this transition locally. Verify it through
	// the same ordered replay path used by reconnecting members, then advance
	// this character's cursor instead of waiting for a later catch-up command.
	if (callfunc("DM_SessionAllows") && .@party_id == $dm_active_party) {
		.@cursor_result = callfunc("DM_ReplayPartyEvents", .@party_id);
		if (.@cursor_result < 0)
			dispbottom("[DM] Campaign quest was applied, but its replay cursor could not be verified; it will retry on the next sync.");
	}
	campaigncatchup;
	return 1;

S_Apply:
	.@a$ = getarg(0);
	.@q = getarg(1);
	.@party_id = getarg(2, 0);

	if (.@a$ == "set") {
		if (questprogress(.@q) == 0)
			setquest(.@q);
	} else if (.@a$ == "complete") {
		if (questprogress(.@q) == 0)
			setquest(.@q);
		if (questprogress(.@q) == 1)
			completequest(.@q);
	} else if (.@a$ == "erase") {
		if (questprogress(.@q) > 0)
			erasequest(.@q);
	}
	if (.@party_id > 0) {
		if (.@a$ == "erase")
			callfunc("DM_StatePut", .@party_id, "quest", .@q + "", -1);
		else
			callfunc("DM_StatePut", .@party_id, "quest", .@q + "", questprogress(.@q));
	}
	return;
}

function	script	DM_PartyQuestComplete	{
	return callfunc("DM_PartyApplyQuest", "complete", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_PartyQuestSet	{
	return callfunc("DM_PartyApplyQuest", "set", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_PartyQuestErase	{
	return callfunc("DM_PartyApplyQuest", "erase", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

// Story flags are character vars too. Use these for party/instance choices so
// later NPC gates see the same branch for every online member in the group.
function	script	DM_PartyApplyFlag	{
	.@flag$ = getarg(0);
	.@value = getarg(1, 1);
	.@party_id = getarg(2, getcharid(CHAR_ID_PARTY));
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (!callfunc("DM_ValidFlagName", .@flag$))
		return 0;
	if (callfunc("DM_SessionAllows") && .@party_id == $dm_active_party) {
		.@event_kind$ = .@value == 0 ? "flag_clear" : "flag_set";
		if (callfunc("DM_CampaignEventAppend", .@party_id, .@event_kind$, .@flag$, .@value) <= 0) {
			dispbottom("[DM] Campaign event could not be saved; story flag was not changed.");
			return 0;
		}
	}

	if (.@party_id <= 0) {
		setd(.@flag$, .@value);
		return 1;
	}

	setd(.@flag$, .@value);
	if (.@value == 0)
		callfunc("DM_StatePut", .@party_id, "flag", .@flag$, -1);
	else
		callfunc("DM_StatePut", .@party_id, "flag", .@flag$, .@value);
	campaignpush;
	.@done = 1;

	// The durable SQL checkpoint advances once per accepted party flag event.
	// Character variables above remain the per-member mirror used by the
	// Session Board; they are never treated as the authority for progression.
	if (.@done > 0 && $dm_active_party > 0 && $dm_active_party == .@party_id)
		callfunc("DM_CheckpointRecordPartyEvent", .@party_id, 1, getcharid(CHAR_ID_CHAR), .@flag$);
	if (callfunc("DM_SessionAllows") && .@party_id == $dm_active_party) {
		.@cursor_result = callfunc("DM_ReplayPartyEvents", .@party_id);
		if (.@cursor_result < 0)
			dispbottom("[DM] Story flag was applied, but its replay cursor could not be verified; it will retry on the next sync.");
	}

	return .@done;
}

function	script	DM_PartySetFlag	{
	return callfunc("DM_PartyApplyFlag", getarg(0), getarg(1, 1), getarg(2, getcharid(CHAR_ID_PARTY)));
}

function	script	DM_PartyClearFlag	{

}

	return callfunc("DM_PartyClearFlag", getarg(0), getarg(1, getcharid(CHAR_ID_PARTY)));
}

// ---------------------------------------------------------------------------
// Hunting contracts.
//
// A contract is filled by handing in items, not by a kill counter. The item
// list lives in the generated dm_hunts.txt table (see tools/gen-hunts.py);
// these helpers are the only thing the arc scripts should call.
//
// Items stay in each character's inventory. A turn-in counts the whole party,
// including people who are logged out, and takes from them only when that
// total is enough. Nothing is removed on a short hand-in. Quest credit stays
// party-wide. Offline stacks are unequipped rows in the inventory table.
// ---------------------------------------------------------------------------

// The talker has the contract open and is short. Says so and prints the
// ledger. Returns 1 when it spoke. Does not take items.
function	script	DM_HuntNoteShort	{
	.@quest_id = getarg(0);
	if (questprogress(.@quest_id) != 1)
		return 0;
	if (callfunc("DM_HuntReady", .@quest_id))
		return 0;

	mes("This contract is not ready. Nothing was taken.");
	mes(callfunc("DM_HuntName", .@quest_id) + ": " + callfunc("DM_HuntBrief", .@quest_id));
	callfunc("DM_HuntLedger", .@quest_id);
	next();
	return 1;
}

// Unequipped stacks for a character who is logged out. Online characters are
// counted with countitem instead, because their live inventory is ahead of SQL.
function	script	DM_HuntOfflineHave	{
	.@char_id = getarg(0);
	.@item = getarg(1);
	if (.@char_id <= 0 || .@item <= 0)
		return 0;
	.@rows = query_sql("SELECT IFNULL(SUM(`amount`),0) FROM `inventory` WHERE `char_id` = " + .@char_id + " AND `nameid` = " + .@item + " AND `equip` = 0", .@sum);
	if (.@rows != 1)
		return 0;
	return .@sum;
}

// Remove up to `need` of an item from a logged-out character. Returns how many
// were removed. Equipped rows are left alone.
function	script	DM_HuntOfflineTake	{
	.@char_id = getarg(0);
	.@item = getarg(1);
	.@need = getarg(2);
	.@left = .@need;
	if (.@char_id <= 0 || .@item <= 0 || .@need <= 0)
		return 0;
	.@rows = query_sql("SELECT `id`, `amount` FROM `inventory` WHERE `char_id` = " + .@char_id + " AND `nameid` = " + .@item + " AND `equip` = 0 AND `amount` > 0 ORDER BY `id`", .@id, .@amount);
	for (.@i = 0; .@i < .@rows && .@left > 0; ++.@i) {
		.@take = .@amount[.@i];
		if (.@take > .@left)
			.@take = .@left;
		if (.@take >= .@amount[.@i])
			query_sql("DELETE FROM `inventory` WHERE `id` = " + .@id[.@i] + " AND `char_id` = " + .@char_id + " AND `equip` = 0");
		else
			query_sql("UPDATE `inventory` SET `amount` = `amount` - " + .@take + " WHERE `id` = " + .@id[.@i] + " AND `char_id` = " + .@char_id + " AND `equip` = 0 AND `amount` >= " + .@take);
		.@left = .@left - .@take;
	}
	return .@need - .@left;
}

// How many of this item the whole party holds, online or not.
function	script	DM_HuntHave	{
	.@item = getarg(0);
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	.@sum = countitem(.@item);

	if (.@party_id <= 0 || .@origin_aid <= 0)
		return .@sum;

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (.@accid[.@i] == .@origin_aid)
			continue;
		.@have = countcharitem(.@charid[.@i], .@item);
		if (.@have < 0)
			.@have = callfunc("DM_HuntOfflineHave", .@charid[.@i], .@item);
		.@sum = .@sum + .@have;
	}
	return .@sum;
}

// Does the party, including logged-out members, hold everything this contract asks for?
function	script	DM_HuntReady	{
	.@quest_id = getarg(0);
	.@n = callfunc("DM_HuntSpec", .@quest_id, .@item, .@need);

	if (.@n <= 0)
		return 0;

	for (.@i = 0; .@i < .@n; ++.@i) {
		if (callfunc("DM_HuntHave", .@item[.@i]) < .@need[.@i])
			return 0;
	}
	return 1;
}

// Take the contract's items from the whole party. Returns 0 without
// removing anything if they are short together. Logged-out members lose the
// items from their saved inventory.
function	script	DM_HuntCollect	{
	.@quest_id = getarg(0);
	.@n = callfunc("DM_HuntSpec", .@quest_id, .@item, .@need);
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);

	if (.@n <= 0)
		return 0;
	if (!callfunc("DM_HuntReady", .@quest_id))
		return 0;

	if (.@party_id > 0) {
		getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
		.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
	}

	for (.@i = 0; .@i < .@n; ++.@i) {
		.@left = .@need[.@i];
		.@take = countitem(.@item[.@i]);
		if (.@take > .@left)
			.@take = .@left;
		if (.@take > 0) {
			delitem(.@item[.@i], .@take);
			.@left = .@left - .@take;
		}
		if (.@party_id > 0) {
			for (.@m = 0; .@m < .@count && .@left > 0; ++.@m) {
				if (.@accid[.@m] == .@origin_aid)
					continue;
				.@take = delcharitem(.@charid[.@m], .@item[.@i], .@left);
				if (.@take < 0)
					.@take = callfunc("DM_HuntOfflineTake", .@charid[.@m], .@item[.@i], .@left);
				.@left = .@left - .@take;
			}
		}
		if (.@left > 0) {
			dispbottom("[DM] The hand-in stopped short. Nothing further was marked complete.");
			return 0;
		}
	}

	// Hunt collection is the authoritative item-consumption boundary. If this
	// character owns the party checkpoint's carried item, clear that ownership
	// only after every required item was successfully removed. A different
	// member cannot consume the checkpoint owner through this path.
	callfunc("DM_CheckpointConsume", getcharid(CHAR_ID_PARTY), getcharid(CHAR_ID_CHAR));

	return 1;
}

// Print the contract's ledger into the open dialogue: what it wants, what the
// talking character is carrying. This is the only progress display the game
// has for a hunt, so it is worth being plain rather than clever.
function	script	DM_HuntLedger	{
	.@quest_id = getarg(0);
	.@n = callfunc("DM_HuntSpec", .@quest_id, .@item, .@need);

	if (.@n <= 0)
		return 0;

	for (.@i = 0; .@i < .@n; ++.@i) {
		.@have = callfunc("DM_HuntHave", .@item[.@i]);
		if (.@have >= .@need[.@i])
			mes("  ^008000" + getitemname(.@item[.@i]) + " " + .@have + " / " + .@need[.@i] + "^000000");
		else
			mes("  " + getitemname(.@item[.@i]) + " ^777777" + .@have + " / " + .@need[.@i] + "^000000");
	}
	return .@n;
}

// The one-line summary an arc uses when it offers a contract.
// Copy quest progress onto this character when the party or the stored row
// is further along. Progress never moves backwards. Arguments are quest ids.
function	script	DM_SyncListedQuests	{
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	.@n = getargcount();

	if (.@party_id <= 0 || .@origin_aid <= 0 || .@n <= 0)
		return 0;

	for (.@q = 0; .@q < .@n; ++.@q) {
		.@quest = getarg(.@q);
		if (.@quest <= 0)
			continue;
		if (!attachrid(.@origin_aid))
			return .@copied;
		.@stored = callfunc("DM_StateGet", .@party_id, "quest", .@quest + "");
		if (.@stored < 0) {
			if (questprogress(.@quest) > 0)
				erasequest(.@quest);
			continue;
		}
		.@mine = questprogress(.@quest);
		.@best = .@mine;
		if (.@stored > .@best)
			.@best = .@stored;
		if (!attachrid(.@origin_aid))
			return .@copied;
		if (.@best > .@mine) {
			if (questprogress(.@quest) == 0 && .@best >= 1)
				setquest(.@quest);
			if (questprogress(.@quest) == 1 && .@best == 2)
				completequest(.@quest);
			++.@copied;
		}
		if (.@best > 0)
			callfunc("DM_StatePut", .@party_id, "quest", .@quest + "", .@best);
	}

	if (.@origin_aid)
		attachrid(.@origin_aid);
	return .@copied;
}

function	script	DM_HuntBrief	{
	.@quest_id = getarg(0);
	.@n = callfunc("DM_HuntSpec", .@quest_id, .@item, .@need);
	.@brief$ = "";

	for (.@i = 0; .@i < .@n; ++.@i) {
		if (.@i > 0)
			.@brief$ += ", ";
		.@brief$ += .@need[.@i] + " " + getitemname(.@item[.@i]);
	}
	return .@brief$;
}

// Print the ledger for every contract in the argument list that the attached
// character is still carrying. Arc turn-in NPCs call this once, so "what do I
// still owe you" is answered in the same breath as the reminder line.
function	script	DM_HuntStanding	{
	.@shown = 0;

	for (.@a = 0; .@a < getargcount(); ++.@a) {
		.@quest_id = getarg(.@a);
		if (.@quest_id <= 0 || questprogress(.@quest_id) != 1)
			continue;
		if (.@shown == 0) {
			mes(" ");
			mes("^777777Still owed:^000000");
		}
		mes("^3355FF" + callfunc("DM_HuntName", .@quest_id) + "^000000");
		callfunc("DM_HuntLedger", .@quest_id);
		++.@shown;
	}
	return .@shown;
}

// Login catch-up. Copies quest progress and story flags from an
// online party member onto a character who does not have them yet.
// Does not overwrite a value this character already has.
function	script	DM_CopyOneFlag	{
	.@flag$ = getarg(0);
	.@party_id = getcharid(CHAR_ID_PARTY);
	.@origin_aid = getcharid(CHAR_ID_ACCOUNT);
	if (.@party_id <= 0 || .@origin_aid <= 0)
		return 0;
	if (!callfunc("DM_ValidFlagName", .@flag$))
		return 0;
	.@mine = getd(.@flag$);
	.@stored = callfunc("DM_StateGet", .@party_id, "flag", .@flag$);
	// -1 is a cleared flag. A character who still has the old value must drop
	// it, and must not write that value back over the clear.
	if (.@stored < 0) {
		if (.@mine != 0)
			setd(.@flag$, 0);
		return .@mine != 0;
	}
	// The story beat only moves forward. A character who was offline for a
	// later chapter must catch up, and must not publish the old beat back.
	if (.@flag$ == "dm_story_beat" || .@flag$ == "dm_catechism_tally") {
		.@best = .@mine;
		if (.@stored > .@best)
			.@best = .@stored;
		if (.@best > .@mine)
			setd(.@flag$, .@best);
		if (getd(.@flag$) > 0)
			callfunc("DM_StatePut", .@party_id, "flag", .@flag$, getd(.@flag$));
		return .@best > .@mine;
	}
	if (@dm_flag_adopt && .@stored > 0 && .@stored != .@mine) {
		setd(.@flag$, .@stored);
		return 1;
	}
	if (.@mine != 0) {
		// Keep this character's choice. Only fill an empty party row from it.
		// A later party choice already in the store is not replaced.
		if (.@stored == 0)
			callfunc("DM_StatePut", .@party_id, "flag", .@flag$, .@mine);
		return 0;
	}
	.@val = .@stored;
	if (.@val != 0 && getd(.@flag$) == 0) {
		setd(.@flag$, .@val);
		return 1;
	}
	return 0;
}

// Pay item, zeny, and experience grants that were queued while this
// character was offline. Each row is marked claimed before the next, so a
// disconnect mid-loop cannot pay the same row twice.
function	script	DM_ClaimPendingGrants	{
	.@char_id = getcharid(CHAR_ID_CHAR);
	if (.@char_id <= 0)
		return 0;
	.@campaign$ = callfunc("DM_CheckpointCampaign");
	.@rows = query_sql("SELECT `id`, `kind`, `amount`, `amount2` FROM `dm_campaign_pending_grant` WHERE `campaign_id` = '" + .@campaign$ + "' AND `char_id` = " + .@char_id + " AND `claimed` = 0", .@id, .@kind$, .@amount, .@amount2);
	for (.@i = 0; .@i < .@rows; ++.@i) {
		query_sql("UPDATE `dm_campaign_pending_grant` SET `claimed` = 1 WHERE `id` = " + .@id[.@i] + " AND `claimed` = 0");
		if (.@kind$[.@i] == "item" && .@amount[.@i] > 0 && .@amount2[.@i] > 0)
			getitem(.@amount[.@i], .@amount2[.@i]);
		else if (.@kind$[.@i] == "zeny" && .@amount[.@i] > 0)
			Zeny += .@amount[.@i];
		else if (.@kind$[.@i] == "exp")
			getexp(.@amount[.@i], .@amount2[.@i]);
	}
	return .@rows;
}

// Replay typed quest/flag transitions for this character. The event cursor is
// advanced only after the character postcondition is observed; retries are
// idempotent and this path never issues rewards.
function	script	DM_ReplayPartyEvents	{
	.@party_id = getarg(0, getcharid(CHAR_ID_PARTY));
	.@char_id = getcharid(CHAR_ID_CHAR);
	.@campaign$ = callfunc("DM_CheckpointCampaign");
	if (.@party_id <= 0 || .@char_id <= 0 || !callfunc("DM_SessionAllows") || .@party_id != $dm_active_party)
		return -1;

	callfunc("DM_CheckpointRememberParty", .@party_id);
	query_sql("INSERT IGNORE INTO `dm_campaign_party_cursor` (`campaign_id`,`char_id`,`party_id`,`last_event_id`) VALUES ('" + .@campaign$ + "'," + .@char_id + "," + .@party_id + ",0)");
	// A character moving to a different active party gets that party's history,
	// never a cursor inherited from their prior party.
	query_sql("UPDATE `dm_campaign_party_cursor` SET `party_id` = " + .@party_id + ", `last_event_id` = 0 WHERE `campaign_id` = '" + .@campaign$ + "' AND `char_id` = " + .@char_id + " AND `party_id` != " + .@party_id);
	.@rows = query_sql("SELECT `last_event_id` FROM `dm_campaign_party_cursor` WHERE `campaign_id` = '" + .@campaign$ + "' AND `char_id` = " + .@char_id + " AND `party_id` = " + .@party_id, .@after);
	if (.@rows != 1)
		return -1;

	.@applied = 0;
	while (1) {
		.@rows = query_sql("SELECT `id`, `kind`, `name`, `value` FROM `dm_campaign_party_event` WHERE `campaign_id` = '" + .@campaign$ + "' AND `party_id` = " + .@party_id + " AND `id` > " + .@after + " ORDER BY `id` LIMIT 128", .@event_id, .@kind$, .@name$, .@value);
		if (.@rows <= 0)
			break;
		for (.@i = 0; .@i < .@rows; ++.@i) {
			.@ok = 0;
			if (.@kind$[.@i] == "quest_set" || .@kind$[.@i] == "quest_complete" || .@kind$[.@i] == "quest_erase") {
				.@quest_id = atoi(.@name$[.@i]);
				if (.@quest_id <= 0)
					return -1;
				if (.@kind$[.@i] == "quest_set") {
					if (questprogress(.@quest_id) == 0)
						setquest(.@quest_id);
					.@ok = questprogress(.@quest_id) > 0;
				} else if (.@kind$[.@i] == "quest_complete") {
					if (questprogress(.@quest_id) == 0)
						setquest(.@quest_id);
					if (questprogress(.@quest_id) == 1)
						completequest(.@quest_id);
					.@ok = questprogress(.@quest_id) == 2;
				} else {
					if (questprogress(.@quest_id) > 0)
						erasequest(.@quest_id);
					.@ok = questprogress(.@quest_id) == 0;
				}
			} else if (.@kind$[.@i] == "flag_set" || .@kind$[.@i] == "flag_clear") {
				if (!callfunc("DM_ValidFlagName", .@name$[.@i]))
					return -1;
				.@expected = .@kind$[.@i] == "flag_clear" ? 0 : .@value[.@i];
				setd(.@name$[.@i], .@expected);
				.@ok = getd(.@name$[.@i]) == .@expected;
			}
			if (!.@ok)
				return -1;
			query_sql("UPDATE `dm_campaign_party_cursor` SET `last_event_id` = " + .@event_id[.@i] + " WHERE `campaign_id` = '" + .@campaign$ + "' AND `char_id` = " + .@char_id + " AND `party_id` = " + .@party_id + " AND `last_event_id` < " + .@event_id[.@i]);
			.@cursor_rows = query_sql("SELECT `last_event_id` FROM `dm_campaign_party_cursor` WHERE `campaign_id` = '" + .@campaign$ + "' AND `char_id` = " + .@char_id + " AND `party_id` = " + .@party_id, .@cursor);
			if (.@cursor_rows != 1 || .@cursor < .@event_id[.@i])
				return -1;
			.@after = .@event_id[.@i];
			++.@applied;
		}
		if (.@rows < 128)
			break;
	}
	return .@applied;
}
