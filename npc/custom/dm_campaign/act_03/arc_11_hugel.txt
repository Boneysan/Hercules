// ── Priest Eadric — The Faithful Witness ───────────────────────────────────
hugel,100,100,5	script	Priest Eadric#dm	4_M_01,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20151) == 2) {
		mes("[Priest Eadric]");
		if (dm_arc11_sigrun_comforted)
			mes("Sigrun died with someone beside her. Randgris is gone. Heaven is quieter now, and I am learning how to grieve it without worshipping the wound.");
		else if (dm_arc11_sigrun_voice)
			mes("Sigrun spent her last words on the laughter behind the order. I do not know what that Voice is. I know it was glad.");
		else
			mes("Heaven fell silent. I am not sure if that is better or worse than what it was saying before.");
		close();
	}

	if (!dm_arc11_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20151, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20151, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20153, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20154, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20155, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("[Priest Eadric]");
	mes("You came from the south. Good. Everything coming from the temple has been carrying a blade.");
	next();
	if (dm_arc10_kiel_killed) {
		mes("[Eadric]");
		mes("A feather was shipped here from Lighthalzen, harvested from something divine as if heaven were another mine. I thought that was blasphemy. Now I think it was evidence.");
		next();
	}
	if (dm_arc09_karsh_exposed || dm_arc09_karsh_deal) {
		mes("[Eadric]");
		mes("Rachel's failures reached my correspondence before you did. I told myself our rites were older, cleaner. Then a Valkyrie descended at a public blessing and killed the man she was meant to bless.");
		next();
	}
	if (dm_arc08_manfred_spared || dm_arc08_manfred_killed) {
		mes("[Eadric]");
		mes("Glast Heim taught us what faith can become when it refuses grief. I have been thinking about that too often.");
		next();
	}
	mes("[Eadric]");
	mes("The Valkyries are not possessed. The temple's line to Odin is still there, but the cascade has filled it with noise. Without a clear signal, they have defaulted to judgment.");
	next();
	switch (select("Help us understand them.", "Just tell us how to survive.", "Your gods are murderers. Why defend them?")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc11_sigrun_comforted", 0);
		mes("[Eadric]");
		mes("Then I will trust you with the old rites. They may not save Randgris, but they may let you reach Sigrun before her light goes out.");
		break;
	case 2:
		mes("[Eadric]");
		mes("Wards, spacing, and humility. Do not meet a Valkyrie's charge like a duel. Survive long enough to learn what is happening.");
		break;
	case 3:
		mes("[Eadric]");
		mes("Because if love becomes denial when it fails us, Bjorn has already won. I will help you. I will also remember the shape of your anger.");
		break;
	}
	next();
	mes("[Eadric]");
	mes("Before anyone reaches the temple, Hugel has to keep breathing. Work the airship muster, log the burned dragon scales from Abyss Lake, and hold the outer ward vigil. Those three reports will tell us whether this is local or the first edge of something worse.");
	callfunc("DM_InstanceSetFlag", "dm_arc11_started", 1);
	callfunc("DM_InstanceQuestStart", 20151);
	callfunc("DM_InstanceQuestStart", 20153);
	callfunc("DM_InstanceQuestStart", 20154);
	callfunc("DM_InstanceQuestStart", 20155);
	next();
	mes("^3355FF[Quest started: When Heaven Turns]^000000");
	mes("^3355FFHugel Airship Muster - clear the Manuk fringe and decide who boards first.^000000");
	mes("^3355FFDragon Scale Turn-In - gather burned scales from Abyss Lake's large dragons.^000000");
	mes("^3355FFTemple Edge Vigil - hold the ward-line against Dolomedes and the night.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20153) == 1 && questprogress(20153, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20153);
		callfunc("DM_PartyExp", 300000, 145000);
		callfunc("DM_GivePartyZeny", 25000);
		.@paid = 1;
		mes("[Eadric]");
		mes("The airship approach is clear. The manifest is not. There are fewer seats than frightened people.");
		next();
		switch (select("Move refugees first.", "Move priests and clergy first.", "Move weapons and supplies first.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc11_airship_refugees", 1);
			mes("[Eadric]");
			mes("Then Hugel will remember you as the people who wrote the vulnerable into the ledger first. That matters when fear is doing the counting.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc11_airship_clergy", 1);
			mes("[Eadric]");
			mes("The rites continue, and so does the organization around them. The ordinary line waits longer. I know why. I still see them waiting.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc11_airship_weapons", 1);
			mes("[Eadric]");
			mes("The militia is better armed against the next descent. Civilians watched crates board ahead of children. That will be remembered too.");
			break;
		}
		next();
	}

	if (questprogress(20154) == 1 && questprogress(20154, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20154);
		callfunc("DM_PartyExp", 300000, 145000);
		callfunc("DM_GivePartyZeny", 25000);
		.@paid = 1;
		mes("[Eadric]");
		mes("The scales are scorched from within. Not fire. Not acid. The same signature as the inverted wards, running through Abyss Lake's dragons.");
		next();
		switch (select("Turn every scale over for study.", "Study some and sell some for bounty.", "Forge warding charms from the scales.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc11_scales_studied", 1);
			mes("[Eadric]");
			mes("Then we map the corruption floor by floor. Randgris will not be a surprise, even if she is still a tragedy.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc11_scales_split", 1);
			mes("[Eadric]");
			mes("Practical. We get a partial pattern, and you get coin enough to keep diving. I will take imperfect truth over no truth.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc11_scale_wards", 1);
			mes("[Eadric]");
			mes("That will protect you. It will also leave the pattern incomplete. Sometimes survival takes the place where understanding should have stood.");
			break;
		}
		next();
	}

	if (questprogress(20155) == 1 && questprogress(20155, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20155);
		callfunc("DM_PartyExp", 300000, 145000);
		callfunc("DM_GivePartyZeny", 25000);
		.@paid = 1;
		mes("[Eadric]");
		mes("The vigil held until dawn. At the weakest ward, the feather you found was burned along its spine. Something laughed where the prayer should have answered.");
		next();
		switch (select("Pray the full rite at each fraying ward.", "Scout the line and reinforce the worst breaks.", "Bait a weak section to draw the threat out.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc11_vigil_prayed", 1);
			mes("[Eadric]");
			mes("Then Bjorn's flock loses its opening. A held vigil is not dramatic. It is only the reason someone eats breakfast tomorrow.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc11_vigil_scouted", 1);
			mes("[Eadric]");
			mes("Useful intelligence, partial strength. The ward-line knows you watched it, even if you did not mend every tear.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc11_vigil_baited", 1);
			mes("[Eadric]");
			mes("You learned quickly. You also thinned the exact wall I asked you to hold. The next descent may remember the invitation.");
			break;
		}
		next();
	}

	if (questprogress(20153) == 2 && questprogress(20154) == 2 && questprogress(20155) == 2 && questprogress(20152) == 0) {
		mes("[Eadric]");
		mes("The reports agree. This is not a local wrath. It is a divine system failing.");
		next();
		mes("[Eadric]");
		mes("One Valkyrie is not attacking. Sigrun is dying in a side chapel, lucid enough to try speaking. If you can kneel instead of strike, she may still tell us what she hears.");
		callfunc("DM_InstanceQuestStart", 20152);
		next();
		mes("^3355FF[Quest started: The Dying Valkyrie's Word]^000000");
		mes("^3355FFApproach Sigrun's chapel without weapons raised. Her last lucidity will not last long.^000000");
		close();
	}

	if (questprogress(20152) == 1) {
		mes("[Eadric]");
		mes("Sigrun waits in the side chapel. Ask what you must, but remember that every answer costs her the strength to remain herself.");
		close();
	}

	if (questprogress(20152) == 2 && !dm_arc11_bjorn_subdued && !dm_arc11_bjorn_forced && !dm_arc11_bjorn_exploited) {
		mes("[Eadric]");
		mes("Sigrun is gone. Bjorn's flock blocks the temple steps now. He wants the wrath to mean his family's deaths had purpose.");
		next();
		mes("[Eadric]");
		mes("Reach the grief if you can. Move him if you must. But do not confuse his doctrine for courage.");
		close();
	}

	if (questprogress(20152) == 2 && (dm_arc11_bjorn_subdued || dm_arc11_bjorn_forced || dm_arc11_bjorn_exploited)) {
		mes("[Eadric]");
		mes("The temple steps are clear. Abyss Lake opens below the failing wards. Randgris waits in the deepest hall.");
		close();
	}

	if (.@paid) {
		mes("[Eadric]");
		mes("Finish the other reports. The airship, the scales, and the vigil are three ways of asking the same question.");
		close();
	}

	mes("[Eadric]");
	if (questprogress(20153) == 1 && questprogress(20154) == 1 && questprogress(20155) == 1)
		mes("The muster, the scale turn-in, and the temple vigil are all still open. Hugel needs evidence before it can survive doctrine.");
	else if (questprogress(20153) == 1)
		mes("Hugel Airship Muster is still open. Clear the Manuk fringe and bring back the manifest.");
	else if (questprogress(20154) == 1)
		mes("Dragon Scale Turn-In is still open. Abyss Lake's large dragons carry the burn pattern.");
	else if (questprogress(20155) == 1)
		mes("Temple Edge Vigil is still open. The ward-line frays near the El Dicastes outskirts.");
	else
		mes("The reports are ready. Sigrun is the next door.");
	close();
}

// ── Valkyrie Sigrun — The Dying Witness ───────────────────────────────────
hugel,104,96,3	script	Valkyrie Sigrun#dm11	4_F_01,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20152) == 0) {
		mes("^3355FF*The side chapel is cold. Something luminous breathes behind the closed screen, too weak to rise and too sacred to ignore.*^000000");
		close();
	}
	if (questprogress(20152) == 2) {
		mes("^3355FF*Sigrun's light is gone. A broken sword-shard rests where her hand fell, humming faintly when the unseen laughter draws near.*^000000");
		close();
	}

	mes("^3355FF*The Valkyrie lies against the altar steps. Her wings are dimming feather by feather.*^000000");
	next();
	mes("[Sigrun]");
	mes("Not... wrath. Noise. A command-road filled with another will.");
	next();
	mes("[Sigrun]");
	mes("It laughs where the All-Father's silence should be. It wants the hall to mistake falling for judgment.");
	next();
	switch (select("Ask her about the Voice.", "Comfort her and let the last word be peace.")) {
	case 1:
		callfunc("DM_Decide", "arc11.sigrun", "voice");
		mes("[Sigrun]");
		mes("A Voice in the order. Not demon. Not mortal. It wears certainty like a mask and laughs when thrones become doors.");
		next();
		mes("^3355FF*Sigrun reaches for a name and fails. The broken sword-shard remains warm for one breath, then cold.*^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc11.sigrun", "comfort");
		mes("[Sigrun]");
		mes("Then tell the priest... we did not mean to turn our blades downward.");
		next();
		mes("^3355FF*She dies before the next question. The broken sword-shard slides free of her hand anyway.*^000000");
		break;
	}
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20152, 1;
	end;
}

// ── Zealot Bjorn — The Man Who Needed Wrath to Mean Something ──────────────
hugel,110,100,5	script	Zealot Bjorn#dm	4_M_02,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc11_started) {
		mes("[Zealot Bjorn]");
		mes("Midgard sleeps. The reckoning approaches and no one watches the horizon.");
		close();
	}
	if (questprogress(20152) != 2) {
		mes("[Zealot Bjorn]");
		mes("The temple is not ready for tourists. Come back when you have heard what heaven sounds like as it dies.");
		close();
	}

	if (dm_arc11_bjorn_subdued || dm_arc11_bjorn_forced || dm_arc11_bjorn_exploited) {
		mes("[Zealot Bjorn]");
		if (dm_arc11_bjorn_joined)
			mes("My family did not die for a reason. That sentence is a wound. I will still walk with you if it keeps another family from needing one.");
		else if (dm_arc11_bjorn_exploited)
			mes("You know the weak wards now. Walk through them if you must. I will bring those who still wish to be found worthy.");
		else
			mes("The steps are clear. That is what you wanted.");
		close();
	}

	mes("[Zealot Bjorn]");
	mes("We deserve their wrath. Every Valkyrie turning her blade on a Midgard citizen is doing what heaven intended.");
	next();
	mes("[Bjorn]");
	mes("My family died under that light. If it was not judgment, then it was nothing. Do you understand what you are asking me to believe?");
	next();
	switch (select("They died for no reason. That is the horror, and it was not your fault.", "Stand aside or be moved.", "Show us the weak wards, then.")) {
	case 1:
		callfunc("DM_Decide", "arc11.bjorn", "subdued");
		mes("[Bjorn]");
		mes("No reason.");
		next();
		mes("[Bjorn]");
		mes("I have been feeding people to the steps because I could not survive that sentence.");
		next();
		mes("^3355FF*Bjorn lowers his crook and tells his flock to go home. Some obey immediately. Some collapse first.*^000000");
		mes("^3355FF[Encounter: Randgris's court will be restrained.]^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc11.bjorn", "forced");
		mes("[Bjorn]");
		mes("Then move me. Let the faithful see whether heroes strike more gently than heaven.");
		next();
		mes("^3355FF*The temple steps clear after an ugly, joyless fight with people who wanted martyrdom more than victory.*^000000");
		mes("^3355FF[Encounter: Randgris will fight with full court pressure.]^000000");
		break;
	case 3:
		callfunc("DM_Decide", "arc11.bjorn", "exploited");
		mes("[Bjorn]");
		mes("So you will enter to be judged. Good. I know where the wards are thin.");
		next();
		mes("^3355FF*Bjorn marks the weak places. His flock follows, believing the party finally understands him.*^000000");
		mes("^3355FF[Encounter: Detardeurus may join the pressure in the deep.]^000000");
		break;
	}
	next();
	mes("^3355FF[The temple steps are open. Descend through Abyss Lake and face Randgris.]^000000");
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20151, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

// ── Valkyrie Hall Set-piece ─────────────────────────────────────────────────
abyss_03,150,150,0	script	Valkyrie Hall#dm	HIDDEN_NPC,{
	end;

OnRandgrisDead:
	killmonster strnpcinfo(NPC_MAP), "Valkyrie Hall#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20151);
	callfunc("DM_InstanceSetFlag", "dm_arc11_randgris_killed", 1);
	callfunc("DM_PartyExp", 1200000, 700000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1199);
	if (dm_arc11_bjorn_joined)
		instance_announce(instance_id(), "^3355FF[DM] Randgris falls. Bjorn catches her as she goes. She says something to him no one else hears. He weeps.^000000", bc_all);
	else if (dm_arc11_sigrun_voice)
		instance_announce(instance_id(), "^3355FF[DM] Valkyrie Randgris falls. The laughter Sigrun warned you about rings once through the hall, then vanishes upward.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] Valkyrie Randgris falls. Heaven's blade is still. No answer comes from above.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The divine signal is unblocked. But Odin remains silent. The New World's wound is the next thread.^000000", bc_all);
	end;

OnAddDead:
	end;
