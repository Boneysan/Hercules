// ── Shared Arc 11 helpers ──────────────────────────────────────────────────

function	script	DM_Arc11AddTrial	{
	.@bit = getarg(0);
	if (!(dm_arc11_trial_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc11_trial_mask", dm_arc11_trial_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc11_trial_mask);
	callfunc("DM_DMJStoryObjective", 20151, 2, "Explore", .@n, 2, (.@n >= 2), getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc11_trial_" + .@bit))
		callfunc("DM_PartyExp", 180000, 87000);
	if (.@n >= 2) {
		if (questprogress(20155) == 0)
			callfunc("DM_InstanceQuestStart", 20155);
		callfunc("DM_PartyMes", "^3355FF[Both trials done. Bjorn has seen all of it. Speak to him at the temple gate.]^000000");
	} else {
		callfunc("DM_PartyMes", "^3355FF[One trial done. The other is on the other side of the village.]^000000");
	}
	return .@n;
}

// mode: 1 joins as an accountable ally, 2 independent. Final.
function	script	DM_Arc11Commit	{
	.@mode = getarg(0);
	if (dm_arc11_bjorn_outcome)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc11_bjorn_outcome", .@mode);
	if (.@mode == 1) {
		callfunc("DM_InstanceClearFlag", "dm_arc11_bjorn_subdued");
		callfunc("DM_InstanceSetFlag", "dm_arc11_bjorn_joined", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc11_civilian_pact", 1);
	} else {
		callfunc("DM_InstanceClearFlag", "dm_arc11_bjorn_joined");
		callfunc("DM_InstanceSetFlag", "dm_arc11_bjorn_subdued", 1);
	}
	callfunc("DM_InstanceQuestComplete", 20155);
	callfunc("DM_DMJStoryObjective", 20155, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc11_resolution"))
		callfunc("DM_PartyExp", 240000, 116000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1105);
	return 1;
}

		callfunc("DM_PartyMes", "Rachel's Prelate Karsh. I heard. The church's failures are not unique to Arunafeltz. The same rot shows up in every institution that mistakes authority for truth. We are not immune here.");
		next();

		callfunc("DM_PartyMes", "[Eadric]");
		callfunc("DM_PartyMes", "I am told the truth about Rachel's rift was kept quiet, and that you were part of that. I will not ask how it felt. I will only say that Bjorn would have had a great deal to say about it.");
		next();
	}
	if (dm_arc08_manfred_killed) {
		callfunc("DM_PartyMes", "[Eadric]");
		callfunc("DM_PartyMes", "The last bishop of Glast Heim. I knew his name from the accounts. His faith was absolute and it died alone in the ruins with him. I keep thinking about that.");
		next();
	}
	if (dm_arc08_manfred_spared) {
		callfunc("DM_PartyMes", "[Eadric]");
		callfunc("DM_PartyMes", "Manfred of Glast Heim is still alive? That is... I had assumed. Good. Good.");
		next();
	}
	if (dm_arc08_manfred_confessed) {
		callfunc("DM_PartyMes", "[Eadric]");
		callfunc("DM_PartyMes", "I am told Manfred admitted that he altered the evacuation order himself, and that you heard him say it. A man that certain, saying that aloud. I find I do not know what to do with it, and I am glad someone does.");

		switch (select("What is their original function?", "What happened to Odin's signal?", "Who is Bjorn?", "We are ready. Where do we start?")) {
		case 1:
			callfunc("DM_PartyMes", "[Eadric]");
			callfunc("DM_PartyMes", "Punishing the faithless. Before the age of temples and prayers, Valkyries were instruments of divine judgment. They chose the worthy among the dead. Everyone else was... judged.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Eadric]");
			callfunc("DM_PartyMes", "The cascade. It is not destroying the connection, it is filling it with noise. Odin's signal exists, but the Valkyries cannot distinguish it from the cascade's frequency. They have been operating blind for weeks.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Eadric]");
			callfunc("DM_PartyMes", "One of ours, or was. He decided the Valkyries are right, that we deserve this. He is standing at the temple gate and he will watch whatever you do. He has been writing it all down.");
			break;
		case 4:
			.@ready = 1;
			break;
		}
		next();
	}

	callfunc("DM_PartyMes", "Two trials, at the ward sites. At the keeper's shrine, the keeper is exposed and the Gryphon packs have turned on it. At the signal obelisk, the signal is damaged, and someone has to read it honestly. Bjorn will be watching both, and he will argue with you in between.");

		callfunc("DM_PartyMes", "Bjorn has made his answer. The gate is open.");

		callfunc("DM_PartyMes", "^3355FF[Optional: Dragon Pack Dispersal - " + callfunc("DM_HuntBrief", 20154) + "]^000000");
	}

	if (questprogress(20153) == 1 && callfunc("DM_HuntCollect", 20153)) {

		callfunc("DM_PartyExp", 100000, 48000);

	if (questprogress(20154) == 1 && callfunc("DM_HuntCollect", 20154)) {
		callfunc("DM_InstanceQuestComplete", 20154);
		callfunc("DM_PartyExp", 100000, 48000);
		callfunc("DM_GivePartyZeny", 25000);
		callfunc("DM_PartyMes", "[Eadric]");

// ── Zealot Bjorn — The Man Who Was Right for the Wrong Reasons ─────────────
hugel,110,100,5	script	Zealot Bjorn#dm	4_M_02,{
	if (!callfunc("DM_SessionGate")) close;
	if (!dm_arc11_started) {
		callfunc("DM_PartyMes", "[Zealot Bjorn]");
		callfunc("DM_PartyMes", "Midgard sleeps. The reckoning approaches and no one watches the horizon.");
		close();
	}
	if (dm_arc11_bjorn_outcome) {
		callfunc("DM_PartyMes", "[Zealot Bjorn]");
		if (dm_arc11_bjorn_joined)
			callfunc("DM_PartyMes", "I walk with you now. Not because the Valkyries were wrong about what we became. But because I think we can still become something else.");
		else
			callfunc("DM_PartyMes", "I will not fight the Valkyries. I will not fight you. I will wait here and see which of us was right.");
		close();
	}

	callfunc("DM_PartyMes", "[Zealot Bjorn]");
	callfunc("DM_PartyMes", "We deserve their wrath. Every Valkyrie turning her blade on a Midgard citizen is doing what heaven intended. We built cities instead of faith. We built commerce instead of virtue.");
	next();
	callfunc("DM_PartyMes", "[Bjorn]");
	callfunc("DM_PartyMes", "I have been saying this for fifteen years. Now the sky confirms it.");
	next();
	// Between stages he argues from what he has seen.
	if (dm_arc11_trial_mask & 1) {
		callfunc("DM_PartyMes", "[Bjorn]");
		callfunc("DM_PartyMes", "You defended the keeper. Fine. Anyone with a blade would. The question is not whether you can fight. The question is whether the thing you defended deserved it.");
		next();
	}
	if (dm_arc11_trial_mask & 2) {
		callfunc("DM_PartyMes", "[Bjorn]");
		if (dm_arc11_signal_partial)
			callfunc("DM_PartyMes", "You read the signal. Roughly. I read it too. It could be Odin's, or it could be noise. You cannot tell me which, and I do not believe you.");
		else
			callfunc("DM_PartyMes", "You read the signal. It is not Odin's voice. It is a frequency that has been laid over it. I have no argument for that, and I resent that I have none.");
		next();
	}
	if ((dm_arc11_trial_mask & 3) != 3) {
		callfunc("DM_PartyMes", "[Bjorn]");
		callfunc("DM_PartyMes", "I have not seen enough to be moved. Do both trials, and I will be at this gate when you come back.");
		close();
	}

	.@done = 0;
	while (!.@done) {
		switch (select("The Valkyries are not judging us. They are malfunctioning.", "You may be right about Midgard's failures. That is not what this is.", "What would you need to stand with us?", "We are ready to decide.", "We need to discuss this.")) {
		case 1:
			callfunc("DM_PartyMes", "[Bjorn]");
			callfunc("DM_PartyMes", "Malfunctioning. A divine instrument of judgment, and you call it broken. ...What if you are right? What if I spent fifteen years calling the instrument what I wished to hear?");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Bjorn]");
			callfunc("DM_PartyMes", "You acknowledge that Midgard is faithless. Then what is your alternative? If the diagnosis is correct and the medicine is wrong, what is the medicine?");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Bjorn]");
			callfunc("DM_PartyMes", "A promise. Out loud, in front of Eadric. If I go into that temple with you, no civilian is the price of it. Not one. If you will not say that, I will stand here and watch, and I will not hinder you.");
			break;
		case 4:
			.@done = 1;
			break;
		case 5:
			callfunc("DM_PartyMes", "[Bjorn]");
			callfunc("DM_PartyMes", "I have been waiting fifteen years. I can wait for you to disagree with each other.");
			close();
		}
		if (!.@done)
			next();
	}
	next();
	switch (select("Promise that no civilian is the price. Ask Bjorn to stand with us.", "Go in without him. Ask him to step aside.", "Not yet. We want to talk it over.")) {
	case 1:
		.@mode = 1;
		break;
	case 2:
		.@mode = 2;
		break;
	case 3:
		callfunc("DM_PartyMes", "[Bjorn]");
		callfunc("DM_PartyMes", "Of course.");
		close();
	}
	next();
	callfunc("DM_PartyMes", "[Bjorn]");
	if (.@mode == 1)
		callfunc("DM_PartyMes", "Say it in front of Eadric, and I will take the crook and go ahead of you. Confirm, and you have promised.");
	else
		callfunc("DM_PartyMes", "You go without me. I step aside and I do not stop you. I will watch which of us was right. Confirm, and I stand down.");
	next();
	if (select("Confirm.", "Wait.") != 1) {
		callfunc("DM_PartyMes", "[Bjorn]");
		callfunc("DM_PartyMes", "Better to wait than to promise lightly.");
		close();
	}
	if (!callfunc("DM_Arc11Commit", .@mode))
		close();
	callfunc("DM_PartyMes", "[Bjorn]");
	if (.@mode == 1) {
		callfunc("DM_PartyMes", "Then I will go ahead of you with this crook. If the faith has any weight left in this place, I will use it. Do not make me regret the promise.");
		callfunc("DM_PartyMes", "^3355FF[Encounter: Bjorn works the Valkyrie ward himself and disrupts a reinforcement wave.]^000000");
	} else {
		callfunc("DM_PartyMes", "I will not block the gate. I will watch from here. You will operate the ward yourselves, and it will take longer.");
		callfunc("DM_PartyMes", "^3355FF[Encounter: the party works the Valkyrie ward through three steps.]^000000");
	}
	next();
	callfunc("DM_PartyMes", "^3355FF[The Zealot's Testimony resolved. The temple gate is open.]^000000");
	close();

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20155, 1;
	end;
}

// ── Trial 1: the exposed keeper ────────────────────────────────────────────
hugel,96,104,4	script	Keeper's Shrine#dm	4_BOARD3,{
	if (!callfunc("DM_SessionGate")) close;
	if (!dm_arc11_started) {
		mes("The party has not brought you into this chapter yet. Join their party, or log in again once someone who has it is online.");
		close;
	}
	if (dm_arc11_trial_mask & 1) {
		callfunc("DM_PartyMes", "^3355FF*The keeper is on her feet again, tending the lamp. She nods once.*^000000");
		close();
	}
	if ('boss_up) {
		callfunc("DM_PartyMes", "^3355FF*The packs are still on the keeper. Hold the shrine.*^000000");
		close();
	}
	if (!'guard_cleared) {
		callfunc("DM_PartyMes", "^3355FF*The keeper kneels at the shrine, unarmed. Bjorn stands at the edge of the lamplight, watching. The Gryphon packs break from the treeline.*^000000");
		.@map$ = strnpcinfo(NPC_MAP);
		getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
		.@self$ = strnpcinfo(NPC_NAME);
		'boss_up = 1;
		callfunc("DM_EncStart", "arc11_keeper", getcharid(CHAR_ID_PARTY), .@map$, .@self$ + "::OnGuardDead", .@self$ + "::OnGuardDead", .@self$);
		callfunc("DM_EncMonster", .@map$, .@x, .@y, "Frightened Gryphon", GRYPHON, .@self$ + "::OnGuardDead", 68, 100, 3, 3);
		close();
	}
	callfunc("DM_PartyMes", "^3355FF*The last Gryphon wheels away. The keeper lifts her head. She is shaking and she is unhurt.*^000000");
	callfunc("DM_Arc11AddTrial", 1);
	close();

OnGuardDead:
	.@left = mobcount(strnpcinfo(NPC_MAP), strnpcinfo(NPC_NAME)+"::OnGuardDead");
	if (.@left > 0)
		end;
	'boss_up = 0;
	'guard_cleared = 1;
	callfunc("DM_EncCleanup", "arc11_keeper");
	end;

OnInit:
	callfunc("DM_AssertWalk");
	end;
}
