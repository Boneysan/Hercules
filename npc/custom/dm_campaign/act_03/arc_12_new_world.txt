// mode: 1 joint supervision, 2 requisition. Final.
function	script	DM_Arc12Commit	{
	.@mode = getarg(0);
	if (dm_arc12_survey_agreement)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc12_survey_agreement", .@mode);
	if (.@mode == 1) {
		callfunc("DM_InstanceClearFlag", "dm_arc12_vance_exposed");
		callfunc("DM_InstanceSetFlag", "dm_arc12_vance_helped", 1);
	} else {
		callfunc("DM_InstanceClearFlag", "dm_arc12_vance_helped");
		callfunc("DM_InstanceSetFlag", "dm_arc12_vance_exposed", 1);
	}
	callfunc("DM_InstanceQuestComplete", 20165);
	callfunc("DM_DMJStoryObjective", 20165, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc12_resolution"))
		callfunc("DM_PartyExp", 280000, 136000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1205);
	return 1;
}

// One stabilized anchor. Bit 1, 2 or 4. Pays once each; at all three, the
// vulnerability window is armed for the encounter.
function	script	DM_Arc12AnchorHeld	{
	.@bit = getarg(0);
	if (!(dm_arc12_anchor_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc12_anchor_mask", dm_arc12_anchor_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc12_anchor_mask);
	callfunc("DM_DMJStoryObjective", 20161, 3, "Interact", .@n, 3, (.@n >= 3), getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc12_anchor_" + .@bit))
		callfunc("DM_PartyExp", 80000, 39000);
	if (.@n >= 3)
		callfunc("DM_PartyMes", "^3355FF[All three anchors hold. Naght Sieger will be vulnerable when the fight begins.]^000000");
	else
		callfunc("DM_PartyMes", "^3355FF[Anchor holds. " + .@n + " of 3.]^000000");
	return .@n;
}

	if ($@dm_arc12_naght_gid <= 0)
		end;

	end;
