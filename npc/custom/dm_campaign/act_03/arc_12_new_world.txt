// ── Envoy Aelith — Voice of the New World ──────────────────────────────────
mid_camp,100,100,5	script	Envoy Aelith#dm	4_F_01,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20161) == 2) {
		mes("[Envoy Aelith]");
		if (dm_new_world_alliance)
			mes("The wound is stitched, for now. You proved humans can cross a threshold without making conquest the first language spoken there.");
		else if (dm_arc12_vance_occupation)
			mes("The Shadow is gone. The wound remains politically useful to people who should know better. That may be the deeper danger.");
		else
			mes("The wound is quieter. I do not yet know whether your people came as healers or merely as frightened soldiers who found a larger battlefield.");
		close();
	}

	if (!dm_arc12_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20161, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20161, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20163, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20164, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20165, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("[Envoy Aelith]");
	mes("You are not soldiers. Good. The soldiers that came before you are part of why some villages now hear the word human and reach for weapons.");
	next();
	if (dm_arc09_gloom_killed || dm_arc09_karsh_exposed) {
		mes("[Aelith]");
		mes("The bleed under Rachel was one mouth of this wound. You closed a symptom. Now you stand on the far side of the sickness.");
		next();
	}
	if (dm_arc11_randgris_killed) {
		mes("[Aelith]");
		mes("Your Valkyrie fell because a divine road filled with noise. Here the roads between worlds are dissolving outright.");
		next();
	}
	mes("[Aelith]");
	mes("The New World is not empty. It has nations, borders, children, debts, and ordinary lives that were not waiting to become your frontier.");
	next();
	switch (select("Teach us how to help.", "Point us at what holds the rift open.", "This is Rune-Midgard's expedition. Stand aside.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_new_world_alliance", 1);
		mes("[Aelith]");
		mes("Then listen before you act. If you keep doing that when it costs you, my people may believe you came to close a wound, not widen it.");
		break;
	case 2:
		mes("[Aelith]");
		mes("Practical. I will help you reach the anchor. Whether the New World helps you in return depends on what you do before you get there.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc12_vance_occupation", 1);
		mes("[Aelith]");
		mes("Then you have already answered one of my questions. I will still tell you where the wound is, because it will kill us both.");
		break;
	}
	next();
	mes("[Aelith]");
	mes("Three small tests first. The Manuk supply manifests are unequal. The Splendide field is not yours to clear like a quarry. The Dicastes border exists even when your paperwork pretends it does not.");
	callfunc("DM_InstanceSetFlag", "dm_arc12_started", 1);
	callfunc("DM_InstanceQuestStart", 20161);
	callfunc("DM_InstanceQuestStart", 20163);
	callfunc("DM_InstanceQuestStart", 20164);
	callfunc("DM_InstanceQuestStart", 20165);
	next();
	mes("^3355FF[Quest started: The Wound With a Keeper]^000000");
	mes("^3355FFManuk Supply Line - clear the Manuk route and decide whose cart receives protection first.^000000");
	mes("^3355FFCornus Mercy Run - handle feral field creatures with the Sapha watching.^000000");
	mes("^3355FFDicastes Border Papers - clear the outskirts and process the crossing writs honestly or otherwise.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20163) == 1 && questprogress(20163, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20163);
		callfunc("DM_PartyExp", 340000, 165000);
		callfunc("DM_GivePartyZeny", 28000);
		.@paid = 1;
		mes("[Aelith]");
		mes("The Manuk route is open. Two carts left the yard today: one full for the barracks, one thin for the Sapha village. Which one did you guard first?");
		next();
		switch (select("The Sapha village cart, every run.", "Sapha and expedition on an even rotation.", "We skimmed a private stockpile from both manifests.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc12_supply_sapha", 1);
			callfunc("DM_InstanceSetFlag", "dm_new_world_alliance", 1);
			mes("[Aelith]");
			mes("Then you noticed what no one asked you to notice. The Blue Supply Knot reads as a debt honored.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc12_supply_even", 1);
			mes("[Aelith]");
			mes("Fair by your ledgers, acceptable by ours. Not a declaration, but not an insult.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc12_supply_skimmed", 1);
			mes("[Aelith]");
			mes("Efficient. The knot reads unpaid. It is possible to steal from two sides and still call it balance, but only in human mathematics.");
			break;
		}
		next();
	}

	if (questprogress(20164) == 1 && questprogress(20164, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20164);
		callfunc("DM_PartyExp", 360000, 175000);
		callfunc("DM_GivePartyZeny", 30000);
		.@paid = 1;
		mes("[Aelith]");
		mes("The Splendide field is quiet. A Sapha herder watched your work from the grass line. What did she see?");
		next();
		switch (select("We killed only what was actively hostile.", "We drove threats off the herding grounds.", "We cleared the field completely and brought proof.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc12_cornus_spared", 1);
			callfunc("DM_InstanceSetFlag", "dm_new_world_alliance", 1);
			mes("[Aelith]");
			mes("Then she will give you a White Horn Splinter, shed rather than harvested. That distinction matters here.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc12_cornus_driven", 1);
			mes("[Aelith]");
			mes("A field kept safe without pretending every frightened creature is a monster. Good enough work earns good enough trust.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc12_cornus_slaughtered", 1);
			callfunc("DM_InstanceSetFlag", "dm_new_world_alliance", 0);
			dm_catechism_tally += 1;
			mes("[Aelith]");
			mes("Proof of restraint cannot be stacked in corpses. You have proved something, but not the thing you meant.");
			break;
		}
		next();
	}

	if (questprogress(20165) == 1 && questprogress(20165, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20165);
		callfunc("DM_PartyExp", 360000, 175000);
		callfunc("DM_GivePartyZeny", 30000);
		.@paid = 1;
		mes("[Aelith]");
		mes("The Dicastes crossing is clear enough to argue at again. That is what borders are, usually: enough safety for paperwork to become the danger.");
		next();
		switch (select("Process every writ by its real terms.", "Forge access for urgent cases.", "Design a process favoring expedition access.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc12_border_honest", 1);
			callfunc("DM_InstanceSetFlag", "dm_new_world_alliance", 1);
			mes("[Aelith]");
			mes("Slow and honest. The people crossing know what they agreed to. That is rarer than your officers think.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc12_border_forged", 1);
			mes("[Aelith]");
			mes("Merciful lawbreaking is still lawbreaking. Several families crossed who would not have. Several officials now have reason to distrust the table.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc12_border_expedition", 1);
			callfunc("DM_InstanceSetFlag", "dm_new_world_alliance", 0);
			mes("[Aelith]");
			mes("Efficient and clever. Also the beginning of occupation written in polite ink.");
			break;
		}
		next();
	}

	if (questprogress(20163) == 2 && questprogress(20164) == 2 && questprogress(20165) == 2 && questprogress(20162) == 0) {
		mes("[Aelith]");
		mes("Supply, mercy, border. You have shown the New World what humans do when watched. Now we address what humans did when no diplomat was present.");
		next();
		mes("[Aelith]");
		mes("A scouting party panicked in a Sapha border-village. A child died. Vance's command has named it a hostile ambush. The village has another name for it.");
		callfunc("DM_InstanceQuestStart", 20162);
		next();
		mes("^3355FF[Quest started: First Contact Done Wrong]^000000");
		mes("^3355FFSpeak to the Sapha elder and decide what justice costs when both sides must keep living afterward.^000000");
		close();
	}

	if (questprogress(20162) == 1) {
		mes("[Aelith]");
		mes("The elder is waiting near the command tents. Walk in with weapons down. If you cannot do that, you already know how the hearing ends.");
		close();
	}

	if (questprogress(20162) == 2 && !dm_arc12_vance_exposed && !dm_arc12_vance_helped && !dm_arc12_vance_occupation) {
		mes("[Aelith]");
		mes("First contact has its verdict. Vance is at the rift-anchor now, trying to decide whether the wound is a threat or a prize.");
		close();
	}

	if (questprogress(20162) == 2 && (dm_arc12_vance_exposed || dm_arc12_vance_helped || dm_arc12_vance_occupation)) {
		mes("[Aelith]");
		mes("The anchor perimeter is open. Nidhoggr's Shadow keeps the wound from closing. Vance wanted it open too. Make sure both keepers are answered.");
		close();
	}

	if (.@paid) {
		mes("[Aelith]");
		mes("Finish the other routes. First contact is not one conversation. It is who gets fed, spared, and translated before the formal apology.");
		close();
	}

	mes("[Aelith]");
	if (questprogress(20163) == 1 && questprogress(20164) == 1 && questprogress(20165) == 1)
		mes("The supply route, mercy run, and border papers are all still open.");
	else if (questprogress(20163) == 1)
		mes("Manuk Supply Line is still open. The carts do not guard themselves.");
	else if (questprogress(20164) == 1)
		mes("Cornus Mercy Run is still open. The Sapha are watching how you clear their field.");
	else if (questprogress(20165) == 1)
		mes("Dicastes Border Papers is still open. The crossing needs safety before it can negotiate.");
	else
		mes("The small answers are recorded. The dead child is the next question.");
	close();
}

// ── Sapha Elder — The Blood-Price Hearing ──────────────────────────────────
mid_camp,104,100,5	script	Sapha Elder#dm12	4_M_04,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20162) == 0) {
		mes("[Sapha Elder]");
		mes("You wear the posture of people who have not yet been told what they owe.");
		close();
	}
	if (questprogress(20162) == 2) {
		mes("[Sapha Elder]");
		if (dm_arc12_first_contact_order)
			mes("Your order is tidy. Our child remains dead inside it.");
		else
			mes("The carved bird is carried. That is not enough. It is something.");
		close();
	}

	mes("[Sapha Elder]");
	mes("A human scout saw a child running and called it an ambush. Your captain's paper says hostile native contact. My house says a name.");
	next();
	mes("^3355FF*The elder places a small carved bird in your hands. It is light enough to forget until everyone in the room looks at it.*^000000");
	next();
	switch (select("Expose the scouts' panic and Vance's cover-up.", "Negotiate a blood-price the village will accept.", "Back the report to keep the expedition united.")) {
	case 1:
		callfunc("DM_Decide", "arc12.first_contact", "truth");
		mes("[Sapha Elder]");
		mes("Truth is not payment. But a debt cannot be paid while hidden under another name.");
		next();
		mes("^3355FF*Aelith accepts the testimony. The expedition loses its clean story; the New World gains a reason to keep speaking.*^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc12.first_contact", "peace");
		mes("[Sapha Elder]");
		mes("Blood-price does not buy the child back. It tells the village the death entered your ledger as more than inconvenience.");
		next();
		mes("^3355FF*The guilty scouts are surrendered to New World justice. No one in the command tent thanks you.*^000000");
		break;
	case 3:
		callfunc("DM_Decide", "arc12.first_contact", "order");
		mes("[Sapha Elder]");
		mes("Then keep your order. We will keep the name.");
		next();
		mes("^3355FF*The carved bird remains in your hand, not accepted, not released.*^000000");
		break;
	}
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20162, 1;
	end;
}

// ── Captain Roderick Vance — The Wrong Expedition ──────────────────────────
mid_camp,110,100,5	script	Captain Vance#dm	4_M_03,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc12_started) {
		mes("[Captain Vance]");
		mes("This is a restricted survey zone. Move along.");
		close();
	}

	if (questprogress(20161) == 2) {
		mes("[Captain Vance]");
		if (dm_arc12_vance_helped)
			mes("I have filed an amended report. 'Dimensional breach sealed by allied action.' It sounds better than admitting I nearly tried to own the wound.");
		else if (dm_arc12_vance_occupation)
			mes("The Shadow is gone. The corridor remains strategically significant. History will decide whether that was wisdom.");
		else
			mes("Nidhoggr's Shadow is dead. I have filed a report. My superiors will dislike the honesty in it.");
		close();
	}

	if (questprogress(20162) != 2) {
		mes("[Captain Vance]");
		mes("I am not discussing the anchor while the first-contact hearing is still making everyone sentimental.");
		close();
	}

	if (dm_arc12_vance_exposed || dm_arc12_vance_helped || dm_arc12_vance_occupation) {
		mes("[Captain Vance]");
		mes("The decision is made. The anchor is waiting.");
		close();
	}

	mes("[Captain Vance]");
	mes("The rift is a strategic corridor. If the world is ending, Rune-Midgard has a duty to hold the high ground that remains.");
	next();
	switch (select("Holding it open is killing two worlds.", "Close it and you are still the hero who sealed the breach.", "What is Rune-Midgard's cut?")) {
	case 1:
		callfunc("DM_Decide", "arc12.vance", "exposed");
		mes("[Vance]");
		mes("One of those worlds is mine.");
		next();
		mes("[Vance]");
		mes("And that was the wrong answer. I heard it when I said it. Pulling back.");
		next();
		mes("^3355FF*Vance withdraws the expedition from the anchor perimeter. He does not forgive you for making him hear himself clearly.*^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc12.vance", "helped");
		mes("[Vance]");
		mes("The officer who sealed the breach. That report would travel well.");
		next();
		mes("[Vance]");
		mes("Fine. We close it. If there is glory in this disaster, I would rather it belong to survival than occupation.");
		next();
		mes("^3355FF*Vance redirects his troops. For once, the flag points toward the same place as mercy.*^000000");
		break;
	case 3:
		callfunc("DM_Decide", "arc12.vance", "occupation");
		mes("[Vance]");
		mes("Access, survey rights, and a defensible corridor when the old maps stop meaning anything.");
		next();
		mes("[Vance]");
		mes("Good. I was beginning to think I was the only patriot left in the camp.");
		next();
		mes("^3355FF*Aelith's people hear the bargain before the ink dries. Some verdicts do not need translation.*^000000");
		break;
	}
	next();
	mes("^3355FF[Vance resolved. Proceed to spl_fild01 and face Nidhoggr's Shadow.]^000000");
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20161, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

// ── Rift Anchor Set-piece ──────────────────────────────────────────────────
spl_fild01,150,150,0	script	Rift Anchor#dm	HIDDEN_NPC,{
	end;

OnRiftStart:
	stopnpctimer;
	'rift_map$ = strnpcinfo(NPC_MAP);
	'rift_party_id = $@dm_arc12_rift_party_id;
	'rift_damage = $@dm_arc12_rift_damage;
	'rift_status = $@dm_arc12_rift_status;
	'rift_status_ms = $@dm_arc12_rift_status_ms;
	if ('rift_party_id > 0) {
		mapannounce 'rift_map$, "[Rift Anchor] The membrane thins. Space pulls toward the wound.", bc_map, 0x66CCFF;
		initnpctimer;
	}
	end;

OnRiftStop:
	stopnpctimer;
	end;

OnTimer7000:
	mapannounce 'rift_map$, "[Rift Anchor] A crosswind from the Ash Vacuum tears through the anchor.", bc_map, 0x66CCFF;
	callfunc("DM_HazardArea", 'rift_map$, 150, 150, 7, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	end;

OnTimer14000:
	mapannounce 'rift_map$, "[Rift Anchor] The horizon folds inward and snaps back.", bc_map, 0x66CCFF;
	callfunc("DM_HazardArea", 'rift_map$, 150, 150, 7, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	end;

OnTimer21000:
	mapannounce 'rift_map$, "[Rift Anchor] Static crawls over the wound in the world.", bc_map, 0x66CCFF;
	callfunc("DM_HazardArea", 'rift_map$, 150, 150, 7, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	end;

OnTimer28000:
	mapannounce 'rift_map$, "[Rift Anchor] The rift exhales one last unstable pressure wave.", bc_map, 0x66CCFF;
	callfunc("DM_HazardArea", 'rift_map$, 150, 150, 7, 'rift_damage, 'rift_status, 'rift_status_ms, 'rift_party_id);
	stopnpctimer;
	end;

OnNidhoggrDead:
	stopnpctimer;
	callfunc("DM_InstanceQuestComplete", 20161);
	callfunc("DM_InstanceSetFlag", "dm_arc12_nidhoggr_killed", 1);
	callfunc("DM_InstanceSetFlag", "dm_arc12_naght_killed", 1);
	callfunc("DM_PartyExp", 1400000, 800000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1299);
	if (dm_new_world_alliance && dm_arc12_vance_helped)
		instance_announce(instance_id(), "^3355FF[DM] Nidhoggr's Shadow collapses. Aelith's people and Vance's soldiers stitch the anchor from opposite sides.^000000", bc_all);
	else if (dm_arc12_vance_occupation)
		instance_announce(instance_id(), "^3355FF[DM] Nidhoggr's Shadow collapses. The rift remains valuable, unstable, and claimed.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] Nidhoggr's Shadow collapses. The wound is cauterized, not healed.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The Ash Vacuum's front is held here. The demon lords on the Nameless Island are watching. They are coordinating.^000000", bc_all);
	end;
}
