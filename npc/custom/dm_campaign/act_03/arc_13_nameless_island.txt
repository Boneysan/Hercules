//= Seal Cascade DM tooling
//===== Description: =========================================
//= The Nameless Island's seals are failing. The demon lords
//= are not leaking through randomly — they are coordinating.
//= Father Quill has held the wards alone for years. The Broker
//= Carrion is offering a treaty, and the treaty is worth reading.
//=
//= REBUILT 2026-10-02 (handoff A3-13). Questions to Carrion never
//= change anything. The party inspects two ward failures and recovers
//= a treaty witness, reads three authored clauses (containment
//= boundary, civilian passage, who monitors breaches), then accepts,
//= accepts with a monitoring condition, or rejects. A treaty only
//= counts once a supervised containment demonstration is completed.
//=
//= Preparations (all three open the decision):
//=   dm_arc13_ward_mask   bits 1,2: ward failures A and B (a ward
//=       record from either reveals the clause-3 loophole)
//=   dm_arc13_witness     the treaty witness, recovered
//= Decision (dm_arc13_treaty_terms, final, equal base EXP):
//=   1 accept as written (knowingly accepts the monitoring risk)
//=   2 accept with Quill's wards monitoring (needs a ward record)
//=   3 reject -> Beelzebub's sanctum, existing encounter
//= Accepting is not completing it: the demonstration is. Only
//= dm_arc13_treaty_verified resolves the arc without the boss.
//= Derived for later arcs: carrion_bribed (a treaty was agreed, shown
//= to players as "treaty agreed"), coalition_deal_honored (verified),
//= beelzebub_killed (the boss was actually fought and killed).
//=
//= Quests (db/quest_db.conf):
//=   20171 The Council of the Drowned  (arc tracker / Beelzebub)
//=   20173 Abbey Bell Silencing  (optional salvage)
//=   20174 Sanctum Sweep         (optional salvage)
//=   20175 The Broker's Terms          (opens when preparations are done)
//=
//= Story flags (persistent character vars, see dm_flags.txt):
//=   dm_arc13_started, dm_arc13_ward_mask, dm_arc13_witness,
//=   dm_arc13_treaty_terms, dm_arc13_treaty_verified,
//=   dm_arc13_carrion_killed, dm_arc13_carrion_bribed,
//=   dm_arc13_coalition_deal_honored, dm_arc13_beelzebub_killed
//===========================================================

// ── Shared Arc 13 helpers ──────────────────────────────────────────────────

// One preparation done. Opens the Broker's Terms once all three are.
function	script	DM_Arc13Prep	{
	.@n = callfunc("DM_CountBits", dm_arc13_ward_mask) + (dm_arc13_witness ? 1 : 0);
	callfunc("DM_DMJStoryObjective", 20171, 2, "Explore", .@n, 3, (.@n >= 3), getcharid(CHAR_ID_PARTY));
	if (.@n >= 3) {
		if (callfunc("DM_ClaimGrant", "arc13_prep"))
			callfunc("DM_InstanceQuestStart", 20175);
		callfunc("DM_PartyMes", "^3355FF[Two ward failures inspected, witness recovered. Carrion's terms can be put to a decision.]^000000");
	} else {
		callfunc("DM_PartyMes", "^3355FF[" + .@n + " of 3: two ward failures, the treaty witness.]^000000");
	}
	return .@n;
}

// mode: 1 accept as written, 2 accept with monitoring, 3 reject. Final.
function	script	DM_Arc13Commit	{
	.@mode = getarg(0);
	if (dm_arc13_treaty_terms)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc13_treaty_terms", .@mode);
	if (.@mode == 3) {
		callfunc("DM_InstanceClearFlag", "dm_arc13_carrion_bribed");
	} else {
		callfunc("DM_InstanceClearFlag", "dm_arc13_carrion_killed");
		callfunc("DM_InstanceSetFlag", "dm_arc13_carrion_bribed", 1);
	}
	callfunc("DM_InstanceQuestComplete", 20175);
	callfunc("DM_DMJStoryObjective", 20175, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc13_resolution"))
		callfunc("DM_PartyExp", 200000, 97000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1305);
	return 1;
}

// The demonstration succeeded: the treaty is real, the boss is skipped, and
// the arc resolves at the same base reward as the fight.
function	script	DM_Arc13Verified	{
	if (dm_arc13_treaty_verified)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc13_treaty_verified", 1);
	callfunc("DM_InstanceSetFlag", "dm_arc13_coalition_deal_honored", 1);
	callfunc("DM_InstanceQuestComplete", 20171);
	if (callfunc("DM_ClaimGrant", "arc13_beelzebub"))
		callfunc("DM_PartyExp", 1600000, 900000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1399);
	callfunc("DM_MapStory", "The demonstration holds. Beelzebub withdraws from the Abbey. The treaty is in force, and so is every term in it.");
	return 1;
}

			callfunc("DM_PartyMes", "Beelzebub. The one in the Abbey sanctum. He is not the strongest of them. He is the one the others will follow.");

// ── The Broker Carrion — The Man With the Contract ─────────────────────────
nameless_i,110,100,5	script	The Broker Carrion#dm	4_M_05,{
	if (!callfunc("DM_SessionGate")) close;
	if (!dm_arc13_started) {
		callfunc("DM_PartyMes", "[The Broker Carrion]");
		callfunc("DM_PartyMes", "I am not taking visitors. Come back when Father Quill has told you what you are walking into.");
		close();
	}
	if (dm_arc13_treaty_terms) {
		callfunc("DM_PartyMes", "[The Broker Carrion]");
		switch (dm_arc13_treaty_terms) {
		case 1:
		case 2:
			if (dm_arc13_treaty_verified)
				callfunc("DM_PartyMes", "The demonstration held and the treaty is in force. I will keep the boundary. I would be grateful if you would remember that I did.");
			else
				callfunc("DM_PartyMes", "The terms are agreed. The demonstration is at the abbey gate whenever you are ready to supervise it. Until it holds, this is only a promise.");
			break;
		case 3:
			callfunc("DM_PartyMes", "You rejected it. That is a legitimate answer. I will be at the Abbey. Beelzebub will be too.");
			break;
		}
		close();
	}

	callfunc("DM_PartyMes", "[The Broker Carrion]");
	callfunc("DM_PartyMes", "The party Quill has been fretting about. I will be plain. The coalition wants the cascade to run its course. I would like to arrange for it not to cost this island.");
	next();
	callfunc("DM_PartyMes", "[Carrion]");
	callfunc("DM_PartyMes", "My terms are three. One: the coalition stays inside the Abbey gate and does not cross it. Two: the ferry keeps running, and no one on the island is harmed in transit. Three: breaches of either are monitored by my coalition, and I report them to you.");
	next();
	.@done = 0;
	while (!.@done) {
		switch (select("What is the catch?", "What happens to the cascade if we take it?", "Who monitors the boundary in clause three?", "We have heard the terms.", "We are done talking.")) {
		case 1:
			callfunc("DM_PartyMes", "[Carrion]");
			callfunc("DM_PartyMes", "That the coalition does not stop the cascade. It stops using this island as its way through. The cascade is somebody else's problem. I would call that an improvement.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Carrion]");
			callfunc("DM_PartyMes", "It continues. It did before I arrived. What changes is where the demon lords stand while it does. Where they stand matters to Quill, so I have bothered to ask.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Carrion]");
			if (dm_arc13_ward_mask)
				callfunc("DM_PartyMes", "...I see you have been to the wards. Yes. Clause three has a hole in it, and the hole is that the party who breaches would also be the party who reports it. That is how I wrote it. I am being open about it because you can see it anyway.");
			else
				callfunc("DM_PartyMes", "My coalition does, and reports to you. I am told that is a weak arrangement. I would not know. I wrote it.");
			break;
		case 4:
			.@done = 1;
			break;
		case 5:
			close();
		}
		if (!.@done)
			next();
	}
	next();
	if ((callfunc("DM_CountBits", dm_arc13_ward_mask) < 2) || !dm_arc13_witness) {
		callfunc("DM_PartyMes", "[Carrion]");
		callfunc("DM_PartyMes", "A treaty without a witness is a promise. And a promise from me, signed by a party that has not even looked at the wards, is a worse one. Inspect the two failed wards and recover the notary, then come back.");
		close();
	}
	callfunc("DM_PartyMes", "[Carrion]");
	callfunc("DM_PartyMes", "You have the witness and you have seen the wards. Then you can answer. Accept the treaty as written. Accept it with a monitoring condition of your own. Or reject it, and see Beelzebub.");
	next();
	switch (select("Accept the treaty as written.", "Accept it, with Quill's wards monitoring breaches.", "Reject the treaty.", "Not yet. We want to talk it over.")) {
	case 1:
		.@mode = 1;
		break;
	case 2:
		.@mode = 2;
		break;
	case 3:
		.@mode = 3;
		break;
	case 4:
		callfunc("DM_PartyMes", "[Carrion]");
		callfunc("DM_PartyMes", "Of course.");
		close();
	}
	next();
	callfunc("DM_PartyMes", "[Carrion]");
	if (.@mode == 1)
		callfunc("DM_PartyMes", "As written. The breach is monitored by those who might breach it, and you are agreeing to that knowingly. Confirm, and then supervise the demonstration.");
	else if (.@mode == 2)
		callfunc("DM_PartyMes", "With the wards watching, then. It costs me. I will agree to it if you will supervise the demonstration. Confirm, and I sign.");
	else
		callfunc("DM_PartyMes", "You will not have the treaty. I will take no offence. Confirm, and I leave.");
	next();
	if (select("Confirm.", "Wait.") != 1) {
		callfunc("DM_PartyMes", "[Carrion]");
		callfunc("DM_PartyMes", "Better to wait than to say yes to me in a hurry.");
		close();
	}
	if (!callfunc("DM_Arc13Commit", .@mode))
		close();
	callfunc("DM_PartyMes", "[Carrion]");
	if (.@mode == 3) {
		callfunc("DM_PartyMes", "Then there is no treaty. Beelzebub is at the Abbey, and the coalition will be at his back.");
		callfunc("DM_PartyMes", "^3355FF[The Broker's Terms resolved. Beelzebub is at the Abbey sanctum.]^000000");
	} else {
		callfunc("DM_PartyMes", "Signed, witnessed, and sealed. The demonstration is at the abbey gate, when you are ready. Only then does it bind. I will be watching it as closely as you.");
		callfunc("DM_PartyMes", "^3355FF[The treaty is agreed, not yet in force. Supervise the containment demonstration at the abbey gate.]^000000");
	}
	close();

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20175, 1;
	end;
}

// ── Ward failure A ─────────────────────────────────────────────────────────
nameless_i,95,104,4	script	Failed Ward West#dm	4_ENERGY_BLUE,{
	callsub(S_Ward, 1);
	end;
OnInit:
	callfunc("DM_AssertWalk");
	end;
OnGuardDead:
	.@left = mobcount(strnpcinfo(NPC_MAP), strnpcinfo(NPC_NAME)+"::OnGuardDead");
	if (.@left > 0)
		end;
	'boss_up = 0;
	'guard_cleared = 1;
	callfunc("DM_EncCleanup", "arc13_ward_west");
	end;
S_Ward:
	.@bit = getarg(0);
	if (!callfunc("DM_SessionGate")) close;
	if (!dm_arc13_started) {
		mes("The party has not brought you into this chapter yet. Join their party, or log in again once someone who has it is online.");
		close;
	}
	if (dm_arc13_ward_mask & .@bit) {
		callfunc("DM_PartyMes", "^3355FF*You have copied what this ward kept.*^000000");
		close();
	}
	if ('boss_up) {
		callfunc("DM_PartyMes", "^3355FF*The shades the ward was holding are still loose.*^000000");
		close();
	}
	callfunc("DM_PartyMes", "^3355FF*A ward stone, cracked clean through. The record it kept is still cut in the base.*^000000");
	next();
	if (select("Read the ward record.", "Leave it.") == 2) close();
	if (!'guard_cleared && callfunc("DM_CheckRoll", "arcana", 12) < 2 && !callfunc("DM_IsDM")) {
		callfunc("DM_PartyMes", "^3355FF*The crack widens. What the ward was holding shakes loose.*^000000");
		.@map$ = strnpcinfo(NPC_MAP);
		getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
		.@self$ = strnpcinfo(NPC_NAME);
		'boss_up = 1;
		callfunc("DM_EncStart", "arc13_ward_west", getcharid(CHAR_ID_PARTY), .@map$, .@self$ + "::OnGuardDead", .@self$ + "::OnGuardDead", .@self$);
		callfunc("DM_EncMonster", .@map$, .@x, .@y, "Loosed Shade", BANSHEE, .@self$ + "::OnGuardDead", 72, 100, 2, 3);
		close();
	}
	callfunc("DM_PartyMes", "^3355FF*The record comes clear. A ward only holds if someone other than the party it contains watches it. That is how it was built.*^000000");
	callfunc("DM_InstanceSetFlag", "dm_arc13_ward_mask", dm_arc13_ward_mask | .@bit);
	if (callfunc("DM_ClaimGrant", "arc13_ward_" + .@bit))
		callfunc("DM_PartyExp", 90000, 44000);
	callfunc("DM_Arc13Prep");
	close();
}

	if (!(dm_arc13_treaty_terms == 1 || dm_arc13_treaty_terms == 2)) {
		callfunc("DM_PartyMes", "^3355FF*The abbey gate. Nothing is happening here. There is no treaty to demonstrate.*^000000");
		close();
	}
	if (dm_arc13_treaty_verified) {
		callfunc("DM_PartyMes", "^3355FF*The gate holds. The coalition stays where the treaty puts it.*^000000");
		close();
	}
	if ('boss_up) {
		callfunc("DM_PartyMes", "^3355FF*The demonstration is under way. Contain the breach.*^000000");
		close();
	}
	callfunc("DM_PartyMes", "^3355FF*The abbey gate. Carrion stands on the far side with his coalition. The notary stands on this one. Both are waiting for you to start.*^000000");
	next();
	if (select("Begin the demonstration.", "Not yet.") == 2) close();
	if (dm_arc13_treaty_terms == 2)
		callfunc("DM_PartyMes", "^3355FF*Quill's wards flare along the wall. Carrion's side is watched. A single test breach slips through.*^000000");
	else
		callfunc("DM_PartyMes", "^3355FF*Nobody watches the breach but the coalition. The test breach is larger than the terms said it would be. Carrion does not look surprised.*^000000");
	.@map$ = strnpcinfo(NPC_MAP);
	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	.@self$ = strnpcinfo(NPC_NAME);
	'boss_up = 1;
	callfunc("DM_EncStart", "arc13_demo", getcharid(CHAR_ID_PARTY), .@map$, .@self$ + "::OnDemoDead", .@self$ + "::OnDemoDead", .@self$);
	callfunc("DM_EncMonster", .@map$, .@x, .@y, "Breaching Shade", BANSHEE, .@self$ + "::OnDemoDead", 72, 100, (dm_arc13_treaty_terms == 2 ? 2 : 3), 3);
	if (dm_arc13_treaty_terms == 1)
		callfunc("DM_EncMonster", .@map$, .@x, .@y, "Unwatched Necromancer", NECROMANCER, .@self$ + "::OnDemoDead", 72, 100, 1, 2);
	close();

OnDemoDead:
	.@left = mobcount(strnpcinfo(NPC_MAP), strnpcinfo(NPC_NAME)+"::OnDemoDead");
	if (.@left > 0)
		end;
	'boss_up = 0;
	callfunc("DM_EncCleanup", "arc13_demo");
	callfunc("DM_Arc13Verified");
	end;

OnInit:
	callfunc("DM_AssertWalk");
	end;
}
