// ── Father Quill the Doorkeeper ───────────────────────────────────────────
nameless_i,100,100,5	script	Father Quill#dm	4_M_04,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20171) == 2) {
		mes("[Father Quill]");
		if (dm_arc13_dead_freed)
			mes("The bell is cracked through. The chapel is quiet. I thought victory would sound louder.");
		else
			mes("The council is broken. The dead are quieter, but not free. I will keep saying their names until I cannot.");
		close();
	}

	if (!dm_arc13_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20171, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20171, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20172, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20173, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20174, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20175, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("[Father Quill]");
	mes("You found the island. That means something. Most people who find it were dead before they arrived.");
	next();
	if (dm_arc02_ancestor_helped) {
		mes("[Quill]");
		mes("Payon's dead remember hands that returned what was stolen. That kindness reached farther than you knew.");
		next();
	} else if (dm_arc02_grove_burned || dm_arc02_voss_spared) {
		mes("[Quill]");
		mes("Payon's dead arrived with questions. Some name you kindly. Some do not.");
		next();
	}
	if (dm_arc12_vance_helped || dm_new_world_alliance) {
		mes("[Quill]");
		mes("You have seen a wall between worlds fail. Here it fails in the direction of the dead.");
		next();
	}
	mes("[Quill]");
	mes("The dead will not rest because the boundary is failing. Worse, they are organized now. Beelzebub is not haunting the Abbey. He is hosting it.");
	next();
	switch (select("Tell us how to lay them down.", "Get us to what is organizing this.", "They're monsters. Point us at the problem.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc13_quill_revered", 1);
		mes("[Quill]");
		mes("Then I give you routes, rites, and names. It is slower. It is also the only work here worth doing.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc13_quill_pragmatic", 1);
		mes("[Quill]");
		mes("I can point you at the council. I will not abandon the door while you take the shorter road.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc13_quill_callous", 1);
		mes("[Quill]");
		mes("That is what the necromancers say too. Start with the bell anyway. Even monsters answer names.");
		break;
	}
	next();
	mes("[Quill]");
	mes("Three routes hold the island's edge: the bell rotation, the drowned coins, and the living roots. Clear them, and I can find the one dead name that came here for you.");
	callfunc("DM_InstanceSetFlag", "dm_arc13_started", 1);
	callfunc("DM_InstanceQuestStart", 20171);
	callfunc("DM_InstanceQuestStart", 20173);
	callfunc("DM_InstanceQuestStart", 20174);
	callfunc("DM_InstanceQuestStart", 20175);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1301);
	next();
	mes("^3355FF[Quest started: The Council of the Drowned]^000000");
	mes("^3355FFSupport routes started: Abbey Bell Rotation, Drowned Coin Tithe, Lasagna Root Wardens.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20173) == 1 && questprogress(20173, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20173);
		callfunc("DM_PartyExp", 380000, 185000);
		callfunc("DM_GivePartyZeny", 32000);
		.@paid = 1;
		mes("[Quill]");
		mes("The bell answered. So did the dead. Now decide what the sound was for.");
		next();
		switch (select("Ring it fully and face them.", "Dampen it and patrol steadily.", "Tune it until one name answers.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_full", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_patrol", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_named", 0);
			mes("[Quill]");
			mes("A true toll spent. The Abbey exhales all at once. I hope it was worth the bronze.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_patrol", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_full", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_named", 0);
			mes("[Quill]");
			mes("Slow, careful, merciful in the way chores can be merciful.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_named", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_full", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_bell_patrol", 0);
			mes("[Quill]");
			mes("You made the bell ask for a specific dead thing. Useful. Ugly. Sometimes both words fit.");
			break;
		}
		next();
	}

	if (questprogress(20174) == 1 && questprogress(20174, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20174);
		callfunc("DM_PartyExp", 390000, 190000);
		callfunc("DM_GivePartyZeny", 33000);
		.@paid = 1;
		mes("[Quill]");
		mes("The drowned coins are older than this chapel. Carrion wants them because old payments still move old doors.");
		next();
		switch (select("Pay the dead properly.", "Pay the ferrymen.", "Hoard the coins against Carrion.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_returned", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_ferrymen", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_hoarded", 0);
			mes("[Quill]");
			mes("The old rite takes time. Time is the only coin the dead still understand.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_ferrymen", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_returned", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_hoarded", 0);
			mes("[Quill]");
			mes("Practical crossings for practical souls. The ferrymen will keep moving.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_hoarded", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_himmelmez_foreshadow", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_returned", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_coins_ferrymen", 0);
			mes("[Quill]");
			mes("Then carry them carefully. Some debts grow colder when they are not paid.");
			break;
		}
		next();
	}

	if (questprogress(20175) == 1 && questprogress(20175, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20175);
		callfunc("DM_PartyExp", 400000, 195000);
		callfunc("DM_GivePartyZeny", 34000);
		.@paid = 1;
		mes("[Quill]");
		mes("The living root still holds. That is rare here. Rarer than courage.");
		next();
		switch (select("Protect the roots and leave them growing.", "Harvest only what the bounty board needs.", "Graft the roots into the Abbey wards.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_protected", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_boundary_stable", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_harvested", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_grafted", 0);
			mes("[Quill]");
			mes("You left the one good thing better than you found it. The island notices.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_harvested", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_boundary_stable", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_protected", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_grafted", 0);
			mes("[Quill]");
			mes("A compromise. I dislike how often survival looks like one.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_grafted", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_boundary_stable", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_protected", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc13_roots_harvested", 0);
			mes("[Quill]");
			mes("Effective triage. The Abbey will hold longer. Somewhere else will not.");
			break;
		}
		next();
	}

	if (questprogress(20173) == 2 && questprogress(20174) == 2 && questprogress(20175) == 2 && questprogress(20172) == 0) {
		mes("[Quill]");
		mes("The three routes are clear. I found the name that came here for you.");
		next();
		mes("[Quill]");
		mes("It is not enough to defeat Beelzebub. The dead have a memory of you now. Go to the grave-niche and answer it.");
		callfunc("DM_InstanceQuestStart", 20172);
		next();
		mes("^3355FF[Quest started: A Name You Knew]^000000");
		mes("^3355FFFind the Familiar Dead near the Abbey's outer grounds.^000000");
		close();
	}

	if (questprogress(20172) == 1) {
		mes("[Quill]");
		mes("The known dead is waiting. Do not bring a weapon first. Bring memory.");
		close();
	}

	if (questprogress(20172) == 2 && !dm_arc13_carrion_killed && !dm_arc13_carrion_bribed) {
		mes("[Quill]");
		mes("Carrion is at the inner gate. He has the council's minutes. If he offers you the apocalypse in tidy handwriting, read the footnotes.");
		close();
	}

	if (questprogress(20172) == 2 && (dm_arc13_carrion_killed || dm_arc13_carrion_bribed)) {
		mes("[Quill]");
		mes("The gate is open. Beelzebub waits in abbey03. Break the venue, not only the host.");
		close();
	}

	if (.@paid) {
		mes("[Quill]");
		mes("Finish all three routes. Then I can name the dead that has been waiting for you.");
		close();
	}

	mes("[Quill]");
	mes("The bell, the coins, and the roots are all still part of the same wound. Close what you can.");
	close();
}

// ── The Familiar Dead — A Name You Knew ───────────────────────────────────
nameless_i,105,96,3	script	The Familiar Dead#dm13	4_GHOST_STAND,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20172) == 0) {
		mes("^3355FF*A cold shape turns its head as if it almost knows you. Not yet.*^000000");
		close();
	}
	if (questprogress(20172) == 2) {
		mes("[The Familiar Dead]");
		if (dm_arc13_known_dead_reckoned)
			mes("Thank you for saying my name correctly.");
		else
			mes("You were always better at endings than at burials.");
		close();
	}

	mes("^3355FF*The dead thing is not a stranger. Quill never gives the name aloud; he lets recognition do the damage.*^000000");
	next();
	if (dm_arc02_ancestor_helped) {
		mes("[The Familiar Dead]");
		mes("You returned a comb once. Some of us remembered that kindness and followed it here.");
	} else if (dm_arc02_grove_burned || dm_catechism_tally > 0) {
		mes("[The Familiar Dead]");
		mes("You have left fast answers behind you. We are one of them.");
	} else {
		mes("[The Familiar Dead]");
		mes("I remember almost enough. Your face. Your hands. The thing you promised, or failed to promise.");
	}
	next();
	switch (select("Reckon honestly and lay them to rest.", "Put them down and keep moving.")) {
	case 1:
		callfunc("DM_Decide", "arc13.known_dead", "reckoned");
		mes("^3355FF*You do not fix the dead. You sit with them long enough that they can stop waiting.*^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc13.known_dead", "refused");
		mes("^3355FF*The shape breaks apart under force. Its face does not leave. It only moves farther ahead.*^000000");
		break;
	}
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20172, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

// ── The Broker Carrion — The Council's Concierge ──────────────────────────
nameless_i,110,100,5	script	The Broker Carrion#dm	4_M_05,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc13_started) {
		mes("[The Broker Carrion]");
		mes("Not yet. Come back when you have seen what Father Quill is protecting.");
		close();
	}
	if (questprogress(20172) != 2) {
		mes("[Broker Carrion]");
		mes("You still have a private appointment with the dead. I try not to interrupt grief. It makes contracts imprecise.");
		close();
	}
	if (questprogress(20171) == 2) {
		mes("[Broker Carrion]");
		if (dm_arc13_carrion_bribed)
			mes("A deal is a deal. The minutes were accurate. What you do with them will be the entertaining part.");
		else
			mes("The council is scattered. You understand that scattered is not the same as gone.");
		close();
	}
	if (dm_arc13_carrion_killed || dm_arc13_carrion_bribed) {
		mes("[Broker Carrion]");
		if (dm_arc13_carrion_bribed)
			mes("You have the seating plan. Beelzebub has the room. Try to keep both facts in mind.");
		else
			mes("...");
		close();
	}

	mes("[Broker Carrion]");
	mes("You made it past Quill and your own dead. Good. I represent the Demon Lords' Coalition as its least sentimental appointment keeper.");
	next();
	if (dm_arc02_voss_spared) {
		mes("[Carrion]");
		mes("And yes. If you are wondering whether Payon's Scholar Voss learned anything after you let him walk away, the answer is: scale.");
		next();
	}
	mes("[Carrion]");
	mes("Beelzebub chairs tonight's council. Satan Morroc has an empty seat. There is also a head-of-table mark no demon here recognizes. I dislike unknown clients.");
	next();
	switch (select("You broker for things that want everyone dead.", "Sell us the minutes.", "You've come a long way from Payon, Scholar.")) {
	case 1:
		callfunc("DM_Decide", "arc13.carrion", "killed");
		mes("[Carrion]");
		mes("Everyone will be dead. I am arranging the seating.");
		next();
		mes("^3355FF*Carrion dissolves into a hard black cloud. His bindings remain, but the inner gate opens.*^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc13.carrion", "bribed");
		mes("[Carrion]");
		mes("Safe passage off the island later. The minutes now. A practical trade, and the apocalypse has excellent handwriting.");
		next();
		mes("^3355FF[You obtain the Coalition's Seating Plan.]^000000");
		break;
	case 3:
		if (dm_arc02_voss_spared) {
			callfunc("DM_Decide", "arc13.carrion", "bribed");
			mes("[Carrion]");
			mes("Remembered! How rare. Then here is the whole agenda, for old research's sake. Do not pretend mercy was free.");
			next();
			mes("^3355FF[You obtain the Coalition's Seating Plan. Carrion is delighted to be remembered.]^000000");
		} else {
			callfunc("DM_Decide", "arc13.carrion", "killed");
			mes("[Carrion]");
			mes("Wrong ghost. But I admire the attempt to make this personal.");
			next();
			mes("^3355FF*Carrion dissolves before the blade lands cleanly. The inner gate opens.*^000000");
		}
		break;
	}
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20171, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

		mes("^3355FF*A lacquered card sits untouched on the stone. It looks important in the way poisons often do.*^000000");
		close();
	}

	mes("^3355FF*A black calling card has been left under a rusted bell weight. The handwriting is graceful enough to make the contents feel ruder.*^000000");

// ── Beelzebub Sanctum Set-piece ───────────────────────────────────────────
abbey03,150,150,0	script	Beelzebub Sanctum#dm	HIDDEN_NPC,{
	end;

OnBeelzebubDead:
	killmonster strnpcinfo(NPC_MAP), "Beelzebub Sanctum#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20171);
	callfunc("DM_InstanceSetFlag", "dm_arc13_beelzebub_killed", 1);
	callfunc("DM_InstanceSetFlag", "dm_arc13_coalition_deal_honored", 0);
	if (dm_arc13_quill_revered && dm_arc13_known_dead_reckoned)
		callfunc("DM_InstanceSetFlag", "dm_arc13_dead_freed", 1);
	else
		callfunc("DM_InstanceSetFlag", "dm_arc13_dead_burned", 1);
	callfunc("DM_PartyExp", 1600000, 900000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1399);
	instance_announce(instance_id(), "^3355FF[DM] Beelzebub falls. The council's room becomes only a ruined abbey again.^000000", bc_all);
	sleep(2000);
	if (dm_arc13_dead_freed)
		instance_announce(instance_id(), "^88CCFF[DM] Quill's last bell rings clean. Thousands of dead names leave the island by the right door.^000000", bc_all);
	else
		instance_announce(instance_id(), "^FF6600[DM] The coalition scatters. The dead remain behind you, quieter but not free.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The Seating Plan points at Surt's fire and the Morroc chair. Veins is next.^000000", bc_all);
	end;

OnAddDead:
	end;
}
