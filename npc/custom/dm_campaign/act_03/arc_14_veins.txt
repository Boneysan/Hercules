// -- Foreman Dunmar ---------------------------------------------------------
veins,100,100,5	script	Foreman Dunmar#dm	4_M_06,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20181) == 2) {
		mes("[Foreman Dunmar]");
		if (dm_arc14_evacuated)
			mes("Ifrit is gone. The tunnels are cooler. Not cool, but cooler. My people got out with names still attached to them.");
		else
			mes("Ifrit is gone. The mountain is quieter. Some names on my board are still crossed out. Do not ask me to call that clean.");
		close();
	}

	if (!dm_arc14_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20181, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20181, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20182, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20183, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20184, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20185, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("[Foreman Dunmar]");
	mes("You are not from Veins. Good. Everyone from Veins has been told there is no problem.");
	next();
	if (dm_arc09_karsh_exposed) {
		mes("[Dunmar]");
		mes("Rachel's prelate admitted the blessing was a rift. Hesma saw that confession and learned exactly the wrong lesson.");
		next();
	} else if (dm_arc09_karsh_deal) {
		mes("[Dunmar]");
		mes("Karsh made a managed deal in Rachel. Hesma has been waiting to see whether a volcano accepts paperwork.");
		next();
	}
	if (dm_arc13_beelzebub_killed || dm_arc13_carrion_minutes) {
		mes("[Dunmar]");
		mes("The Nameless Island agenda pointed here. The Sigil Ring does not chill near this mountain anymore. It scalds.");
		next();
	}
	mes("[Dunmar]");
	mes("The rock is breathing on a schedule. Hesma calls that miner superstition. My ledger says three families are still in the shaft tenements and two crews are on the deep shift.");
	next();
	switch (select("We'll get your people out.", "Show us the practical routes.", "Point us at the fire.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc14_dunmar_solidarity", 1);
		mes("[Dunmar]");
		mes("Then you get maps, names, and my best warding gear. People first. Proof second. Revenge never.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc14_dunmar_pragmatic", 1);
		mes("[Dunmar]");
		mes("Practical is good. Practical keeps people alive when sermons do not.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc14_dunmar_hardline", 1);
		mes("[Dunmar]");
		mes("You will get there faster than my miners can climb out. Remember that when the mountain asks for names.");
		break;
	}
	next();
	mes("[Dunmar]");
	mes("Three jobs make the evacuation real: update the ledger, scout the ash route, and cut the Magmaring firebreak. When those are done, we go after the deep shift.");
	callfunc("DM_InstanceSetFlag", "dm_arc14_started", 1);
	callfunc("DM_InstanceQuestStart", 20181);
	callfunc("DM_InstanceQuestStart", 20183);
	callfunc("DM_InstanceQuestStart", 20184);
	callfunc("DM_InstanceQuestStart", 20185);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1401);
	next();
	mes("^3355FF[Quest started: The Herald in the Magma]^000000");
	mes("^3355FFSupport routes started: Veins Evacuation Ledger, Bifrost Ash Scouting, Magmaring Firebreak.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20183) == 1 && questprogress(20183, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20183);
		callfunc("DM_PartyExp", 430000, 205000);
		callfunc("DM_GivePartyZeny", 35000);
		.@paid = 1;
		mes("[Dunmar]");
		mes("The ledger is current. That is not a miracle. It is names, addresses, and someone willing to carry them through smoke.");
		next();
		switch (select("Move families first.", "Move equipment with partial evacuations.", "Move fighters and able miners first.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_families", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_equipment", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_fighters", 0);
			mes("[Dunmar]");
			mes("The old and the children leave first. That is the order I would have written if Hesma ever let me post it.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_equipment", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_families", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_fighters", 0);
			mes("[Dunmar]");
			mes("Tools and people together. Slower, but Veins might still have work after it has survivors.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_fighters", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_families", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ledger_equipment", 0);
			mes("[Dunmar]");
			mes("Cold math. It gives me more hands near the volcano and worse dreams after.");
			break;
		}
		next();
	}

	if (questprogress(20184) == 1 && questprogress(20184, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20184);
		callfunc("DM_PartyExp", 470000, 225000);
		callfunc("DM_GivePartyZeny", 39000);
		.@paid = 1;
		mes("[Dunmar]");
		mes("The Bifrost ash fused into glass on a repeating cycle. Heat does not breathe unless something below it does.");
		next();
		switch (select("Map safe routes methodically.", "Balance scouting with bounty work.", "Follow the heat signature directly.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_routes", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_bounty", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_source", 0);
			mes("[Dunmar]");
			mes("A complete map. That saves lives when panic makes every corridor look the same.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_bounty", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_routes", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_source", 0);
			mes("[Dunmar]");
			mes("Useful work, useful coin, and a few blank places on the map. I can work with that.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_source", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_routes", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_ash_bounty", 0);
			mes("[Dunmar]");
			mes("You found the breathing pattern faster than anyone. The evacuees will not be walking that pattern. They will be walking blind spots.");
			break;
		}
		next();
	}

	if (questprogress(20185) == 1 && questprogress(20185, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20185);
		callfunc("DM_PartyExp", 450000, 215000);
		callfunc("DM_GivePartyZeny", 37000);
		.@paid = 1;
		mes("[Dunmar]");
		mes("The Magmaring clusters pulled back from the tenement wall. For now. The burn rings say they were reacting to something underneath them.");
		next();
		switch (select("Cut the firebreak at full width.", "Rescue miners caught in the open.", "Harvest the flame for reagents.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_full", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_rescue", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_reagents", 0);
			mes("[Dunmar]");
			mes("A real firebreak. Not pretty, not profitable. Useful.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_rescue", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_full", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_reagents", 0);
			mes("[Dunmar]");
			mes("You saved the ones in front of you. The wall is still weaker than I want. Both facts are true.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_reagents", 1);
			callfunc("DM_InstanceSetFlag", "dm_catechism_tally", dm_catechism_tally + 1);
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_full", 0);
			callfunc("DM_InstanceSetFlag", "dm_arc14_firebreak_rescue", 0);
			mes("[Dunmar]");
			mes("Weapon-grade flame and a firebreak that is mostly wishful thinking. I hope the reagent was worth the scorch.");
			break;
		}
		next();
	}

	if (questprogress(20183) == 2 && questprogress(20184) == 2 && questprogress(20185) == 2 && questprogress(20182) == 0) {
		mes("[Dunmar]");
		mes("The ledger is current, the ash route is mapped enough to risk, and the firebreak is cut. Now the mountain takes the deep shift.");
		next();
		mes("[Dunmar]");
		mes("Hesma has the sealed survey data. She can authorize the descent and admit what she buried, or she can give us the seal and keep lying.");
		callfunc("DM_InstanceQuestStart", 20182);
		next();
		mes("^3355FF[Quest started: The Deep Shift]^000000");
		mes("^3355FFConfront Prelate Hesma, then descend toward the Magma Cathedral.^000000");
		close();
	}

	if (questprogress(20182) == 1 && !dm_arc14_hesma_exposed && !dm_arc14_hesma_bribed) {
		mes("[Dunmar]");
		mes("Hesma is at the temple office. She has not left in two days. She knows something is coming.");
		close();
	}

	if (questprogress(20182) == 2) {
		mes("[Dunmar]");
		mes("The deep approach is open. Thor's Volcano, Cathedral level. Ifrit is only the herald. Remember that when victory starts sounding too clean.");
		close();
	}

	if (.@paid) {
		mes("[Dunmar]");
		mes("Finish all three support routes. Then we start crossing names off the deep-shift board, one way or another.");
		close();
	}

	mes("[Dunmar]");
	mes("The ledger, ash route, and firebreak all matter. This is what saving people looks like before it looks like heroics.");
	close();
}

// -- Prelate Hesma ----------------------------------------------------------
veins,110,100,5	script	Prelate Hesma#dm	4_M_06,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc14_started) {
		mes("[Prelate Hesma]");
		mes("Veins's sacred fires are a place of peace and reflection. Please respect the sanctity of the grounds.");
		close();
	}

	if (questprogress(20181) == 2) {
		mes("[Prelate Hesma]");
		if (dm_arc14_hesma_exposed)
			mes("Ifrit is gone. My confession is public. I will cooperate with the inquiry.");
		else
			mes("The city is alive. Some of them are alive because they were calm. I know which names that sentence avoids.");
		close();
	}

	if (questprogress(20182) != 1) {
		mes("[Prelate Hesma]");
		mes("Foreman Dunmar has work for you before this office becomes useful.");
		close();
	}

	mes("[Prelate Hesma]");
	mes("Foreman Dunmar sent you. He is a good man who is very angry with me. He is entitled to be.");
	next();
	if (dm_arc09_karsh_exposed) {
		mes("[Hesma]");
		mes("Karsh signed his confession. I watched Rachel's faith crack and told myself Veins deserved a gentler lie.");
		next();
	} else if (dm_arc09_karsh_deal) {
		mes("[Hesma]");
		mes("Karsh survived by managing the truth. I thought I could do the same with a volcano. That sentence sounds worse aloud.");
		next();
	}
	mes("[Hesma]");
	mes("I have known about the deep thermal readings for six months. I sealed them because panic kills people. So does fire. I chose which killer got more time.");
	next();
	switch (select("Release the survey data and authorize evacuation.", "Give us the seal and keep the city calm.", "Certainty is the curse, Prelate.")) {
	case 1:
		callfunc("DM_Decide", "arc14.hesma", "exposed");
		mes("[Hesma]");
		mes("Yes. The records, the authorization, the confession. All of it. I should have done this before the mountain started counting for me.");
		next();
		mes("^3355FF*Hesma signs the evacuation order and releases the sealed survey data.*^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc14.hesma", "bribed");
		mes("[Hesma]");
		mes("The seal is yours. The city account will remain a geological emergency. People move better when they are not screaming about gods.");
		next();
		mes("^3355FF*The deep approach seal changes hands. The official lie survives another hour.*^000000");
		break;
	case 3:
		callfunc("DM_Decide", "arc14.hesma", "exposed");
		mes("[Hesma]");
		mes("Aldric's old lesson. I hated it when Rachel used it as a sermon. I hate more that it is useful.");
		next();
		mes("^3355FF*Hesma opens the sealed records and writes her name under the evacuation order.*^000000");
		break;
	}
	next();
	mes("^3355FF[The Deep Shift resolved. Descend to thor_v03 and face Ifrit.]^000000");
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20182, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

// Hesma's withheld notice - records the cover-up in her own words.
veins,112,96,3	script	Withheld Notice#dm14	2_BOARD1,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc14_started) {
		mes("^3355FF*A sealed civic notice hangs here, unsigned and undelivered.*^000000");
		close();
	}

	mes("^3355FF*A notice meant for the city council has been folded and refolded until the edges are white. It was never posted.*^000000");
	next();
	mes("^0000AA[Draft Emergency Notice]^000000");
	mes("Pilgrimage activity is to be suspended pending review of thermal instability below the Cathedral routes. Civil evacuation planning should begin immediately.");
	next();
	mes("^0000AA[Draft Emergency Notice]^000000");
	mes("Projected consequence of immediate disclosure: loss of revenue, panic flight, institutional embarrassment, survivable political damage.");
	next();
	mes("^0000AA[Draft Emergency Notice]^000000");
	mes("Projected consequence of continued delay: miner fatalities, route collapse, eventual manifestation event no rhetoric will survive.");
	close();
}

// -- Magma Cathedral Set-piece ---------------------------------------------
thor_v03,150,150,0	script	Magma Cathedral#dm	HIDDEN_NPC,{
	end;

OnMagmaStart:
	stopnpctimer;
	'magma_map$ = strnpcinfo(NPC_MAP);
	'magma_party_id = $@dm_arc14_magma_party_id;
	'magma_damage = $@dm_arc14_magma_damage;
	if ('magma_party_id > 0) {
		mapannounce 'magma_map$, "[Magma Cathedral] The floor begins to pulse with Surt's heat.", bc_map, 0xFF6600;
		initnpctimer;
	}
	end;

OnMagmaStop:
	stopnpctimer;
	end;

OnTimer8000:
	mapannounce 'magma_map$, "[Magma Cathedral] A ring of heat rolls out from the seal.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'magma_map$, 150, 150, 8, 'magma_damage, 0, 0, 'magma_party_id);
	end;

OnTimer16000:
	mapannounce 'magma_map$, "[Magma Cathedral] The seal vents another furnace breath.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'magma_map$, 150, 150, 8, 'magma_damage, 0, 0, 'magma_party_id);
	end;

OnTimer24000:
	mapannounce 'magma_map$, "[Magma Cathedral] The lava channels brighten underfoot.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'magma_map$, 150, 150, 8, 'magma_damage, 0, 0, 'magma_party_id);
	end;

OnTimer32000:
	mapannounce 'magma_map$, "[Magma Cathedral] Surt's pressure hammers the chamber again.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'magma_map$, 150, 150, 8, 'magma_damage, 0, 0, 'magma_party_id);
	end;

OnTimer40000:
	mapannounce 'magma_map$, "[Magma Cathedral] The cathedral spends its last stored heat.", bc_map, 0xFF6600;
	callfunc("DM_HazardArea", 'magma_map$, 150, 150, 8, 'magma_damage, 0, 0, 'magma_party_id);
	stopnpctimer;
	end;

OnIfritDead:
	stopnpctimer;
	callfunc("DM_InstanceQuestComplete", 20181);
	callfunc("DM_InstanceSetFlag", "dm_arc14_ifrit_killed", 1);
	callfunc("DM_InstanceSetFlag", "dm_act03_complete", 1);
	callfunc("DM_InstanceSetFlag", "dm_arc14_cinder_taken", 1);
	callfunc("DM_PartyExp", 2000000, 1000000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1499);
	if (dm_arc14_hesma_exposed)
		instance_announce(instance_id(), "^3355FF[DM] Ifrit is banished. Dunmar's people evacuate under a public order, and Hesma's confession burns hotter than the ash.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] Ifrit is banished. The herald is gone, but some miners die calm inside a lie that outlived its usefulness.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] You carry a Cinder of the Sleeping Fire. It will not cool. Act III is complete; Act IV begins with Thanatos.^000000", bc_all);
	end;
}
