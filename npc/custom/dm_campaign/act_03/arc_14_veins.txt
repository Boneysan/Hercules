// ── Shared Arc 14 helpers ──────────────────────────────────────────────────

function	script	DM_Arc14Stage	{
	.@stage = getarg(0);
	if (dm_arc14_evacuation_stage >= .@stage)
		return dm_arc14_evacuation_stage;
	callfunc("DM_InstanceSetFlag", "dm_arc14_evacuation_stage", .@stage);
	callfunc("DM_DMJStoryObjective", 20181, 2, "Explore", .@stage, 3, (.@stage >= 3), getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc14_stage_" + .@stage)) {
		if (.@stage == 1) {
			callfunc("DM_PartyExp", 100000, 48000);
			callfunc("DM_InstanceQuestStart", 20185);
		} else if (.@stage == 2) {
			callfunc("DM_PartyExp", 100000, 48000);
		} else {
			callfunc("DM_PartyExp", 160000, 78000);
		}
	}
	if (.@stage >= 3)
		callfunc("DM_PartyMes", "^3355FF[Evacuation complete. Both groups are at Dunmar's camp. The workers have something to tell you.]^000000");
	else
		callfunc("DM_PartyMes", "^3355FF[Evacuation step " + .@stage + " of 3 done.]^000000");
	return .@stage;
}

// mode: 1 public authorization, 2 private access with accountability. Final.
function	script	DM_Arc14Commit	{
	.@mode = getarg(0);
	if (dm_arc14_authorization)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc14_authorization", .@mode);
	if (.@mode == 1) {
		callfunc("DM_InstanceClearFlag", "dm_arc14_hesma_bribed");
		callfunc("DM_InstanceSetFlag", "dm_arc14_hesma_exposed", 1);
	} else {
		callfunc("DM_InstanceClearFlag", "dm_arc14_hesma_exposed");
		callfunc("DM_InstanceSetFlag", "dm_arc14_hesma_bribed", 1);
	}
	callfunc("DM_InstanceQuestComplete", 20185);
	callfunc("DM_DMJStoryObjective", 20185, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc14_resolution"))
		callfunc("DM_PartyExp", 240000, 116000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1405);
	return 1;
}

// ── Prelate Hesma — The Prelate's Silence ──────────────────────────────────

			callfunc("DM_PartyMes", "I signed in front of the camp. It will be said that I was forced. It will also be said that I signed. I would prefer the second.");
		else
			callfunc("DM_PartyMes", "The approach is open. Let it be understood that I did not do this out of weakness. I will want to know how it is recorded.");
		close();
	}

	callfunc("DM_PartyMes", "[Prelate Hesma]");
	callfunc("DM_PartyMes", "Dunmar sent you. He would. He is a good man who has never once had to defend a building.");
	next();
	if (dm_arc09_karsh_exposed || dm_arc09_secret_kept) {
		callfunc("DM_PartyMes", "[Hesma]");
		callfunc("DM_PartyMes", (dm_arc09_secret_kept ? "I am told you kept Karsh's secret in Rachel. I am glad. We understand each other, then." : "I am told you exposed Karsh in Rachel. I assume that is why you have come to see me before you did anything else."));
		next();
	}
	callfunc("DM_PartyMes", "[Hesma]");
	callfunc("DM_PartyMes", "The cathedral is failing. The Church knows. The Church is not going to say so. If the miners leave in a panic, the cathedral falls with no one left to hold what is under it. My silence is not indifference.");
	next();
	if (dm_arc14_evacuation_stage < 1) {
		callfunc("DM_PartyMes", "[Hesma]");
		callfunc("DM_PartyMes", "You have not even surveyed the route. Come back with something I would have to answer.");
		close();
	}
	.@done = 0;
	while (!.@done) {
		switch (select("What is under the cathedral?", "What does a public authorization cost you?", "What does private access cost the workers?", "We are ready to decide.", "We need to discuss this.")) {
		case 1:
			callfunc("DM_PartyMes", "[Hesma]");
			callfunc("DM_PartyMes", "Surt. Or what is left of the thing that was sealed here. Ifrit is the first of it to stir. There will be more.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Hesma]");
			callfunc("DM_PartyMes", "My post, and in public, which is worse. The Church will say I was forced and then it will say I was wrong. The crews move at once, in view of everyone, with the whole camp as witness.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Hesma]");
			callfunc("DM_PartyMes", "Nothing, in the tunnels. They move just as fast. What it costs is later: I open a private approach, and someone has to write down that I did and why. If nobody does, it is a favor. If somebody does, it is a record.");
			break;
		case 4:
			.@done = 1;
			break;
		case 5:
			callfunc("DM_PartyMes", "[Hesma]");
			callfunc("DM_PartyMes", "Do. The tunnels will not wait much longer, but I will wait for people who think before they act.");
			close();
		}
		if (!.@done)
			next();
	}
	next();
	switch (select("Force a public evacuation authorization.", "Use his private approach, and document it afterward.", "Not yet. We want to talk it over.")) {
	case 1:
		.@mode = 1;
		break;
	case 2:
		.@mode = 2;
		break;
	case 3:
		callfunc("DM_PartyMes", "[Hesma]");
		callfunc("DM_PartyMes", "Of course.");
		close();
	}
	next();
	callfunc("DM_PartyMes", "[Hesma]");
	if (.@mode == 1)
		callfunc("DM_PartyMes", "Publicly, then. In front of the camp. Everyone will see what I was made to sign. Confirm, and I walk to the fire and read it out.");
	else
		callfunc("DM_PartyMes", "Privately. The approach opens at my word, and you write down that it did. Confirm, and I open it.");
	next();
	if (select("Confirm.", "Wait.") != 1) {
		callfunc("DM_PartyMes", "[Hesma]");
		callfunc("DM_PartyMes", "Better to wait than to be wrong in a camp full of people.");
		close();
	}
	if (!callfunc("DM_Arc14Commit", .@mode))
		close();
	callfunc("DM_PartyMes", "[Hesma]");
	if (.@mode == 1)
		callfunc("DM_PartyMes", "Then it is done. I have read it out. The crews can move now, openly.");
	else
		callfunc("DM_PartyMes", "The approach is open. Move your crews. And write it down, because I will want it written.");
	next();
	callfunc("DM_PartyMes", "^3355FF[The Prelate's Silence resolved. The crews can be moved.]^000000");
	close();

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20185, 1;
	end;
}

		close();

		callfunc("DM_PartyMes", "^3355FF*The golems that woke in the rubble are still moving.*^000000");
		close();
	}
	callfunc("DM_PartyMes", "^3355FF*The gallery that blocks the way out. Half the roof is down. The other half is wondering whether to follow.*^000000");
	next();
	switch (select("Brace the roof and clear the way.", "Clear it by force.", "Leave it.")) {
	case 1:
		if (callfunc("DM_CheckRoll", "craft", 12) < 2 && !callfunc("DM_IsDM")) {
			callfunc("DM_PartyMes", "^3355FF*The bracing is not quite right. Something in the rubble stirs: lava golems, woken by the noise.*^000000");
			break;
		}
		callfunc("DM_PartyMes", "^3355FF*You set the braces one at a time. The roof settles on them and does not argue.*^000000");
		callfunc("DM_Arc14Stage", 2);
		close();
	case 2:
		callfunc("DM_PartyMes", "^3355FF*You break the rubble open. The noise wakes what was sleeping in it.*^000000");
		break;
	case 3:
		close();
	}
	if (!'guard_cleared) {
		.@map$ = strnpcinfo(NPC_MAP);
		getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
		.@self$ = strnpcinfo(NPC_NAME);
		'boss_up = 1;
		callfunc("DM_EncStart", "arc14_gallery", getcharid(CHAR_ID_PARTY), .@map$, .@self$ + "::OnGuardDead", .@self$ + "::OnGuardDead", .@self$);
		callfunc("DM_EncMonster", .@map$, .@x, .@y, "Woken Lava Golem", LAVA_GOLEM, .@self$ + "::OnGuardDead", 74, 100, 2, 3);
	}
	close();

OnGuardDead:
	.@left = mobcount(strnpcinfo(NPC_MAP), strnpcinfo(NPC_NAME)+"::OnGuardDead");
	if (.@left > 0)
		end;
	'boss_up = 0;
	'guard_cleared = 1;
	callfunc("DM_EncCleanup", "arc14_gallery");
	// The golems were the obstruction: with them down, the way is clear.
	callfunc("DM_Arc14Stage", 2);
	end;

OnInit:
	callfunc("DM_AssertWalk");
	end;
}

	end;

	stopnpctimer;
	'magma_map$ = strnpcinfo(NPC_MAP);
	'magma_party_id = $@dm_arc14_magma_party_id;
	'magma_damage = $@dm_arc14_magma_damage;

	// and hit softer. The rule is on the announcement the party saw.
