// ── Shared Arc 8 helpers ───────────────────────────────────────────────────

function	script	DM_Arc08AddMemory	{
	.@bit = getarg(0);
	if (!(dm_arc08_memory_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc08_memory_mask", dm_arc08_memory_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc08_memory_mask);
	callfunc("DM_DMJStoryObjective", 20121, 2, "Explore", .@n, 2, (.@n >= 2), getcharid(CHAR_ID_PARTY));
	if (.@n == 2) {
		if (callfunc("DM_ClaimGrant", "arc08_memories")) {
			callfunc("DM_PartyExp", 300000, 144000);
			callfunc("DM_InstanceQuestStart", 20124);
			callfunc("DM_PartyMes", "^3355FF[Two memories. You can tell the evacuation order from the command that overrode it. Brother Manfred will talk about the watch now.]^000000");
		}
	} else if (.@n >= 3) {
		if (callfunc("DM_ClaimGrant", "arc08_memories_third"))
			callfunc("DM_PartyExp", 40000, 19000);
		callfunc("DM_PartyMes", "^3355FF[Three memories. The command was altered by hand: Manfred's seal on it, and the courier's mark beside it. Aldric's.]^000000");
	} else {
		callfunc("DM_PartyMes", "^3355FF[One memory recovered. Another opens the decision.]^000000");
	}
	return .@n;
}

// mode: 1 release, 2 rebind, 3 leave them. First commitment is final.
function	script	DM_Arc08Commit	{
	.@mode = getarg(0);
	if (dm_arc08_watch_resolution)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc08_watch_resolution", .@mode);
	if (.@mode == 3) {
		callfunc("DM_InstanceClearFlag", "dm_arc08_manfred_spared");
		callfunc("DM_InstanceSetFlag", "dm_arc08_manfred_killed", 1);
	} else {
		callfunc("DM_InstanceClearFlag", "dm_arc08_manfred_killed");
		callfunc("DM_InstanceSetFlag", "dm_arc08_manfred_spared", 1);
	}
	// Going in knowing who altered the order is on the record.
	if ((dm_arc08_memory_mask & 7) == 7)
		callfunc("DM_InstanceSetFlag", "dm_arc08_manfred_confessed", 1);
	callfunc("DM_InstanceQuestComplete", 20124);
	callfunc("DM_DMJStoryObjective", 20124, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc08_resolution"))
		callfunc("DM_PartyExp", 200000, 96000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 805);
	return 1;
}

// ── Brother Manfred — The Last Bishop ──────────────────────────────────────
gl_in01,120,120,5	script	Brother Manfred#dm	4_M_04,{
	if (!callfunc("DM_SessionGate")) close;
	if (!dm_arc08_started) {
		callfunc("DM_PartyMes", "[Brother Manfred]");
		callfunc("DM_PartyMes", "This is holy ground. Outsiders are not welcome in the Kingdom's halls.");
		close();
	}

	if (dm_arc08_watch_resolution) {
		callfunc("DM_PartyMes", "[Brother Manfred]");
		switch (dm_arc08_watch_resolution) {
		case 1:
			callfunc("DM_PartyMes", "They are at rest. I did not think I would be glad of it. Go. Face what our faith made.");
			break;
		case 2:
			callfunc("DM_PartyMes", "They are standing with you, because you asked and they agreed. That is not how I would have done it thirty years ago.");
			break;
		case 3:
			callfunc("DM_PartyMes", "...");
			break;
		}
		close();
	}

	callfunc("DM_PartyMes", "[Brother Manfred]");
	callfunc("DM_PartyMes", "The kingdom endures. It endures in the stones, in the silence, in the faith that outlasted every sword. Do you understand what faith means when it has nothing left to lose?");
	next();
	callfunc("DM_PartyMes", "[Manfred]");
	callfunc("DM_PartyMes", "The Dark Lord is not evil. It is the test. Midgard turned away from the Twelve. The cascade is their answer. I will not let heretics unmake what the Twelve have sent.");
	next();
	if (callfunc("DM_CountBits", dm_arc08_memory_mask) < 2) {
		callfunc("DM_PartyMes", "[Manfred]");
		callfunc("DM_PartyMes", "You speak of the watch as though you knew what it was ordered. You do not. Come back when you do.");
		close();
	}
	callfunc("DM_PartyMes", "[Manfred]");
	callfunc("DM_PartyMes", "You have been at the old posts. You have read what the watch was told. Then you know the question is not whether I am right. It is what you intend to do to them.");
	next();
	.@done = 0;
	while (!.@done) {
		switch (select("What does each choice cost the watch?", "What does it cost you?", "Who altered the order?", "We are ready to decide.", "We need to discuss this.")) {
		case 1:
			callfunc("DM_PartyMes", "[Manfred]");
			callfunc("DM_PartyMes", "Release them: they lay down their arms and sleep, the court loses them, and I lose the only thing that has made sense of thirty years. Rebind them: they stand once more, with their own consent, until the Dark Lord falls, and then they are released. If you walk away, they stay as they are, and so do I.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Manfred]");
			callfunc("DM_PartyMes", "My purpose. If they sleep, I have nothing to guard. If they stand, I am only their witness. I am not asking you to spare me that.");
			break;
		case 3:
			if ((dm_arc08_memory_mask & 7) == 7) {
				callfunc("DM_PartyMes", "[Manfred]");
				callfunc("DM_PartyMes", "I did. The evacuation order and the command to hold in prayer were two sheets, and I put one over the other and sealed it. I believed the Twelve would hold the gate. The courier did not read it. I did not tell him to.");
			} else {
				callfunc("DM_PartyMes", "[Manfred]");
				callfunc("DM_PartyMes", "You do not have all of it yet. I will not hand you the last piece for free.");
			}
			break;
		case 4:
			.@done = 1;
			break;
		case 5:
			callfunc("DM_PartyMes", "[Manfred]");
			callfunc("DM_PartyMes", "Talk. The watch has waited this long. It can wait for people who are thinking.");
			close();
		}
		if (!.@done)
			next();
	}
	next();
	switch (select("Release the watch from its oath.", "Ask the watch to stand once more, with their consent.", "Leave them as they are.", "Not yet. We want to talk it over.")) {
	case 1:
		.@mode = 1;
		break;
	case 2:
		.@mode = 2;
		break;
	case 3:
		.@mode = 3;
		break;
	case 4:
		callfunc("DM_PartyMes", "[Manfred]");
		callfunc("DM_PartyMes", "Of course.");
		close();
	}
	next();
	callfunc("DM_PartyMes", "[Manfred]");
	if (.@mode == 1)
		callfunc("DM_PartyMes", "They will sleep, and I will not stop you. Confirm it, and the oath is lifted.");
	else if (.@mode == 2)
		callfunc("DM_PartyMes", "They will be asked, not commanded. If they say yes, they stand until the Dark Lord falls and then they rest. Confirm it, and I ask them.");
	else
		callfunc("DM_PartyMes", "You will walk past. They will stand, and I will pray, and the Dark Lord will have a larger court for it. Confirm it, and that is how it is.");
	next();
	if (select("Confirm.", "Wait.") != 1) {
		callfunc("DM_PartyMes", "[Manfred]");
		callfunc("DM_PartyMes", "Better to wait than to be wrong in front of the dead.");
		close();
	}
	if (!callfunc("DM_Arc08Commit", .@mode))
		close();
	callfunc("DM_PartyMes", "[Manfred]");
	switch (.@mode) {
	case 1:
		callfunc("DM_PartyMes", "I release them. Go and rest, all of you. It was the wrong order, and you obeyed it faithfully.");
		callfunc("DM_PartyMes", "^3355FF[Encounter: the Dark Lord's court is smaller; the watch will not stand against you.]^000000");
		break;
	case 2:
		callfunc("DM_PartyMes", "They said yes. Not all of them. Enough. Take the banner they carried and set it where you can reach it.");
		callfunc("DM_PartyMes", "^3355FF[Encounter: a recovery banner stands in the abbey; the watch helps hold the court.]^000000");
		break;
	case 3:
		callfunc("DM_PartyMes", "Then go. I will keep praying.");
		callfunc("DM_PartyMes", "^3355FF[Encounter: Manfred's despair joins the Dark Lord's court.]^000000");
		break;
	}
	next();
	callfunc("DM_PartyMes", "^3355FF[The Weight of Faith resolved. The cathedral is open.]^000000");
	close();

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20124, 1;
	end;
}

// ── Memory 1: the watch post ───────────────────────────────────────────────
glast_01,130,130,4	script	Watch Post#dm	4_BOARD3,{
	callsub(S_Memory, 1, RAYDRIC_ARCHER, "Archers of the Watch", 3, "^3355FF*A scrap of the standing orders, still pinned: HOLD THE GATE. NO ONE LEAVES. THE TWELVE WILL PROVIDE.*^000000");
	end;
OnInit:
	callfunc("DM_AssertWalk");
	end;
OnGuardDead:
	.@left = mobcount(strnpcinfo(NPC_MAP), strnpcinfo(NPC_NAME)+"::OnGuardDead");
	if (.@left > 0)
		end;
	'boss_up = 0;
	'guard_cleared = 1;
	callfunc("DM_EncCleanup", "arc08_watch");
	end;

S_Memory:
	.@bit = getarg(0);
	.@mob = getarg(1);
	.@label$ = getarg(2);
	.@count = getarg(3);
	.@text$ = getarg(4);
	if (!callfunc("DM_SessionGate")) close;
	if (!dm_arc08_started) {
		mes("The party has not brought you into this chapter yet. Join their party, or log in again once someone who has it is online.");
		close;
	}
	if (dm_arc08_memory_mask & .@bit) {
		callfunc("DM_PartyMes", "^3355FF*Nothing more here. You have what it kept.*^000000");
		close();
	}
	if ('boss_up) {
		callfunc("DM_PartyMes", "^3355FF*The guard is still on its feet.*^000000");
		close();
	}
	if (!'guard_cleared) {
		callfunc("DM_PartyMes", "^3355FF*Something here still believes the order. It turns toward you.*^000000");
		.@map$ = strnpcinfo(NPC_MAP);
		getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
		.@self$ = strnpcinfo(NPC_NAME);
		'boss_up = 1;
		callfunc("DM_EncStart", "arc08_watch", getcharid(CHAR_ID_PARTY), .@map$, .@self$ + "::OnGuardDead", .@self$ + "::OnGuardDead", .@self$);
		callfunc("DM_EncMonster", .@map$, .@x, .@y, .@label$, .@mob, .@self$ + "::OnGuardDead", 60, 100, .@count, 3);
		close();
	}
	callfunc("DM_PartyMes", .@text$);
	callfunc("DM_Arc08AddMemory", .@bit);
	close();
}

	end;

OnDarkLordDead:
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc08_darklord", .@party))
		end;
	killmonster strnpcinfo(NPC_MAP), "Glast Heim Abbey#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20121);
	callfunc("DM_InstanceSetFlag", "dm_arc08_darklord_killed", 1);
	callfunc("DM_PartyExp", 700000, 400000, .@party);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 899);
	if (dm_arc08_watch_resolution == 2)
		instance_announce(instance_id(), "^3355FF[DM] The Dark Lord disperses. The watch lowers its banners and, one by one, lies down. Their oath is finished.^000000", bc_all);
	else if (dm_arc08_manfred_spared)
		instance_announce(instance_id(), "^3355FF[DM] The Dark Lord disperses. In the quiet after, Manfred's voice is the only sound.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] The Dark Lord is banished. The cursed kingdom finally falls silent.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The same institutional rot lives on in Rachel. The cascade has a church to feed on next.^000000", bc_all);
	end;

OnAddDead:
	end;
}
