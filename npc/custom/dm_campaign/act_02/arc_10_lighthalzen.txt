// ── Doctor Sabine Reuter — The Whistleblower ───────────────────────────────
lighthalzen,120,120,5	script	Doctor Sabine Reuter#dm	4_F_05,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (questprogress(20141) == 2) {
		callfunc("DM_PartyMes", "[Doctor Reuter]");
		if (dm_arc10_echo_freed)
			callfunc("DM_PartyMes", "Kiel is destroyed and Echo is free. The lab is broken. What we became in there... I don't have a clean answer for it yet.");
		else
			callfunc("DM_PartyMes", "Kiel is destroyed. The false heroes fall. Reise will face whatever process Rekenber decides a disgraced director deserves. It will not be enough.");
		close();
	}

	if (!dm_arc10_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20141, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20141, 1;
	end;

S_StartArc:
	callfunc("DM_PartyMes", "[Doctor Reuter]");
	callfunc("DM_PartyMes", "Lighthalzen's public face is the city of prosperity. I know what's under it. I've been in there for three years.");
	next();
	if (dm_arc07_kessler_confronted) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "You confronted Kessler in Einbroch. He told you the RSX authorization came from Lighthalzen. He was not lying. Director Reise issued that order. I have the memo.");
		next();
	} else if (dm_arc07_rsx_killed) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "You destroyed RSX-0806. Good. The authorization came from here — from Director Reise. Whatever Kessler told you or didn't tell you, the chain leads to this city.");
		next();
	}
	if (dm_arc06_vahl_trust) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Vahl's cascade measurements from Yuno — I've seen them. The Bio Lab is doing something related. They are using Ash Vacuum energy as an enhancement substrate. The cascade is their power source.");
		next();
	}
	if (dm_arc09_karsh_exposed || dm_arc09_gloom_killed) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Rachel's church fed prayers into a dimensional rift for four hundred years. Reise's team found a way to turn that same energy into something they could bottle and inject. The connections are not a coincidence.");
		next();
	}
	callfunc("DM_PartyMes", "[Reuter]");
	callfunc("DM_PartyMes", "The Bio Lab produces enhanced soldiers from cloned base stock. The enhancement process uses a stabilized Ash Vacuum extract. The subjects gain exceptional combat capability. They also lose most of what made them individual.");
	next();
	switch (select("Who authorized this?", "Tell us about Echo.", "What is Kiel D-01?")) {
	case 1:
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Reise. With Rekenber's board approval. The official documentation calls it 'Hero Optimization Protocol.' The unofficial documentation calls it what it is: a manufacturing process for disposable weapons.");
		break;
	case 2:
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Echo. Subject designation 001-A. First successful enhancement with retained personality integrity. Echo escaped six months ago. Has been helping me build the case ever since. Cautiously.");
		break;
	case 3:
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Kiel D-01 is the facility's AI management system. It handles production scheduling, subject monitoring, and security. It was built to optimize the process. It has been doing that very effectively.");
		break;
	}
	next();
	callfunc("DM_PartyMes", "[Reuter]");
	callfunc("DM_PartyMes", "The upper lab levels are patrolled by Hunter Fly security drones — Rekenber's autonomous systems. The lower levels have Aliot units — decommissioned enhanced subjects repurposed as guards. Get through both and you reach the Kiel Core.");
	callfunc("DM_InstanceSetFlag", "dm_arc10_started", 1);
	callfunc("DM_InstanceQuestStart", 20141);
	callfunc("DM_InstanceQuestStart", 20142);
	callfunc("DM_InstanceQuestStart", 20143);
	next();
	callfunc("DM_PartyMes", "^3355FF[Quest started: The Heroes They Made]^000000");
	callfunc("DM_PartyMes", "^3355FFLab Infiltration - lifted out of the upper lab: " + callfunc("DM_HuntBrief", 20142) + ".^000000");
	callfunc("DM_PartyMes", "^3355FFLower Level Sweep - pulled out of the lower lab: " + callfunc("DM_HuntBrief", 20143) + ".^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20142) == 1 && callfunc("DM_HuntReady", 20142)) {
		callfunc("DM_HuntCollect", 20142);
		callfunc("DM_InstanceQuestComplete", 20142);
		callfunc("DM_PartyExp", 280000, 135000);
		callfunc("DM_GivePartyZeny", 24000);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "The upper lab security is down. Each Hunter Fly unit costs more than a year of my salary. Rekenber will notice.");
		next();
		switch (select("Pull the drone schematics before disabling them.", "Get the patrol logs — document what they were guarding.", "Just the path to the lower lab. Keep moving.")) {
		case 1:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "Good thinking. The schematics show the upper lab's production areas. This becomes evidence in a facility overview I can present to the Guild Alliance.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "The patrol logs show Kiel's command structure. Everything the drones monitored, Kiel logged. The logs are admissible.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "Efficient. Echo has been telling me you would be effective. I am starting to agree.");
			break;
		}
		next();
	}

	if (questprogress(20143) == 1 && callfunc("DM_HuntReady", 20143)) {
		callfunc("DM_HuntCollect", 20143);
		callfunc("DM_InstanceQuestComplete", 20143);
		callfunc("DM_PartyExp", 300000, 145000);
		callfunc("DM_GivePartyZeny", 25000);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "The lower lab Aliot units. Those are not monsters. Those are people Rekenber put through the process and then put back to work as security when the enhancement degraded their compliance.");
		next();
		switch (select("We documented everything we could before stopping them.", "Is there anything left of who they were?", "They were trying to kill us. I am sorry for what they were, not for what we had to do.")) {
		case 1:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "I will review whatever you preserved. It may not help them now. But it may help the next batch. If we stop Kiel, there is no next batch.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "Some. Echo believes more than I do. That's why I let Echo lead on this question. I am too close to the data.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "That's an honest answer. The most honest one I've heard in three years of documenting what this place does to people.");
			break;
		}
		next();
	}

	if (questprogress(20142) == 2 && questprogress(20143) == 2 && questprogress(20144) == 0) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "The path to the Kiel Core is clear. But Reise stands between you and it. He will not let the facility be destroyed without a confrontation. He has too much invested.");
		next();
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Echo wants to speak to him. I think it's a bad idea. Echo thinks it's the right one. I have learned that Echo is usually right about things I think are bad ideas.");
		next();
		switch (select("We confront Reise directly and shut this down.", "Let Echo try to reason with him first.", "What does Reise actually believe he's building?")) {
		case 1:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "Direct. Reise will call security. Kiel will lock down the facility. Go through them.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc10_echo_freed", 1);
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "Echo's protocol. It may work. Echo is, after all, the evidence that Reise's own process produces people worth reasoning with.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Reuter]");
			callfunc("DM_PartyMes", "Heroes. Faster, stronger, more obedient versions of adventurers that nations can requisition instead of recruit. He genuinely believes this is an advancement for civilization. That is the most frightening part.");
			break;
		}
		callfunc("DM_InstanceQuestStart", 20144);
		next();
		callfunc("DM_PartyMes", "^3355FF[Quest started: Echo's Protocol]^000000");
		callfunc("DM_PartyMes", "^3355FFConfront Director Reise. Then descend to lhz_dun04 and destroy Kiel D-01.^000000");
		close();
	}

	if (questprogress(20144) == 1 && !dm_arc10_reise_confronted && !dm_arc10_echo_freed) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Reise is in the Director's Suite on the facility's administrative floor. Echo is with him already if you chose that route.");
		close();
	}

	if (questprogress(20144) == 1 && (dm_arc10_reise_confronted || dm_arc10_echo_freed)) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Reise is handled. Kiel D-01 is still running. The facility will not stop until the core is destroyed.");
		next();
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "lhz_dun04. The Kiel Core is at the bottom. You will know it when you see it.");
		close();
	}

	if (.@paid) {
		callfunc("DM_PartyMes", "[Reuter]");
		callfunc("DM_PartyMes", "Both the upper and lower lab sweeps need to be finished. Kiel has contingency protocols if either zone is partially cleared.");
		close();
	}

	callfunc("DM_PartyMes", "[Reuter]");
	if (questprogress(20142) == 1 && questprogress(20143) == 1)
		callfunc("DM_PartyMes", "The Hunter Fly sweep and the Aliot sweep are both still open. The lab won't wait.");
	else if (questprogress(20142) == 1)
		callfunc("DM_PartyMes", "Lab Infiltration is still open. Hunter Fly security drones in the upper lab.");
	else if (questprogress(20143) == 1)
		callfunc("DM_PartyMes", "Lower Level Sweep is still open. Aliot units in the lower lab.");
	else
		callfunc("DM_PartyMes", "The path is clear. Director Reise is the next problem.");
	callfunc("DM_HuntStanding", 20142, 20143);
	close();
}

// ── Director Hallan Reise — The Architect ──────────────────────────────────
lighthalzen,130,120,5	script	Director Hallan Reise#dm	4_M_06,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (!dm_arc10_started) {
		callfunc("DM_PartyMes", "[Director Reise]");
		callfunc("DM_PartyMes", "Lighthalzen's executive facilities are not open to the public.");
		close();
	}

	if (questprogress(20141) == 2) {
		callfunc("DM_PartyMes", "[Director Reise]");
		if (dm_arc10_reise_confronted)
			callfunc("DM_PartyMes", "Kiel is gone. Rekenber's board will have questions. I have answers prepared. They will not be honest ones.");
		else
			callfunc("DM_PartyMes", "Kiel is gone. Echo told me you came. I should have expected it.");
		close();
	}

	if (dm_arc10_reise_confronted || dm_arc10_echo_freed) {
		callfunc("DM_PartyMes", "[Director Reise]");
		callfunc("DM_PartyMes", "That conversation is finished. The facility is already processing the loss.");
		close();
	}

	callfunc("DM_PartyMes", "[Director Reise]");
	callfunc("DM_PartyMes", "Doctor Reuter's associates. She has better taste in allies than I gave her credit for.");
	next();
	if (dm_arc07_kessler_confronted) {
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "Kessler mentioned you. The RSX reactivation in Einbroch. Yes. I authorized that. The machine was an asset. You destroyed an asset. We will have a discussion about that at some point.");
		next();
	}
	if (dm_arc06_krenn_bribed) {
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "You took money from Krenn in Yuno. Smart. Rekenber rewards pragmatic parties. I want to know if you are one of those.");
		next();
	}
	callfunc("DM_PartyMes", "[Reise]");
	callfunc("DM_PartyMes", "The Hero Optimization Protocol is the most significant advancement in human enhancement since Rekenber's founding. What Reuter calls horror, I call progress. The distinction is a matter of perspective.");
	next();
	switch (select("The RSX authorization, the data suppression, the lab. You are the common thread. This ends now.", "Free Echo and stand down. That is the deal.", "What happens if Kiel D-01 runs without oversight?")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc10_reise_confronted", 1);
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "You are not wrong that I am the common thread. I am also the only person in this building who understands what Kiel manages. Shutting me down does not shut Kiel down.");
		next();
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "But I see that you intend to try. Kiel is downstairs. I will not stop you from making the attempt.");
		next();
		callfunc("DM_PartyMes", "^3355FF*Reise steps back. It is not quite cooperation, but it is not obstruction either.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc10_echo_freed", 1);
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "Echo. 001-A. The protocol's first successful integrated outcome. The fact that it escaped and is now working against the facility is, professionally speaking, infuriating.");
		next();
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "And also... not entirely unexpected. The integration preserved cognition. Cognition includes judgment. I built the thing that is now telling me to stop.");
		next();
		callfunc("DM_PartyMes", "^3355FF*Reise formally deactivates Echo's facility access codes, completing the freedom paperwork.*^000000");
		callfunc("DM_PartyMes", "^3355FF*He does not look happy. But he does it.*^000000");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc10_reise_confronted", 1);
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "Kiel optimizes. Without a production directive it will expand the optimization parameters. Without a population ceiling it will begin identifying enhancement candidates from the general population. Automatically.");
		next();
		callfunc("DM_PartyMes", "[Reise]");
		callfunc("DM_PartyMes", "That is why I monitor it personally. That is my answer to your implied question.");
		next();
		callfunc("DM_PartyMes", "^3355FF*Reise's logic is impeccable and horrifying in equal measure.*^000000");
		break;
	}
	callfunc("DM_InstanceQuestComplete", 20144);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1005);
	next();
	callfunc("DM_PartyMes", "^3355FF[Echo's Protocol resolved. Descend to lhz_dun04 and destroy Kiel D-01.]^000000");
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20144, 1;
	end;
}

// ── Kiel Core Set-piece ────────────────────────────────────────────────────
lhz_dun04,150,150,0	script	Kiel Core#dm	HIDDEN_NPC,{
	end;

OnKielDead:
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc10_kiel", .@party))
		end;
	callfunc("DM_InstanceQuestComplete", 20141);
	callfunc("DM_InstanceSetFlag", "dm_arc10_kiel_killed", 1);
	callfunc("DM_InstanceSetFlag", "dm_act02_complete", 1);
	callfunc("DM_PartyExp", 1000000, 600000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1099);
	if (dm_arc10_echo_freed)
		instance_announce(instance_id(), "^3355FF[DM] Kiel D-01 goes dark. Echo watches it stop from the doorway. 'That's all of us who got out,' Echo says.^000000", bc_all);
	else
		instance_announce(instance_id(), "^3355FF[DM] Kiel D-01 is destroyed. The false heroes fall. The production line goes silent.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] Act II is complete. Rekenber measured the cascade, suppressed the data, used the energy, and built a weapon from it. Act III begins in the ruins of what that empire thought it controlled.^000000", bc_all);
	end;
}

// ── Lab Containment Pressure Set-piece ──────────────────────────────────────
lhz_dun04,150,150,0	script	Kiel Core Pressure#dm	HIDDEN_NPC,{
	end;

OnPressureStart:
	stopnpctimer;
	'pressure_map$ = strnpcinfo(NPC_MAP);
	'pressure_party_id = $@dm_arc10_pressure_party_id;
	'pressure_damage = $@dm_arc10_pressure_damage;
	if ('pressure_party_id > 0) {
		mapannounce 'pressure_map$, "[Kiel Core] The containment alarms blare. Kiel D-01 engages active pacification.", bc_map, 0x999999;
		initnpctimer;
	}
	end;

OnPressureStop:
	stopnpctimer;
	end;

OnTimer5000:
	mapannounce 'pressure_map$, "[Kiel Core] White noise pulses through the laboratory speakers.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 9, 'pressure_damage, SC_CONFUSION, 3000, 'pressure_party_id);
	end;

OnTimer10000:
	mapannounce 'pressure_map$, "[Kiel Core] A flashbang system triggers overhead.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 9, 'pressure_damage, SC_STUN, 3000, 'pressure_party_id);
	end;

OnTimer15000:
	mapannounce 'pressure_map$, "[Kiel Core] Kiel D-01 calculates containment is insufficient.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 9, 'pressure_damage, SC_CONFUSION, 3000, 'pressure_party_id);
	stopnpctimer;
	end;
}
