		next();

		next();

		next();

	next();

		break;

		mes("[Reuter]");
		mes("The first prototype whose deletion failed. Echo was built to obey, woke up with a conscience, and walked out. The lab calls that a recall problem.");
		break;

		next();

	callfunc("DM_InstanceSetFlag", "dm_arc10_started", 1);
	callfunc("DM_InstanceQuestStart", 20141);
	callfunc("DM_InstanceQuestStart", 20143);
	callfunc("DM_InstanceQuestStart", 20144);
	callfunc("DM_InstanceQuestStart", 20145);
	next();
	mes("^3355FF[Quest started: The Factory of Heroes]^000000");
	mes("^3355FFSlum Sample Run - clear 20 Remover while tracing the clinic blood-card route.^000000");
	mes("^3355FFScaraba Chitin Order - collect 10 Antler and 10 Rake Scaraba plates from Scaraba Hole.^000000");
	mes("^3355FFClone Name Registry - clear Bio Lab F2 clone-stock and recover names from the cells.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20143) == 1 && questprogress(20143, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20143);
		callfunc("DM_PartyExp", 280000, 135000);
		callfunc("DM_GivePartyZeny", 24000);
		.@paid = 1;
		mes("[Reuter]");
		mes("The clinic cards match the missing-patient drawer. Those Remover were not lab accidents. They were what was left after the useful parts were taken.");
		next();
		switch (select("Turn the collectors away and treat patients for free.", "Sell surplus samples to fund the real clinic.", "Swap cards to shield one patient from selection.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc10_patients_protected", 1);
			mes("[Reuter]");
			mes("That costs us medicine and rent, but fewer people vanish this week. I can work with a loss. I cannot work with another empty bed.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc10_clinic_funded", 1);
			mes("[Reuter]");
			mes("A grim trade. The clinic stays open because the machine gets a little more blood. I hate that I understand the arithmetic.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc10_sample_swapped", 1);
			mes("[Reuter]");
			mes("You saved one name by moving the hunger elsewhere. Effective, personal, and not clean. The slums know that kind of mercy too well.");
			break;
		}
		next();
	}

	if (questprogress(20144) == 1 && questprogress(20144, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20144);
		callfunc("DM_PartyExp", 300000, 145000);
		callfunc("DM_GivePartyZeny", 25000);

		mes("[Reuter]");
		mes("The chitin plates are cut to a size that matches no lab furnace or armor rack I know. They are making something shaped like a person and armored like a product.");
		next();
		switch (select("Split the haul with the junior crews working the tunnel.", "Guard the crews without slowing our own pace.", "Overharvest and sell the surplus to the quartermaster.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc10_chitin_shared", 1);
			mes("[Reuter]");
			mes("Then Scaraba Hole stops being a race to see which desperate crew dies first. The lower city remembers practical kindness.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc10_juniors_guarded", 1);
			mes("[Reuter]");
			mes("Goodwill, no evidence. Sometimes that is still worth having when the streets start closing around you.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc10_chitin_overharvested", 1);
			mes("[Reuter]");
			mes("Fast money, and more material for the armor I am trying to prove exists. Reise will thank you in the language of invoices.");
			break;
		}
		next();
	}

	if (questprogress(20145) == 1 && questprogress(20145, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20145);
		callfunc("DM_PartyExp", 320000, 155000);
		callfunc("DM_GivePartyZeny", 26000);
		.@paid = 1;
		mes("[Reuter]");
		mes("Bio Lab F2 used serials where names should have been. The tags you recovered are the first funeral some of those people will ever get.");
		next();

		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc10_clone_names_restored", 1);

			mes("[Reuter]");
			mes("Echo will hear about that. A person made by a factory watches closely when strangers decide whether the dead deserve names.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc10_clone_serials_logged", 1);
			mes("[Reuter]");
			mes("Clean, fast, admissible. Also exactly as cold as the original system, just less corrupt.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc10_registry_erased", 1);
			mes("[Reuter]");
			mes("Sabotage works. Memory does not always survive it. I will not pretend I know which mattered more tonight.");
			break;
		}
		next();
	}

	if (questprogress(20143) == 2 && questprogress(20144) == 2 && questprogress(20145) == 2 && questprogress(20142) == 0) {
		mes("[Reuter]");
		mes("Cards, chitin, names. Three threads, one factory. Echo has agreed to meet you.");
		next();
		mes("[Reuter]");
		mes("Understand this before you speak: Echo does not want to become your weapon. Echo wants to be nobody, safely, for the first time.");
		callfunc("DM_InstanceQuestStart", 20142);
		next();
		mes("^3355FF[Quest started: The One Who Said No]^000000");
		mes("^3355FFFind Echo in the Lighthalzen slums and decide whether a manufactured fighter has the right to refuse the fight.^000000");
		close();
	}

	if (questprogress(20142) == 1) {
		mes("[Reuter]");
		mes("Echo is near the field kit. Reise's recall teams are already sweeping the alleys. Speak carefully.");
		close();
	}

		mes("[Reuter]");
		if (dm_arc10_echo_freed)
			mes("Echo is free. Reise is exposed. Kiel D-01 is still running below, and every template it holds is another Echo that may never get to say no.");
		else if (dm_arc10_echo_returned)
			mes("Echo is back in custody. Reise will pretend that proves the program works. Kiel D-01 still has to be stopped before he can turn that proof into a production order.");

		next();

		mes("The core is in lhz_dun04. The false heroes will wear familiar patterns. Do not let that make them familiar people.");

	if (.@paid) {
		mes("[Reuter]");
		mes("Finish the other two routes. The lab's story only becomes undeniable when the cards, plates, and names point to the same door.");
		close();
	}

	mes("[Reuter]");
	if (questprogress(20143) == 1 && questprogress(20144) == 1 && questprogress(20145) == 1)
		mes("Clinic cards, Scaraba plates, and the clone registry are all still open. The lower city is counting on us to make the pattern visible.");
	else if (questprogress(20143) == 1)
		mes("Slum Sample Run is still open. Remover in lhz_dun01, and blood cards behind every cleanup.");
	else if (questprogress(20144) == 1)
		mes("Scaraba Chitin Order is still open. Antler and Rake Scaraba in Scaraba Hole.");
	else if (questprogress(20145) == 1)
		mes("Clone Name Registry is still open. Bio Lab F2, and every cell with a number where a name should be.");
	else
		mes("The evidence is ready. Echo is the next door.");
	close();
}

		next();
		mes("^3355FF*Reise signs away Echo's facility claim with the expression of a man writing off a priceless instrument.*^000000");

		callfunc("DM_Decide", "arc10.echo", "returned");
		mes("[Reise]");

		next();

		next();
		mes("^3355FF*Somewhere below, a recall code closes around the first made thing that asked not to be used.*^000000");
		break;

	next();

	if ('pressure_party_id > 0) {
		mapannounce 'pressure_map$, "[Kiel Core] The containment alarms blare. Kiel D-01 engages active pacification.", bc_map, 0x999999;
		initnpctimer;
	}
	end;

OnPressureStop:
	stopnpctimer;
	end;

OnTimer5000:
	mapannounce 'pressure_map$, "[Kiel Core] White noise pulses through the laboratory speakers.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 9, 'pressure_damage, SC_CONFUSION, 3000, 'pressure_party_id);
	end;

OnTimer10000:
	mapannounce 'pressure_map$, "[Kiel Core] A flashbang system triggers overhead.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 9, 'pressure_damage, SC_STUN, 3000, 'pressure_party_id);
	end;

OnTimer15000:
	mapannounce 'pressure_map$, "[Kiel Core] Kiel D-01 calculates containment is insufficient.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 9, 'pressure_damage, SC_CONFUSION, 3000, 'pressure_party_id);
	stopnpctimer;
	end;
}
