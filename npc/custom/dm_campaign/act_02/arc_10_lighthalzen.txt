// Progress marker after any preparation. Starts Echo's Protocol once the
// containment control is open.
function	script	DM_Arc10Prep	{
	.@n = (dm_arc10_patrol_disabled ? 1 : 0) + (dm_arc10_evidence_secured ? 1 : 0) + (dm_arc10_containment_opened ? 1 : 0);
	callfunc("DM_DMJStoryObjective", 20141, 2, "Explore", .@n, 3, (.@n >= 3), getcharid(CHAR_ID_PARTY));
	if (dm_arc10_containment_opened) {
		if (callfunc("DM_ClaimGrant", "arc10_route")) {
			callfunc("DM_PartyExp", 360000, 174000);
			callfunc("DM_InstanceQuestStart", 20144);
			callfunc("DM_PartyMes", "^3355FF[Echo's containment is open. Speak to Echo.]^000000");
		}
	} else {
		callfunc("DM_PartyMes", "^3355FF[" + .@n + " of 3 done: patrol terminal, research record, containment control.]^000000");
	}
	return .@n;
}

// Echo's exit resolved: only now is she free.
function	script	DM_Arc10EchoFree	{
	if (dm_arc10_echo_freed)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc10_echo_freed", 1);
	callfunc("DM_DMJStoryObjective", 20144, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc10_echo"))
		callfunc("DM_PartyExp", 240000, 114000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1005);
	return 1;
}

// mode: 1 preserve the work as testimony, 2 destroy it. Final.
function	script	DM_Arc10Commit	{
	.@mode = getarg(0);
	if (dm_arc10_lab_resolution)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc10_lab_resolution", .@mode);
	callfunc("DM_InstanceQuestComplete", 20144);
	if (callfunc("DM_ClaimGrant", "arc10_resolution"))
		callfunc("DM_PartyExp", 240000, 114000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 1006);
	return 1;
}

	close();
