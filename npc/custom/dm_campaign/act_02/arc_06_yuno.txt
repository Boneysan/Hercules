// Evidence score. Partial migration readings count for 1, clean for 2.
function	script	DM_Arc06Score	{
	.@s = 0;
	if (dm_arc06_evidence_mask & 1)
		.@s += (dm_arc06_post_fail ? 1 : 2);
	if (dm_arc06_evidence_mask & 2)
		.@s += 2;
	if (dm_arc06_evidence_mask & 4)
		.@s += 2;
	return .@s;
}

// Record one source and open the hearing the first time the score reaches 4.
function	script	DM_Arc06AddEvidence	{
	.@bit = getarg(0);
	if (!(dm_arc06_evidence_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc06_evidence_mask", dm_arc06_evidence_mask | .@bit);
	.@score = callfunc("DM_Arc06Score");
	callfunc("DM_DMJStoryObjective", 20101, 2, "Explore", callfunc("DM_CountBits", dm_arc06_evidence_mask), 2, (.@score >= 4), getcharid(CHAR_ID_PARTY));
	if (.@score >= 4) {
		if (callfunc("DM_ClaimGrant", "arc06_evidence")) {
			callfunc("DM_PartyExp", 360000, 171000);
			callfunc("DM_InstanceQuestStart", 20102);
			callfunc("DM_PartyMes", "^3355FF[Evidence enough for a Council hearing. Return to Doctor Vahl.]^000000");
		} else {
			callfunc("DM_PartyMes", "^3355FF[The case is stronger now.]^000000");
		}
	} else {
		callfunc("DM_PartyMes", "^3355FF[Evidence: " + .@score + " of 4 needed for a hearing.]^000000");
	}
	return .@score;
}

// The one commitment path. mode: 1 named, 2 anonymous, 3 protected (with
// Krenn), 4 buried (Krenn's money). Mutually exclusive: the first commitment
// is final. Pays equal base story EXP whichever way the party goes.
function	script	DM_Arc06Commit	{
	.@mode = getarg(0);
	if (dm_arc06_disclosure_mode)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc06_disclosure_mode", .@mode);
	if (.@mode == 1 || .@mode == 2)
		callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_exposed", 1);
	if (.@mode == 3)
		callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_cooperated", 1);
	if (.@mode == 4) {
		callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_bribed", 1);
		if (callfunc("DM_ClaimGrant", "arc06_krenn_zeny"))
			callfunc("DM_GivePartyZeny", 25000);
	}
	// Vahl trusts people who put their names to it, or who brought all three sources.
	if (.@mode == 1 || (dm_arc06_evidence_mask & 7) == 7)
		if (.@mode != 4)
			callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
	callfunc("DM_InstanceQuestComplete", 20102);
	callfunc("DM_DMJStoryObjective", 20102, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc06_resolution"))
		callfunc("DM_PartyExp", 240000, 114000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 605);
	return 1;
}

S_Post:
	.@bit = getarg(0);
	.@tag$ = getarg(1);
	if (!callfunc("DM_SessionGate")) close;
	if (!dm_arc06_started) {
		mes("The party has not brought you into this chapter yet. Join their party, or log in again once someone who has it is online.");

		callfunc("DM_PartyMes", "^3355FF*Calibrated. The needle sits where Vahl said it would.*^000000");

	next();
