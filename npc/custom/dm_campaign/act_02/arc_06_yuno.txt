// ── Doctor Ingrid Vahl — Yuno Atmospheric Researcher ───────────────────────
yuno,155,185,5	script	Doctor Ingrid Vahl#dm	4_F_02,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (questprogress(20101) == 2) {
		callfunc("DM_PartyMes", "[Doctor Vahl]");
		if (dm_arc06_mistress_killed)
			callfunc("DM_PartyMes", "The readings have stabilized. Not safe — stable. There is a difference, and Yuno still does not want to know it.");
		else
			callfunc("DM_PartyMes", "The Mistress has gone quiet. Something else holds the shard now. I am not certain it is better.");
		next();
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Keep the number. Someone has to carry it.");
		close();
	}

	if (!dm_arc06_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20101, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20101, 1;
	end;

S_StartArc:
	callfunc("DM_PartyMes", "[Doctor Vahl]");
	callfunc("DM_PartyMes", "You are not Rekenber's private security. That is already better than most of my visitors today.");
	next();
	if (countitem(50001) > 0 || dm_arc01_sigil_ring_obtained) {
		callfunc("DM_PartyMes", "^3355FF*The Sigil Ring grows warmer here, not colder. Yuno hums on a frequency the ring agrees with.*^000000");
		next();
	}
	if (dm_arc05_northern_account || dm_arc05_brode_threatened) {
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Brode's gray courier and the northern account? That money leads here. Director Krenn's suppression budget comes from the same Yuno investor chain. You already pulled that thread.");
		next();
	}
	callfunc("DM_PartyMes", "[Vahl]");
	callfunc("DM_PartyMes", "I am Doctor Vahl. I measure the dimensional instability in the Ymir shard that holds this city up.");
	next();
	callfunc("DM_PartyMes", "[Vahl]");
	callfunc("DM_PartyMes", "The Republic built everything on a fragment the old gods used as a hinge. Fine. But the cascade is interfering with the fragment's resonance. I have the number now. The city has years, not decades.");
	next();
	switch (select("What does Yuno need from us?", "Who is paying to suppress this?", "Show us the data.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Atmospheric readings from the Harpy migration routes. They sense the disturbance before instruments do. And machine parts from Juperos — the ruins below are still reacting to the fragment.");
		break;
	case 2:
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Director Krenn. Rekenber pays him to file my readings as 'within acceptable parameters.' The city floats on denial as much as the shard.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "This graph. The resonance frequency is diverging from baseline. That line will intersect the cascade frequency in... I am still writing the decimal. It is not a long one.");
		break;
	}
	next();
	callfunc("DM_PartyMes", "[Vahl]");
	callfunc("DM_PartyMes", "Help me gather field evidence. Harpy weather logs from the sky migration, and Venatu components from the Juperos ruins. Then I can finish the report and we see what Krenn does with it.");
	callfunc("DM_InstanceSetFlag", "dm_arc06_started", 1);
	callfunc("DM_InstanceQuestStart", 20101);
	callfunc("DM_InstanceQuestStart", 20103);
	callfunc("DM_InstanceQuestStart", 20104);
	next();
	callfunc("DM_PartyMes", "^3355FF[Quest started: The Heart That Holds the City]^000000");
	callfunc("DM_PartyMes", "^3355FFHarpy Weather Log - gather migration data from 20 Harpy in the Yuno fields.^000000");
	callfunc("DM_PartyMes", "^3355FFJuperos Parts Requisition - recover 20 Venatu components from the Juperos ruins.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20103) == 1 && questprogress(20103, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20103);
		callfunc("DM_PartyExp", 180000, 85000);
		callfunc("DM_GivePartyZeny", 15000);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Harpy migration data. They have been banking south for three weeks. They feel the shard warming before the instruments register it.");
		next();
		switch (select("Log the readings and file them.", "Publish the migration data early.", "Give the raw data to Krenn as leverage.")) {
		case 1:
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "Filed. When the crisis report is complete, this becomes the paper trail he cannot burn.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_exposed", 1);
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "The naturalist journals will run it before Krenn can kill the suppression memo. That is either brave or reckless. Possibly both.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "Giving Krenn advance data to use as cover for his own suppression. That is a bet on his gratitude, which I would not take.");
			break;
		}
		next();
	}

	if (questprogress(20104) == 1 && questprogress(20104, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20104);
		callfunc("DM_PartyExp", 220000, 105000);
		callfunc("DM_GivePartyZeny", 18000);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Venatu components. The ruins are vibrating at the same frequency as the shard. The Juperos builders knew. They built resonators into the ruins to monitor the fragment from below.");
		next();
		switch (select("Calibrate them for your instruments.", "Sell them to Schwarzwald engineers.", "Keep one — the Sigil Ring reacts to it.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "With these calibrated, the instruments will read the fragment directly. The number I give the Council will be accurate enough to make them genuinely afraid.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "They will learn what they need from the parts. And we get the budget to finish the report.");
			break;
		case 3:
			if (dm_arc01_sigil_ring_obtained || countitem(50001) > 0) {
				callfunc("DM_PartyMes", "[Vahl]");
				callfunc("DM_PartyMes", "The resonator and the ring hum at each other. The Juperos builders and whoever forged that ring understood the same thing about sealed things.");
			} else {
				callfunc("DM_PartyMes", "[Vahl]");
				callfunc("DM_PartyMes", "A souvenir from the ruins. They make a noise that no one else in the party hears.");
			}
			break;
		}
		next();
	}

	if (questprogress(20103) == 2 && questprogress(20104) == 2 && questprogress(20102) == 0) {
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Both datasets are in. The report is written: resonance divergence, migration anomaly, and a timeline the Republic will not enjoy.");
		next();
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "I am calling it The Number Nobody Wants. Krenn has to either sign the transmission or formally bury it. Either way it ends his comfortable suppression.");
		next();
		switch (select("Deliver it over Krenn's head to the Republic Council.", "Let us carry it past Krenn ourselves.", "Use it as leverage against Krenn.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_exposed", 1);
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "The Council will not enjoy it. But they will not be able to bury it with you watching.");
			break;
		case 2:
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "Krenn's office is between us and the Council chamber. He will intercept it. Be ready for that conversation.");
			break;
		case 3:
			callfunc("DM_PartyMes", "[Vahl]");
			callfunc("DM_PartyMes", "Leverage only works if you are willing to use it, and willing to survive what happens after you do. But yes. It is leverage.");
			break;
		}
		callfunc("DM_InstanceQuestStart", 20102);
		next();
		callfunc("DM_PartyMes", "^3355FF[Quest started: The Number Nobody Wants]^000000");
		callfunc("DM_PartyMes", "^3355FFConfront Director Krenn. Then descend to Juperos and face the Mistress.^000000");
		close();
	}

	if (questprogress(20102) == 1 && !dm_arc06_krenn_bribed && !dm_arc06_krenn_exposed) {
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Krenn is the wall between the data and the city that needs it. He posts near the Grand Library most mornings.");
		close();
	}

	if ((questprogress(20102) == 1 || questprogress(20102) == 0) && (dm_arc06_krenn_bribed || dm_arc06_krenn_exposed)) {
		if (questprogress(20102) == 1) callfunc("DM_InstanceQuestComplete", 20102);
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Krenn is dealt with. Now the fragment below needs dealing with.");
		next();
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "The Mistress. An old guardian spirit from the Juperos era — it was supposed to watch over the shard. The cascade has been feeding it. Whatever it is watching over now, it is no longer the shard.");
		next();
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "The entrance is below the plateau, south of the Republic's lower docks.");
		close();
	}

	if (.@paid) {
		callfunc("DM_PartyMes", "[Vahl]");
		callfunc("DM_PartyMes", "Bring back both sets of field data. The report writes itself after that.");
		close();
	}

	callfunc("DM_PartyMes", "[Vahl]");
	if (questprogress(20103) == 1 && questprogress(20104) == 1)
		callfunc("DM_PartyMes", "The Harpy logs and the Juperos parts are both still open. The shard readings do not improve while we wait.");
	else if (questprogress(20103) == 1)
		callfunc("DM_PartyMes", "The Harpy migration contract is still open. They are banking south from the outer fields.");
	else if (questprogress(20104) == 1)
		callfunc("DM_PartyMes", "The Juperos parts requisition is still open. The Venatu down there know what they are guarding.");
	else
		callfunc("DM_PartyMes", "Finish both contracts and come back. The number is waiting.");
	close();
}

// ── Director Aldous Krenn — The Republic's Money ───────────────────────
yuno,160,185,5	script	Director Aldous Krenn#dm	4_M_01,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (!dm_arc06_started) {
		callfunc("DM_PartyMes", "[Director Krenn]");
		callfunc("DM_PartyMes", "The Research Bureau handles appointments. I handle approvals. You require neither.");
		close();
	}

	if (dm_arc06_krenn_bribed || dm_arc06_krenn_exposed) {
		callfunc("DM_PartyMes", "[Director Krenn]");
		if (dm_arc06_krenn_bribed)
			callfunc("DM_PartyMes", "You took the money. Good. Now let us pretend this was a professional arrangement.");
		else
			callfunc("DM_PartyMes", "The Council has the document. I hope you know what you have started.");
		close();
	}

	callfunc("DM_PartyMes", "[Director Krenn]");
	callfunc("DM_PartyMes", "Vahl's numbers. Yes. I have seen them.");
	next();
	if (dm_arc04_cassell_unmasked) {
		callfunc("DM_PartyMes", "[Krenn]");
		callfunc("DM_PartyMes", "Brother Cassell sends his regards, incidentally. He was very specific: 'parties carrying unusual seals are to receive every courtesy.' So. Here we are.");
		next();
	}
	callfunc("DM_PartyMes", "[Krenn]");
	callfunc("DM_PartyMes", "The investors who built the Republic's infrastructure take a dim view of public panic. My job is managed uncertainty. Vahl's report is unmanaged certainty. You see the problem.");
	next();
	switch (select("Sign the document. This city deserves the truth.", "We will take the data to a buyer who is not Rekenber.", "What does staying quiet pay?")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_exposed", 1);
		callfunc("DM_PartyMes", "[Krenn]");
		callfunc("DM_PartyMes", "You want me to sign my own career's end so Yuno can panic in an orderly fashion. No. But you do not need my signature.");
		next();
		callfunc("DM_PartyMes", "^3355FF*Krenn's refusal goes on record. The document moves to the Council over his objection.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_exposed", 1);
		callfunc("DM_PartyMes", "[Krenn]");
		callfunc("DM_PartyMes", "The Schwarzwald naturalist journals will print anything. And I will find a reason to have you removed from the city before morning.");
		next();
		callfunc("DM_PartyMes", "^3355FF*The threat is hollow. Vahl's data leaves through the naturalist circuit.*^000000");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_bribed", 1);
		if (callfunc("DM_ClaimGrant", "arc06_krenn_zeny"))
			callfunc("DM_GivePartyZeny", 25000);
		callfunc("DM_PartyMes", "[Krenn]");
		callfunc("DM_PartyMes", "Now you are speaking the Republic's language. Twenty-five thousand zeny. Consider it a retainer for services rendered in the public interest.");
		next();
		callfunc("DM_PartyMes", "^3355FF*25,000 zeny changes hands. Krenn files the report under 'acceptable variance.'*^000000");
		break;
	}
	callfunc("DM_InstanceQuestComplete", 20102);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 605);
	next();
	callfunc("DM_PartyMes", "^3355FF[The Number Nobody Wants resolved. The Juperos set-piece is now accessible.]^000000");
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20102, 1;
	end;
}

// ── Juperos Vault Set-piece ────────────────────────────────────────────────
juperos_01,150,150,0	script	Juperos Vault#dm	HIDDEN_NPC,{
	end;

OnMistressDead:
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc06_mistress", .@party))
		end;
	callfunc("DM_InstanceQuestComplete", 20101);
	callfunc("DM_InstanceSetFlag", "dm_arc06_mistress_killed", 1);
	callfunc("DM_PartyExp", 500000, 300000, .@party);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 699);
	instance_announce(instance_id(), "^3355FF[DM] Mistress falls. The Ymir fragment shudders, but the city above holds... for now.^000000", bc_all);
	sleep(2000);
	instance_announce(instance_id(), "^FF6600[DM] The cascade has a measured number. The Ash Vacuum thread opens. Act II continues north.^000000", bc_all);
	end;
}
