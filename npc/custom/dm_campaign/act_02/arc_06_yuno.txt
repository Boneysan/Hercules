// ── Doctor Ingrid Vahl — Yuno Atmospheric Researcher ───────────────────────
yuno,155,185,5	script	Doctor Ingrid Vahl#dm	4_F_02,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20101) == 2) {
		mes("[Doctor Vahl]");
		if (dm_arc06_mistress_killed)
			mes("The readings have stabilized. Not safe — stable. There is a difference, and Yuno still does not want to know it.");
		else
			mes("The Mistress has gone quiet. Something else holds the shard now. I am not certain it is better.");
		next();
		mes("[Vahl]");
		mes("Keep the number. Someone has to carry it.");
		close();
	}

	if (!dm_arc06_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20101, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20101, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20103, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20104, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("^3355FF*A woman stares intently at a vibrating brass instrument, not looking up as you approach.*^000000");
	next();
	mes("[Doctor Vahl]");
	mes("You feel it, don't you? The Guild down south thinks it's just monster migrations, but the math doesn't lie. The dimensional weight of the world is dropping. I have measured how long we have left, and the Academy filed my number as a typo.");
	next();
	if (countitem(50001) > 0 || dm_arc01_sigil_ring_obtained) {
		mes("^3355FF*The Sigil Ring grows warmer here, not colder. Yuno hums on a frequency the ring agrees with.*^000000");
		next();
	}
	if (dm_arc05_northern_account || dm_arc05_brode_threatened) {
		mes("[Vahl]");
		mes("Brode's gray courier and the northern account? That money leads here. Director Krenn's suppression budget comes from the same Yuno investor chain. You already pulled that thread.");
		next();
	}
	switch (select("We know. The seals are weakening. We're trying to stop it.", "Are you hiring us to fix a machine, or hunt a monster?", "Who could siphon energy from a floating city?")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
		mes("[Vahl]");
		mes("Seals? Magic? Don't be primitive. It's physics! The harmonic frequency of the Ymir Fragment keeping this city afloat is de-syncing. Someone is actively drawing on the continent's anchor — and the readings spiked the week Director Krenn's 'sampling' rig went down the shaft.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
		mes("[Vahl]");
		mes("Both! A massive energy signature is surging from the Juperos Ruins. If they drain the Ymir Fragment, Yuno drops out of the sky. I need you to go down there and unplug whatever Krenn plugged in.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
		mes("[Vahl]");
		mes("Someone with a board seat and no fear of consequences — Director Aldous Krenn. And someone who understands pre-industrial tech better than we do: my assistants found a gray robe in the maintenance shafts. Your 'magic' cult has discovered science, it seems. The two are not unrelated.");
		break;
	}
	next();
	mes("[Vahl]");
	mes("Help me gather field evidence. Harpy weather logs from the sky migration, Venatu components from the Juperos ruins, and a heat index from the Norgroad vents. Then I can finish the report and we see what Krenn does with it.");
	callfunc("DM_InstanceSetFlag", "dm_arc06_started", 1);
	callfunc("DM_InstanceQuestStart", 20101);
	callfunc("DM_InstanceQuestStart", 20103);
	callfunc("DM_InstanceQuestStart", 20104);
	callfunc("DM_InstanceQuestStart", 20105);
	next();
	mes("^3355FF[Quest started: The Heart That Holds the City]^000000");
	mes("^3355FFHarpy Weather Log - gather migration data from 20 Harpy in the Yuno fields.^000000");
	mes("^3355FFJuperos Parts Requisition - recover 20 Venatu components from the Juperos ruins.^000000");
	mes("^3355FFNorgroad Heat Index - log the vent pulse from 20 Anacondaq in Norgroad.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20103) == 1 && questprogress(20103, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20103);
		callfunc("DM_PartyExp", 180000, 85000);
		callfunc("DM_GivePartyZeny", 15000);
		.@paid = 1;
		mes("[Vahl]");
		mes("Harpy migration data. They have been banking south for three weeks. They feel the shard warming before the instruments register it.");
		next();
		switch (select("Log the readings and file them.", "Publish the migration data early.", "Give the raw data to Krenn as leverage.")) {
		case 1:
			mes("[Vahl]");
			mes("Filed. When the crisis report is complete, this becomes the paper trail he cannot burn.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_exposed", 1);
			mes("[Vahl]");
			mes("The naturalist journals will run it before Krenn can kill the suppression memo. That is either brave or reckless. Possibly both.");
			break;
		case 3:
			mes("[Vahl]");
			mes("Giving Krenn advance data to use as cover for his own suppression. That is a bet on his gratitude, which I would not take.");
			break;
		}
		next();
	}

	if (questprogress(20104) == 1 && questprogress(20104, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20104);
		callfunc("DM_PartyExp", 220000, 105000);
		callfunc("DM_GivePartyZeny", 18000);
		.@paid = 1;
		mes("[Vahl]");
		mes("Venatu components. The ruins are vibrating at the same frequency as the shard. The Juperos builders knew. They built resonators into the ruins to monitor the fragment from below.");
		next();
		switch (select("Calibrate them for your instruments.", "Sell them to Schwarzwald engineers.", "Keep one — the Sigil Ring reacts to it.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc06_vahl_trust", 1);
			mes("[Vahl]");
			mes("With these calibrated, the instruments will read the fragment directly. The number I give the Council will be accurate enough to make them genuinely afraid.");
			break;
		case 2:
			mes("[Vahl]");
			mes("They will learn what they need from the parts. And we get the budget to finish the report.");
			break;
		case 3:
			if (dm_arc01_sigil_ring_obtained || countitem(50001) > 0) {
				mes("[Vahl]");
				mes("The resonator and the ring hum at each other. The Juperos builders and whoever forged that ring understood the same thing about sealed things.");
			} else {
				mes("[Vahl]");
				mes("A souvenir from the ruins. They make a noise that no one else in the party hears.");
			}
			break;
		}
		next();
	}

	if (questprogress(20105) == 1 && questprogress(20105, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20105);
		callfunc("DM_PartyExp", 200000, 95000);
		callfunc("DM_GivePartyZeny", 16000);
		.@paid = 1;
		mes("[Vahl]");
		mes("The heat-glass vials. Look — the wax gradations melted into a clean sine wave. Norgroad isn't just hot. It's breathing. In time with the city's failing anchor.");
		next();
		switch (select("Stabilize the dangerous vents.", "Sell vent-taps to the foundries.", "Overcharge the readings for funding.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc06_norgroad_choice", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc06_vents_stabilized", 1);
			mes("[Vahl]");
			mes("Slower work. Safer dungeon. The facilities office will sign your names to the savings and forget them by next quarter.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc06_norgroad_choice", 2);
			callfunc("DM_InstanceSetFlag", "dm_arc06_needle_degraded", 1);
			mes("[Vahl]");
			mes("Excellent zeny. Also a changed rhythm. Whatever clean reading I had from those vents, I no longer have.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc06_norgroad_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc06_audit_incoming", 1);
			mes("[Vahl]");
			mes("The money will arrive. So will an audit. Krenn's faction times those well.");
			break;
		}
		next();
	}

	if (questprogress(20103) == 2 && questprogress(20104) == 2 && questprogress(20105) == 2 && questprogress(20102) == 0) {
		mes("[Vahl]");
		mes("Both datasets are in. The report is written: resonance divergence, migration anomaly, and a timeline the Republic will not enjoy.");
		next();
		mes("[Vahl]");
		mes("I am calling it The Number Nobody Wants. Krenn has to either sign the transmission or formally bury it. Either way it ends his comfortable suppression.");
		next();
		switch (select("Deliver it over Krenn's head to the Republic Council.", "Let us carry it past Krenn ourselves.", "Use it as leverage against Krenn.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc06_krenn_exposed", 1);
			mes("[Vahl]");
			mes("The Council will not enjoy it. But they will not be able to bury it with you watching.");
			break;
		case 2:
			mes("[Vahl]");
			mes("Krenn's office is between us and the Council chamber. He will intercept it. Be ready for that conversation.");
			break;
		case 3:
			mes("[Vahl]");
			mes("Leverage only works if you are willing to use it, and willing to survive what happens after you do. But yes. It is leverage.");
			break;
		}
		callfunc("DM_InstanceQuestStart", 20102);
		next();
		mes("^3355FF[Quest started: The Number Nobody Wants]^000000");
		mes("^3355FFConfront Director Krenn. Then descend to Juperos and face the Mistress.^000000");
		close();
	}

	if (questprogress(20102) == 1 && !dm_arc06_krenn_bribed && !dm_arc06_krenn_exposed) {
		mes("[Vahl]");
		mes("Krenn is the wall between the data and the city that needs it. He posts near the Grand Library most mornings.");
		close();
	}

	if ((questprogress(20102) == 1 || questprogress(20102) == 0) && (dm_arc06_krenn_bribed || dm_arc06_krenn_exposed)) {
		if (questprogress(20102) == 1) callfunc("DM_InstanceQuestComplete", 20102);
		mes("[Vahl]");
		mes("Krenn is dealt with. Now the fragment below needs dealing with.");
		next();
		mes("[Vahl]");
		mes("The Mistress. An old guardian spirit from the Juperos era — it was supposed to watch over the shard. The cascade has been feeding it. Whatever it is watching over now, it is no longer the shard.");
		next();
		mes("[Vahl]");
		mes("The entrance is below the plateau, south of the Republic's lower docks.");
		close();
	}

	if (.@paid) {
		mes("[Vahl]");
		mes("Bring back both sets of field data. The report writes itself after that.");
		close();
	}

	mes("[Vahl]");
	if (questprogress(20103) == 1 || questprogress(20104) == 1 || questprogress(20105) == 1)
		mes("The Harpy logs, the Juperos parts, and the Norgroad index are all field evidence. The shard readings do not improve while we wait.");
	else
		mes("Finish the contracts and come back. The number is waiting.");
	close();
}
