// ── Acolyte Naima — The Believer Who Looked Too Closely ────────────────────
rachel,120,120,5	script	Acolyte Naima#dm	4_F_04,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20131) == 2) {
		mes("[Acolyte Naima]");
		if (dm_arc09_karsh_exposed)
			mes("The truth is out. The Prelate is finished. The church will survive this. I have to believe that.");
		else
			mes("The bleeding stopped. The rift is quiet. Karsh walks free. I do not know if what we did was right.");
		close();
	}

	if (!dm_arc09_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20131, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20131, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20133, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20134, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20135, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("[Acolyte Naima]");
	mes("You are not pilgrims. Good. Every pilgrim I tell this to walks away faster.");
	next();
	if (dm_arc08_manfred_spared || dm_arc08_manfred_killed) {
		mes("[Naima]");
		mes("You came from Glast Heim. The kingdom that loved its faith to death. I have studied the accounts. Rachel is slower. But I think we are walking the same road.");
		next();
	}
	if (dm_arc08_darklord_killed) {
		mes("[Naima]");
		mes("What the cascade did to Glast Heim — amplifying a kingdom's collective despair into something that walks around — I think it is doing something similar here. Except the thing under the ice has been listening to prayers instead of grief.");
		next();
	}
	mes("[Naima]");
	mes("The Blessing of the Twelve. The sacred rift beneath the ice. Every child in Arunafeltz learns the story. A divine aperture where the Twelve's will pours into the world.");
	next();
	mes("[Naima]");
	mes("I took ice core samples three months ago. The rift is not divine. It is dimensional. It matches every measurement I have of the Ash Vacuum. And the church has been blessing pilgrims with its power for years without knowing what it is handing out.");
	next();
	switch (select("How long has the church known?", "What is Gloom Under Night?", "What do you need us to do?")) {
	case 1:
		mes("[Naima]");
		mes("Karsh has been sitting on my predecessor's identical findings for twenty years. The paperwork exists. He archived it under a false classification code. I found it.");
		break;
	case 2:
		mes("[Naima]");
		mes("The first thing that heard the prayers. When an entire civilization prays into a dimensional rift for four hundred years, something eventually answers. Gloom Under Night is that answer.");
		break;
	case 3:
		mes("[Naima]");
		mes("Three kinds of work, all of them quiet. Walk the pilgrim road. Cut ice cores. And sit in the confession corridor after dark. Evidence first. Theology later.");
		break;
	}
	next();
	mes("[Naima]");
	mes("The pilgrim road to the Sanctuary runs through Roween country — escort the caravans and keep your eyes open; every finished walk ends with a blessed bead I need tested. The ice cores grow in the Karu caverns where the Nereid nest. And the confession booths... clear the Vanberk from Sanctuary's first hall and listen to what the booths are keeping.");
	callfunc("DM_InstanceSetFlag", "dm_arc09_started", 1);
	callfunc("DM_InstanceQuestStart", 20131);
	callfunc("DM_InstanceQuestStart", 20133);
	callfunc("DM_InstanceQuestStart", 20134);
	callfunc("DM_InstanceQuestStart", 20135);
	next();
	mes("^3355FF[Quest started: The Descending Light]^000000");
	mes("^3355FFRoween Pilgrim Road - escort the pilgrims and clear 20 Roween from the Rachel fields.^000000");
	mes("^3355FFIce Core Collections - cut cores from the Karu ice; clear 20 Nereid.^000000");
	mes("^3355FFSanctuary Confessionals - clear 20 Vanberk and work the confession corridor.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20133) == 1 && questprogress(20133, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20133);
		callfunc("DM_PartyExp", 250000, 120000);
		callfunc("DM_GivePartyZeny", 20000);
		.@paid = 1;
		mes("[Naima]");
		mes("The caravans arrived whole. Here — one of the road beads, palmed from the blessing line. Test it against my instruments and it reads as a faint echo of the rift itself. The church is consecrating pilgrims with power it does not understand it is giving them.");
		next();
		switch (select("We walked at the pace of their weakest. They trusted us.", "We mined the road for rumors about the descending light.", "We used the road as bait — someone has been diverting pilgrims.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc09_pilgrim_trust", 1);
			mes("[Naima]");
			mes("Then they will talk to you again, and to me. A pilgrim who felt protected trusts what comes next. That trust is worth more than any sample I own.");
			break;
		case 2:
			mes("[Naima]");
			mes("Useful. The rumors line up with my readings. But the pilgrims arrived feeling handled rather than protected. They notice the difference. So do I.");
			break;
		case 3:
			mes("[Naima]");
			mes("And you caught one — an unlicensed 'healing' stop run off the clergy's books. Corruption in the middle ranks. I will add it to the file. The pilgrim you used as the lure will not walk this road again, though.");
			break;
		}
		next();
	}

	if (questprogress(20134) == 1 && questprogress(20134, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20134);
		callfunc("DM_PartyExp", 250000, 120000);
		callfunc("DM_GivePartyZeny", 20000);
		.@paid = 1;
		mes("[Naima]");
		mes("The cores. Look — this one does not melt. Does not warm. On my instruments it reads as a smaller echo of the bleed-site itself. The church calls this phenomenon holy and mines it like a resource. The workshop takes delivery by the crate.");
		next();
		switch (select("Give the cores to your healers. Treat the frostbite cases.", "Deliver the commissioned crates to the temple workshop.", "We destroyed the unstable ones in the field.")) {
		case 1:
			mes("[Naima]");
			mes("The pilgrim road produces shock and frostbite by the week, and the cold in these can be used safely if you know what it is. Real good, done quietly. Thank you.");
			break;
		case 2:
			mes("[Naima]");
			mes("Standard pay, standard paperwork. And one more batch of 'miracle' components for the thing they are building down there. I will not pretend that sits easily. You will see why soon.");
			break;
		case 3:
			mes("[Naima]");
			mes("No reward for that. But the workshop's manifest now has a gap where 'my' cores should be — and a paper gap is a question a prelate has to answer. Well played, if that was the intent.");
			break;
		}
		next();
	}

	if (questprogress(20135) == 1 && questprogress(20135, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20135);
		callfunc("DM_PartyExp", 250000, 120000);
		callfunc("DM_GivePartyZeny", 20000);
		.@paid = 1;
		mes("[Naima]");
		mes("The corridor is clear. Now the part I could not put in a contract. The confession booths keep no records — that is the promise. I have found dozens of slips. Private fears, transcribed by a junior prelate, cross-referenced against family names, debts, standing.");
		next();
		mes("[Naima]");
		mes("A filing system for blackmail, dressed as pastoral care. What did you do with the night's slips?");
		next();
		switch (select("Burned them. And warned the vulnerable ones directly.", "Copied them. They are evidence against Karsh's faction.", "We kept one junior priest's slip. He is... cooperative now.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc09_confessors_protected", 1);
			mes("[Naima]");
			mes("Then the faithful stay whole while the prelates fall. That is the only version of this I can live with. Rachel's people will remember who protected them.");
			break;
		case 2:
			mes("[Naima]");
			mes("Fast and damning. A case built on files we did not create but chose to use. I will take it. I will also remember what it cost the people in those booths.");
			break;
		case 3:
			dm_catechism_tally += 1;
			mes("[Naima]");
			mes("...");
			next();
			mes("^3355FF*Naima says nothing. Her silence is louder than a lecture: the party has run the prelates' own play, just once, for a good reason.*^000000");
			break;
		}
		next();
	}

	if (questprogress(20133) == 2 && questprogress(20134) == 2 && questprogress(20135) == 2 && questprogress(20132) == 0) {
		mes("[Naima]");
		mes("Road, ice, and corridor. Three threads, one knot. And my censer found where they tie together: a warded door in the Sanctuary's service tunnels. Not holy ground. Just a workshop.");
		next();
		mes("[Naima]");
		mes("Whatever the church's angel is, it is maintained down there. Go backstage, and see how the miracle is made.");
		callfunc("DM_InstanceQuestStart", 20132);
		next();
		mes("^3355FF[Quest started: The Angel in the Workshop]^000000");
		mes("^3355FFFind the Vesper workshop in the Sanctuary's service tunnels and learn what keeps the angel running.^000000");
		close();
	}

	if (questprogress(20132) == 1) {
		mes("[Naima]");
		mes("The workshop is behind the warded door in the service tunnels. There will be someone inside keeping it running. Remember — the person who maintains a lie is not always the person who told it.");
		close();
	}

	if (questprogress(20132) == 2 && !dm_arc09_karsh_exposed && !dm_arc09_karsh_deal) {
		mes("[Naima]");
		if (dm_arc09_vesper_disable)
			mes("You have the schematics and a witness. The miracle is manufacture, and we can prove it. Now Karsh. He is at his study in the Sanctuary. He will be expecting something. He is always expecting something.");
		else
			mes("The workshop is rubble and the evidence with it. Karsh will call it heretic vandalism, and half of Rachel will believe him. Go to him anyway. The ice cores and the road beads still make a case.");
		close();
	}

	if ((dm_arc09_karsh_exposed || dm_arc09_karsh_deal) && questprogress(20131) == 1) {
		mes("[Naima]");
		mes("Karsh is resolved. Whatever he chose, the record is what it is.");
		next();
		mes("[Naima]");
		mes("Gloom Under Night remains. It does not care about Karsh or the data or what we choose. It is four hundred years of answered prayers. It will take something harder than truth to quiet it.");
		close();
	}

	if (.@paid) {
		mes("[Naima]");
		mes("The other contracts still need finishing. The case is not complete without all three threads.");
		close();
	}

	mes("[Naima]");
	if (questprogress(20133) == 1 && questprogress(20134) == 1 && questprogress(20135) == 1)
		mes("The road, the ice, and the corridor are all still open. Quiet work, all three.");
	else if (questprogress(20133) == 1)
		mes("Roween Pilgrim Road is still open. Roween packs in the Rachel fields.");
	else if (questprogress(20134) == 1)
		mes("Ice Core Collections is still open. Nereid in the Karu ice caverns.");
	else if (questprogress(20135) == 1)
		mes("Sanctuary Confessionals is still open. Vanberk in the Sanctuary's first hall.");
	else
		mes("The data is waiting. Come back when the contracts are done.");
	close();
}

	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20132) != 1) {
		if (questprogress(20132) == 2) {
			mes("[Lay-Technician Osric]");
			if (dm_arc09_engineer_spared)
				mes("I keep the maintenance log now for your Naima. If the Council ever holds a hearing... I will say what I saw. All of it.");
			else
				mes("...");
			close();

		close();
	case 2:
		callfunc("DM_InstanceQuestComplete", 20132);
		callfunc("DM_InstanceSetFlag", "dm_arc09_workshop_smashed", 1);
		dm_catechism_tally += 1;
		callfunc("DM_PartyExp", 300000, 150000);
		callfunc("DM_GivePartyZeny", 30000);
		mes("^3355FF*It is fast, and it is cathartic, and the crash of the benches sounds exactly like the 'violent heretics' Karsh preaches about. Osric does not run. He stands by the door with his eyes shut until it is over.*^000000");
		next();
		mes("[Osric]");
		mes("The medicine came from that bench. But you know that. You decided it anyway. That is what deciding looks like, I suppose.");
		next();

// ── High Prelate Karsh — The Church's Secret Keeper ────────────────────────
rachel,130,120,5	script	High Prelate Karsh#dm	4_M_05,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc09_started) {
		mes("[High Prelate Karsh]");
		mes("The Sanctuary is not a public venue. Present credentials or leave.");
		close();
	}

	if (questprogress(20131) == 2) {
		mes("[High Prelate Karsh]");
		if (dm_arc09_karsh_exposed)
			mes("The Council of Arunafeltz has been informed. My statement stands. I was managing a sensitive theological matter responsibly.");
		else
			mes("An arrangement was reached. The church will manage the transition in its own way.");
		close();
	}

	if (dm_arc09_karsh_exposed || dm_arc09_karsh_deal) {
		mes("[High Prelate Karsh]");
		mes("We have concluded our business. Whatever you believe you have accomplished, consider carefully what happens to institutions when they are publicly embarrassed.");
		close();
	}

	if (questprogress(20132) != 2) {
		mes("[High Prelate Karsh]");
		mes("Naima's errand-runners. I know what she is collecting and I know what she thinks it proves. Come back when you have seen everything. I prefer arguments at full strength.");
		close();
	}

	mes("[High Prelate Karsh]");
	mes("Naima sent you. Of course she did.");
	next();
	if (dm_arc06_krenn_exposed) {
		mes("[Karsh]");
		mes("You were in Yuno as well. Krenn and I have mutual acquaintances. They tell me you have a pattern of... institutional interference.");
		next();
	}
	if (dm_arc08_manfred_spared) {
		mes("[Karsh]");
		mes("You left Manfred of Glast Heim alive. A mercy or a mistake. I will refrain from judging. The man spent thirty years in ruins because of his faith. I have some sympathy for that particular affliction.");
		next();
	}
	if (dm_arc09_vesper_disable) {
		mes("[Karsh]");
		mes("And you have been in my workshop. Yes, I know about the workshop. I signed for the crates. The angel comforts a hundred thousand people. You would be surprised what a man will sign for that.");
		next();
	} else if (dm_arc09_workshop_smashed) {
		mes("[Karsh]");
		mes("You destroyed the workshop. Do you know what that bought me? Every pulpit in Rachel, preaching about violent heretics. You have made my case better than I ever could.");
		next();
	}
	mes("[Karsh]");
	mes("I have the documentation Naima's predecessor produced. I have Naima's own notes from three months ago. I have twenty years of archives that say exactly what she claims they say.");
	next();
	mes("[Karsh]");
	mes("The question is what you intend to do with that fact.");
	next();
	switch (select("Sign a public confession. The people of Arunafeltz deserve the truth.", "What would it take for you to manage this transition yourself?", "We are not interested in the church. We need Gloom Under Night stopped.")) {
	case 1:
		callfunc("DM_Decide", "arc09.karsh", "exposed");
		mes("[Karsh]");
		mes("A public confession. Signed, dated, distributed. The Council will remove me. The church will enter a crisis it may not survive.");
		next();
		mes("[Karsh]");
		mes("...");
		next();
		mes("[Karsh]");
		mes("Give me the paper.");
		next();
		mes("^3355FF*Karsh signs. His hand does not tremble.*^000000");
		break;
	case 2:
		callfunc("DM_Decide", "arc09.karsh", "deal");
		mes("[Karsh]");
		mes("A managed transition. I formally reclassify the Blessing as a dimensional anomaly under theological study. The Council is briefed privately. No public confession. The institution survives.");
		next();
		mes("[Karsh]");
		mes("In return, Naima's findings are incorporated into official doctrine on a five-year timeline. And I retire from active duty at year's end.");
		next();
		mes("^3355FF*The deal is struck. The truth moves at the church's pace. That may be the best available outcome.*^000000");
		break;
	case 3:
		callfunc("DM_Decide", "arc09.karsh", "deal");
		mes("[Karsh]");
		mes("Then we have more in common than I expected. Gloom Under Night is what happens when four hundred years of prayer hits a dimensional aperture with nothing on the other side to receive it. I need it gone as much as you do.");
		next();
		mes("[Karsh]");
		mes("I will cooperate with whatever you need to descend into ice_dun03. The theology is a separate question. One problem at a time.");
		next();
		mes("^3355FF*Karsh's cooperation is practical rather than principled. But it opens the ice dungeon approach.*^000000");
		break;
	}
	next();
	mes("^3355FF[Karsh is resolved. Descend to ice_dun03 and face Gloom Under Night.]^000000");
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}
