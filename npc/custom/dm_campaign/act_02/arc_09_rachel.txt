// ── Shared Arc 9 helpers ───────────────────────────────────────────────────

function	script	DM_Arc09SecureWitness	{
	callfunc("DM_InstanceSetFlag", "dm_arc09_witness_secured", getarg(0));
	callfunc("DM_DMJStoryObjective", 20134, 2, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc09_witness"))
		callfunc("DM_PartyExp", 156000, 75000);
	callfunc("DM_PartyMes", "^3355FF[Naima is secured and her testimony is on record. Return to her.]^000000");
	return 1;
}

// mode: 1 disclose now, 2 supervised evacuation then disclose, 3 keep the secret.
function	script	DM_Arc09Commit	{
	.@mode = getarg(0);
	if (dm_arc09_disclosure_schedule)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc09_disclosure_schedule", .@mode);
	if (.@mode == 1) {
		callfunc("DM_InstanceSetFlag", "dm_arc09_karsh_exposed", 1);
	} else if (.@mode == 2) {
		callfunc("DM_InstanceSetFlag", "dm_arc09_approach_safe", 1);
	} else {
		callfunc("DM_InstanceSetFlag", "dm_arc09_karsh_deal", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc09_secret_kept", 1);
	}
	callfunc("DM_InstanceQuestComplete", 20134);
	callfunc("DM_DMJStoryObjective", 20134, 3, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc09_resolution"))
		callfunc("DM_PartyExp", 208000, 100000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 905);
	return 1;
}

	}
	close();
