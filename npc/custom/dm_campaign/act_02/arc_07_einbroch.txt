// Open the labor decision the first time two preparations are done, and pay a
// bonus for the third.
function	script	DM_Arc07AddPrep	{
	.@bit = getarg(0);
	if (!(dm_arc07_prep_mask & .@bit))
		callfunc("DM_InstanceSetFlag", "dm_arc07_prep_mask", dm_arc07_prep_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc07_prep_mask);
	callfunc("DM_DMJStoryObjective", 20111, 2, "Explore", .@n, 2, (.@n >= 2), getcharid(CHAR_ID_PARTY));
	if (.@n == 2) {
		if (callfunc("DM_ClaimGrant", "arc07_prep")) {
			callfunc("DM_PartyExp", 360000, 171000);
			callfunc("DM_InstanceQuestStart", 20112);
			callfunc("DM_PartyMes", "^3355FF[Two preparations done. Greta can put the decision to the foundry council.]^000000");
		}
	} else if (.@n >= 3) {
		if (callfunc("DM_ClaimGrant", "arc07_prep_bonus"))
			callfunc("DM_PartyExp", 60000, 28000);
		callfunc("DM_PartyMes", "^3355FF[All three preparations done.]^000000");
	} else {
		callfunc("DM_PartyMes", "^3355FF[One preparation done. One more opens the decision.]^000000");
	}
	return .@n;
}

// The one commitment path. mode: 1 support, 2 monitored shutdown, 3 enforce.
// First commitment is final, and exactly one strike flag ends up set.
function	script	DM_Arc07Commit	{
	.@mode = getarg(0);
	if (dm_arc07_labor_resolution)
		return 0;
	callfunc("DM_InstanceSetFlag", "dm_arc07_labor_resolution", .@mode);
	if (.@mode == 1)
		callfunc("DM_InstanceSetFlag", "dm_arc07_strike_supported", 1);
	else if (.@mode == 2)
		callfunc("DM_InstanceSetFlag", "dm_arc07_strike_negotiated", 1);
	else
		callfunc("DM_InstanceSetFlag", "dm_arc07_strike_broken", 1);
	// Presenting the order is how an accusation becomes a fact on the record.
	if (dm_arc07_prep_mask & 4)
		callfunc("DM_InstanceSetFlag", "dm_arc07_kessler_confronted", 1);
	callfunc("DM_InstanceQuestComplete", 20112);
	callfunc("DM_DMJStoryObjective", 20112, 1, "Interact", 1, 1, 1, getcharid(CHAR_ID_PARTY));
	if (callfunc("DM_ClaimGrant", "arc07_resolution"))
		callfunc("DM_PartyExp", 240000, 114000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 705);
	return 1;
}

ein_dun02,150,150,0	script	Reactivation Bay#dm	HIDDEN_NPC,{
	end;

OnPressureStart:
	stopnpctimer;
	'pressure_map$ = strnpcinfo(NPC_MAP);
	'pressure_party_id = $@dm_arc07_pressure_party_id;
	'pressure_damage = $@dm_arc07_pressure_damage;
	if ('pressure_party_id > 0) {
		mapannounce 'pressure_map$, "[Reactivation Bay] Smoke stacks vent across the mine floor as RSX-0806 comes online.", bc_map, 0x999999;
		initnpctimer;
	}
	end;

OnPressureStop:
	stopnpctimer;
	end;

OnTimer5000:
	mapannounce 'pressure_map$, "[Reactivation Bay] Hot smoke rolls between the rail pylons.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 7, 'pressure_damage, SC_BLIND, 4000, 'pressure_party_id);
	end;

	mapannounce 'pressure_map$, "[Reactivation Bay] The old boilers cough dust and machine oil.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 7, 'pressure_damage, SC_BLIND, 4000, 'pressure_party_id);
	end;

OnTimer15000:
	mapannounce 'pressure_map$, "[Reactivation Bay] A pressure valve bursts, then seals itself with a shriek.", bc_map, 0x999999;
	callfunc("DM_HazardArea", 'pressure_map$, 150, 150, 7, 'pressure_damage, SC_BLIND, 4000, 'pressure_party_id);
	stopnpctimer;
	end;

OnRSXDead:
	.@party = callfunc("DM_SessionParty");
	if (!callfunc("DM_ClaimGrant", "arc07_rsx", .@party))
		end;
	stopnpctimer;
	callfunc("DM_InstanceQuestComplete", 20111);
	callfunc("DM_InstanceSetFlag", "dm_arc07_rsx_killed", 1);
	callfunc("DM_PartyExp", 600000, 350000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 799);
	instance_announce(instance_id(), "^3355FF[DM] RSX-0806 collapses in a shower of sparks. The machine that wouldn't stay dead is finally quiet.^000000", bc_all);
	sleep(2000);
	if (dm_arc07_kessler_confronted)
		instance_announce(instance_id(), "^FF6600[DM] The reactivation order traces to Lighthalzen. All roads in Act II lead north.^000000", bc_all);
	else
		instance_announce(instance_id(), "^FF6600[DM] The clearance logs on the machine trace upward. Rekenber is connected to the cascade... and all roads lead to Lighthalzen.^000000", bc_all);
	end;
}
