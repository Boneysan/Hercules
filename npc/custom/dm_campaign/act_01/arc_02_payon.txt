// Sun-Hwa - shrine-medium and Arc 2 quest-giver
payon,181,104,4	script	Sun-Hwa#dm	4_F_THAISHAMAN,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (questprogress(20012) == 2) {
		if (questprogress(20007) == 1)
			callfunc("DM_Arc02Complete");
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		if (dm_arc02_grove_restored)
			callfunc("DM_PartyMes", "The incense burns warm again. Not bright. Warm is enough.");
		else
			callfunc("DM_PartyMes", "Payon is quiet. That is not the same thing as peace, but it is what remains.");
		next();
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "Keep Voss's notebook. What he wrote in clean ink, others will try in blood.");
		close();
	}

	if (!dm_arc02_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20007, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20007, 1;
	end;

S_StartArc:
	if (!callfunc("DM_SceneClaim", "arc02_sunhwa")) {
		callfunc("DM_SceneFail", "arc02_sunhwa");
		close();
	}
	callfunc("DM_PartyMes", "[Sun-Hwa]");
	callfunc("DM_PartyMes", "You carry a cold thing.");
	next();
	if (countitem(50001) > 0 || dm_arc01_sigil_ring_obtained) {
		callfunc("DM_PartyMes", "^3355FF*The Sigil Ring in your pack chills until your fingers ache. It points down - toward Payon's cave-tombs.*^000000");
		next();
	}
	callfunc("DM_PartyMes", "[Sun-Hwa]");
	callfunc("DM_PartyMes", "I am Sun-Hwa, voice to the ancestors. Lately, the dead speak without being asked.");
	next();
	callfunc("DM_PartyMes", "[Sun-Hwa]");
	callfunc("DM_PartyMes", "A scholar named Voss came to honor our rites. Since then, graves empty, lanterns burn blue, and the cave breathes like a wounded animal.");
	next();
	switch (select("We'll bring peace to them.", "What does Payon pay for proof?", "Tell us why they are waking.", "We need to discuss this.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc02_rite_path", 1);
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "Then we begin with respect. The dead answer slowly to those who do not command them.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc02_rite_path", 2);
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "The Guild pays for work. Payon pays for care. Do both and no one will call you mercenary.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc02_rite_path", 2);
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "Someone is harvesting the oldest souls first. To know who, we must ask one of them directly.");
		break;
	case 4:
		callfunc("DM_SceneRelease", "arc02_sunhwa");
		close();
	}
	next();
	callfunc("DM_PartyMes", "[Sun-Hwa]");
	callfunc("DM_PartyMes", "Walk the memorials. Two names, spoken aloud, will open the grove. The mushroom rings and bone tags are coin, not keys.");
	callfunc("DM_InstanceSetFlag", "dm_arc02_started", 1);
	callfunc("DM_InstanceQuestStart", 20007);
	next();
	callfunc("DM_SceneRelease", "arc02_sunhwa");
	callfunc("DM_PartyMes", "^3355FF[Quest started: The Sleeping Forest]^000000");
	callfunc("DM_PartyMes", "^3355FFRead two of the three memorials near this shrine, or speak to the family witness. Walk the blue lanterns. Optional contracts pay. They do not name the dead.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20008) == 1 && callfunc("DM_HuntReady", 20008)) {
		callfunc("DM_HuntCollect", 20008);
		callfunc("DM_InstanceQuestComplete", 20008);
		callfunc("DM_PartyExp", 7000, 2800);
		callfunc("DM_GivePartyZeny", 2500);
		callfunc("DM_GivePartyItem", Strawberry, 5);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "The mushroom ring is clear. Keep the charm; proof that even field work can disturb a sacred place.");
		next();
		switch (select("Clear the ring and burn the rot.", "Preserve samples for Sun-Hwa.", "Let Voss inspect the samples.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc02_mushroom_choice", 1);
			callfunc("DM_PartyMes", "[Sun-Hwa]");
			callfunc("DM_PartyMes", "A clean answer. Not always a complete one.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc02_mushroom_choice", 2);
			callfunc("DM_PartyMes", "[Sun-Hwa]");
			callfunc("DM_PartyMes", "Careful hands. These spores carry grave-cold sap.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc02_mushroom_choice", 3);
			callfunc("DM_PartyMes", "[Sun-Hwa]");
			callfunc("DM_PartyMes", "Do not hand knives to people who call wounds data.");
			break;
		}
		next();
	}
	if (questprogress(20009) == 1 && callfunc("DM_HuntReady", 20009)) {
		callfunc("DM_HuntCollect", 20009);
		callfunc("DM_InstanceQuestComplete", 20009);
		callfunc("DM_PartyExp", 9000, 3500);
		callfunc("DM_GivePartyZeny", 3000);
		callfunc("DM_GivePartyItem", Skel_Bone, 3);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "The bone tags are names. Not trophies. Names.");
		next();
		switch (select("Return the tags to their families.", "Sell them for faster gear.", "Give the tags to Voss.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc02_bone_tags_choice", 1);
			callfunc("DM_PartyMes", "[Sun-Hwa]");
			callfunc("DM_PartyMes", "Then one ancestor will answer willingly when the time comes.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc02_bone_tags_choice", 2);
			callfunc("DM_PartyMes", "[Sun-Hwa]");
			callfunc("DM_PartyMes", "Payon remembers prices. So do the dead.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc02_bone_tags_choice", 3);
			callfunc("DM_PartyMes", "[Sun-Hwa]");
			callfunc("DM_PartyMes", "He will thank you in ink. That should worry you.");
			break;
		}
		next();
	}

	if (questprogress(20010) == 0) {
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "The lanterns along this road burn blue. Walk them. I will not relight them from this cushion.");
		next();
		if (select("We'll walk the lanterns.", "Later.") == 1) {
			callfunc("DM_InstanceQuestStart", 20010);
			callfunc("DM_PartyMes", "^3355FF[Lanterns for the Lost. Three blue lanterns around this shrine. Two are enough.]^000000");
			close();
		}
	}

	if (questprogress(20011) == 0) {
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "A family waits by the east memorial. Their grandmother's grave is empty. The comb is not in this room.");
		next();
		if (select("We'll go to the family.", "Later.") == 1) {
			callfunc("DM_InstanceQuestStart", 20011);
			callfunc("DM_PartyMes", "^3355FF[The Emptied Graves. Speak to the cliff family, then the empty stone.]^000000");
			close();
		}
	}

	.@bits = dm_arc02_memorial_known;
	.@known = 0;
	if (.@bits & 1) ++.@known;
	if (.@bits & 2) ++.@known;
	if (.@bits & 4) ++.@known;

	if (.@known >= 2 && questprogress(20012) == 0) {
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "Two names. That is enough. The grove is in the lowest cave. Voss is feeding a Flower that was never meant to bloom underground.");
		next();
		if (dm_arc02_ancestor_helped) {
			callfunc("DM_PartyMes", "[Sun-Hwa]");
			callfunc("DM_PartyMes", "Because you returned the grandmother whole, she will answer without force.");
			callfunc("DM_InstanceSetFlag", "dm_arc02_rite_path", 1);
		} else {
			switch (select("Respect the taboo; let hints guide us.", "Ask Sun-Hwa to carry the mark. We understand the cost.", "We'll do the rite ourselves - tell us the price first.", "Later.")) {
			case 1:
				callfunc("DM_InstanceSetFlag", "dm_arc02_rite_path", 1);
				callfunc("DM_PartyMes", "[Sun-Hwa]");
				callfunc("DM_PartyMes", "Then we go slowly.");
				break;
			case 2:
				callfunc("DM_PartyMes", "[Sun-Hwa]");
				callfunc("DM_PartyMes", "The mark does not wash off. I will still do it if you ask me plainly.");
				next();
				if (select("We ask. Carry it.", "No. We find another way.") == 1) {
					callfunc("DM_InstanceSetFlag", "dm_arc02_rite_path", 2);
					callfunc("DM_InstanceSetFlag", "dm_arc02_sunhwa_marked", 1);
					callfunc("DM_PartyMes", "[Sun-Hwa]");
					callfunc("DM_PartyMes", "Then I carry it. Let the ancestors judge me kindly.");
				}
				break;
			case 3:
				callfunc("DM_PartyMes", "[Sun-Hwa]");
				callfunc("DM_PartyMes", "If you take the rite, you take the mark. That is the cost. It is not a door you walk through twice.");
				next();
				if (select("We accept the mark.", "Not that cost.") == 1) {
					callfunc("DM_InstanceSetFlag", "dm_arc02_rite_path", 3);
					callfunc("DM_InstanceSetFlag", "dm_arc02_sunhwa_marked", 1);
				}
				break;
			case 4:
				close();
			}
		}
		next();
		callfunc("DM_InstanceQuestStart", 20012);
		callfunc("DM_PartyMes", "^3355FF[Quest started: What the Ancestors Won't Say]^000000");
		callfunc("DM_PartyMes", "^3355FFPayon Cave F5. Cut the grove conduits you can see before you choose restore or burn.^000000");
		close();
	}

	switch (select("How is the investigation?", "Take a field contract.", "That's all.")) {
	case 1:
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", callfunc("DM_Act01BoardLine"));
		callfunc("DM_HuntStanding", 20008, 20009);
		close();
	case 2:
		callfunc("DM_PartyMes", "[Sun-Hwa]");
		callfunc("DM_PartyMes", "Optional. The dead do not need mushroom spores.");
		next();
		switch (select("Mushroom Ring Patrol.", "Bone Tag Turn-In.", "Neither.")) {
		case 1:
			if (questprogress(20008) == 0) {
				callfunc("DM_InstanceQuestStart", 20008);
				callfunc("DM_PartyMes", "^3355FF[Optional: " + callfunc("DM_HuntBrief", 20008) + "]^000000");
			}
			close();
		case 2:
			if (questprogress(20009) == 0) {
				callfunc("DM_InstanceQuestStart", 20009);
				callfunc("DM_PartyMes", "^3355FF[Optional: " + callfunc("DM_HuntBrief", 20009) + "]^000000");
			}
			close();
		case 3:
			close();
		}
	case 3:
		close();
	}
}

// Scholar Voss - human villain and Moonlight Flower set-piece
pay_dun04,120,115,4	script	Scholar Voss#dm	4_M_SCIENCE,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (questprogress(20012) != 1) {
		if (questprogress(20012) == 2) {
			callfunc("DM_PartyMes", "^3355FF*The grove is bruised but quiet. Voss's neat chalk circles are scuffed into useless dust.*^000000");
			close();
		}
		callfunc("DM_PartyMes", "[Scholar Voss]");
		callfunc("DM_PartyMes", "Not now. Field work is most easily spoiled by footsteps.");
		close();
	}
	if ('boss_up) {
		callfunc("DM_PartyMes", "[Scholar Voss]");
		callfunc("DM_PartyMes", "Do try not to contaminate the sample while it is blooming.");
		close();
	}

	callfunc("DM_PartyMes", "^3355FF*Roots pulse under the cave floor. Moonlight gathers in the grove like a wound pretending to be beautiful.*^000000");
	next();
	callfunc("DM_PartyMes", "[Scholar Voss]");
	callfunc("DM_PartyMes", "Ah. Prontera's sewer party. Cassell said the Guild might send dogs after the second trial.");
	next();
	if (!callfunc("DM_SceneClaim", "arc02_voss")) {
		callfunc("DM_SceneFail", "arc02_voss");
		close();
	}
	switch (select("These were people, Voss.", "Walk away and leave the notes.", "Show us how the harvest works.", "We need to discuss this.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc02_voss_approach", 1);
		callfunc("DM_PartyMes", "[Voss]");
		callfunc("DM_PartyMes", "These were data. People are the storage medium.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc02_voss_approach", 2);
		callfunc("DM_PartyMes", "[Voss]");
		callfunc("DM_PartyMes", "A practical offer. I keep my life, you keep the notebook.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc02_voss_approach", 3);
		callfunc("DM_PartyMes", "[Voss]");
		callfunc("DM_PartyMes", "A bound spirit resonates with the seal when fed memory-rich souls. Acoustics, not theology.");
		break;
	case 4:
		callfunc("DM_SceneRelease", "arc02_voss");
		close();
	}
	next();
	if (!dm_arc02_notebook_obtained) {
		callfunc("DM_InstanceSetFlag", "dm_arc02_notebook_obtained", 1);
		callfunc("DM_PartyMes", "^3355FF[Voss's Notebook obtained.]^000000");
		callfunc("DM_PartyMes", "^3355FFThe final page mentions: 'the cold north, at scale.'^000000");
		next();
	}

	if (!dm_arc02_conduits_severed) {
		callfunc("DM_PartyMes", "^777777The blue roots in this grove can be cut at the two conduit stones. Doing it here from a menu is the rushed version.^000000");
		next();
		switch (select("Sever the soul-conduits first.", "Rush the Flower now.", "We need to discuss this.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc02_conduits_severed", 1);
			callfunc("DM_PartyMes", "^3355FF*You cut the remaining roots. The Flower shrieks, starving.*^000000");
			break;
		case 2:
			callfunc("DM_PartyMes", "^3355FF*The grove still drinks. This will hurt.*^000000");
			break;
		case 3:
			callfunc("DM_SceneRelease", "arc02_voss");
			close();
		}
		next();
	}

	switch (select("Lay the Flower to rest.", "Burn the grove and end it.", "We need to discuss this.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc02_grove_restored", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc02_grove_burned", 0);
		callfunc("DM_PartyMes", "^3355FF*Sun-Hwa's rite anchors the stolen souls. Mercy has weight; someone has to carry it.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc02_grove_burned", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc02_grove_restored", 0);
		callfunc("DM_PartyMes", "^3355FF*Fire is quick. Not all things released by fire are freed.*^000000");
		break;
	case 3:
		callfunc("DM_SceneRelease", "arc02_voss");
		close();
	}
	next();
	callfunc("DM_SceneTouch", "arc02_voss");

	.@map$ = strnpcinfo(NPC_MAP);
	getmapxy(.@m$, .@vx, .@vy, UNITTYPE_NPC);
	.@adds = 5;
	if (dm_arc02_conduits_severed)
		.@adds -= 2;
	if (dm_arc02_ancestor_helped)
		.@adds -= 1;
	if (dm_arc02_voss_approach != 1)
		.@adds -= 1;
	if (.@adds < 1)
		.@adds = 1;

	'boss_up = 1;
	.@self$ = strnpcinfo(NPC_NAME);
	.@boss$ = .@self$ + "::OnMoonlightDead";
	.@add$ = .@self$ + "::OnAddKilled";
	callfunc("DM_EncStart", "arc02_voss", getcharid(CHAR_ID_PARTY), .@map$, .@boss$, .@add$, .@self$);
	if (dm_arc02_conduits_severed)
		mapannounce .@map$, "[Moonlit Grove] The conduits snap. The Flower fights hungry and diminished.", bc_map, 0x88CCFF;
	else
		mapannounce .@map$, "[Moonlit Grove] The conduits flare. The Flower drinks from every open grave.", bc_map, 0xCC66FF;

	.@pct = 100;
	if (dm_arc02_conduits_severed)
		.@pct = 70;
	callfunc("DM_EncMonster", .@map$, .@vx, .@vy, "Moonlight Flower", MOONLIGHT,
		.@boss$, 28, .@pct, 1, 0);
	if (.@adds > 0)
		callfunc("DM_EncMonster", .@map$, .@vx, .@vy, "Restless Dead", SOHEE,
			.@add$, 28, .@pct, .@adds, 5);
	if (dm_arc02_voss_approach == 1)
		callfunc("DM_EncMonster", .@map$, .@vx, .@vy, "Voss's Raised Guard", SOLDIER_SKELETON,
			.@add$, 28, .@pct, 2, 4);
	close();

OnAddKilled:
	end;

OnMoonlightDead:
	if (!callfunc("DM_EncOnDeath", "arc02_voss"))
		end;
	.@party = getd("$dm_enc_p_arc02_voss");
	if (.@party <= 0)
		end;
	if (dm_arc02_voss_approach == 1) {
		callfunc("DM_InstanceSetFlag", "dm_arc02_voss_killed", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc02_voss_spared", 0);
	} else {
		callfunc("DM_InstanceSetFlag", "dm_arc02_voss_spared", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc02_voss_killed", 0);
	}
	if (!callfunc("DM_ClaimGrant", "arc02_moonlight", .@party)) {
		callfunc("DM_EncCleanup", "arc02_voss");
		end;
	}
	'boss_up = 0;
	.@map$ = strnpcinfo(NPC_MAP);
	killmonster .@map$, strnpcinfo(NPC_NAME)+"::OnAddKilled";
	callfunc("DM_PartyExp", 44000, 18000, .@party);
	callfunc("DM_Arc02Complete", .@party);
	if (dm_arc02_grove_restored)
		mapannounce .@map$, "[Moonlit Grove] Silver bells fade. The stolen souls find the shrine road home.", bc_map, 0x88CCFF;
	else
		mapannounce .@map$, "[Moonlit Grove] The grove burns clean. Payon's dead go silent all at once.", bc_map, 0xFFAA33;
	dispbottom "[Seal Cascade] Arc 2 complete. Voss's notebook names the pattern: bound spirits fed with souls to weaken seals.";
	end;

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20012, 1;
	end;
}

function	script	DM_Arc02Lantern	{
	.@bit = getarg(0);
	if (!callfunc("DM_SessionAllows") || !dm_arc02_started) {
		close();
		return 0;
	}
	if (dm_arc02_lantern_mask & .@bit) {
		callfunc("DM_PartyMes", "^3355FF*This lantern already remembers warmth.*^000000");
		close();
		return 0;
	}
	if (questprogress(20010) == 0)
		callfunc("DM_InstanceQuestStart", 20010);
	callfunc("DM_PartyMes", "^3355FF*Blue fire. A name trying to find the road home.*^000000");
	next();
	if (select("Relight it warm.", "Leave it.") == 2) {
		close();
		return 0;
	}
	callfunc("DM_InstanceSetFlag", "dm_arc02_lantern_mask", dm_arc02_lantern_mask | .@bit);
	.@n = callfunc("DM_CountBits", dm_arc02_lantern_mask);
	callfunc("DM_PartyMes", "^3355FF[Lantern " + .@n + "/3.]^000000");
	if (.@n >= 2 && questprogress(20010) == 1) {
		next();
		switch (select("Guide the spirits home.", "Leave one dark as a trap for Voss.", "Keep a cold lantern.", "Later.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc02_lantern_choice", 1);
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc02_lantern_choice", 2);
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc02_lantern_choice", 3);
			callfunc("DM_GivePartyItem", Lantern, 1);
			break;
		case 4:
			close();
			return 0;
		}
		callfunc("DM_InstanceQuestComplete", 20010);
		if (callfunc("DM_ClaimGrant", "arc02_lanterns")) {
			callfunc("DM_PartyExp", 6000, 2500);
			callfunc("DM_GivePartyZeny", 1500);
		}
		callfunc("DM_PartyMes", "^3355FF[Lanterns for the Lost complete.]^000000");
	}
	close();
	return 1;
}
