// Captain Mara - ally, ferry captain, and Arc 5 quest-giver
alberta,184,150,4	script	Captain Mara#dm	4_F_01,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (questprogress(20030) == 2) {
		callfunc("DM_PartyMes", "[Captain Mara]");
		if (dm_arc05_manifest_taken)
			callfunc("DM_PartyMes", "Every port on Brode's ledger is a storm waiting for a name.");
		else
			callfunc("DM_PartyMes", "The deep is quiet. I do not trust quiet water, but I will take it.");
		close();
	}

	if (!dm_arc05_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20025, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20025, 1;
	end;

S_StartArc:
	callfunc("DM_PartyMes", "[Captain Mara]");
	callfunc("DM_PartyMes", "Welcome to Alberta. Smells like salt, tar, fish, and men pretending cargo has no conscience.");
	next();
	if (countitem(50001) > 0 || dm_arc01_sigil_ring_obtained) {
		callfunc("DM_PartyMes", "^3355FF*The Sigil Ring sweats cold at the waterline. The cold moves with the tide.*^000000");
		next();
	}
	if (dm_arc03_sabra_deal) {
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "Sabra's people are aboard the Saltmother. Alive, because somebody paid for mercy before it was profitable.");
		next();
	} else if (dm_arc03_sabra_exposed) {
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "Some of Morroc's survivors made it. Starving. Angry. Justice must have looked cleaner from shore.");
		next();
	}
	callfunc("DM_PartyMes", "[Captain Mara]");
	callfunc("DM_PartyMes", "Brode owns the blind spots between Izlude and Alberta. I know them because I used to sail them.");
	next();
	switch (select("We're here to stop smuggling, not refugees.", "Just point us at Brode.", "Stay out of our way, smuggler.", "We need to discuss this.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc05_mara_trust", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_mara_network", 1);
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "Good. Then my crew is yours, and every honest deckhand I know hears your name kindly.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc05_mara_pragmatic", 1);
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "Brode's counting-house. Locked holds. Inspection exemptions. That is the route.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc05_mara_insulted", 1);
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "I've heard worse from better sailors. You get directions, not shelter.");
		break;
	case 4:
		close();
	}
	next();
	callfunc("DM_PartyMes", "[Captain Mara]");
	callfunc("DM_PartyMes", "Do the harbor work first. Ferry tokens, Byalan contract tags, and Sunken Ship pages will show you how Brode moves people like freight.");
	callfunc("DM_InstanceSetFlag", "dm_arc05_started", 1);
	callfunc("DM_InstanceQuestStart", 20025);
	callfunc("DM_InstanceQuestStart", 20026);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 501);
	next();
	callfunc("DM_PartyMes", "^3355FF[Quest started: Tides and Trade]^000000");
	callfunc("DM_PartyMes", "^3355FFTwo ferry departures at the pier. Two diver sites. Then Brode's hold. Optional hunts pay; they do not open the lock.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20026) == 1) {
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "The pier handles departures. I will not pretend a conversation is a sailing. Go to the Saltmother's plank.");
		next();
	}

	if (questprogress(20027) == 1 && callfunc("DM_HuntReady", 20027)) {
		callfunc("DM_HuntCollect", 20027);
		callfunc("DM_InstanceQuestComplete", 20027);
		callfunc("DM_PartyExp", 70000, 32000);
		callfunc("DM_GivePartyZeny", 8500);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "Salt-eaten contract tags. Byalan is not just dangerous now. It is organized wrong.");
		next();
		switch (select("Clear sea monsters.", "Rescue trapped divers.", "Trace cult crates underwater.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc05_byalan_choice", 1);
			callfunc("DM_PartyMes", "[Captain Mara]");
			callfunc("DM_PartyMes", "Clean lanes save lives. Boring heroism is still heroism.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc05_byalan_choice", 2);
			callfunc("DM_InstanceSetFlag", "dm_arc05_divers_rescued", 1);
			callfunc("DM_PartyMes", "[Captain Mara]");
			callfunc("DM_PartyMes", "Those divers will drink to your names until memory improves them.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc05_byalan_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc05_sea_cult_evidence", 1);
			callfunc("DM_PartyMes", "[Captain Mara]");
			callfunc("DM_PartyMes", "Crates below the tide. Brode's kind always thinks water hides invoices.");
			break;
		}
		next();
	}

	if (questprogress(20028) == 1 && callfunc("DM_HuntReady", 20028)) {
		callfunc("DM_HuntCollect", 20028);
		callfunc("DM_InstanceQuestComplete", 20028);
		callfunc("DM_PartyExp", 76000, 35000);
		callfunc("DM_GivePartyZeny", 9000);
		.@paid = 1;
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "Waterlogged manifest pages. Brode's ink runs, but his greed stays legible.");
		next();
		switch (select("Return cargo records.", "Sell salvage.", "Expose Brode early.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc05_manifest_choice", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc05_records_returned", 1);
			callfunc("DM_PartyMes", "[Captain Mara]");
			callfunc("DM_PartyMes", "Records go home. Some families will finally have names for grief.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc05_manifest_choice", 2);
			callfunc("DM_PartyMes", "[Captain Mara]");
			callfunc("DM_PartyMes", "Salvage pays. Just do not become the sort of person Brode would hire.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc05_manifest_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc05_brode_exposed_early", 1);
			callfunc("DM_PartyMes", "[Captain Mara]");
			callfunc("DM_PartyMes", "Then his lawyers wake before his conscience. We move tonight.");
			break;
		}
		next();
	}

	if (dm_arc05_departures_used >= 1 && questprogress(20029) == 0) {
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "Brode has one hold listed as ballast. Ballast does not knock back.");
		next();
		switch (select("Start Cargo of Souls.", "Later.")) {
		case 1:
			callfunc("DM_InstanceQuestStart", 20029);
			callfunc("DM_PartyApplyFlag", "dm_story_beat", 529);
			callfunc("DM_PartyMes", "^3355FF[Quest started: Cargo of Souls]^000000");
			callfunc("DM_PartyMes", "^3355FFFind the inspection-exempt hold on the Sunken Ship route.^000000");
			close();
		case 2:
			close();
		}
	}

	if (questprogress(20029) == 2 && questprogress(20030) == 0) {
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", "The ledger says Brode dumped heavy crates past Byalan. The sea turned wrong after that.");
		next();
		switch (select("Start What Turned Over in the Deep.", "Later.")) {
		case 1:
			callfunc("DM_InstanceQuestStart", 20030);
			callfunc("DM_PartyApplyFlag", "dm_story_beat", 530);
			callfunc("DM_PartyMes", "^3355FF[Quest started: What Turned Over in the Deep]^000000");
			callfunc("DM_PartyMes", "^3355FFDive to Turtle Island's deep shelf and stop what the cargo woke.^000000");
			close();
		case 2:
			close();
		}
	}

	switch (select("How is the investigation?", "Take a field contract.", "That's all.")) {
	case 1:
		callfunc("DM_PartyMes", "[Captain Mara]");
		callfunc("DM_PartyMes", callfunc("DM_Act01BoardLine"));
		callfunc("DM_HuntStanding", 20027, 20028);
		close();
	case 2:
		next();
		switch (select("Byalan Tide Contract.", "Sunken Ship Manifest.", "Neither.")) {
		case 1:
			if (questprogress(20027) == 0) callfunc("DM_InstanceQuestStart", 20027);
			close();
		case 2:
			if (questprogress(20028) == 0) callfunc("DM_InstanceQuestStart", 20028);
			close();
		case 3:
			close();
		}
	case 3:
		close();
	}
}

// Smuggler-Baron Brode - logistics villain
alberta,101,65,4	script	Smuggler-Baron Brode#dm	4_M_03,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (!dm_arc05_started) {
		callfunc("DM_PartyMes", "[Brode]");
		callfunc("DM_PartyMes", "Appointments, invoices, or threats. If you have none of those, leave.");
		close();
	}

	if (questprogress(20029) == 2) {
		callfunc("DM_PartyMes", "[Brode]");
		if (dm_arc05_brode_sold_manifest)
			callfunc("DM_PartyMes", "You bought a ledger. Do not pretend that makes you cleaner than the man who sold it.");
		else if (dm_arc05_brode_threatened)
			callfunc("DM_PartyMes", "Sink one ship and ten hungry men buy rope. Remember that.");
		else if (dm_arc05_brode_ruined)
			callfunc("DM_PartyMes", "You ruined me. That means I have time now.");
		else
			callfunc("DM_PartyMes", "Freight moves. Heroes shout. Ports stay hungry.");
		close();
	}

	if (dm_arc05_departures_used < 1) {
		callfunc("DM_PartyMes", "[Brode]");
		callfunc("DM_PartyMes", "The Saltmother hasn't even left. Come back when a departure is on the water. I don't sell ledgers to tourists.");
		close();
	}
	callfunc("DM_PartyMes", "[Brode]");
	callfunc("DM_PartyMes", "People keep saying cult like that changes the bill of lading. I ship freight. Freight pays.");
	next();
	if (dm_arc05_manifest_taken) {
		callfunc("DM_PartyMes", "[Brode]");
		callfunc("DM_PartyMes", "You already bought or stole the paper. Stop shopping.");
		close();
	}
	if (!callfunc("DM_SceneClaim", "arc05_brode")) {
		callfunc("DM_SceneFail", "arc05_brode");
		close();
	}
	switch (select("You're trafficking people for a death-cult.", "We'll outbid the cult for the ledger.", "Talk, or we sink everything you own.", "We need to discuss this.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc05_brode_ruined", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_brode_enemy", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_manifest_taken", 1);
		callfunc("DM_PartyMes", "[Brode]");
		callfunc("DM_PartyMes", "Then come take your righteous inventory. The cargo's conscience was never on my manifest.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc05_brode_sold_manifest", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_manifest_taken", 1);
		callfunc("DM_PartyMes", "[Brode]");
		callfunc("DM_PartyMes", "Gold for paper. Finally, someone speaks an honest language.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc05_brode_threatened", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_brode_enemy", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_manifest_taken", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_northern_account", 1);
		callfunc("DM_PartyMes", "[Brode]");
		callfunc("DM_PartyMes", "Fine. Gray courier. Northern account. Yuno money. Now stop looking at my ships like kindling.");
		break;
	case 4:
		callfunc("DM_SceneRelease", "arc05_brode");
		close();
	}
	callfunc("DM_InstanceSetFlag", "dm_arc05_cassell_courier_drop", 1);
	callfunc("DM_InstanceQuestStart", 20029);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 525);
	next();
	callfunc("DM_PartyMes", "^3355FF*Brode's Manifest lists ports and dates forward into places you have not yet visited.*^000000");
	callfunc("DM_SceneRelease", "arc05_brode");
	close();

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20029, 1;
	end;
}

// Cargo of Souls - locked hold reveal on the Sunken Ship route
treasure01,153,160,4	script	Exempt Hold#dm	4_TREASURE_BOX,{
	if (!callfunc("DM_SessionAllows")) { close; }
	if (questprogress(20029) != 1) {
		if (questprogress(20029) == 2) {
			callfunc("DM_PartyMes", "^3355FF*The hold is open. Saltwater washes over empty shackles and broken crate seals.*^000000");
			close();
		}
		callfunc("DM_PartyMes", "^3355FF*This hold is sealed by maritime exemption and better locks than a wreck deserves.*^000000");
		close();
	}
	if ('boss_up) {
		callfunc("DM_PartyMes", "^3355FF*Brode's crew is already fighting to keep the hold closed.*^000000");
		close();
	}

	if (!callfunc("DM_SceneClaim", "arc05_hold")) {
		callfunc("DM_SceneFail", "arc05_hold");
		close();
	}
	callfunc("DM_PartyMes", "^3355FF*The manifest lists this hold as ballast. Ballast does not breathe, pray, or knock back.*^000000");
	next();
	if (!callfunc("DM_Arc05QuietTools"))
		callfunc("DM_PartyMes", "^777777Quiet needs Mara's picks, a diver's prybar, or the informant's stamp. You do not have them yet.*^000000");
	next();
	switch (select("Crack the hold quietly (tools, fewer blades).", "Break it open loudly (everyone hears).", "We need to discuss this.")) {
	case 1:
		if (!callfunc("DM_Arc05QuietTools")) {
			callfunc("DM_PartyMes", "^3355FF*The exemption lock laughs at bare hands. Walk the pier or the Byalan lines, or go loud.*^000000");
			callfunc("DM_SceneRelease", "arc05_hold");
			close();
		}
		callfunc("DM_InstanceSetFlag", "dm_arc05_hold_quiet", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_hold_loud", 0);
		callfunc("DM_PartyMes", "^3355FF*The lock gives with a soft click. Mara's cover survives another hour.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc05_hold_loud", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_hold_quiet", 0);
		callfunc("DM_PartyMes", "^3355FF*The door breaks. Every hired blade on the route hears it.*^000000");
		break;
	case 3:
		callfunc("DM_SceneRelease", "arc05_hold");
		close();
	}
	next();
	callfunc("DM_SceneTouch", "arc05_hold");

	if (dm_arc03_sabra_deal) {
		callfunc("DM_InstanceSetFlag", "dm_arc05_sabra_refugees_safe", 1);
		callfunc("DM_PartyMes", "^3355FF*Sabra's people are inside, alive. The old debt comes due as gratitude, not accusation.*^000000");
	} else if (dm_arc03_sabra_exposed) {
		callfunc("DM_InstanceSetFlag", "dm_arc05_sabra_refugees_starved", 1);
		callfunc("DM_PartyMes", "^3355FF*The survivors inside are starving. They look at you like justice had a face once.*^000000");
	} else {
		callfunc("DM_PartyMes", "^3355FF*Refugees and cult cargo share one manifest. The image explains the whole arc.*^000000");
	}
	next();

	.@map$ = strnpcinfo(NPC_MAP);
	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	'boss_up = 1;
	.@self$ = strnpcinfo(NPC_NAME);
	callfunc("DM_EncStart", "arc05_hold", getcharid(CHAR_ID_PARTY), .@map$, .@self$ + "::OnCrewDead", .@self$ + "::OnCrewDead", .@self$);
	mapannounce .@map$, "[Cargo of Souls] Brode's sealed hold opens. His crew tries to close it with blood.", bc_map, 0xFFAA33;
	.@knives = 2;
	.@hydras = 1;
	if (dm_arc05_hold_loud) {
		.@knives = 6;
		.@hydras = 4;
	}
	callfunc("DM_EncMonster", .@map$, .@x, .@y, "Brode's Knife", 1071,
		.@self$ + "::OnCrewDead", 58, 100, .@knives, 6);
	callfunc("DM_EncMonster", .@map$, .@x, .@y, "Cargo Hydra", 1068,
		.@self$ + "::OnCrewDead", 58, 100, .@hydras, 6);
	close();

OnCrewDead:
	.@left = mobcount(strnpcinfo(NPC_MAP), strnpcinfo(NPC_NAME)+"::OnCrewDead");
	if (.@left > 0)
		end;
	if (!callfunc("DM_EncOnDeath", "arc05_hold"))
		end;
	.@party = getd("$dm_enc_p_arc05_hold");
	if (.@party <= 0)
		end;
	if (!callfunc("DM_ClaimGrant", "arc05_crew", .@party)) {
		callfunc("DM_EncCleanup", "arc05_hold");
		end;
	}

	'boss_up = 0;
	callfunc("DM_InstanceQuestComplete", 20029);
	callfunc("DM_InstanceSetFlag", "dm_arc05_refugees_freed", 1);
	callfunc("DM_InstanceSetFlag", "dm_arc05_manifest_taken", 1);
	callfunc("DM_PartyExp", 110000, 50000);
	callfunc("DM_GivePartyZeny", 12000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 529);
	mapannounce strnpcinfo(NPC_MAP), "[Cargo of Souls] The hold is open. Brode's Manifest names the route north.", bc_map, 0x88CCFF;
	end;

OnInit:
	callfunc("DM_AssertWalk");
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20029, 1;
	end;
}

// Tao Gunka set-piece - Act I finale

	if (questprogress(20030) != 1) {

		callfunc("DM_SceneFail", "arc05_tao");
		close();
	}
	callfunc("DM_PartyMes", "^3355FF*Below the shelf, the water turns like a sleeping island rolling in its dream. Brode's dumped crates are wedged into the trench.*^000000");

	switch (select("Recover crates, then surrender them after.", "Recover crates and keep them for leverage.", "Leave the crates. Meet the beast.", "We need to discuss this.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc05_relic_crates_recovered", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_relics_surrendered", 1);

	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc05_relic_crates_recovered", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_relics_kept", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc05_relics_surrendered", 0);
		if (dm_arc04_pilgrim_catechism)
			callfunc("DM_InstanceSetFlag", "dm_catechism_tally", dm_catechism_tally + 1);
		callfunc("DM_PartyMes", "^3355FF*You keep the same cargo Brode trafficked. The difference is motive. For now.*^000000");

	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	.@adds = 5;
	if (dm_arc05_relic_crates_recovered)
		.@adds -= 2;
	if (dm_arc05_mara_trust)
		.@adds -= 1;
	if (dm_arc05_hold_loud || dm_arc05_brode_exposed_early)
		.@adds += 2;
	if (dm_arc05_mara_insulted)
		.@adds += 1;
	if (.@adds < 1)
		.@adds = 1;

	'boss_up = 1;
	.@self$ = strnpcinfo(NPC_NAME);
	callfunc("DM_EncStart", "arc05_tao", getcharid(CHAR_ID_PARTY), .@map$, .@self$ + "::OnTaoDead", .@self$ + "::OnSideDead", .@self$);
	mapannounce .@map$, "[Deep Trench Wake] Tao Gunka rises from the disturbed deep. Act I ends under black water.", bc_map, 0xFFAA33;
	callfunc("DM_EncMonster", .@map$, .@x, .@y, "Tao Gunka", 1583,
		.@self$ + "::OnTaoDead", 58, 100, 1, 0);
	if (!dm_arc05_relic_crates_recovered)
		callfunc("DM_EncMonster", .@map$, .@x + 3, .@y + 3, "Drake", 1112,
			.@self$ + "::OnSideDead", 58, 100, 1, 0);
	if (dm_arc05_hold_loud || dm_arc05_relics_kept)
		callfunc("DM_EncMonster", .@map$, .@x - 3, .@y - 3, "Kraken", 2202,
			.@self$ + "::OnSideDead", 58, 100, 1, 0);
	callfunc("DM_EncMonster", .@map$, .@x, .@y, "Trench Wake", 1045,
		.@self$ + "::OnSideDead", 58, 100, .@adds, 6);
	close();

OnSideDead:
	end;

OnTaoDead:
	if (!callfunc("DM_EncOnDeath", "arc05_tao"))
		end;
	.@party = getd("$dm_enc_p_arc05_tao");
