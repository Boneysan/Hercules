// Mira Ashkey - Refugee Camp and Arc 3 quest-giver
morocc,159,97,4	script	Mira Ashkey#dm	4_F_02,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20018) == 2) {
		mes("[Mira]");
		if (dm_arc03_mira_scarf_returned) {
			mes("^3355FF*Mira wraps the torn blue scarf around her brass house-key without a word.*^000000");
			next();
			mes("[Mira]");
			mes("He didn't suffer long. That's... that's something. Thank you. Thank you for not leaving him in the dark.");
		} else {
			mes("You returned. Did you find him? Did you find his scarf?");
			next();
			mes("^3355FF*You shake your head. She turns away, looking out over the endless desert.*^000000");
		}
		close();
	}

	if (!dm_arc03_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20013, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20013, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20014, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20015, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20016, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("^3355FF*Mira clutches a brass house-key wrapped in red thread.*^000000");
	next();
	mes("[Mira]");
	mes("You're Guild, aren't you? Please. My brother went into the desert to find water two days ago. He didn't come back. The sand... the sand is swallowing people.");
	next();
	switch (select("We will find him, Mira. We'll search the desert.", "The ground is shaking here too? Like Prontera?", "We have bounties in the Sphinx. If we see him, we'll send him back.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc03_mira_promise", 1);
		mes("[Mira]");
		mes("Thank you! The guards won't even look. They say the desert takes what it wants when the ground shakes.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc03_mira_promise", 2);
		mes("[Mira]");
		mes("Worse. At night, it sounds like... breathing. Huge, heavy breaths coming from deep beneath the Pyramids. It's driving the monsters mad.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc03_mira_promise", 3);
		mes("[Mira]");
		mes("Just... look for a blue scarf. If... if you only find the scarf, please bring it back to me.");
		break;
	}
	next();
	mes("[Mira]");
	mes("The Guild bounties are still active across the desert, but the Pyramids... that's where the breathing comes from. Please be careful.");
	callfunc("DM_InstanceSetFlag", "dm_arc03_started", 1);
	callfunc("DM_InstanceQuestStart", 20013);
	callfunc("DM_InstanceQuestStart", 20014); // Muka
	callfunc("DM_InstanceQuestStart", 20015); // Ant Eggs
	callfunc("DM_InstanceQuestStart", 20016); // Zerom
	callfunc("DM_InstanceQuestStart", 20017); // Osiris
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 301);
	next();
	mes("^3355FF[Quest started: Sand and Whispers]^000000");
	mes("^3355FFInvestigate the Pyramids for Mira's brother.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20014) == 1 && questprogress(20014, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20014);
		callfunc("DM_PartyExp", 16000, 7000);
		callfunc("DM_GivePartyZeny", 3500);
		.@paid = 1;
		mes("[Mira]");
		mes("Thank you for clearing the caravans. Every drop of water helps the camp.");
		next();
	}

	if (questprogress(20015) == 1 && questprogress(20015, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20015);
		callfunc("DM_PartyExp", 22000, 9000);
		callfunc("DM_GivePartyZeny", 4500);
		.@paid = 1;
		mes("[Mira]");
		mes("The ant tunnels are safer. But it doesn't stop the breathing from the Pyramids.");
		next();
	}

	if (questprogress(20016) == 1 && questprogress(20016, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20016);
		callfunc("DM_PartyExp", 30000, 12000);
		callfunc("DM_GivePartyZeny", 5500);
		.@paid = 1;
		mes("[Mira]");
		mes("The Sphinx is clear. But my brother isn't there... he must be in the Pyramids.");
		next();
	}

	if (.@paid) {
		mes("[Mira]");
		mes("The Guild will transfer your bounty rewards. Please... find him.");
		close();
	}

	mes("[Mira]");
	mes("He was wearing a blue scarf. The Pyramids... please check the Pyramids.");
	close();
}

// Guild Assassin - Pyramids Entrance
moc_ruins,75,165,4	script	Guild Assassin#dm	1_M_ASSASSIN,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc03_started) {
		mes("[Guild Assassin]");
		mes("Move along. Nothing for you here.");
		close();
	}
	if (dm_arc03_assassin_met) {
		mes("[Guild Assassin]");
		mes("I warned you. Don't come crying to me when the mummy tears you apart.");
		close();
	}

	mes("^3355FF*A cloaked figure steps out of stealth, arms crossed.*^000000");
	next();
	mes("[Guild Assassin]");
	mes("Turn around, Guild-rats. This ruin is claimed by the shadows today.");
	next();
	switch (select("People are dying in there. We have to stop it.", "We have a bounty on Osiris. You want to share the kill, or get out of our way?", "Move, or you're the first thing I kill today.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc03_assassin_path", 1);
		mes("[Guild Assassin]");
		mes("You're out of your depth. There are robed fanatics in there trying to wake up a pharaoh. They think it's a god. Idiots. It's just a corpse with a curse.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc03_assassin_path", 2);
		mes("[Guild Assassin]");
		mes("(Chuckles) Confident. Fine. But the cultists inside are using some kind of harmonic frequency. Every time they chant, the whole city shudders. If you're going in, kill the chanters first.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc03_assassin_path", 3);
		mes("[Guild Assassin]");
		mes("Big words. Let's see if you can back them up when the mummy starts tearing you apart.");
		next();
		mes("^3355FF*The Assassin fades into stealth.*^000000");
		mes("[Guild Assassin]");
		mes("Don't say I didn't warn you.");
		callfunc("DM_InstanceSetFlag", "dm_arc03_assassin_met", 1);
		close();
	}
	next();
	mes("[Guild Assassin]");
	mes("I'm not getting paid to babysit. If you go in, you're on your own.");
	callfunc("DM_InstanceSetFlag", "dm_arc03_assassin_met", 1);
	close();
}

// High Priest - Pyramids Lowest Level (moc_pryd06)
moc_pryd06,102,85,4	script	High Priest#dm	4_M_BISHOP,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20018) == 2) {
		mes("^3355FF*The cultists are gone. The shaft behind the throne breathes cold air from far below.*^000000");
		close();
	}
	if ('amon_up) {
		mes("[High Priest]");
		mes("The Great Despair awakens!");
		close();
	}
	if ('osiris_up) {
		mes("[High Priest]");
		mes("The Pharaoh judges you!");
		close();
	}

	mes("^3355FF*Cultists surround a massive sarcophagus. The ground is violently shaking. The High Priest holds a tuning fork that hums with dark magic.*^000000");
	next();
	mes("[High Priest]");
	mes("The sleeping demon stirs! Its dreams corrupt the dead! Rise, Osiris! Show them the power of the great seal!");
	next();
	switch (select("You don't know what you're messing with! Stop the ritual!", "What demon? What is beneath Morroc?", "We're shutting this down. Now.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc03_priest_path", 1);
		mes("[High Priest]");
		mes("We know exactly what we are doing! We are testing the locks on the cage!");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc03_priest_path", 2);
		mes("[High Priest]");
		mes("The Great Despair! The One Beneath! We do not speak its name - the sand speaks it for us, every time it shakes! The seal is cracking, and the pharaohs are only the first to feel its breath!");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc03_priest_path", 3);
		mes("[High Priest]");
		mes("You cannot shut down the inevitable! Rise, Osiris!");
		break;
	}
	next();
	
	.@map$ = strnpcinfo(NPC_MAP);
	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	
	'osiris_up = 1;
	mapannounce .@map$, "[High Priest] The Pharaoh rises! Defend the ritual!", bc_map, 0xFFAA33;
	
	// Summon Osiris and cultists
	monster .@map$, .@x, .@y, "Osiris", 1038, 1, "High Priest#dm::OnOsirisDead";
	areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Whisperer Cultist", 1041, 4, "High Priest#dm::OnAddDead";
	close();

OnAddDead:
	end;

OnOsirisDead:
	'osiris_up = 0;
	.@map$ = strnpcinfo(NPC_MAP);
	killmonster .@map$, "High Priest#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20017);
	callfunc("DM_InstanceQuestStart", 20018); // Start Amon Ra phase
	
	mapannounce .@map$, "[High Priest] Osiris has fallen... but the deep doors are grinding open! The King Beneath rises!", bc_map, 0xFF0000;
	sleep2 3000;
	
	// Summon Amon Ra directly after Osiris
	'amon_up = 1;
	mapannounce .@map$, "[The King Beneath] Amon Ra emerges from the deepest tomb!", bc_map, 0xFFAA33;
	monster .@map$, .@x, .@y, "Amon Ra", 1511, 1, "High Priest#dm::OnAmonDead";
	areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Whisperer Cultist", 1041, 4, "High Priest#dm::OnAddDead";
	end;

OnAmonDead:
	'amon_up = 0;
	.@map$ = strnpcinfo(NPC_MAP);
	killmonster .@map$, "High Priest#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20018);
	callfunc("DM_InstanceSetFlag", "dm_arc03_mira_scarf_returned", 1);
	callfunc("DM_PartyExp", 150000, 65000);
	callfunc("DM_GivePartyZeny", 14000);
	
	mapannounce .@map$, "[Pyramids] The tremor stops, but a low, distant heartbeat can be felt through the floor.", bc_map, 0x88CCFF;
	sleep2 3000;
	mapannounce .@map$, "[Pyramids] Among the cult's 'offerings', you find a torn blue scarf.", bc_map, 0x88CCFF;
	
	dispbottom "[Seal Cascade] Arc 3 complete. You have recovered Mira's brother's scarf. Return to Morroc.";
	end;

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20018, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}
