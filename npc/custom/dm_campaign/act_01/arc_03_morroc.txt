// Rashid - tomb-guide and Arc 3 quest-giver
morocc,159,97,4	script	Rashid the Guide#dm	4_M_ORIENT02,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20018) == 2) {
		mes("[Rashid]");
		if (dm_arc03_rashid_mocked)
			mes("So. You heard it too. Old fool, was I?");
		else
			mes("You came back from below the kings. That means the thing I felt was not age or wine.");
		next();
		mes("[Rashid]");
		mes("Keep the lid-stone rubbing. Someday someone will try to open what it covers.");
		close();
	}

	if (!dm_arc03_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20013, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20013, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20014, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20015, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20016, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("[Rashid]");
	mes("Morroc laughs at the demon under the sand because laughter is cheaper than moving.");
	next();
	if (countitem(50001) > 0 || dm_arc01_sigil_ring_obtained) {
		mes("^3355FF*The Sigil Ring bites colder here than it did in Payon. It does not point north, east, or west. It points down.*^000000");
		next();
	}
	mes("[Rashid]");
	mes("I am Rashid. I robbed these tombs before your grandparents learned fear, and I put half of it back before fear learned my name.");
	next();
	switch (select("Tell it your way, grandfather.", "Just the fastest way down.", "There is no demon. You are an old fool.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc03_rashid_trust", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_rashid_route", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_ward_charm", 1);
		mes("[Rashid]");
		mes("Good ears. Good manners. Take the route, take the charm, and when the stone breathes, move before it screams.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc03_rashid_route", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_ward_charm", 0);
		mes("[Rashid]");
		mes("Fastest route, then. You will get there. Whether you arrive whole is not included.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc03_rashid_mocked", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_rashid_route", 0);
		callfunc("DM_InstanceSetFlag", "dm_arc03_ward_charm", 0);
		mes("[Rashid]");
		mes("Coin buys a map. Respect buys the warnings. You have purchased only paper.");
		break;
	}
	next();
	mes("[Rashid]");
	mes("Start at the last wells. The relief mission feeds refugees, and someone there is digging below kindness.");
	callfunc("DM_InstanceSetFlag", "dm_arc03_started", 1);
	callfunc("DM_InstanceQuestStart", 20013);
	callfunc("DM_InstanceQuestStart", 20014);
	callfunc("DM_InstanceQuestStart", 20015);
	callfunc("DM_InstanceQuestStart", 20016);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 301);
	next();
	mes("^3355FF[Quest started: Sand and Whispers]^000000");
	mes("^3355FFCaravan Water Debt - cull 20 Muka along Sograt's water routes.^000000");
	mes("^3355FFAnt Hell Survey - break 15 Ant Eggs and bring back amber-sand samples.^000000");
	mes("^3355FFSphinx Night Watch - cull 20 Zerom in the Sphinx.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20014) == 1 && questprogress(20014, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20014);
		callfunc("DM_PartyExp", 16000, 7000);
		callfunc("DM_GivePartyZeny", 3500);
		.@paid = 1;
		mes("[Rashid]");
		mes("Cracked water seals. Proof that even thirst has a ledger.");
		next();
		switch (select("Save water for refugees.", "Sell it to caravans.", "Use it to expose cult maps.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc03_water_choice", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc03_refugee_turnout", 1);
			mes("[Rashid]");
			mes("Mercy is slow, but people remember who brought water.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc03_water_choice", 2);
			mes("[Rashid]");
			mes("Coin is honest. It never pretends to be kind.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc03_water_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc03_cult_maps_found", 1);
			mes("[Rashid]");
			mes("Good. Their routes follow water because lies still need mouths.");
			break;
		}
		next();
	}

	if (questprogress(20015) == 1 && questprogress(20015, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20015);
		callfunc("DM_PartyExp", 22000, 9000);
		callfunc("DM_GivePartyZeny", 4500);
		.@paid = 1;
		mes("[Rashid]");
		mes("Amber-sand from Ant Hell. The tunnels are feeling for something older than ants.");
		next();
		switch (select("Collapse the tunnels.", "Map them.", "Mark the forbidden route.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc03_ant_choice", 1);
			mes("[Rashid]");
			mes("The desert will swallow them again. For now.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc03_ant_choice", 2);
			mes("[Rashid]");
			mes("A map is a promise to return. Be careful what you promise.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc03_ant_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc03_forbidden_route_marked", 1);
			mes("[Rashid]");
			mes("Now you know where not to go. That is almost wisdom.");
			break;
		}
		next();
	}

	if (questprogress(20016) == 1 && questprogress(20016, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20016);
		callfunc("DM_PartyExp", 30000, 12000);
		callfunc("DM_GivePartyZeny", 5500);
		.@paid = 1;
		mes("[Rashid]");
		mes("Zerom eye beads. The Sphinx is not watching outward anymore.");
		next();
		switch (select("Clear the watch.", "Listen to the whispers.", "Follow them toward the sealed door.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc03_sphinx_choice", 1);
			mes("[Rashid]");
			mes("Clean work. But clean does not mean complete.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc03_sphinx_choice", 2);
			mes("[Rashid]");
			mes("Then you heard the corridor breathing before the quake. Remember that.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc03_sphinx_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc03_shaft_heard", 1);
			mes("[Rashid]");
			mes("You followed the sound under the maps. That sound has been waiting longer than Morroc.");
			break;
		}
		next();
	}

	if (questprogress(20014) == 2 && questprogress(20015) == 2 && questprogress(20017) == 0) {
		mes("[Rashid]");
		mes("A dead king has climbed out of the sand and started holding court. Put Osiris back before the Pyramid remembers him.");
		next();
		switch (select("We'll ride for the tomb-court.", "Later.")) {
		case 1:
			callfunc("DM_InstanceQuestStart", 20017);
			callfunc("DM_PartyApplyFlag", "dm_story_beat", 317);
			mes("^3355FF[Quest started: What the Desert Spat Up]^000000");
			mes("^3355FFFind Osiris in Pyramid F4. Break the court before the king.^000000");
			close();
		case 2:
			close();
		}
	}

	if (questprogress(20016) == 2 && questprogress(20018) == 0) {
		mes("[Rashid]");
		mes("The route ends below Amon Ra. His chamber is not the bottom of the Pyramid. It is the lid.");
		next();
		if (dm_arc03_ward_charm)
			mes("You still carry my ward-charm. Let it go cold before you trust the floor.");
		else
			mes("You did not take the charm. So when the room trembles, move before pride does.");
		next();
		switch (select("Start The King Beneath.", "Later.")) {
		case 1:
			callfunc("DM_InstanceQuestStart", 20018);
			callfunc("DM_PartyApplyFlag", "dm_story_beat", 318);
			mes("^3355FF[Quest started: The King Beneath]^000000");
			mes("^3355FFGo to Pyramid B2. Amon Ra guards the shaft that must not open.^000000");
			close();
		case 2:
			close();
		}
	}

	if (.@paid) {
		mes("[Rashid]");
		mes("The support work is done. The old kings will not wait politely.");
		close();
	}

	mes("[Rashid]");
	mes("Clear the water route, survey Ant Hell, and watch the Sphinx. The desert is trying to tell you what the tombs already know.");
	close();
}

// Mother Sabra - relief mission leader and human villain
moc_fild01,84,45,4	script	Mother Sabra#dm	4_F_SISTER,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (!dm_arc03_started) {
		mes("[Mother Sabra]");
		mes("If you need water, take water. If you need a sermon, I have no strength left for one.");
		close();
	}

	if (questprogress(20018) == 2) {
		mes("[Mother Sabra]");
		if (dm_arc03_sabra_deal)
			mes("The tents stand. The shaft is quiet. Debts are quieter still.");
		else if (dm_arc03_sabra_exposed)
			mes("The kingdom closed the dig. It also closed its eyes to the people who slept above it.");
		else
			mes("You saw the shaft and came back with fear. Good. Fear keeps hands from doors.");
		close();
	}

	mes("[Mother Sabra]");
	mes("I bury the children who do not survive the road. Then I feed the parents. If that makes me a monster, Morroc should have made kinder monsters.");
	next();
	switch (select("Your dig will wake it.", "These people need you. Walk away from the dig.", "What is really down there?")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc03_sabra_exposed", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_sabra_deal", 0);
		mes("[Mother Sabra]");
		mes("Better awake and ended than this endless half-sleep, shaking us apart by inches.");
		next();
		mes("^3355FF*Exposing the dig will close the mission. The refugees will scatter, some toward worse hands.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc03_sabra_deal", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_refugees_protected", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_sabra_exposed", 0);
		mes("[Mother Sabra]");
		mes("Guarantee their safety and I abandon the shaft. Not the cause. The shaft.");
		next();
		mes("^3355FF*The refugees keep the tents. Sabra keeps your debt for Arc 5.*^000000");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc03_sabra_marked_party", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_sabra_revealed_cascade", 1);
		mes("[Mother Sabra]");
		mes("Not a demon. A wound. The gods bandaged it with seals and called the rot peace.");
		next();
		mes("[Mother Sabra]");
		mes("We do not need to break one seal. We weaken all of them, and the world breathes.");
		break;
	}
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 325);
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

// Osiris - side tomb-court set-piece
moc_pryd04,100,92,4	script	Osiris's Court#dm	4_M_NFDEADMAN,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20017) != 1) {
		if (questprogress(20017) == 2) {
			mes("^3355FF*The sand-drowned court is still. A gold mask lies warm only in memory.*^000000");
			close();
		}
		mes("^3355FF*The court has not convened for you yet. Rashid must name the dead king first.*^000000");
		close();
	}
	if ('boss_up) {
		mes("^3355FF*Osiris's court is already rising. Break the court before the king.*^000000");
		close();
	}

	mes("^3355FF*Cold seal-light leaks through a sand-drowned tomb-court. Osiris sits where no throne should be.*^000000");
	next();
	switch (select("Lay Osiris and his court to rest.", "Loot the court and leave.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc03_osiris_restored", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_osiris_looted", 0);
		mes("^3355FF*You speak the names Rashid taught you. The dead answer poorly, but they answer.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc03_osiris_looted", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc03_osiris_restored", 0);
		mes("^3355FF*Gold comes away easily. Rest does not.*^000000");
		break;
	}
	next();

	.@map$ = strnpcinfo(NPC_MAP);
	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	'boss_up = 1;
	mapannounce .@map$, "[Osiris's Court] The dead king rises. His court shields him.", bc_map, 0xFFAA33;
	areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Sand Court", 1165, 4, "Osiris's Court#dm::OnCourtDead";
	areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Buried Knife", 1119, 2, "Osiris's Court#dm::OnCourtDead";
	monster .@map$, .@x, .@y, "Osiris", 1038, 1, "Osiris's Court#dm::OnOsirisDead";
	close();

OnCourtDead:
	end;

OnOsirisDead:
	'boss_up = 0;
	.@map$ = strnpcinfo(NPC_MAP);
	killmonster .@map$, "Osiris's Court#dm::OnCourtDead";
	callfunc("DM_InstanceQuestComplete", 20017);
	callfunc("DM_InstanceSetFlag", "dm_arc03_funeral_mask", 1);
	callfunc("DM_PartyExp", 90000, 38000);
	callfunc("DM_GivePartyZeny", 9000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 399);
	mapannounce .@map$, "[Osiris's Court] The dead king falls. The Funeral Mask is warm with seal-light.", bc_map, 0x88CCFF;
	end;

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20017, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

// Amon Ra - main Pyramid set-piece
moc_pryd06,102,85,4	script	Amon Ra's Lid#dm	4_M_CRU_HEAD,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20018) != 1) {
		if (questprogress(20018) == 2) {
			mes("^3355FF*Behind the throne, the cult-cut shaft remains sealed. It breathes cold air from far below.*^000000");
			close();
		}
		mes("^3355FF*Amon Ra's chamber is not ready to reveal what it covers. Speak with Rashid first.*^000000");
		close();
	}
	if ('boss_up) {
		mes("^3355FF*The god-king is awake. The lid-stone groans behind him.*^000000");
		close();
	}

	mes("^3355FF*Amon Ra's chamber is not the bottom of the Pyramid. Behind the throne, a shaft has been cut around a capstone older than the tomb.*^000000");
	next();
	if (dm_arc03_rashid_trust)
		mes("^3355FF*Rashid's warning reaches you before the tremor. You have one breath to move.*^000000");
	else
		mes("^3355FF*The floor convulses without warning. Something below is not dreaming. It is waiting.*^000000");
	next();

	switch (select("Drive off Sabra's diggers first.", "Fight through king and cult together.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc03_cult_diggers_cleared", 1);
		mes("^3355FF*You scatter the diggers from the shaft. Amon Ra wakes with fewer voices feeding him.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc03_cult_diggers_cleared", 0);
		mes("^3355FF*The cult keeps chanting as the god-king opens his eyes.*^000000");
		break;
	}
	next();

	if (!dm_arc03_sabra_deal && !dm_arc03_sabra_exposed) {
		mes("^3355FF*Sabra's choice still waits above. The shaft will not stay quiet unless someone pays for quiet.*^000000");
		next();
	}

	.@map$ = strnpcinfo(NPC_MAP);
	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	.@adds = 5;
	if (dm_arc03_cult_diggers_cleared)
		.@adds -= 2;
	if (dm_arc03_ward_charm)
		.@adds -= 1;
	if (dm_arc03_osiris_restored)
		.@adds -= 1;
	if (.@adds < 1)
		.@adds = 1;

	'boss_up = 1;
	mapannounce .@map$, "[Amon Ra's Lid] The tomb is only a lid. The king beneath the king wakes.", bc_map, 0xFFAA33;
	monster .@map$, .@x, .@y, "Amon Ra", 1511, 1, "Amon Ra's Lid#dm::OnAmonDead";
	areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Seal Digger", 1041, .@adds, "Amon Ra's Lid#dm::OnAddDead";
	close();

OnAddDead:
	end;

OnAmonDead:
	'boss_up = 0;
	.@map$ = strnpcinfo(NPC_MAP);
	killmonster .@map$, "Amon Ra's Lid#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20018);
	callfunc("DM_InstanceSetFlag", "dm_arc03_lidstone_obtained", 1);
	callfunc("DM_InstanceSetFlag", "dm_arc03_shaft_seen", 1);
	callfunc("DM_PartyExp", 150000, 65000);
	callfunc("DM_GivePartyZeny", 14000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 398);
	mapannounce .@map$, "[Amon Ra's Lid] Amon Ra falls. The sealed shaft stays shut, and that is the most frightening thing about it.", bc_map, 0x88CCFF;
	dispbottom "[Seal Cascade] Arc 3 complete. The rumor is true: something vast waits under Morroc.";
	end;

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20018, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}
