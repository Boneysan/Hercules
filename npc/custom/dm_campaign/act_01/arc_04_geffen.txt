// Archmage Vella - Mages' Guild gatekeeper, Scene 1 flavor
geffen,127,100,4	script	Archmage Vella#dm	4_F_02,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (dm_arc04_vella_met) {
		mes("[Archmage Vella]");
		if (dm_arc04_vella_choice == 3)
			mes("Only a handful of people have that clearance. Archmagus Doran holds one of them. Do not make me regret naming him.");
		else
			mes("The door is already open. That has not stopped being the problem.");
		close();
	}

	mes("^3355FF*A robed woman paces frantically in front of the tower stair.*^000000");
	next();
	mes("[Archmage Vella]");
	mes("This is an internal guild matter! The Adventurers' Guild has no jurisdiction in the Geffen Tower!");
	next();
	switch (select("We have proof the Whisperer cult is operating here. Let us help.", "Wynne sent us. We're getting paid to clear the dungeon. Are you going to open the door?", "Who disabled the wards?")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc04_vella_choice", 1);
		mes("[Archmage Vella]");
		mes("You have proof? ...Fine. My senior enchanters are missing. And the warding seals in the basement — the ones keeping the Geffen Dungeon locked — have been systematically disabled from the inside.");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc04_vella_choice", 2);
		mes("[Archmage Vella]");
		mes("The door is already open. That's the problem. The magic holding back the horrors below has been unravelled by someone who knows our highest secrets.");
		break;
	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc04_vella_choice", 3);
		mes("[Archmage Vella]");
		mes("Only a handful of people have that clearance. Archmagus Doran holds one of them — and he has been 'studying' the lower vault for months with a visiting scholar he vouched for personally. A soft-spoken man. In gray robes.");
		break;
	}
	callfunc("DM_InstanceSetFlag", "dm_arc04_vella_met", 1);
	close();

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}

// Apprentice Elsbeth - Geffen ally and Arc 4 support handler
geffen,120,100,4	script	Apprentice Elsbeth#dm	4_F_01,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20024) == 2) {
		mes("[Apprentice Elsbeth]");
		if (dm_arc04_seal_reinforced)
			mes("The tower lamps are dimmer. The city is angry. The seal is alive.");
		else if (dm_arc04_overflow_kept)
			mes("The tower shines beautifully tonight. Please do not ask me what it costs.");
		else
			mes("The vault is quiet for now. I am learning to distrust quiet things.");
		close();
	}

	if (!dm_arc04_started) {
		callsub(S_StartArc);
		end;
	}

	callsub(S_TurnIns);
	end;

OnInit:
	questinfo QTYPE_QUEST, 1;
	setquestinfo QINFO_QUEST, 20019, 0;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20019, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20020, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20021, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20022, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;

S_StartArc:
	mes("[Apprentice Elsbeth]");
	mes("Everyone calls Geffen the city of spellcraft. No one says what the craft is chained to.");
	next();
	if (countitem(50001) > 0 || dm_arc01_sigil_ring_obtained) {
		mes("^3355FF*The Sigil Ring vibrates under the tower's bells. It does not point toward a dungeon. It points under the city itself.*^000000");
		next();
	}
	mes("[Apprentice Elsbeth]");
	mes("I am Elsbeth. Third-year apprentice. Keeper of keys I am not supposed to understand.");
	next();
	switch (select("Stay safe and give us the ward-key.", "Come with us and explain the vault.", "I do not trust Guild people.")) {
	case 1:
		callfunc("DM_Decide", "arc04.elsbeth", "protect");
		mes("[Apprentice Elsbeth]");
		mes("Then take this ward-key. If the lower door asks a question, answer with silence.");
		break;
	case 2:
		callfunc("DM_Decide", "arc04.elsbeth", "vault");
		mes("[Apprentice Elsbeth]");
		mes("I know the prayers they carved into the locks. I will come, and I will regret being useful.");
		break;
	case 3:
		callfunc("DM_Decide", "arc04.elsbeth", "distrust");
		mes("[Apprentice Elsbeth]");
		mes("Fair. Trust the key, then, if not the hand that offers it. No? Then you will force the old way.");
		break;
	}
	next();
	mes("[Apprentice Elsbeth]");
	mes("Start with the work the Guild pretends is ordinary. Chalk sigils, orc banners, black silk. They all hum with the same prison-song.");
	callfunc("DM_InstanceSetFlag", "dm_arc04_started", 1);
	callfunc("DM_InstanceQuestStart", 20019);
	callfunc("DM_InstanceQuestStart", 20020);
	callfunc("DM_InstanceQuestStart", 20021);
	callfunc("DM_InstanceQuestStart", 20022);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 401);
	next();
	mes("^3355FF[Quest started: The City Above the Beast]^000000");
	mes("^3355FFTower Apprentice Drills - clear 20 Poison Spores from Geffen's training route.^000000");
	mes("^3355FFOrc Village Bounty - break 20 Orc Warriors at the village edge.^000000");
	mes("^3355FFArgiope Silk Run - gather black silk evidence from 20 Argiope.^000000");
	close();

S_TurnIns:
	.@paid = 0;
	if (questprogress(20020) == 1 && questprogress(20020, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20020);
		callfunc("DM_PartyExp", 38000, 17000);
		callfunc("DM_GivePartyZeny", 5500);
		.@paid = 1;
		mes("[Apprentice Elsbeth]");
		mes("Apprentice chalk covered in spore dust. Wonderful. Also horrifying.");
		next();
		switch (select("Teach restraint.", "Teach power.", "Let students test forbidden diagrams.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc04_drills_choice", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc04_students_warned", 1);
			mes("[Apprentice Elsbeth]");
			mes("Restraint is slower. It also leaves more apprentices alive.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc04_drills_choice", 2);
			mes("[Apprentice Elsbeth]");
			mes("Power first, explanations later. The Guild will pretend it disapproves.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc04_drills_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc04_forbidden_diagrams", 1);
			mes("[Apprentice Elsbeth]");
			mes("Those diagrams are Doran's. If they answer you, do not answer back.");
			break;
		}
		next();
	}

	if (questprogress(20021) == 1 && questprogress(20021, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20021);
		callfunc("DM_PartyExp", 52000, 23000);
		callfunc("DM_GivePartyZeny", 6500);
		.@paid = 1;
		mes("[Apprentice Elsbeth]");
		mes("Chipped orc banners. The sigil marks are not orcish. They are bait.");
		next();
		switch (select("Take the bounty cleanly.", "Arm frightened villagers.", "Use cult bait to draw more monsters.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc04_orc_choice", 1);
			mes("[Apprentice Elsbeth]");
			mes("Clean work still leaves blood. But it does not multiply it.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc04_orc_choice", 2);
			callfunc("DM_InstanceSetFlag", "dm_arc04_villagers_armed", 1);
			mes("[Apprentice Elsbeth]");
			mes("Good. People near Geffen should know what the city is doing to them.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc04_orc_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc04_cult_bait_used", 1);
			mes("[Apprentice Elsbeth]");
			mes("Effective. Doran would call that word a pardon.");
			break;
		}
		next();
	}

	if (questprogress(20022) == 1 && questprogress(20022, HUNTING) == 2) {
		callfunc("DM_InstanceQuestComplete", 20022);
		callfunc("DM_PartyExp", 62000, 28000);
		callfunc("DM_GivePartyZeny", 8000);
		.@paid = 1;
		mes("[Apprentice Elsbeth]");
		mes("Black silk thread. Every strand maps back to the lower vault.");
		next();
		switch (select("Deliver the silk to the mages.", "Burn the contaminated silk.", "Keep it for ritual tracing.")) {
		case 1:
			callfunc("DM_InstanceSetFlag", "dm_arc04_silk_choice", 1);
			callfunc("DM_InstanceSetFlag", "dm_arc04_seal_pattern_evidence", 1);
			mes("[Apprentice Elsbeth]");
			mes("Evidence enters the archive. Whether the archive opens again is a political question.");
			break;
		case 2:
			callfunc("DM_InstanceSetFlag", "dm_arc04_silk_choice", 2);
			mes("[Apprentice Elsbeth]");
			mes("The smoke stinks like hymnals. That is new.");
			break;
		case 3:
			callfunc("DM_InstanceSetFlag", "dm_arc04_silk_choice", 3);
			callfunc("DM_InstanceSetFlag", "dm_arc04_ritual_thread_kept", 1);
			mes("[Apprentice Elsbeth]");
			mes("Then you can trace Doran's route. You can also be traced.");
			break;
		}
		next();
	}

	if (questprogress(20020) == 2 && questprogress(20021) == 2 && questprogress(20022) == 2 && questprogress(20023) == 0) {
		mes("[Apprentice Elsbeth]");
		mes("The proof is complete. Archmagus Doran has gone below the west locks. The lower vault is praying wrong.");
		next();
		switch (select("Start The Vault That Prays.", "Later.")) {
		case 1:
			callfunc("DM_InstanceQuestStart", 20023);
			callfunc("DM_PartyApplyFlag", "dm_story_beat", 423);
			mes("^3355FF[Quest started: The Vault That Prays]^000000");
			mes("^3355FFEnter Geffen Dungeon F3. Break Doran's glyphs and drive Baphomet off the seal.^000000");
			close();
		case 2:
			close();
		}
	}

	if (.@paid) {
		mes("[Apprentice Elsbeth]");
		mes("That is all the ordinary work. Which means the dangerous work is ready.");
		close();
	}

	mes("[Apprentice Elsbeth]");
	mes("Bring back the chalk sigil, orc banner, and black silk. Geffen's problem is hiding in its chores.");
	close();
}

		if (questprogress(20023) == 2) {
			mes("[Archmagus Doran]");
			if (dm_arc04_doran_stood_down)
				mes("Publish the proof, then. Let Geffen hate the truth in daylight.");
			else
				mes("The Guild will call me traitor because that word is cheaper than history.");
			close();
		}

		close();

	if (dm_arc04_doran_confronted) {
		mes("[Archmagus Doran]");
		mes("The glyphs are placed. The beast will wake whether you flatter or threaten me.");
		close();
	}

	mes("^3355FF*Brother Cassell stands before a massive, glowing arcane lock, methodically dissolving it with counter-magic. Cassell turns as the party approaches, removing his hood fully for the first time.*^000000");

		mes("[Deacon Holt]");
		mes("That voice. That's him. That's the gray almsman.");
		next();
	}
	mes("[Brother Cassell]");
	mes("Ah. The heroes of Prontera. And Payon. And Morroc. You are remarkably persistent. And predictably late.");
	next();
	switch (select("It's over, Cassell. We know you're trying to break the seal.", "Why do this? What's the point of weakening every seal at once?", "Teach me how the seals work.")) {

		callfunc("DM_InstanceSetFlag", "dm_arc04_cassell_path", 1);
		mes("[Brother Cassell]");
		mes("Break this seal? Heavens, no. Baphomet is a brute. If I break his seal completely, Geffen falls, and the other nations panic. I am merely... opening a window. A crack to let the pressure out.");

		callfunc("DM_InstanceSetFlag", "dm_arc04_cassell_path", 2);
		mes("[Brother Cassell]");
		mes("Because if one seal breaks, humanity rallies and fixes it. If they all break at once... the slate is wiped clean. The Voice demands a new world, and the old one must be cleared away.");

	case 3:
		callfunc("DM_InstanceSetFlag", "dm_arc04_cassell_path", 3);
		mes("[Brother Cassell]");
		mes("(Smiles, genuinely pleased) A seeker of knowledge. How refreshing. Not here — not shouted over a dying ward. Survive what is below, and I will pour the tea.");
		break;
	}
	next();
	mes("^3355FF*Cassell bows and is gone. Doran does not look up from the anchor-stones.*^000000");
	next();
	mes("[Archmagus Doran]");
	mes("I proved the tower runs on seal overflow. They buried the proof and promoted the cowards.");

	case 1:
		callfunc("DM_Decide", "arc04.doran", "condemn");
		mes("[Archmagus Doran]");
		mes("I'll wake it.");
		break;
	case 2:
		callfunc("DM_Decide", "arc04.doran", "surrender");
		mes("[Archmagus Doran]");
		mes("If the city reads my proof with the lamps dimmed, perhaps truth will finally have a cost.");
		break;
	case 3:
		callfunc("DM_Decide", "arc04.doran", "teaching");
		mes("[Archmagus Doran]");
		mes("The gray man explains it better than I ever could. Ask him, if you survive him.");
		break;
	}
	next();
	mes("^3355FF*The vault door opens below him. The air smells like old candles and wet fur.*^000000");
	close();

// Baphomet's Seal - main Geffen set-piece
gef_dun02,214,212,4	script	Baphomet's Seal#dm	4_ENERGY_RED,{
	if (!$dm_mode || getcharid(CHAR_ID_PARTY) != $dm_active_party) { close; }
	if (questprogress(20023) != 1) {
		if (questprogress(20023) == 2) {
			mes("^3355FF*The cracked keystone still hums where the beast struck it.*^000000");
			close();
		}
		mes("^3355FF*The lower vault is locked by Guild wards. Elsbeth and Doran must move the story here first.*^000000");
		close();
	}
	if (!(dm_arc04_puzzle_1 && dm_arc04_puzzle_2 && dm_arc04_puzzle_3)) {
		mes("^3355FF*The inner vault is protected by an outer glyph sequence. You must align the three outer glyphs before approaching the center.*^000000");
		close();
	}
	if ('boss_up) {
		mes("^3355FF*Baphomet is already battering the seal. Survive the prayer.*^000000");
		close();
	}

	mes("^3355FF*The chamber is a chapel built to face inward. Every glyph prays at Baphomet's prison, begging it to stay closed and keep feeding Geffen.*^000000");
	next();
	if (!dm_arc04_doran_confronted) {
		mes("^3355FF*Doran's work is still hidden in the glyphs. The set-piece will be harsher unless someone confronts him first.*^000000");
		next();
	}
	if (dm_arc04_elsbeth_with_party) {
		mes("^3355FF*Elsbeth reads the lock aloud. Something below hears her name.*^000000");
		next();
	}
	switch (select("Reinforce the seal, even if Geffen dims.", "Keep the overflow flowing and hold the beast down.")) {
	case 1:
		callfunc("DM_InstanceSetFlag", "dm_arc04_seal_reinforced", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc04_city_dimmed", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc04_overflow_kept", 0);
		mes("^3355FF*You turn the glyphs inward. The tower lights above begin to go out one floor at a time.*^000000");
		break;
	case 2:
		callfunc("DM_InstanceSetFlag", "dm_arc04_overflow_kept", 1);
		callfunc("DM_InstanceSetFlag", "dm_arc04_seal_reinforced", 0);
		callfunc("DM_InstanceSetFlag", "dm_arc04_city_dimmed", 0);
		mes("^3355FF*You keep Geffen bright. The seal accepts the compromise with a sound like cracking teeth.*^000000");
		break;
	}
	next();

	.@map$ = strnpcinfo(NPC_MAP);
	getmapxy(.@m$, .@x, .@y, UNITTYPE_NPC);
	.@adds = 5;
	if (dm_arc04_ward_key)
		.@adds -= 2;
	if (dm_arc04_doran_stood_down)
		.@adds -= 1;
	if (dm_arc04_students_warned || dm_arc04_villagers_armed)
		.@adds -= 1;
	if (dm_arc02_rite_path == 3 || dm_arc02_voss_spared || dm_arc03_sabra_revealed_cascade || dm_arc03_sabra_marked_party || dm_arc04_forbidden_diagrams || dm_arc04_cult_bait_used || dm_arc04_ritual_thread_kept) {
		.@adds += 2;
		callfunc("DM_InstanceSetFlag", "dm_arc04_doppelganger_stronger", 1);
	}
	if (dm_arc04_elsbeth_with_party) {
		.@adds += 1;
		callfunc("DM_InstanceSetFlag", "dm_arc04_elsbeth_hurt", 1);
	}
	if (.@adds < 1)
		.@adds = 1;

	'boss_up = 1;
	mapannounce .@map$, "[Baphomet's Seal] The prison-prayer breaks rhythm. Baphomet answers.", bc_map, 0xFFAA33;
	monster .@map$, .@x, .@y, "Baphomet", 1039, 1, "Baphomet's Seal#dm::OnBaphometDead";
	monster .@map$, .@x + 2, .@y + 2, "Doppelganger", 1046, 1, "Baphomet's Seal#dm::OnDoppelDead";
	areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Prison Echo", 1109, .@adds, "Baphomet's Seal#dm::OnAddDead";
	areamonster .@map$, .@x - 5, .@y - 5, .@x + 5, .@y + 5, "Tower Marionette", 1143, 2, "Baphomet's Seal#dm::OnAddDead";
	close();

OnAddDead:
	end;

OnDoppelDead:
	mapannounce strnpcinfo(NPC_MAP), "[Baphomet's Seal] The false hero collapses into old seal-light.", bc_map, 0x88CCFF;
	end;

OnBaphometDead:
	'boss_up = 0;
	.@map$ = strnpcinfo(NPC_MAP);
	killmonster .@map$, "Baphomet's Seal#dm::OnAddDead";
	callfunc("DM_InstanceQuestComplete", 20023);
	callfunc("DM_InstanceSetFlag", "dm_arc04_keystone_shard", 1);
	callfunc("DM_InstanceQuestStart", 20024);
	callfunc("DM_PartyExp", 210000, 90000);
	callfunc("DM_GivePartyZeny", 18000);
	callfunc("DM_PartyApplyFlag", "dm_story_beat", 498);
	mapannounce .@map$, "[Baphomet's Seal] Baphomet is driven back. A Keystone Shard breaks loose from the vault.", bc_map, 0x88CCFF;
	dispbottom "[Seal Cascade] The Pilgrim's Offer is now available near the cracked seal.";
	end;

OnInit:
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20023, 1;
	questinfo QTYPE_QUEST2, 1;
	setquestinfo QINFO_QUEST, 20000, 1;
	end;
}
