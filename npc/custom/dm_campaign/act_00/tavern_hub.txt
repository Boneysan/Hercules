prt_in,163,22,5	script	Tavern Keeper	4_M_01,{
	mes "[Tavern Keeper]";
	mes "Welcome to the Seal Cascade inn. Rest up, adventurers.";
	next;
	if (callfunc("DM_IsDM")) {
		mes "DM options:";
		if (select("Rest Party", "Nothing") == 1) {
			callfunc("DM_Rest");
		}
		close;
	}
	
	if (dm_campaign_complete) {
		mes "The rooms are quiet again. Funny how a city sounds after people stop waiting for the world to end.";
		close;
	}
	if (dm_wynne_lost) {
		mes "Nobody sits in Wynne's old corner anymore. The ledger is still there. Nobody likes touching it.";
		close;
	}
	if (dm_wynne_rescued && dm_arc10_kiel_killed) {
		mes "Wynne came through after Lighthalzen, limping and swearing. Best sound this room heard all week.";
		close;
	}
	if (dm_wynne_captured) {
		mes "A Guild courier came in asking after Wynne, then left when nobody answered. That is never a small problem.";
		close;
	}
	if (dm_arc19_started) {
		mes "Everyone in the common room is pretending Morroc is far away. Nobody believes it.";
		close;
	}
	if (dm_arc16_started) {
		mes "Half the guests are officials who suddenly discovered they had relatives elsewhere. The other half are refugees who already knew how packing works.";
		close;
	}
	if (dm_arc12_started) {
		mes "The foreign traders stopped arguing about prices and started arguing about which sky is the real one. That's usually a bad sign.";
		close;
	}
	mes "We have rooms available if you need them. Pay up front. Apocalypses are bad for bookkeeping.";
	close;
}

function	script	DM_RestItem	{
	percentheal 100, 100;
	savepoint "prt_in", 163, 22;
	sc_start SC_INC_AGI, 600000, 10;
	sc_start SC_BLESSING, 600000, 10;
	return;
}

function	script	DM_Rest	{
	.@done = callfunc("DM_PartyForEach", "DM_RestItem", $dm_active_party);
	dispbottom("[DM] Party rested (" + .@done + " members). HP/SP restored, savepoint updated, buffs applied.");
	return 1;
}
