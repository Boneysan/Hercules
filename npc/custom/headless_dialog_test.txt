//===== Hercules Script ======================================
//= Headless Dialogue Test Helper NPC
//============================================================

prontera,160,200,4	script	Dialogue Test NPC	4_F_01,{
	mes "[Dialogue Test NPC]";
	mes "Hello! I am a helper NPC for headless client testing.";
	next;
	mes "Select a test scenario:";
	switch(select("Linear Dialogue", "Number Input", "String Input", "Dialogue Warp")) {
		case 1:
			mes "You chose Linear Dialogue.";
			next;
			mes "This is the next screen.";
			close;
		case 2:
			mes "Please input a number:";
			next;
			input(.@num);
			mes "You entered number: " + .@num;
			close;
		case 3:
			mes "Please input a string:";
			next;
			input(.@str$);
			mes "You entered string: " + .@str$;
			close;
		case 4:
			mes "Warping you to payon...";
			close2;
			warp "payon",150,150;
			end;
	}
}

// Disposable-runner-only quest award fixture. It exercises the same authored
// DM_PartyExp path as campaign turn-ins without changing a production quest.
prontera,163,200,4	script	EXP Quest Award Test NPC	4_F_01,{
	mes "[EXP Quest Award Test]";
	mes "A small authored quest award fixture.";
	next;
	if (!questprogress(2000))
		setquest 2000;
	if (questprogress(2000) == 1) {
		callfunc("DM_PartyExp", 1000, 500);
		completequest 2000;
		mes "Quest awarded 1000 base / 500 job EXP.";
		close;
	}
	mes "Quest already complete.";
	close;
}
